100 // ===========================================================================
101 // XC nuclear gradient at fixed density matrix, part 1: basis-function
102 // (Pulay) and grid-point translation terms, for LDA and GGA (the GGA sigma
103 // terms use AOShell::EvalAOspaceHessian and vanish for LDA, where
104 // df_dsigma stays 0). Despite the name it computes both terms, which
105 // share one AO evaluation per grid point. Derivation in vxc_potential.cc.
106 // ===========================================================================
107 Eigen::MatrixXd PulayGradient(const Eigen::MatrixXd& density_matrix,
108 const AOBasis& dftbasis) const;
109
110 // ===========================================================================
111 // XC nuclear gradient, part 2: derivative of the SSW grid weights. Only the
112 // sum PulayGradient + GridWeightGradient equals the derivative of the XC
113 // energy of IntegrateVXC (checked by finite differences in
114 // test_xcgradient.cc). Cost is O(Natoms^3) per grid point, one power of
115 // Natoms more than the SSW partition itself.
116 // ===========================================================================
117 Eigen::MatrixXd GridWeightGradient(const Eigen::MatrixXd& density_matrix,
118 const QMMolecule& atoms) const;
119
120 // ===========================================================================
121 // First-order change of PulayGradient + GridWeightGradient along a
122 // symmetric density direction P: d/dt [PulayGradient + GridWeightGradient]
123 // (D + t P) at t = 0, analytically in one grid pass (the XC part of
124 // relaxed excited-state gradients). Needs second functional derivatives
125 // from libxc. Checked against finite differences in test_xcgradient.cc.
126 // ===========================================================================
127 Eigen::MatrixXd GradientResponse(const Eigen::MatrixXd& density_matrix,
128 const Eigen::MatrixXd& direction,
129 const QMMolecule& atoms) const;
130
131 // ===========================================================================
132 // UKS (spin-polarized) analogs of PulayGradient/GridWeightGradient
133 // above, for LDA and GGA.
134 //
135 // Basis-type and translation-type rho terms, for each spin channel
136 // s in {a,b}:
137 // d(rho_s,p)/dR_A|_basis = -2 * sum_{mu in A} temp_s,mu * grad(chi_mu)
138 // d(rho_s,p)/dR_A|_translation = grad_s,p if A == owner(p), else 0
139 // with temp_s = dmat_s * ao.values and grad_s = 2*(ao.derivatives^T *
140 // temp_s), as in IntegrateVXCSpin (UKS spin density matrices are not
141 // doubled, unlike DMAT_here = 2*P for RKS), contracted with vrho_a/vrho_b
142 // and summed over both spins.
143 //
144 // GGA: the three sigma_aa/sigma_ab/sigma_bb terms reduce to the per-spin
145 // machinery of the restricted GGA case (Gmat_s, Hessian_rho_s) applied
146 // once per spin, with the effective gradient vectors
147 // V_a = 2*vsigma_aa*rho_a_grad + vsigma_ab*rho_b_grad
148 // V_b = 2*vsigma_bb*rho_b_grad + vsigma_ab*rho_a_grad
149 // Basis-type sigma contribution per mu in A:
150 // -[grad(chi_mu)*(Gmat_a.V_a)_mu + temp_a,mu*(Hessian_mu.V_a)
151 // + grad(chi_mu)*(Gmat_b.V_b)_mu + temp_b,mu*(Hessian_mu.V_b)]
152 // Translation-type sigma contribution (A==owner(p) only):
153 // Hessian_rho_a . V_a + Hessian_rho_b . V_b
154 // where Gmat_s = 2*dmat_s*ao.derivatives and Hessian_rho_s is the
155 // restricted Hessian_rho per spin channel.
156 //
157 // The weight-derivative term uses the same dw/dR as GridWeightGradient;
158 // only the energy density changes to (rho_a+rho_b)*f_xc, evaluated with
159 // sigma_aa/sigma_ab/sigma_bb from the density gradients, as in the energy
160 // of IntegrateVXCSpin.
161 Eigen::MatrixXd PulayGradientUKS(const Eigen::MatrixXd& dmat_alpha,
162 const Eigen::MatrixXd& dmat_beta,
163 const AOBasis& dftbasis) const;
164 Eigen::MatrixXd GridWeightGradientUKS(const Eigen::MatrixXd& dmat_alpha,
165 const Eigen::MatrixXd& dmat_beta,
166 const QMMolecule& atoms) const;
