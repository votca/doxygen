20// Analytic nuclear gradient of a BSE excitation energy (TDA or full) with
21// static RPA screening and fixed (geometry-independent) QP corrections.
22//
23// Notation. p, q, r: MOs; i, j, k: occupied, a, b: virtual (all MOs of the
24// closed-shell KS reference); v, c: occupied/virtual levels of the BSE window
25// [vmin, cmax]; k, d: occupied/virtual levels of the RPA window
26// [rpamin, rpamax]. B^P_pq = sum_R L_RP (R|pq) with L L^T = V^-1 (here
27// L = V^-1/2; the energy is invariant under orthogonal rotations of the aux
28// index, so VOTCA's Pseudo_InvSqrt_GWBSE frame gives the same Omega).
29//
30// Energy, written here for the TDA (X normalized, kappa = 1 for singlets, 0
31// for triplets; the full BSE adds Y, see the amplitude section below):
32// Omega = sum_vc X_vc^2 (e_c - e_v) + 2 kappa |x|^2
33// - sum_{vc,v'c'} X_vc X_v'c' B_vv'^T M B_cc',
34// x^P = sum_vc X_vc B^P_vc, M = eps^-1,
35// eps = 1 + sum_kd f(Delta_kd) b_kd b_kd^T, f(D) = 4 D / (D^2 + eta^2),
36// Delta_kd = e'_d - e'_k,
37// with e_p = eps_p + a_p (BSE) and e'_p = eps_p + pi_p (RPA), a and pi fixed.
38// By Hellmann-Feynman dOmega = X^T dH X, so Omega depends on the geometry
39// through B (integrals and MO coefficients) and the KS energies eps_p:
40// dOmega = sum_P g^P . dB^P + sum_p w_p deps_p,
41// g^P = dOmega/dB^P (exchange, direct, and screening response through M),
42// w_p = dOmega/deps_p (diagonal of the BSE difference density and of the
43// screening response density, cf. Holzer, arXiv:2611.12617).
44//
45// dB^P has an integral part (three-centre and metric derivatives at fixed
46// MO coefficients, ContractThreeCenterDerivatives and
47// ContractCoulombMetricDerivatives with
48// Gamma^R = C (sum_P L_RP g^P) C^T and G = -1/2 L (g . B)_sym L^T) and an
49// orbital part sum_rp U_rp Q_rp with Q_rp = sum_{P,q} B^P_rq (g^P_pq +
50// g^P_qp) for the MO response C -> C (1 + U). The canonical KS orbitals
51// respond as
52// same space, r != p: U_rp = (F'_rp - eps_p S_rp) / (eps_p - eps_r)
53// diagonal: U_pp = -S_pp / 2, deps_p = F'_pp - eps_p S_pp
54// occupied-virtual: U_ia = -S_ia - U_ai, U_ai from coupled-perturbed KS
55// with S_pq = C_p^T dS C_q, F' = F^(R) + G[dD], F^(R) the derivative of the
56// Fock matrix at fixed AO density and G the closed-shell Fock response
57// (J + 1/2 a_x K + f_xc) to the density change dD (dD_ai = dD_ia = 2 U_ai,
58// dD_jk = -2 S_jk). Collecting terms:
59// D_rp = (Q_rp - Q_pr) / (2 (eps_p - eps_r)) same space, r != p
60// D_pp = w_p
61// L_ai = Q_ai - Q_ia + 4 G_ai[D], (A + B) Z = L (Z-vector)
62// Ptilde = D (occ-occ, virt-virt), Ptilde_ai = Ptilde_ia = -Z_ai / 2
63// W_rp = (Q_rp eps_p - Q_pr eps_r) / (2 (eps_p - eps_r)) same space
64// W_pp = Q_pp / 2 + w_p eps_p, W_jk += 2 G_jk[Ptilde],
65// W_ai = W_ia = (Q_ia - Z_ai eps_i) / 2
66// and
67// dOmega/dR = (integral part) + sum Ptilde_AO dF/dR|_D - sum W_AO dS/dR.
68// The Fock derivative at fixed density contracted with Ptilde is split into
69// the one-electron part, RI-J and RI-K (exact, both are bilinear) and XC
70// (d/dt of the ground-state XC gradient at D + t Ptilde, analytic in
71// Vxc_Potential::GradientResponse), which contains the f_xc term of the moving
72// basis functions and grid. No g_xc is needed: the BSE kernel has no XC.
73//
74// Degenerate KS levels within a space get D_rp = 0 (the antisymmetric part
75// of Q vanishes for symmetry-related partners, not for accidental
76// degeneracies).

106Eigen::MatrixXd ContractAODerivatives(const std::vector<AOMatrixDerivative>& d,
107 const Eigen::MatrixXd& P) {
108 Eigen::MatrixXd grad = Eigen::MatrixXd::Zero(Index(d.size()), 3);
109 for (Index a = 0; a < Index(d.size()); ++a) {
110 for (Index xyz = 0; xyz < 3; ++xyz) {
111 grad(a, xyz) = P.cwiseProduct(d[a][xyz]).sum();
112 }
113 }
114 return grad;
115}

117// A <- A B for a square B, by blocks of rows (scratch of one block)
118void RightMultiplyInPlace(Eigen::MatrixXd& A, const Eigen::MatrixXd& B) {
119 const Index rows = A.rows();
120 const Index block = std::max<Index>(1, std::min<Index>(rows, 4096));
121 for (Index r0 = 0; r0 < rows; r0 += block) {
122 const Index n = std::min(block, rows - r0);
123 Eigen::MatrixXd tmp = A.middleRows(r0, n) * B;
124 A.middleRows(r0, n) = tmp;
125 }
126}

130DFTEngine::BSEGradientResult DFTEngine::BSEExcitationGradient(
131 const Orbitals& orb, const BSEGradientInput& in) {
132 if (auxbasis_name_.empty()) {
133 throw std::runtime_error(
134 "BSEExcitationGradient: needs the RI ground state (auxbasisset)");
135 }
136 if (!ecp_name_.empty()) {
137 throw std::runtime_error("BSEExcitationGradient: ECPs are not supported");
138 }
139 if (!HasLibint2DerivativeSupport()) {
140 throw std::runtime_error(
141 "BSEExcitationGradient: libint2 lacks derivative integrals");
142 }
143 Orbitals work = orb;
144 Prepare(work);
145 if (num_alpha_electrons_ != num_beta_electrons_) {
146 throw std::runtime_error(
147 "BSEExcitationGradient: closed-shell ground states only");
148 }
149 const QMMolecule& mol = work.QMAtoms();
150 const Vxc_Potential<Vxc_Grid> vxc = SetupVxc(mol);
151 // wall time of the phases, reported at the end
152 std::vector<std::pair<std::string, double>> phase_times;
153 auto phase_start = std::chrono::steady_clock::now();
154 auto phase = [&](const std::string& name) {
155 const auto now = std::chrono::steady_clock::now();
156 phase_times.emplace_back(
157 name, std::chrono::duration<double>(now - phase_start).count());
158 phase_start = now;
159 };
160
161 const Eigen::MatrixXd& C = orb.MOs().eigenvectors();
162 const Eigen::VectorXd& eps = orb.MOs().eigenvalues();
163 const Index nmo = C.cols();
164 const Index nocc = num_docc_;
165 const Index nvirt = nmo - nocc;
166 const Index homo = nocc - 1;
167 const Eigen::MatrixXd C_occ = C.leftCols(nocc);
168 const Eigen::MatrixXd C_virt = C.rightCols(nvirt);
169 const Eigen::MatrixXd D0 = 2.0 * C_occ * C_occ.transpose();
170
171 // windows
172 if (in.vmin < 0 || in.vmin > homo || in.cmax <= homo || in.cmax >= nmo ||
173 in.rpamin < 0 || in.rpamin > homo || in.rpamax <= homo ||
174 in.rpamax >= nmo) {
175 throw std::runtime_error("BSEExcitationGradient: invalid windows");
176 }
177 const Index vt = homo - in.vmin + 1;
178 const Index ct = in.cmax - homo;
179 // targets: the BSE excitation energy, or the energies of QP levels
180 const bool has_bse = in.qp_levels.empty();
181 if ((has_bse && in.X.size() != vt * ct) ||
182 (has_bse && in.Y.size() != 0 && in.Y.size() != vt * ct) ||
183 in.energies_bse.size() != in.cmax - in.vmin + 1 ||
184 in.energies_rpa.size() != in.rpamax - in.rpamin + 1) {
185 throw std::runtime_error("BSEExcitationGradient: input sizes");
186 }
187 if (in.qp_response &&
188 (in.qpmin < in.rpamin || in.qpmax > in.rpamax || in.qpmin > in.qpmax ||
189 in.qp_energies.size() != in.qpmax - in.qpmin + 1)) {
190 throw std::runtime_error(
191 "BSEExcitationGradient: QP window must lie in the RPA window");
192 }
193 for (Index level : in.qp_levels) {
194 if (level < in.vmin || level > in.cmax) {
195 throw std::runtime_error(
196 "BSEExcitationGradient: QP level outside the window");
197 }
198 }
199 const Index lo = std::min(in.vmin, in.rpamin);
200 const Index hi = std::max(in.cmax, in.rpamax);
201 const Index nw = hi - lo + 1;
202 const Eigen::MatrixXd Cw = C.middleCols(lo, nw);
203 const Index v0 = in.vmin - lo; // window offsets
204 const Index c0 = homo + 1 - lo; // first virtual
205 const Index k0 = in.rpamin - lo;
206 const Index nk = homo - in.rpamin + 1;
207 const Index nd = in.rpamax - homo;
208 const double kappa = in.singlet ? 1.0 : 0.0;
209
210 // energies used in the Hamiltonian, by window position
211 Eigen::VectorXd eA = Eigen::VectorXd::Zero(nw);
212 eA.segment(v0, in.cmax - in.vmin + 1) = in.energies_bse;
213 Eigen::VectorXd eP = Eigen::VectorXd::Zero(nw);
214 eP.segment(k0, in.rpamax - in.rpamin + 1) = in.energies_rpa;
215
216 // ---- B^P_pq over the window, L = V^-1/2
217 AOCoulomb aocoulomb;
218 aocoulomb.Fill(auxbasis_);
219 const Index naux = auxbasis_.AOBasisSize();
220 Eigen::MatrixXd L;
221 {
222 const SymmetricEigenSystem es = SymmetricEigen(aocoulomb.Matrix());
223 if (es.values.minCoeff() <= 0.0) {
224 throw std::runtime_error(
225 "BSEExcitationGradient: aux Coulomb metric not positive definite");
226 }
227 L = es.vectors * es.values.cwiseInverse().cwiseSqrt().asDiagonal() *
228 es.vectors.transpose();
229 }
230 // (R|pq) first, then B = J L in place by blocks of rows: one array of
231 // nw^2 x Naux (the size of the GW-BSE integrals) instead of two
232 Eigen::MatrixXd Bmat(nw * nw, naux);
233 ForEachThreeCenterIntegral(
234 auxbasis_, dftbasis_,
235 [&Cw, &Bmat, nw](Index R, const Eigen::MatrixXd& I) {
236 Eigen::MatrixXd m = Cw.transpose() * (I * Cw);
237 Bmat.col(R) = Eigen::Map<const Eigen::VectorXd>(m.data(), nw * nw);
238 });
239 RightMultiplyInPlace(Bmat, L);
240 auto B = [&Bmat, nw](Index P) {
241 return Eigen::Map<const Eigen::MatrixXd>(Bmat.col(P).data(), nw, nw);
242 };
243
244 // X (and Y for the full BSE) as (v, c); the energy functional
245 // Omega = (X A X + Y A Y + 2 X B Y) / (X X - Y Y)
246 // is stationary in X, Y, so dOmega = (X dA X + Y dA Y + 2 X dB Y) / N
247 const bool full = has_bse && in.Y.size() > 0;
248 const Eigen::MatrixXd Xvc =
249 has_bse ? Eigen::MatrixXd(
250 Eigen::Map<const Eigen::MatrixXd>(in.X.data(), ct, vt)
251 .transpose())
252 : Eigen::MatrixXd::Zero(vt, ct);
253 const Eigen::MatrixXd Yvc =
254 full ? Eigen::MatrixXd(
255 Eigen::Map<const Eigen::MatrixXd>(in.Y.data(), ct, vt)
256 .transpose())
257 : Eigen::MatrixXd::Zero(vt, ct);
258 const double norm = has_bse ? Xvc.squaredNorm() - Yvc.squaredNorm() : 1.0;
259 if (norm <= 0.0) {
260 throw std::runtime_error("BSEExcitationGradient: X X - Y Y <= 0");
261 }
262
263 phase("three-index integrals B");
264 // ---- energy and its partial derivatives g^P (nw x nw per P), w
265 Eigen::MatrixXd gmat = Eigen::MatrixXd::Zero(nw * nw, naux);
266 auto g = [&gmat, nw](Index P) {
267 return Eigen::Map<Eigen::MatrixXd>(gmat.col(P).data(), nw, nw);
268 };
269 Eigen::VectorXd w = Eigen::VectorXd::Zero(nw); // dOmega/deps, window
270
271 // dOmega/de of the BSE energies (wA) and of the RPA energies (w)
272 Eigen::VectorXd wA = Eigen::VectorXd::Zero(nw);
273 double omega_qp = 0.0;
274 for (Index v = 0; v < vt; ++v) {
275 for (Index c = 0; c < ct; ++c) {
276 const double x2 = Xvc(v, c) * Xvc(v, c) + Yvc(v, c) * Yvc(v, c);
277 omega_qp += x2 * (eA(c0 + c) - eA(v0 + v));
278 wA(c0 + c) += x2;
279 wA(v0 + v) -= x2;
280 }
281 }
282
283 // exchange, A and B: 2 kappa |x + y|^2
284 const Eigen::MatrixXd XpY = Xvc + Yvc;
285 Eigen::VectorXd x(naux);
286 for (Index P = 0; P < naux; ++P) {
287 x(P) = XpY.cwiseProduct(B(P).block(v0, c0, vt, ct)).sum();
288 }
289 const double omega_x = 2.0 * kappa * x.squaredNorm();
290 for (Index P = 0; P < naux; ++P) {
291 g(P).block(v0, c0, vt, ct) += 4.0 * kappa * x(P) * XpY;
292 }
293
294 // static RPA screening
295 Eigen::MatrixXd Bkd(nk * nd, naux);
296 Eigen::VectorXd f(nk * nd);
297 Eigen::VectorXd fprime(nk * nd);
298 const double eta2 = in.eta * in.eta;
299 for (Index d = 0; d < nd; ++d) {
300 for (Index k = 0; k < nk; ++k) {
301 const double Delta = eP(c0 + d) - eP(k0 + k);
302 const double den = Delta * Delta + eta2;
303 f(k + nk * d) = 4.0 * Delta / den;
304 fprime(k + nk * d) = 4.0 * (eta2 - Delta * Delta) / (den * den);
305 }
306 }
307 for (Index P = 0; P < naux; ++P) {
308 const auto b = B(P);
309 for (Index d = 0; d < nd; ++d) {
310 Bkd.col(P).segment(nk * d, nk) = b.block(k0, c0 + d, nk, 1);
311 }
312 }
313 Eigen::MatrixXd epsilon = Bkd.transpose() * f.asDiagonal() * Bkd;
314 epsilon.diagonal().array() += 1.0;
315 Eigen::MatrixXd M;
316 {
317 SymmetricEigenSystem es = SymmetricEigen(epsilon);
318 for (Index i = 0; i < es.values.size(); ++i) {
319 es.values(i) = (es.values(i) > 1e-8) ? 1.0 / es.values(i) : 0.0;
320 }
321 M = es.vectors * es.values.asDiagonal() * es.vectors.transpose();
322 }
323
324 double omega_d = 0.0;
325 if (has_bse) {
326 // direct term of A: Y^Q = X B^Q_cc X^T + Y B^Q_cc Y^T (v x v),
327 // Z^P = X^T B^P_vv X + Y^T B^P_vv Y (c x c)
328 Eigen::MatrixXd Bvv(vt * vt, naux);
329 Eigen::MatrixXd Ymat(vt * vt, naux);
330 Eigen::MatrixXd Zmat(ct * ct, naux);
331 for (Index P = 0; P < naux; ++P) {
332 const auto b = B(P);
333 const Eigen::MatrixXd bvv = b.block(v0, v0, vt, vt);
334 const Eigen::MatrixXd bcc = b.block(c0, c0, ct, ct);
335 Eigen::MatrixXd Yq = Xvc * bcc * Xvc.transpose();
336 Eigen::MatrixXd Zc = Xvc.transpose() * bvv * Xvc;
337 if (full) {
338 Yq += Yvc * bcc * Yvc.transpose();
339 Zc += Yvc.transpose() * bvv * Yvc;
340 }
341 Bvv.col(P) = Eigen::Map<const Eigen::VectorXd>(bvv.data(), vt * vt);
342 Ymat.col(P) = Eigen::Map<const Eigen::VectorXd>(Yq.data(), vt * vt);
343 Zmat.col(P) = Eigen::Map<const Eigen::VectorXd>(Zc.data(), ct * ct);
344 }
345 Eigen::MatrixXd GammaW = Bvv.transpose() * Ymat; // G_PQ = B^P . Y^Q
346 {
347 const Eigen::MatrixXd MY = Ymat * M;
348 const Eigen::MatrixXd MZ = Zmat * M;
349 for (Index P = 0; P < naux; ++P) {
350 g(P).block(v0, v0, vt, vt) -=
351 Eigen::Map<const Eigen::MatrixXd>(MY.col(P).data(), vt, vt);
352 g(P).block(c0, c0, ct, ct) -=
353 Eigen::Map<const Eigen::MatrixXd>(MZ.col(P).data(), ct, ct);
354 }
355 }
356 // direct term of B (full BSE): 2 X Hd2 Y with
357 // Hd2(v1c1, v2c2) = -B_{v1c2}^T M B_{v2c1}, i.e.
358 // -2 sum_PQ M_PQ B^P_vc . T^Q, T^Q = X (B^Q_vc)^T Y (v x c)
359 // and dT^Q/dB^Q gives T'^P = Y (B^P_vc)^T X.
360 if (full) {
361 Eigen::MatrixXd Bvc(vt * ct, naux);
362 Eigen::MatrixXd Tmat(vt * ct, naux);
363 Eigen::MatrixXd Tpmat(vt * ct, naux);
364 for (Index P = 0; P < naux; ++P) {
365 const Eigen::MatrixXd bvc = B(P).block(v0, c0, vt, ct);
366 const Eigen::MatrixXd T = Xvc * bvc.transpose() * Yvc;
367 const Eigen::MatrixXd Tp = Yvc * bvc.transpose() * Xvc;
368 Bvc.col(P) = Eigen::Map<const Eigen::VectorXd>(bvc.data(), vt * ct);
369 Tmat.col(P) = Eigen::Map<const Eigen::VectorXd>(T.data(), vt * ct);
370 Tpmat.col(P) = Eigen::Map<const Eigen::VectorXd>(Tp.data(), vt * ct);
371 }
372 GammaW += 2.0 * Bvc.transpose() * Tmat;
373 const Eigen::MatrixXd MT = (Tmat + Tpmat) * M;
374 for (Index P = 0; P < naux; ++P) {
375 g(P).block(v0, c0, vt, ct) -=
376 2.0 * Eigen::Map<const Eigen::MatrixXd>(MT.col(P).data(), vt, ct);
377 }
378 }
379 omega_d = -(M.cwiseProduct(GammaW)).sum();
380 GammaW = 0.5 * (GammaW + GammaW.transpose()).eval();
381 // screening response: dOmega = Tr(Gbar d eps), Gbar = M GammaW M
382 {
383 const Eigen::MatrixXd Gbar = M * GammaW * M;
384 const Eigen::MatrixXd GB = Bkd * Gbar; // (kd x P)
385 for (Index P = 0; P < naux; ++P) {
386 for (Index d = 0; d < nd; ++d) {
387 g(P).block(k0, c0 + d, nk, 1) +=
388 2.0 *
389 f.segment(nk * d, nk).cwiseProduct(GB.col(P).segment(nk * d, nk));
390 }
391 }
392 const Eigen::VectorXd bGb = Bkd.cwiseProduct(GB).rowwise().sum();
393 for (Index d = 0; d < nd; ++d) {
394 for (Index k = 0; k < nk; ++k) {
395 const double t = fprime(k + nk * d) * bGb(k + nk * d);
396 w(c0 + d) += t;
397 w(k0 + k) -= t;
398 }
399 }
400 }
401 }
402
403 phase("BSE energy and partial derivatives");
404 // normalization of the full BSE (1 for TDA and normalized vectors)
405 gmat /= norm;
406 w /= norm;
407 wA /= norm;
408 BSEGradientResult result;
409 result.omega = (omega_qp + omega_x + omega_d) / norm;
410
411 // u_p: coefficients of sum_p u_p <p|V_xc|p> in the energy (QP response)
412 Eigen::VectorXd u = Eigen::VectorXd::Zero(nmo);
413
414 // ---- from the partial derivatives g, w, u of one target to its gradient
415 auto downstream = [&](bool last) -> Eigen::MatrixXd {
416 // ---- integral part of sum g . dB
417 // metric term first (needs g and B), then B is released (after the last
418 // target) and g becomes
419 // gL (column R: sum_P L_RP g^P) in place; the integral derivatives and
420 // the orbital part Q (with s^R = gL^R + gL^R^T) both use gL
421 Eigen::MatrixXd grad;
422 {
423 Eigen::MatrixXd G = gmat.transpose() * Bmat;
424 if (last) {
425 Bmat.resize(0, 0);
426 }
427 G = (0.5 * (G + G.transpose())).eval();
428 grad = -0.5 * ContractCoulombMetricDerivatives(auxbasis_, L * G * L);
429 }
430 RightMultiplyInPlace(gmat, L);
431 const Eigen::MatrixXd& gL = gmat;
432 grad += ContractThreeCenterDerivatives(
433 auxbasis_, dftbasis_, [&gL, &Cw, nw](Index R, Eigen::MatrixXd& Gamma) {
434 Gamma = Cw *
435 (Eigen::Map<const Eigen::MatrixXd>(gL.col(R).data(), nw, nw) *
436 Cw.transpose());
437 });
438
439 phase("integral derivatives of B");
440 // ---- orbital part: Q_rp = sum_{P,q} B^P_rq s^P_qp, s^P = g^P + g^P^T,
441 // for all r (all MOs), p in the window
442 Eigen::MatrixXd Q = Eigen::MatrixXd::Zero(nmo, nmo);
443 {
444 Eigen::MatrixXd Qw = Eigen::MatrixXd::Zero(nmo, nw);
445 ForEachThreeCenterIntegral(
446 auxbasis_, dftbasis_, [&](Index R, const Eigen::MatrixXd& I) {
447 const Eigen::Map<const Eigen::MatrixXd> gLR(gL.col(R).data(), nw,
448 nw);
449 const Eigen::MatrixXd sLR = gLR + gLR.transpose();
450 const Eigen::MatrixXd t = C.transpose() * (I * (Cw * sLR));
451#pragma omp critical(bsegradient_Q)
452 { Qw += t; }
453 });
454 Q.middleCols(lo, nw) = Qw;
455 }
456 gmat.resize(0, 0);
457 // v_xc expectation values (QP response): orbital part 2 u_p Vxc_rp
458 const bool has_u = u.cwiseAbs().maxCoeff() > 0.0;
459 if (has_u) {
460 const Eigen::MatrixXd Vxc_mo =
461 C.transpose() * vxc.IntegrateVXC(D0).matrix() * C;
462 Q += 2.0 * Vxc_mo * u.asDiagonal();
463 }
464 Eigen::VectorXd wfull = Eigen::VectorXd::Zero(nmo);
465 wfull.segment(lo, nw) = w;
466
467 phase("orbital part Q");
468 // ---- D, Theta (same-space blocks and diagonal)
469 const double kDegenerate = 1e-8;
470 Eigen::MatrixXd Dmo = Eigen::MatrixXd::Zero(nmo, nmo);
471 Eigen::MatrixXd Wmo = Eigen::MatrixXd::Zero(nmo, nmo);
472 auto same_space = [&](Index first, Index n) {
473 for (Index r = first; r < first + n; ++r) {
474 for (Index p = first; p < first + n; ++p) {
475 if (r == p) {
476 Dmo(p, p) = wfull(p);
477 Wmo(p, p) = 0.5 * Q(p, p) + wfull(p) * eps(p);
478 continue;
479 }
480 const double de = eps(p) - eps(r);
481 if (std::abs(de) < kDegenerate) {
482 Wmo(r, p) = 0.25 * (Q(r, p) + Q(p, r));
483 continue;
484 }
485 Dmo(r, p) = (Q(r, p) - Q(p, r)) / (2.0 * de);
486 Wmo(r, p) = (Q(r, p) * eps(p) - Q(p, r) * eps(r)) / (2.0 * de);
487 }
488 }

493 // closed-shell Fock response G[X] (MO in, MO out) to a symmetric MO
494 // density change X
495 auto fock_response = [&](const Eigen::MatrixXd& Xmo) {
496 const Eigen::MatrixXd Xao = C * Xmo * C.transpose();
497 Eigen::MatrixXd G = vxc.IntegrateFXC(D0, {Xao})[0];
498 G += ERIs_.CalculateERIs_3c(Xao);
499 if (ScaHFX_ > 0) {
500 G += 0.5 * ScaHFX_ * ERIs_.CalculateEXX_3c(Eigen::MatrixXd(0, 0), Xao);
501 }
502 return Eigen::MatrixXd(C.transpose() * G * C);
503 };

505 // XC-kernel response alone (the v_xc expectation values), MO in and out
506 Eigen::MatrixXd Gu;
507 if (has_u) {
508 const Eigen::MatrixXd Uao = C * u.asDiagonal() * C.transpose();
509 Gu = C.transpose() * vxc.IntegrateFXC(D0, {Uao})[0] * C;
510 }
511
512 phase("Fock responses");
513 // ---- Z-vector: (A + B) Z = L
514 Eigen::MatrixXd Zvo;
515 {
516 const Eigen::MatrixXd GD = fock_response(Dmo);
517 Eigen::MatrixXd Lvo = Q.block(nocc, 0, nvirt, nocc) -
518 Q.block(0, nocc, nocc, nvirt).transpose() +
519 4.0 * GD.block(nocc, 0, nvirt, nocc);
520 if (has_u) {
521 Lvo += 4.0 * Gu.block(nocc, 0, nvirt, nocc);
522 }
523 // StaticResponse solves (A + B) U = -C_v^T h C_o
524 AOOverlap overlap;
525 overlap.Fill(dftbasis_);
526 const Eigen::MatrixXd& S = overlap.Matrix();
527 Eigen::MatrixXd h = -(S * C_virt * Lvo * C_occ.transpose() * S);
528 h = (h + h.transpose()).eval();
529 // the gradient error is linear in the Z residual: 1e-6 relative changed
530 // the forces of pyridine + 6 H2O by 1e-8 Ha/Bohr against 1e-8 (three
531 // iterations fewer), far below the FD agreement and optimizer thresholds
532 std::vector<Eigen::MatrixXd> dP = StaticResponse(orb, {h}, 1e-6, 200);
533 Zvo = 0.5 * C_virt.transpose() * S * dP[0] * S * C_occ;
534 }
535
536 phase("Z-vector");
537 // ---- relaxed density Ptilde and energy-weighted density W
538 Eigen::MatrixXd Pmo = Dmo;
539 Pmo.block(nocc, 0, nvirt, nocc) = -0.5 * Zvo;
540 Pmo.block(0, nocc, nocc, nvirt) = -0.5 * Zvo.transpose();
541 {
542 const Eigen::MatrixXd GP = fock_response(Pmo);
543 Wmo.topLeftCorner(nocc, nocc) += 2.0 * GP.topLeftCorner(nocc, nocc);
544 if (has_u) {
545 Wmo.topLeftCorner(nocc, nocc) += 2.0 * Gu.topLeftCorner(nocc, nocc);
546 }
547 }
548 for (Index i = 0; i < nocc; ++i) {
549 for (Index a = 0; a < nvirt; ++a) {
550 const double wai = 0.5 * (Q(i, nocc + a) - Zvo(a, i) * eps(i));
551 Wmo(nocc + a, i) = wai;
552 Wmo(i, nocc + a) = wai;
553 }
554 }
555 const Eigen::MatrixXd Pao = C * Pmo * C.transpose();
556 const Eigen::MatrixXd Wao = C * Wmo * C.transpose();
557
558 phase("energy-weighted density");
559 // ---- sum Ptilde dF/dR|_D - sum W dS/dR
560 grad += ContractAODerivatives(ComputeKineticDerivatives(dftbasis_), Pao);
561 grad += ContractAODerivatives(
562 ComputeNuclearAttractionDerivatives(dftbasis_, mol), Pao);
563 grad -= ContractAODerivatives(ComputeOverlapDerivatives(dftbasis_), Wao);
564 phase("one-electron and overlap derivatives");
565 // RI-J, exact: the Hartree energy is quadratic in the density
566 grad += 0.5 * (DFTGradient::RIJGradient(D0 + Pao, auxbasis_, dftbasis_) -
567 DFTGradient::RIJGradient(D0 - Pao, auxbasis_, dftbasis_));
568 if (ScaHFX_ > 0) {
569 const SymmetricEigenSystem es = SymmetricEigen(Pao);
570 const double cutoff = 1e-12 * es.values.cwiseAbs().maxCoeff();
571 std::vector<Index> keep;
572 for (Index k = 0; k < es.values.size(); ++k) {
573 if (std::abs(es.values(k)) > cutoff) {
574 keep.push_back(k);
575 }
576 }
577 Eigen::MatrixXd U(es.vectors.rows(), Index(keep.size()));
578 Eigen::VectorXd lambda(Index(keep.size()));
579 for (Index k = 0; k < Index(keep.size()); ++k) {
580 U.col(k) = es.vectors.col(keep[std::size_t(k)]);
581 lambda(k) = es.values(keep[std::size_t(k)]);
582 }
583 grad -= ScaHFX_ * DFTGradient::RIPairGradient(U, lambda, C_occ, auxbasis_,
584 dftbasis_);
585 }
586 phase("RI-J and RI-K");
587 // XC: d/dt of the ground-state XC gradient at D0 + t (Ptilde + U), with
588 // U = sum_p u_p |p><p| from the v_xc expectation values (analytic, one
589 // grid pass)
590 {
591 const Eigen::MatrixXd Pxc = Pao + C * u.asDiagonal() * C.transpose();
592 grad += vxc.GradientResponse(D0, Pxc, mol);
593 }
594 phase("XC");
595 return grad;

615 u.setZero();

642 run_targets([&] { w += wA; });
643 } else {
644 // ---- response of the QP corrections, plasmon-pole model of VOTCA
645 // (Sigma_PPM, PPM::PPM_construct_parameters). The QP energies E of
646 // qpmin..qpmax solve E_p = eps_p + sigma_p(E_p; R), sigma = (1 - a_x)
647 // Sigma_x - v_xc + Sigma_c, with the screening and the poles of G at
648 // the RPA energies R: G0W0: R = KS energies plus fixed shifts; evGW:
649 // R = E in the QP window and, outside, KS energies plus the
650 // out-of-window shift of the corrections E - eps (OutOfWindowShift).
651 // With the weights W of dOmega on E, the adjoint (1 - J)^T v = W,
652 // J = dsigma/dE (through omega_p = E_p and, for evGW, R), turns
653 // W . dE into sum_p v_p (deps_p + dsigma_p|_{omega, R}) plus the
654 // explicit dependence of R on eps; G0W0: v_p = Z_p W_p.
655 // Sigma_c(omega) = sum_i a_i sum_n (Bt^i_pn)^2 h(omega - e_n +- Omega_i),
656 // h(t) = t / (t^2 + eta^2), a_i = w_i Omega_i / 2, Bt = B Phi with
657 // eps(0) = Phi diag(lambda) Phi^T, w_i = 1 - 1/lambda_i,
658 // Omega_i^2 = -nu^2 (y_i - 1) / (y_i - 1 + w_i), y_i = phi_i^T eps(i nu)^-1
659 // phi_i; + for occupied, - for virtual n. The reverse pass goes through
660 // all of these down to B and R.
661 const Index q0 = in.qpmin - lo;
662 const Index nq = in.qpmax - in.qpmin + 1;
663 const Index r0 = k0;
664 const Index nr = in.rpamax - in.rpamin + 1;
665 const Index qr = in.qpmin - in.rpamin; // QP level p is RPA level qr + p
666 const double nu2 = in.ppm_imag * in.ppm_imag;
667 const double seta2 = in.sigma_eta * in.sigma_eta;
668 const double ax = ScaHFX_;
669
670 using ReversePass =
671 std::function<Eigen::VectorXd(const Eigen::VectorXd&, bool)>;
672 // the adjoint and the bookkeeping of the energy weights, the same for
673 // all self-energy models: dsigma = Sigma_c'(E_p), reverse(v, accumulate)
674 // the reverse pass of sum_p v_p sigma_p(omega_p; R) at fixed omega
675 // (returns d/dR; with accumulate also adds d/dB, d/deps_p and v_xc),
676 // poles (nr x nq): column p the part of d sigma_p/dR through the pole
677 // energies of G alone (cheap, preconditioner of the evGW adjoint)
678 auto finish = [&](const Eigen::VectorXd& dsigma, const ReversePass& reverse,
679 const Eigen::MatrixXd& poles) {
680 // weights of dOmega: wE on the QP energies, wR on the RPA energies
681 // (screening of the BSE; BSE levels outside the QP window use R)
682 Eigen::VectorXd wR = w.segment(r0, nr);
683 w.setZero();
684 Eigen::VectorXd wE = wA.segment(q0, nq);
685 for (Index j = 0; j < nw; ++j) {
686 if (j >= q0 && j < q0 + nq) {
687 continue;
688 }
689 if (j < r0 || j >= r0 + nr) {
690 if (wA(j) != 0.0) {
691 throw std::runtime_error(
692 "BSEExcitationGradient: BSE level outside the RPA window");
693 }
694 continue;
695 }
696 wR(j - r0) += wA(j);
697 }
698
699 if (!in.qp_selfconsistent) {
700 // G0W0: R = eps + fixed shifts
701 const Eigen::VectorXd v =
702 wE.cwiseQuotient((1.0 - dsigma.array()).matrix());
703 const Eigen::VectorXd rho = reverse(v, true);
704 w.segment(r0, nr) += wR + rho;
705 } else {
706 // evGW: Jacobian of the out-of-window RPA energies with respect to
707 // the corrections c = E - eps of the QP window (piecewise linear in c)
708 OutOfWindowShift shift;
709 shift.mode = OutOfWindowShift::Parse(in.out_of_window_shift);
710 shift.fit_fraction = in.out_of_window_fit_fraction;
711 // R(c, eps): the linear shift also depends on the KS energies (the
712 // abscissae of its fit), so both Jacobians are taken
713 auto rpa_energies_eps = [&](const Eigen::VectorXd& c,
714 const Eigen::VectorXd& e) {
715 Eigen::VectorXd R = e.segment(in.rpamin, nr);
716 R.segment(qr, nq) += c;
717 shift.Apply(R, e, in.rpamin, in.rpamax, homo, in.qpmin, in.qpmax);
718 return R;
719 };
720 auto rpa_energies = [&](const Eigen::VectorXd& c) {
721 return rpa_energies_eps(c, eps);
722 };
723 const Eigen::VectorXd c_ref =
724 in.qp_energies - eps.segment(in.qpmin, nq);
725 const Eigen::VectorXd R_ref = rpa_energies(c_ref);
726 result.rpa_rebuild_error =
727 (R_ref - in.energies_rpa).cwiseAbs().maxCoeff();
728 Eigen::MatrixXd Js = Eigen::MatrixXd::Zero(nr, nq); // dR/dc
729 const double dc = 1e-7;
730 for (Index q = 0; q < nq; ++q) {
731 Eigen::VectorXd cp = c_ref;
732 cp(q) += dc;
733 Eigen::VectorXd cm = c_ref;
734 cm(q) -= dc;
735 Js.col(q) = (rpa_energies(cp) - rpa_energies(cm)) / (2.0 * dc);
736 }
737 Js.middleRows(qr, nq).setZero(); // inside the window R = E itself
738 // W . dE with W_q = wE_q + wR_{window q} + sum_o wR_o Js_oq
739 const Eigen::VectorXd Wbar =
740 wE + wR.segment(qr, nq) + Js.transpose() * wR;
741 // adjoint: v_q (1 - Sigma_c'_q) - rho_q(v) - sum_o rho_o(v) Js_oq =
742 // Wbar_q, linear in v. A fixed-point iteration need not converge (it
743 // mirrors the evGW iteration, which VOTCA stabilizes by mixing), so
744 // GMRES, right-preconditioned with the same operator restricted to
745 // the pole energies of G (screening frozen); one reverse pass per
746 // iteration. Fallback: the matrix column by column (nq passes).
747 const Eigen::VectorXd diag1 = (1.0 - dsigma.array()).matrix();
748 auto apply = [&](const Eigen::VectorXd& vin) {
749 const Eigen::VectorXd rho = reverse(vin, false);
750 return Eigen::VectorXd(vin.cwiseProduct(diag1) - rho.segment(qr, nq) -
751 Js.transpose() * rho);
752 };
753 Eigen::MatrixXd Mpre =
754 -poles.middleRows(qr, nq) - Js.transpose() * poles;
755 Mpre.diagonal() += diag1;
756 const Eigen::PartialPivLU<Eigen::MatrixXd> luM(Mpre);
757 const double bnorm = Wbar.norm();
758 const double tol = 1e-11 * std::max(1.0, bnorm);
759 Eigen::VectorXd v = Eigen::VectorXd::Zero(nq);
760 Index iterations = 0;
761 bool converged = (bnorm == 0.0);
762 if (!converged) {
763 const Index max_krylov = std::min<Index>(nq, 60);
764 Eigen::MatrixXd Vk(nq, max_krylov + 1);
765 Eigen::MatrixXd Hk =
766 Eigen::MatrixXd::Zero(max_krylov + 1, max_krylov);
767 Vk.col(0) = Wbar / bnorm;
768 for (Index j = 0; j < max_krylov; ++j) {
769 Eigen::VectorXd z = apply(luM.solve(Vk.col(j)));
770 for (Index pass = 0; pass < 2; ++pass) { // Gram-Schmidt, twice
771 for (Index i = 0; i <= j; ++i) {
772 const double hij = Vk.col(i).dot(z);
773 Hk(i, j) += hij;
774 z -= hij * Vk.col(i);
775 }
776 }
777 Hk(j + 1, j) = z.norm();
778 iterations = j + 1;
779 // least squares min |bnorm e1 - H y|
780 Eigen::VectorXd rhs = Eigen::VectorXd::Zero(j + 2);
781 rhs(0) = bnorm;
782 const Eigen::MatrixXd Hj = Hk.topLeftCorner(j + 2, j + 1);
783 const Eigen::VectorXd y = Hj.colPivHouseholderQr().solve(rhs);
784 const double res = (rhs - Hj * y).norm();
785 if (res < tol || Hk(j + 1, j) < 1e-14 * bnorm) {
786 v = luM.solve(Vk.leftCols(j + 1) * y);
787 converged = true;
788 break;
789 }
790 Vk.col(j + 1) = z / Hk(j + 1, j);
791 }
792 }
793 if (converged &&
794 (apply(v) - Wbar).norm() > 1e-8 * std::max(1.0, bnorm)) {
795 converged = false;
796 }
797 if (converged) {
798 XTP_LOG(Log::error, *pLog_)
799 << TimeStamp() << " evGW adjoint: GMRES converged in "
800 << iterations << " iterations" << std::flush;
801 } else {
802 XTP_LOG(Log::error, *pLog_)
803 << TimeStamp() << " evGW adjoint: GMRES did not converge in "
804 << iterations << " iterations, building the matrix" << std::flush;
805 Eigen::MatrixXd A(nq, nq);
806 for (Index q = 0; q < nq; ++q) {
807 Eigen::VectorXd e = Eigen::VectorXd::Zero(nq);
808 e(q) = 1.0;
809 A.col(q) = apply(e);
810 }
811 const Eigen::PartialPivLU<Eigen::MatrixXd> lu(A);
812 v = lu.solve(Wbar);
813 if ((A * v - Wbar).cwiseAbs().maxCoeff() >
814 1e-8 * std::max(1.0, Wbar.cwiseAbs().maxCoeff())) {
815 throw std::runtime_error(
816 "BSEExcitationGradient: the evGW adjoint equation is "
817 "singular");
818 }
819 }
820 const Eigen::VectorXd rho = reverse(v, true);
821 // explicit eps dependence of the out-of-window R at fixed E (the dE
822 // part is in the adjoint): K_ok = dR_o/deps_k with c = E - eps
823 Eigen::VectorXd wRo = wR + rho;
824 wRo.segment(qr, nq).setZero();
825 Eigen::MatrixXd K = Eigen::MatrixXd::Zero(nr, nr);
826 const double de = 1e-7;
827 for (Index k = 0; k < nr; ++k) {
828 Eigen::VectorXd ep = eps;
829 ep(in.rpamin + k) += de;
830 Eigen::VectorXd em = eps;
831 em(in.rpamin + k) -= de;
832 const Eigen::VectorXd cp = in.qp_energies - ep.segment(in.qpmin, nq);
833 const Eigen::VectorXd cm = in.qp_energies - em.segment(in.qpmin, nq);
834 K.col(k) = (rpa_energies_eps(cp, ep) - rpa_energies_eps(cm, em)) /
835 (2.0 * de);
836 }
837 w.segment(r0, nr) += K.transpose() * wRo;
838 }
839 };

843 const SymmetricEigenSystem es0 = SymmetricEigen(epsilon);
844 const Eigen::MatrixXd& Phi = es0.vectors;
845 const Eigen::VectorXd& lam = es0.values;
846 Eigen::VectorXd fi(nk * nd);
847 Eigen::VectorXd fiprime(nk * nd);
848 for (Index d = 0; d < nd; ++d) {

863 Eigen::VectorXd nom = Eigen::VectorXd::Zero(naux);
864 Eigen::VectorXd sgn = Eigen::VectorXd::Ones(naux);

873 sgn(i) = (frac < 0.0) ? -1.0 : 1.0;
874 Om(i) = std::sqrt(std::abs(frac));
875 a(i) = 0.5 * wi * Om(i);
876 }
877 const Eigen::VectorXd epole = eP.segment(r0, nr);

884 // per QP level: B_pn (rows of B), Bt = B_p Phi (stored, nr x Naux per
885 // level), pole factors h, h' (recomputed when needed), Sigma_c'(E_p)
886 auto gatherB = [&](Index p) {
887 Eigen::MatrixXd Bp(nr, naux);
888 for (Index n = 0; n < nr; ++n) {
889 Bp.row(n) = Bmat.row((q0 + p) + nw * (r0 + n));
890 }
891 return Bp;
892 };
893 auto poleFactors = [&](Index p, Eigen::ArrayXXd& H, Eigen::ArrayXXd& Hp) {
894 const double omega_p = in.qp_energies(p);
895 H.resize(nr, naux);
896 Hp.resize(nr, naux);
897 for (Index i = 0; i < naux; ++i) {
898 for (Index n = 0; n < nr; ++n) {
899 const double t = omega_p - epole(n) + s_n(n) * Om(i);
900 const double den = t * t + seta2;
901 H(n, i) = t / den;
902 Hp(n, i) = (seta2 - t * t) / (den * den);
903 }
904 }
905 };
906 std::vector<Eigen::MatrixXd> Bts(static_cast<std::size_t>(nq));
907 Eigen::VectorXd dsigma(nq);
908 Eigen::MatrixXd poles(nr, nq);
909 for (Index p = 0; p < nq; ++p) {
910 const Index pabs = in.qpmin + p;
911 const double omega_p = in.qp_energies(p);
912 const Eigen::MatrixXd Bp = gatherB(p);
913 Bts[std::size_t(p)] = Bp * Phi;
914 Eigen::ArrayXXd H, Hp;
915 poleFactors(p, H, Hp);
916 const Eigen::ArrayXXd aBt2 =
917 Bts[std::size_t(p)].array().square().rowwise() *
918 a.transpose().array();
919 const double sigma_c = (aBt2 * H).sum();
920 dsigma(p) = (aBt2 * Hp).sum();
921 poles.col(p) = -(aBt2 * Hp).rowwise().sum().matrix();
922 const double sigma_x = -(1.0 - ax) * Bp.topRows(nk).squaredNorm();
923 const double e_rebuilt =
924 eps(pabs) + sigma_x - Vxc_mo(pabs, pabs) + sigma_c;
925 result.qp_rebuild_error =
926 std::max(result.qp_rebuild_error, std::abs(e_rebuilt - omega_p));
927 }

929 // reverse pass of sum_p v_p sigma_p(omega_p; R) at fixed omega: returns
930 // its derivative with respect to R (rho, RPA window); with accumulate,
931 // also adds the derivatives with respect to B (g), eps_p and v_xc
932 auto reverse = [&](const Eigen::VectorXd& vq, bool accumulate) {
933 Eigen::VectorXd rho = Eigen::VectorXd::Zero(nr);
934 Eigen::VectorXd gOm = Eigen::VectorXd::Zero(naux);
935 Eigen::VectorXd gw = Eigen::VectorXd::Zero(naux);
936 Eigen::MatrixXd GPhi = Eigen::MatrixXd::Zero(naux, naux);
937 for (Index p = 0; p < nq; ++p) {
938 const double v = vq(p);
939 if (v == 0.0) {
940 continue;
941 }
942 const Eigen::MatrixXd Bp = gatherB(p);
943 const Eigen::MatrixXd& Bt = Bts[std::size_t(p)];
944 Eigen::ArrayXXd H, Hp;
945 poleFactors(p, H, Hp);
