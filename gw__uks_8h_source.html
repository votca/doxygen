131 QPFunc(Index gw_level, const Sigma_base_UKS& sigma, double offset)
132 : gw_level_(gw_level), offset_(offset), sigma_c_func_(sigma) {}
133
134 std::pair<double, double> operator()(double frequency) const {
135 std::pair<double, double> result;
136 result.first = value(frequency, EvalStage::Other);
137 result.second = deriv(frequency);
138 return result;
139 }
140
141 double sigma(double frequency, EvalStage stage = EvalStage::Other) const {
142 const std::uint64_t key = FrequencyKey(frequency);
143
144 auto insert_result = seen_frequencies_.insert(key);
145 if (!insert_result.second) {
146 ++stats_.sigma_repeat_calls;
147 } else {
148 ++stats_.sigma_unique_frequencies;
149 }
150
151 CountSigmaStage(stage);
152 auto cached = cache_.find(key);
153 if (cached != cache_.end()) {
154 return cached->second;
155 }
156 return sigma_c_func_.CalcCorrelationDiagElement(gw_level_, frequency);
157 }
158
162 void Prefetch(const std::vector<double>& frequencies) const {
163 std::vector<double> todo;
164 for (double f : frequencies) {
165 if (cache_.find(FrequencyKey(f)) == cache_.end()) {
166 todo.push_back(f);
167 }
168 }
169 if (todo.empty()) {
170 return;
171 }
172 const Eigen::VectorXd values = sigma_c_func_.CalcCorrelationDiagElements(
173 gw_level_,
174 Eigen::Map<const Eigen::VectorXd>(todo.data(), Index(todo.size())));
175 for (std::size_t i = 0; i < todo.size(); ++i) {
176 cache_[FrequencyKey(todo[i])] = values(Index(i));
177 }
178 }
179
180 double value(double frequency, EvalStage stage = EvalStage::Other) const {
181 return sigma(frequency, stage) + offset_ - frequency;
182 }
183
184 double deriv(double frequency) const {
185 ++stats_.deriv_calls;
186 return sigma_c_func_.CalcCorrelationDiagElementDerivative(gw_level_,
187 frequency) -
188 1.0;
189 }
190
191 const QPStats& GetStats() const { return stats_; }
192
193 private:
194 static std::uint64_t FrequencyKey(double x) {
195 std::uint64_t key = 0;
196 static_assert(sizeof(double) == sizeof(std::uint64_t),
197 "Unexpected double size");
198 std::memcpy(&key, &x, sizeof(double));
199 return key;
200 }
201
202 void CountSigmaStage(EvalStage stage) const {
203 switch (stage) {
204 case EvalStage::Scan:
205 ++stats_.sigma_scan_calls;
206 break;
207 case EvalStage::Refine:
208 ++stats_.sigma_refine_calls;
209 break;
210 case EvalStage::Derivative:
211 ++stats_.sigma_derivative_calls;
212 break;
213 case EvalStage::Other:
214 default:
215 ++stats_.sigma_other_calls;
216 break;
217 }
218 }
219
220 Index gw_level_;
221 double offset_;
222 const Sigma_base_UKS& sigma_c_func_;
223
224 mutable std::unordered_set<std::uint64_t> seen_frequencies_;
225 mutable std::unordered_map<std::uint64_t, double> cache_;
226 mutable QPStats stats_;
227 };
