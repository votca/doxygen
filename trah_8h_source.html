57struct StepOptions {
58 // The micro-iterations stop when |(H - mu) x + g| <= eta |g| with the
59 // forcing term eta = min(max_forcing, |g|), so the macro-iterations
60 // converge quadratically close to the minimum.

62 double residual_floor = 1e-10;

64 Index max_subspace = 60;
65 double alpha_max = 1e4;

67};
