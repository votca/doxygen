32void Sigma_Exact::PrepareScreening() {
33 RPA::rpa_eigensolution rpa_solution;
34 {
35 OptionalTiming t(timings_, "screening: RPA pair eigenproblem");
36 rpa_solution = rpa_.Diagonalize_H2p();
37 }
38 OptionalTiming t(timings_, "screening: residues");
39 rpa_omegas_ = rpa_solution.omega;
40 // residues of level m: M_m Z with the screening modes
41 // Z = sum_v M_v,virt^T (X+Y)_v, once for all levels
42 Eigen::MatrixXd modes = ScreeningModes(rpa_solution.XpY);
43 const Index B = Mmn_.nsize();
44 const Index S = rpa_omegas_.size();
45 residues_.resize(B * S, qptotal_);
46 const Index qpoffset = opt_.qpmin - opt_.rpamin;
47#pragma omp parallel for schedule(dynamic)
48 for (Index gw_level = 0; gw_level < qptotal_; gw_level++) {
49 Eigen::Map<Eigen::MatrixXd>(residues_.col(gw_level).data(), B, S)
50 .noalias() = Mmn_[gw_level + qpoffset] * modes;
51 }
52 if (keep_screening_data_) {
53 modes_ = std::move(modes);
54 screening_energies_ = rpa_.getRPAInputEnergies();
55 }
56}

63Sigma_base::ScreeningData Sigma_Exact::ScreeningDump(
64 const Eigen::VectorXd& nu) const {
65 if (modes_.size() == 0) {
66 throw std::runtime_error(
67 "Sigma_Exact::ScreeningDump: screening modes were not kept "
68 "(setKeepScreeningData before PrepareScreening)");
69 }
70 ScreeningData data;
71 data.omegas = rpa_omegas_;
72 data.energies = screening_energies_;
73 // W_c(i nu) = -4 sum_s Z_s Z_s^T Omega_s / (nu^2 + Omega_s^2) in the aux
74 // frame of the integrals (2 for the spin sum times 2 from the two poles
75 // +-Omega_s), so eps^-1(0) - 1 = -4 Z Omega^-1 Z^T; its eigenvectors are
76 // those of eps(0), i.e. the plasmon-pole frame, eigenvalues -w_i.
77 const Eigen::VectorXd inv_omega = rpa_omegas_.cwiseInverse();
78 // A = 4 Z Omega^-1 Z^T as a rank-k update (lower triangle, which is all
79 // SymmetricEigen reads; dsyevd with MKL or Accelerate)
80 const Index naux = modes_.rows();
81 Eigen::MatrixXd A = Eigen::MatrixXd::Zero(naux, naux);
82 A.selfadjointView<Eigen::Lower>().rankUpdate(
83 modes_ * (4.0 * inv_omega).cwiseSqrt().asDiagonal());
84 const SymmetricEigenSystem es = SymmetricEigen(A);
85 data.frame_weights = es.values;
86 const Eigen::MatrixXd a =
87 4.0 * (es.vectors.transpose() * modes_).array().square().matrix();
88 data.strength = (a.colwise().sum().transpose()).cwiseProduct(inv_omega);
89 FrameSums(a, rpa_omegas_, nu, data);
90 return data;
91}

98double Sigma_Exact::CalcCorrelationDiagElement(Index gw_level,
99 double frequency) const {
100 CountDiagEval();
101 // factor 2: both (identical) spin channels
102 return 2.0 * exact_pole_sums::Value(Poles(), Residues(gw_level), frequency);

120double Sigma_Exact::CalcCorrelationOffDiagElement(Index gw_level1,
121 Index gw_level2,
122 double frequency1,
123 double frequency2) const {
124 // 2 (both spins) x 1/2 (symmetrised in the two frequencies)
125 const Eigen::Vector2d w(frequency1, frequency2);
126 const ExactPoles poles = Poles();
127 const Eigen::VectorXd r12 =
128 residues_.col(gw_level1).cwiseProduct(residues_.col(gw_level2));
129 const Eigen::VectorXd sums =
130 exact_pole_sums::ValuesDirectWeighted(poles, r12.data(), w);
131 return sums(0) + sums(1);
132}

139Eigen::MatrixXd Sigma_Exact::ScreeningModes(const Eigen::MatrixXd& XpY) const {
140 const Index lumo = opt_.homo + 1;
141 const Index n_occ = lumo - opt_.rpamin;
142 const Index n_unocc = opt_.rpamax - opt_.homo;
143 const Index rpasize = n_occ * n_unocc;
144 vc2index vc = vc2index(0, 0, n_unocc);
145 // rows (v, c) of the virtual rows of the hole slices
146 Eigen::MatrixXd M(rpasize, Mmn_.auxsize());
147#pragma omp parallel for schedule(dynamic)
148 for (Index v = 0; v < n_occ; v++) {
149 M.middleRows(vc.I(v, 0), n_unocc) = Mmn_[v].middleRows(n_occ, n_unocc);
150 }
151 return M.transpose() * XpY;
152}
