30void Sigma_Exact::PrepareScreening() {
31 RPA::rpa_eigensolution rpa_solution;
32 {
33 OptionalTiming t(timings_, "screening: RPA pair eigenproblem");
34 rpa_solution = rpa_.Diagonalize_H2p();
35 }
36 OptionalTiming t(timings_, "screening: residues");
37 rpa_omegas_ = rpa_solution.omega;
38 // residues of level m: M_m Z with the screening modes
39 // Z = sum_v M_v,virt^T (X+Y)_v, once for all levels
40 const Eigen::MatrixXd modes = ScreeningModes(rpa_solution.XpY);
41 const Index B = Mmn_.nsize();
42 const Index S = rpa_omegas_.size();
43 residues_.resize(B * S, qptotal_);
44 const Index qpoffset = opt_.qpmin - opt_.rpamin;
45#pragma omp parallel for schedule(dynamic)
46 for (Index gw_level = 0; gw_level < qptotal_; gw_level++) {
47 Eigen::Map<Eigen::MatrixXd>(residues_.col(gw_level).data(), B, S)
48 .noalias() = Mmn_[gw_level + qpoffset] * modes;
49 }
50}

57double Sigma_Exact::CalcCorrelationDiagElement(Index gw_level,
58 double frequency) const {
59 CountDiagEval();
60 // factor 2: both (identical) spin channels
61 return 2.0 * exact_pole_sums::Value(Poles(), Residues(gw_level), frequency);
62}

64Eigen::VectorXd Sigma_Exact::CalcCorrelationDiagElements(

79double Sigma_Exact::CalcCorrelationOffDiagElement(Index gw_level1,
80 Index gw_level2,
81 double frequency1,
82 double frequency2) const {
83 // 2 (both spins) x 1/2 (symmetrised in the two frequencies)
84 const Eigen::Vector2d w(frequency1, frequency2);
85 const ExactPoles poles = Poles();
86 const Eigen::VectorXd r12 =
87 residues_.col(gw_level1).cwiseProduct(residues_.col(gw_level2));
88 const Eigen::VectorXd sums =
89 exact_pole_sums::ValuesDirectWeighted(poles, r12.data(), w);
90 return sums(0) + sums(1);
91}

98Eigen::MatrixXd Sigma_Exact::ScreeningModes(const Eigen::MatrixXd& XpY) const {
99 const Index lumo = opt_.homo + 1;
100 const Index n_occ = lumo - opt_.rpamin;
101 const Index n_unocc = opt_.rpamax - opt_.homo;
102 const Index rpasize = n_occ * n_unocc;
103 vc2index vc = vc2index(0, 0, n_unocc);
104 // rows (v, c) of the virtual rows of the hole slices
105 Eigen::MatrixXd M(rpasize, Mmn_.auxsize());
106#pragma omp parallel for schedule(dynamic)
107 for (Index v = 0; v < n_occ; v++) {
108 M.middleRows(vc.I(v, 0), n_unocc) = Mmn_[v].middleRows(n_occ, n_unocc);
109 }
110 return M.transpose() * XpY;
111}
