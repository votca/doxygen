41Eigen::MatrixXd Sigma_base::PairContraction(
42 Index nrows,
43 const std::function<void(Index, Eigen::MatrixXd&)>& makeX) const {
44 const Index qpmin = opt_.qpmin - opt_.rpamin;
45 const Index auxsize = Mmn_.auxsize();
46 // X of a block of levels kept at once, and the chunk of all M_n gathered
47 // for one product; the block size sets how often the integrals are read.
48 const double kBlockBytes = block_bytes_;
49 const double kChunkBytes = chunk_bytes_;
50 const Index block = std::max<Index>(
51 1, std::min<Index>(qptotal_, Index(kBlockBytes / (8.0 * double(nrows) *
52 double(auxsize)))));
53 const Index chunk = std::max<Index>(
54 1, std::min<Index>(auxsize, Index(kChunkBytes / (8.0 * double(nrows) *
55 double(qptotal_)))));
56 Eigen::MatrixXd A = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
57 std::vector<Eigen::MatrixXd> X(static_cast<std::size_t>(block));
58 // chunk buffers kept across chunks (no reallocation per chunk)
59 Eigen::MatrixXd Mc;
60 Eigen::MatrixXd Xc;
61 for (Index m0 = 0; m0 < qptotal_; m0 += block) {
62 const Index nb = std::min(block, qptotal_ - m0);
63#pragma omp parallel for schedule(dynamic)
64 for (Index j = 0; j < nb; ++j) {
65 makeX(m0 + j, X[std::size_t(j)]);
66 }
67 for (Index s0 = 0; s0 < auxsize; s0 += chunk) {
68 const Index ns = std::min(chunk, auxsize - s0);
69 Mc.resize(nrows * ns, qptotal_);
70 Xc.resize(nrows * ns, nb);
71#pragma omp parallel for schedule(static)
72 for (Index n = 0; n < qptotal_; ++n) {
73 Eigen::Map<Eigen::MatrixXd>(Mc.col(n).data(), nrows, ns) =
74 Mmn_[n + qpmin].block(0, s0, nrows, ns);
75 }
76 for (Index j = 0; j < nb; ++j) {
77 Eigen::Map<Eigen::MatrixXd>(Xc.col(j).data(), nrows, ns) =
78 X[std::size_t(j)].middleCols(s0, ns);
79 }
80 A.middleRows(m0, nb).noalias() += Xc.transpose() * Mc;
81 }
82 }
83 return A;
84}

98void Sigma_base::FrameSums(const Eigen::MatrixXd& a,
99 const Eigen::VectorXd& omegas,
100 const Eigen::VectorXd& nu, ScreeningData& data) {
101 const Index S = omegas.size();
102 // columns: Omega^-1, Omega^0, Omega^1, then Omega/(nu^2+Omega^2)
103 Eigen::MatrixXd g(S, 3 + nu.size());
104 for (Index s = 0; s < S; ++s) {
105 const double w = omegas(s);
106 g(s, 0) = 1.0 / w;
107 g(s, 1) = 1.0;
108 g(s, 2) = w;
109 for (Index j = 0; j < nu.size(); ++j) {
110 g(s, 3 + j) = -w / (nu(j) * nu(j) + w * w);
111 }
112 }
113 const Eigen::MatrixXd sums = a * g;
114 data.frame_moments = sums.leftCols(3);
115 data.frame_eps_inv_imag = sums.rightCols(nu.size());
116}

118Eigen::MatrixXd Sigma_base::CalcExchangeMatrix() const {
119 const Index occlevel = opt_.homo - opt_.rpamin + 1;
120 const Index qpmin = opt_.qpmin - opt_.rpamin;
121 // Exchange is with the bare v, which is M M^T only as long as the
122 // auxiliary frame is orthogonal. Once the index is dressed for an
123 // environment (TCMatrix_gwbse::DressAuxIndex) it takes the bare
124 // interaction in the current frame as an explicit kernel.
125 const bool bare = Mmn_.AuxFrameIsOrthogonal();
126 Eigen::MatrixXd v;
127 if (!bare) {
128 v = Mmn_.ToCurrentAuxFrame(
129 Eigen::MatrixXd::Identity(Mmn_.auxsize(), Mmn_.auxsize()));
130 }
131 // Sigma_x(m,n) = - sum_{i occ} sum_P M_m(i,P) [v] M_n(i,P)
132 const Eigen::MatrixXd A =
133 PairContraction(occlevel, [&](Index m, Eigen::MatrixXd& X) {
134 if (bare) {
135 X = Mmn_[m + qpmin].topRows(occlevel);
136 } else {
137 X = Mmn_[m + qpmin].topRows(occlevel) * v;
138 }
139 });
140 return -0.5 * (A + A.transpose());
141}

143Eigen::MatrixXd Sigma_base::CalcHartreeMatrix(const Eigen::MatrixXd& R) const {
144 const Index occlevel = opt_.homo - opt_.rpamin + 1;
145 const Index qpmin = opt_.qpmin - opt_.rpamin;
146 // d_P = sum_{i occ} M_ii,P (the density in the auxiliary basis)
147 Eigen::VectorXd d = Eigen::VectorXd::Zero(Mmn_.auxsize());
148 for (Index i = 0; i < occlevel; ++i) {
149 d += Mmn_[i].row(i).transpose();
150 }
151 // Kernel 1 (bare v) or 1 + R (u = v + v_reac), given in the fill-time
152 // frame and brought into the current one (see CalcExchangeMatrix). On
153 // dressed integrals F F^T = 1 + R, so the screened kernel is close to the
154 // identity there; it is still taken explicitly, which keeps this right
155 // in any frame.
156 if (R.size() > 0) {
157 if (R.rows() != Mmn_.auxsize() || R.cols() != Mmn_.auxsize()) {
158 throw std::runtime_error(
159 "Sigma_base::CalcHartreeMatrix: reaction field is " +
160 std::to_string(R.rows()) + "x" + std::to_string(R.cols()) +
161 ", auxiliary basis has " + std::to_string(Mmn_.auxsize()) +
162 " functions.");
163 }
164 d = Mmn_.ToCurrentAuxFrame(
165 Eigen::MatrixXd::Identity(Mmn_.auxsize(), Mmn_.auxsize()) + R) *
166 d;
167 } else if (!Mmn_.AuxFrameIsOrthogonal()) {
168 d = Mmn_.ToCurrentAuxFrame(
169 Eigen::MatrixXd::Identity(Mmn_.auxsize(), Mmn_.auxsize())) *
170 d;
171 }
172 Eigen::MatrixXd J(qptotal_, qptotal_);
173#pragma omp parallel for schedule(dynamic)
174 for (Index m = 0; m < qptotal_; ++m) {
175 J.col(m) = 2.0 * Mmn_[m + qpmin].middleRows(qpmin, qptotal_) * d;
176 }
177 return 0.5 * (J + J.transpose());
178}

180Eigen::MatrixXd Sigma_base::CalcReactionFieldMatrix(
181 const Eigen::MatrixXd& R) const {
182 const Eigen::MatrixXd Rc = Mmn_.ToCurrentAuxFrame(R);
183 const Index occlevel = opt_.homo - opt_.rpamin + 1;
184 const Index qpmin = opt_.qpmin - opt_.rpamin;
185 // Row l of X_m is s_l M_m(l,:) R: the occupation sign folded in once.
186 const Eigen::MatrixXd A =
187 PairContraction(Mmn_.nsize(), [&](Index m, Eigen::MatrixXd& X) {
188 X = Mmn_[m + qpmin] * Rc;
189 X.topRows(occlevel) *= -1.0;
190 });
191 return 0.25 * (A + A.transpose());
192}

194Eigen::VectorXd Sigma_base::CalcCorrelationDiag(
195 const Eigen::VectorXd& frequencies) const {
196 Eigen::VectorXd result = Eigen::VectorXd::Zero(qptotal_);
197#pragma omp parallel for schedule(dynamic)
198 for (Index gw_level = 0; gw_level < qptotal_; gw_level++) {
199 result(gw_level) =
200 CalcCorrelationDiagElement(gw_level, frequencies[gw_level]);
201 }
202 return result;
203}

205Eigen::MatrixXd Sigma_base::CalcCorrelationOffDiag(
206 const Eigen::VectorXd& frequencies) const {
207 Eigen::MatrixXd result = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
208#pragma omp parallel for schedule(dynamic)
209 for (Index gw_level1 = 0; gw_level1 < qptotal_; gw_level1++) {
210 for (Index gw_level2 = gw_level1 + 1; gw_level2 < qptotal_; gw_level2++) {
211 double sigma_c = CalcCorrelationOffDiagElement(
212 gw_level1, gw_level2, frequencies[gw_level1], frequencies[gw_level2]);
213 result(gw_level2, gw_level1) = sigma_c;
214 }
215 }
216 result = result.selfadjointView<Eigen::Lower>();
217 return result;
218}
