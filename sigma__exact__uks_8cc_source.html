38void Sigma_Exact_UKS::PrepareScreening() {
39
40 // Build Coulomb-active screening modes in the auxiliary basis from the full
41 // unrestricted H2p eigenvectors. This suppresses numerically dark / spin-like
42 // modes that can appear in the explicit alpha+beta transition basis but
43 // should not contribute to the screened Coulomb interaction W.
44 const Eigen::VectorXd* cached_omegas = nullptr;
45 const std::vector<Eigen::VectorXd>* cached_modes = nullptr;
46 rpa_.GetCachedScreeningModes(cached_omegas, cached_modes);
47
48 rpa_omegas_ = *cached_omegas;
49 screening_modes_ = *cached_modes;
50 // residues of level m: M_m times the modes, one GEMM per level
51 Eigen::MatrixXd modes(Mmn_.auxsize(), rpa_omegas_.size());
52 for (Index s = 0; s < rpa_omegas_.size(); s++) {
53 modes.col(s) = screening_modes_[std::size_t(s)];
54 }
55 residues_ = std::vector<Eigen::MatrixXd>(qptotal_);
56 const Index qpoffset = opt_.qpmin - opt_.rpamin;
57#pragma omp parallel for schedule(dynamic)
58 for (Index gw_level = 0; gw_level < qptotal_; gw_level++) {
59 residues_[gw_level] = Mmn_[gw_level + qpoffset] * modes;
60 }
61}

96double Sigma_Exact_UKS::CalcCorrelationOffDiagElement(Index gw_level1,
97 Index gw_level2,
98 double frequency1,
99 double frequency2) const {
100 const ExactPoles poles =
101 UKSPoles(getSpinRPAInputEnergies(), opt_.homo + 1 - opt_.rpamin,
102 rpa_omegas_, opt_.eta);
103 const Eigen::MatrixXd r12 =
104 residues_[gw_level1].cwiseProduct(residues_[gw_level2]);
105 const Eigen::VectorXd sums = exact_pole_sums::ValuesDirectWeighted(
106 poles, r12.data(), Eigen::Vector2d(frequency1, frequency2));
107 return 0.5 * (sums(0) + sums(1));
108}
