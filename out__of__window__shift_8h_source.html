101 private:
102 // Shifts levels [out_first, out_last] using the corrections of the window
103 // levels [win_first, win_last] of the same kind; below is true for the
104 // occupied side (boundary win_first), false for the virtual side (boundary
105 // win_last).
106 void ShiftSide(Eigen::VectorXd& energies, const Eigen::VectorXd& dft,
107 Index rpamin, Index out_first, Index out_last, Index win_first,
108 Index win_last, bool below) const {
109 const Index nwin = win_last - win_first + 1;
110 auto correction = [&](Index level) {
111 return energies(level - rpamin) - dft(level);
112 };
113 auto out = energies.segment(out_first - rpamin, out_last - out_first + 1);
114 if (nwin <= 0) {
115 return;
116 }
117 if (mode == Mode::Max) {
118 double max_correction = 0.0;
119 for (Index l = win_first; l <= win_last; ++l) {
120 max_correction = std::max(max_correction, std::abs(correction(l)));
121 }
122 out.array() += below ? -max_correction : max_correction;
123 return;
124 }
125 const Index boundary = below ? win_first : win_last;
126 const Index nfit =
127 (mode == Mode::Nearest)
128 ? 1
129 : std::min(nwin,
130 std::max<Index>(
131 2, Index(std::ceil(fit_fraction * double(nwin)))));
132 if (nfit < 2) {
133 out.array() += correction(boundary);
134 return;
135 }
136 // least squares D = a + b x over the nfit levels next to the boundary,
137 // x = e - e_boundary
138 const Index fit_first = below ? win_first : win_last - nfit + 1;
139 const double e_b = dft(boundary);
140 double sx = 0, sy = 0, sxx = 0, sxy = 0, span = 0;
141 for (Index l = fit_first; l < fit_first + nfit; ++l) {
142 const double x = dft(l) - e_b;
143 const double y = correction(l);
144 sx += x;
145 sy += y;
146 sxx += x * x;
147 sxy += x * y;
148 span = std::max(span, std::abs(x));
149 }
150 const double n = double(nfit);
151 const double var = sxx - sx * sx / n;
152 const double b = (var > 1e-12 * n) ? (sxy - sx * sy / n) / var : 0.0;
153 const double a = (sy - b * sx) / n;
154 for (Index l = out_first; l <= out_last; ++l) {
155 const double x = std::clamp(dft(l) - e_b, -span, span);
156 energies(l - rpamin) += a + b * x;
157 }
158 }
159};
