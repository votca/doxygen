136 // ===========================================================================
137 // STATUS: written but NOT yet compiled/tested (see aoshell.cc for the
138 // detailed derivation). Added for GGA XC-gradient support -- the
139 // existing AOValues/EvalAOspace (first derivatives only) is untouched;
140 // this is a new, separate struct/function reusing the SAME per-shell
141 // polynomial structure, extended with second r-derivatives (Hessians),
142 // needed to differentiate sigma=|grad(rho)|^2 for GGA functionals.
143 //
144 // Supports S, P, D, F, G shells (l=0..4), matching EvalAOspace's own
145 // supported range exactly -- not a new limitation, same shells this
146 // codebase's AO evaluator has always supported for values/gradients.
147 //
148 // General formula (verified symbolically/exactly against direct
149 // double differentiation for representative polynomials spanning
150 // every supported shell, via sympy, before writing any of this):
151 // for chi(r) = exp(-alpha*d^2) * P(center), d^2=|center|^2,
152 // H_ij = prefactor * d2P/dxi_dxj
153 // - 2*alpha*delta_ij*AOvalue
154 // - 2*alpha*xi*grad_j - 2*alpha*xj*grad_i
155 // - 4*alpha^2*xi*xj*AOvalue
156 // where prefactor, AOvalue, grad_i are exactly the quantities
157 // EvalAOspace already computes for the SAME shell -- only the new
158 // d2P/dxi_dxj terms (the polynomial P's own second derivatives,
159 // exact per shell, also sympy-verified) are new per-shell content.
160 // ===========================================================================
161 struct AOValuesHessian {
162 AOValuesHessian(Index size) {
163 values = Eigen::VectorXd::Zero(size);
164 derivatives = Eigen::MatrixX3d::Zero(size, 3);
165 hessians = std::vector<Eigen::Matrix3d>(size, Eigen::Matrix3d::Zero());
166 }
167 Eigen::VectorXd values;
168 Eigen::MatrixX3d derivatives;
169 std::vector<Eigen::Matrix3d> hessians;
170 };
171
172 AOValuesHessian EvalAOspaceHessian(const Eigen::Vector3d& grid_pos) const;
173

176 GaussianIterator begin() const { return gaussians_.begin(); }
177 GaussianIterator end() const { return gaussians_.end(); }
