47// conjugate gradients with the Jacobi preconditioner; the hitting times are
48// dominated by the slow modes (traps), so the residual has to be driven far
49// down (default 1e-14) before they are resolved
50Eigen::VectorXd PCG(const SpMatR& A, const Eigen::VectorXd& b,
51 const Settings& settings, Index& its, bool& converged) {
52 Eigen::VectorXd dinv = A.diagonal().cwiseInverse();
53 Eigen::VectorXd x = Eigen::VectorXd::Zero(b.size());
54 Eigen::VectorXd r = b;
55 double bnorm = b.norm();
56 its = 0;
57 converged = true;
58 if (bnorm == 0.0) {
59 return x;
60 }
61 Eigen::VectorXd z = dinv.cwiseProduct(r);
62 Eigen::VectorXd d = z;
63 Eigen::VectorXd Ad(b.size());
64 double rz = r.dot(z);
65 const double threshold2 =
66 settings.tolerance * settings.tolerance * bnorm * bnorm;
67 // stop at a rounding floor just above the tolerance instead of running
68 // into the iteration limit; the refinement decides. Only close to the
69 // tolerance: further up, plateaus of the residual are normal while the
70 // slow trap modes are resolved.
71 const double floor2 = 1e6 * threshold2; // 1000 x tolerance
72 double best2 = r.squaredNorm();
73 Index last_improvement = 0;
74 while (r.squaredNorm() > threshold2) {
75 if (its >= settings.max_iterations) {
76 converged = false;
77 break;
78 }
79 if (best2 < floor2 && its - last_improvement > 1000) {
80 break;
81 }
82 Ad.noalias() = A * d;
83 double alpha = rz / d.dot(Ad);
84 x += alpha * d;
85 r -= alpha * Ad;
86 z = dinv.cwiseProduct(r);
87 double rz_new = r.dot(z);
88 d = z + (rz_new / rz) * d;
89 rz = rz_new;
90 ++its;
91 double r2 = r.squaredNorm();
92 if (r2 < 0.81 * best2) { // residual down by 10 %
93 best2 = r2;
94 last_improvement = its;
95 }
96 }
97 return x;
98}

105class GeneratorSolver {
106 public:
107 GeneratorSolver(const SpMatR& Q, const Eigen::VectorXd& h, Solver solver,
108 const Settings& settings)
109 : Q_(Q), settings_(settings) {
110 Qabs_ = Q_.cwiseAbs();
111 Qt_ = Q_.transpose();
112 Qtabs_ = Qt_.cwiseAbs();
113 const Index m = Q_.rows();
114
115 bool symmetric = (solver == Solver::Cholesky || solver == Solver::CG ||
116 (solver == Solver::Auto && h.size() == m));
117 if ((solver == Solver::Cholesky || solver == Solver::CG) && h.size() != m) {
118 throw std::runtime_error(
119 "GraphRandomWalk: the cholesky and cg solvers need rates with "
120 "detailed balance, use lu or bicgstab");
121 }
122 if (symmetric && h.cwiseAbs().maxCoeff() > 600.0) {
123 if (solver != Solver::Auto) {
124 throw std::runtime_error(
125 "GraphRandomWalk: the potential range is too large for the "
126 "symmetrized solvers, use lu or bicgstab");
127 }
128 symmetric = false;
129 }
130 if (solver == Solver::Auto) {
131 bool direct = m <= settings_.direct_limit;
132 if (symmetric) {
133 solver = direct ? Solver::Cholesky : Solver::CG;
134 } else {
135 solver = direct ? Solver::LU : Solver::BiCGSTAB;
136 }
137 }
138 iterative_ = (solver == Solver::CG || solver == Solver::BiCGSTAB);
139
140 if (symmetric) {
141 p_ = h.array().exp();
142 pinv_ = p_.cwiseInverse();
143 // S = P Q P^-1 is symmetric; Q = P^-1 S P, Q^T = P S P^-1
144 SpMatR S = p_.asDiagonal() * Q_ * pinv_.asDiagonal();
145 SpMatR St = S.transpose();
146 S = 0.5 * (S + St);
147 if (solver == Solver::Cholesky) {
148 name = "cholesky (LDLT of the symmetrized generator)";
149 ldlt_.compute(SpMat(S));
150 if (ldlt_.info() != Eigen::Success) {
151 throw std::runtime_error(
152 "GraphRandomWalk: Cholesky factorization failed");
153 }
154 solveS_ = [this](const Eigen::VectorXd& b) {
155 return Eigen::VectorXd(ldlt_.solve(b));
156 };
157 } else {
158 name = "cg (Jacobi preconditioned, symmetrized generator)";
159 S_ = std::move(S);
160 solveS_ = [this](const Eigen::VectorXd& b) {
161 Index its = 0;
162 bool ok = true;
163 Eigen::VectorXd y = PCG(S_, b, settings_, its, ok);
164 iterations += its;
165 converged = converged && ok;
166 return y;
167 };
168 }
169 solve_ = [this](const Eigen::VectorXd& b) {
170 Eigen::VectorXd y = solveS_((p_.array() * b.array()).matrix());
171 return Eigen::VectorXd(pinv_.array() * y.array());
172 };
173 solve_adjoint_ = [this](const Eigen::VectorXd& b) {
174 Eigen::VectorXd y = solveS_((pinv_.array() * b.array()).matrix());
175 return Eigen::VectorXd(p_.array() * y.array());
176 };
177 } else if (solver == Solver::LU) {
178 name = "lu";
179 Qc_ = Q_;
180 lu_.analyzePattern(Qc_);
181 lu_.factorize(Qc_);
182 if (lu_.info() != Eigen::Success) {
183 throw std::runtime_error("GraphRandomWalk: LU factorization failed: " +
184 lu_.lastErrorMessage());
185 }
186 solve_ = [this](const Eigen::VectorXd& b) {
187 return Eigen::VectorXd(lu_.solve(b));
188 };
189 solve_adjoint_ = [this](const Eigen::VectorXd& b) {
190 return Eigen::VectorXd(lu_.transpose().solve(b));
191 };
192 } else {
193 name = "bicgstab (incomplete LU preconditioned)";
194 for (auto* b : {&bicg_, &bicg_adjoint_}) {
195 b->setTolerance(settings_.tolerance);
196 b->setMaxIterations(settings_.max_iterations);
197 // a cheap incomplete factorization; accuracy comes from the
198 // iterations and the refinement
199 b->preconditioner().setDroptol(1e-4);
200 b->preconditioner().setFillfactor(5);
201 }
202 // the iterative solvers keep a reference to the matrix
203 Qc_ = Q_;
204 Qtc_ = Qt_;
205 bicg_.compute(Qc_);
206 bicg_adjoint_.compute(Qtc_);
207 auto make = [this](BiCG& b) {
208 return [this, &b](const Eigen::VectorXd& rhs) {
209 bool solved = false;
210 Eigen::VectorXd x = ScaledSolve(b, rhs, &solved);
211 if (!solved) {
212 return x; // zero right-hand side
213 }
214 iterations += b.iterations();
215 // only the iteration limit counts here: Eigen versions differ in
216 // reporting a residual floor just above the tolerance or a restart
217 // as NoConvergence/NumericalIssue; the refinement against Q and the
218 // backward error decide whether the result is usable
219 converged = converged && b.iterations() < b.maxIterations();
220 return x;
221 };
222 };
223 solve_ = make(bicg_);
224 solve_adjoint_ = make(bicg_adjoint_);
225 }
226 }
227
228 Eigen::VectorXd Solve(const Eigen::VectorXd& b) {
229 return Refine(Q_, Qabs_, solve_, b);
230 }
231 Eigen::VectorXd SolveAdjoint(const Eigen::VectorXd& b) {
232 return Refine(Qt_, Qtabs_, solve_adjoint_, b);
233 }
234
235 std::string name;
236 Index iterations = 0;
237 bool converged = true;
238 double backward_error = 0.0;
239 Index refinement_steps = 0;
240
241 private:
242 using BiCG = Eigen::BiCGSTAB<SpMat, Eigen::IncompleteLUT<double>>;
243
244 // Solve A x = b with an approximate solver and iterative refinement
245 // against A; tracks the largest componentwise backward error
246 // |b - A x|_i / (|A| |x| + |b|)_i.
247 Eigen::VectorXd Refine(const SpMatR& A, const SpMatR& Aabs,
248 const SolveFn& solve, const Eigen::VectorXd& b) {
249 // direct solvers: refine to rounding level; iterative solvers: 1e-12
250 // suffices, each step is a full solve
251 double target =
252 iterative_ ? std::max(1e-12, 100.0 * settings_.tolerance) : 1e-14;
253 auto error = [&A, &Aabs, &b](const Eigen::VectorXd& y) {
254 Eigen::VectorXd r = b - A * y;
255 Eigen::VectorXd scale = Aabs * y.cwiseAbs() + b.cwiseAbs();
256 double e = 0.0;
257 for (Index i = 0; i < r.size(); ++i) {
258 if (scale[i] > 0.0) {
259 e = std::max(e, std::abs(r[i]) / scale[i]);
260 }
261 }
262 return std::make_pair(e, Eigen::VectorXd(r));
263 };
264 Eigen::VectorXd x = solve(b);
265 Index steps = 0;
266 auto [err, r] = error(x);
267 // refine until the backward error is at the target or stops improving
268 while (err > target && steps < 5) {
269 Eigen::VectorXd x_new = x + solve(r);
270 auto [err_new, r_new] = error(x_new);
271 ++steps;
272 if (err_new > 0.5 * err) {
273 if (err_new < err) {
274 x = x_new;
275 err = err_new;
276 }
277 break;
278 }
279 x = x_new;
280 err = err_new;
281 r = r_new;
282 }
283 if (!x.allFinite()) {
284 throw std::runtime_error("GraphRandomWalk: solution is not finite");
285 }
286 backward_error = std::max(backward_error, err);
287 refinement_steps = std::max(refinement_steps, steps);
288 return x;
289 }
290
291 const SpMatR& Q_;
292 Settings settings_;
293 SpMatR Qabs_;
294 SpMatR Qt_;
295 SpMatR Qtabs_;
296 bool iterative_ = false;
297 Eigen::VectorXd p_;
298 Eigen::VectorXd pinv_;
299 SpMatR S_;
300 SpMat Qc_; // column-major copies for LU and BiCGSTAB
301 SpMat Qtc_;
302 Eigen::SimplicialLDLT<SpMat, Eigen::Lower, Eigen::AMDOrdering<int>> ldlt_;
303 Eigen::SparseLU<SpMat, Eigen::COLAMDOrdering<int>> lu_;
304 BiCG bicg_;
305 BiCG bicg_adjoint_;
306 SolveFn solveS_;
307 SolveFn solve_;
308 SolveFn solve_adjoint_;
309};

311// row-major sparse matrix from rows of (column, value), duplicates summed
312class RowBuilder {
313 public:
314 explicit RowBuilder(Index rows) : rows_(rows) {
315 outer_.reserve(rows + 1);
316 outer_.push_back(0);
317 }
318 void Add(Index col, double value) { row_.emplace_back(col, value); }
319 void FinishRow() {
320 std::sort(row_.begin(), row_.end());
321 for (std::size_t k = 0; k < row_.size(); ++k) {
322 if (k > 0 && row_[k].first == row_[k - 1].first) {
323 values_.back() += row_[k].second;
324 } else {
325 inner_.push_back(row_[k].first);
326 values_.push_back(row_[k].second);
327 }
328 }
329 row_.clear();
330 outer_.push_back(Index(inner_.size()));
331 }
332 SpMatR Build(Index cols) const {
333 if (Index(outer_.size()) != rows_ + 1) {
334 throw std::runtime_error("RowBuilder: incomplete matrix");
335 }
336 SpMatR A(rows_, cols);
337 A.reserve(Index(inner_.size()));
338 for (Index i = 0; i < rows_; ++i) {
339 A.startVec(i);
340 for (Index k = outer_[i]; k < outer_[i + 1]; ++k) {
341 A.insertBack(i, inner_[k]) = values_[k];
342 }
343 }
344 A.finalize();
345 return A;
346 }
347
348 private:
349 Index rows_;
350 std::vector<std::pair<Index, double>> row_;
351 std::vector<Index> outer_;
352 std::vector<Index> inner_;
353 std::vector<double> values_;
354};

356inline Index PairRank(Index i, Index j) {
357 if (i > j) {
358 std::swap(i, j);
359 }
360 return j * (j - 1) / 2 + i;
361}

392GraphRandomWalk::Transient GraphRandomWalk::FindTransient() const {
393 const Index n = graph_.NumberOfSites();
394 const std::vector<ToFGraph::Site>& sites = graph_.Sites();
395 Transient t;
396 t.index_of.assign(n, -1);
397 // transient sites: not a sink, sink reachable
398 for (Index i = 0; i < n; ++i) {
399 if (sites[i].role != ToFGraph::Role::Sink && sites[i].reaches_sink) {
400 t.index_of[i] = Index(t.site_of.size());
401 t.site_of.push_back(i);
402 }
403 }
404 const Index m = Index(t.site_of.size());
405 if (m == 0) {
406 throw std::runtime_error("GraphRandomWalk: no transient sites");
407 }
408 for (Index i : t.site_of) {
409 if (!(graph_.EscapeRate(i) > 0.0)) {
410 throw std::runtime_error("GraphRandomWalk: site " + std::to_string(i) +
411 " has no hops");
412 }
413 // a carrier can enter a site from which no sink is reachable
414 ToFGraph::EdgeRange out = graph_.OutEdges(i);
415 const ToFGraph::Edge* trap =
416 std::find_if(out.begin(), out.end(), [&](const ToFGraph::Edge& e) {
417 return e.rate > 0.0 && sites[e.to].role != ToFGraph::Role::Sink &&
418 t.index_of[e.to] < 0;
419 });
420 if (trap != out.end()) {
421 throw std::runtime_error(
422 "GraphRandomWalk: site " + std::to_string(trap->to) +
423 " traps carriers (no path to a sink), the time of flight is "
424 "infinite");
425 }
426 }
427 // symmetrization weights, only with detailed balance
428 if (graph_.isReversible()) {
429 Eigen::VectorXd U(m);
430 for (Index k = 0; k < m; ++k) {
431 U[k] = sites[t.site_of[k]].potential;
432 }
433 std::vector<double> sorted(U.data(), U.data() + m);
434 std::nth_element(sorted.begin(), sorted.begin() + m / 2, sorted.end());
435 double Uref = sorted[m / 2];
436 t.half_log_weight = -(U.array() - Uref) / (2.0 * graph_.Temperature());
437 }
438 return t;
439}

441GraphRandomWalk::Result GraphRandomWalk::SingleCarrier(Solver solver) const {
442 const Index n = graph_.NumberOfSites();
443 const std::vector<ToFGraph::Site>& sites = graph_.Sites();
444 Transient tr = FindTransient();
445 const Index m = Index(tr.site_of.size());
446
447 // Q = diag(D) - W on the transient sites
448 RowBuilder rows(m);
449 for (Index t = 0; t < m; ++t) {
450 Index i = tr.site_of[t];
451 rows.Add(t, graph_.EscapeRate(i));
452 for (const ToFGraph::Edge& e : graph_.OutEdges(i)) {
453 if (e.rate > 0.0 && sites[e.to].role != ToFGraph::Role::Sink) {
454 rows.Add(tr.index_of[e.to], -e.rate);
455 }
456 }
457 rows.FinishRow();
458 }
459 SpMatR Q = rows.Build(m);
460 GeneratorSolver gs(Q, tr.half_log_weight, solver, settings_);
461
462 Eigen::VectorXd tau = gs.Solve(Eigen::VectorXd::Ones(m));
463 Eigen::VectorXd m2 = gs.Solve(2.0 * tau);
464 const std::vector<Index>& sources = graph_.SourceSites();
465 Eigen::VectorXd w = Eigen::VectorXd::Zero(m);
466 for (Index s : sources) {
467 w[tr.index_of[s]] = 1.0 / double(sources.size());
468 }
469 Eigen::VectorXd g = gs.SolveAdjoint(w);
470
471 Result result;
472 result.solver = gs.name;
473 result.carriers = 1;
474 result.unknowns = m;
475 result.backward_error = gs.backward_error;
476 result.refinement_steps = gs.refinement_steps;
477 result.iterations = gs.iterations;
478 result.converged = gs.converged && gs.backward_error < 1e-10;
479
480 Index ns = Index(sources.size());
481 result.tau.resize(ns);
482 result.tau_std.resize(ns);
483 double mean2 = 0.0;
484 for (Index k = 0; k < ns; ++k) {
485 Index t = tr.index_of[sources[k]];
486 result.sources.push_back({sources[k]});
487 result.tau[k] = tau[t];
488 result.tau_std[k] = std::sqrt(std::max(m2[t] - tau[t] * tau[t], 0.0));
489 mean2 += m2[t];
490 }
491 result.tau_mean = result.tau.mean();
492 mean2 /= double(ns);
493 result.tau_std_injection =
494 std::sqrt(std::max(mean2 - result.tau_mean * result.tau_mean, 0.0));
495 result.tau_harmonic = double(ns) / result.tau.cwiseInverse().sum();
496
497 result.occupation_time = Eigen::VectorXd::Zero(n);
498 for (Index t = 0; t < m; ++t) {
499 result.occupation_time[tr.site_of[t]] = g[t];
500 }
501 return result;
502}

504GraphRandomWalk::Result GraphRandomWalk::TwoCarriers(Observable observable,
505 Solver solver) const {
506 const Index n = graph_.NumberOfSites();
507 const std::vector<ToFGraph::Site>& sites = graph_.Sites();
508 Transient tr = FindTransient();
509 const Index m = Index(tr.site_of.size());
510 if (m < 2) {
511 throw std::runtime_error(
512 "GraphRandomWalk: two carriers need at least two transient sites");
513 }
514 const Index M = m * (m - 1) / 2;
515 if (M > settings_.max_states || M > Index(std::numeric_limits<int>::max())) {
516 throw std::runtime_error(
517 "GraphRandomWalk: " + std::to_string(M) +
518 " pair states exceed the limit of " +
519 std::to_string(settings_.max_states) +
520 " (maxstates); use kinetic Monte Carlo for this system");
521 }
522 const std::vector<Index>& sources = graph_.SourceSites();
523 if (sources.size() < 2) {
524 throw std::runtime_error(
525 "GraphRandomWalk: two carriers need at least two source sites");
526 }
527 const bool transit = (observable == Observable::Transit);
528
529 // pair block Q_PP; for Transit also W_PS (pair -> remaining single carrier)
530 RowBuilder pp(M);
531 RowBuilder ps(M);
532 for (Index j = 1; j < m; ++j) {
533 for (Index i = 0; i < j; ++i) {
534 double D = 0.0;
535 const std::array<std::array<Index, 2>, 2> movers = {
536 {{i, j}, {j, i}}}; // {moving carrier, other carrier}
537 for (const auto& mv : movers) {
538 Index a = mv[0];
539 Index b = mv[1];
540 for (const ToFGraph::Edge& e : graph_.OutEdges(tr.site_of[a])) {
541 if (e.rate <= 0.0 || e.to == tr.site_of[b]) {
542 continue; // blocked: site occupied by the other carrier
543 }
544 D += e.rate;
545 if (sites[e.to].role == ToFGraph::Role::Sink) {
546 if (transit) {
547 ps.Add(b, e.rate);
548 }
549 } else {
550 pp.Add(PairRank(tr.index_of[e.to], b), -e.rate);
551 }
552 }
553 }
554 pp.Add(PairRank(i, j), D);
555 pp.FinishRow();
556 ps.FinishRow();
557 }
558 }
559 SpMatR Q = pp.Build(M);
560
561 // symmetrization weights of a pair: product of the site weights
562 Eigen::VectorXd h;
563 if (tr.half_log_weight.size() == m) {
564 h.resize(M);
565 for (Index j = 1; j < m; ++j) {
566 for (Index i = 0; i < j; ++i) {
567 h[PairRank(i, j)] = tr.half_log_weight[i] + tr.half_log_weight[j];
568 }
569 }
570 }
571 GeneratorSolver gs(Q, h, solver, settings_);
572
573 // injection: uniform over pairs of distinct source sites
574 std::vector<Index> src(sources.size());
575 std::transform(sources.begin(), sources.end(), src.begin(),
576 [&tr](Index s) { return tr.index_of[s]; });
577 std::vector<std::array<Index, 2>> pairs;
578 for (std::size_t a = 0; a < src.size(); ++a) {
579 for (std::size_t b = a + 1; b < src.size(); ++b) {
580 pairs.push_back({src[a], src[b]});
581 }
582 }
583 Eigen::VectorXd w = Eigen::VectorXd::Zero(M);
584 for (const auto& pr : pairs) {
585 w[PairRank(pr[0], pr[1])] = 1.0 / double(pairs.size());
586 }
587
588 Result result;
589 result.carriers = 2;
590 result.observable = observable;
591 result.unknowns = M;
592 Eigen::VectorXd tau;
593 Eigen::VectorXd m2;
594 Eigen::VectorXd g;
595 Eigen::VectorXd g_single;
596 std::string single_solver;
597 double single_error = 0.0;
598 if (!transit) {
599 tau = gs.Solve(Eigen::VectorXd::Ones(M));
600 m2 = gs.Solve(2.0 * tau);
601 g = gs.SolveAdjoint(w);
602 } else {
603 // remaining single carrier: Q1 tau_S = 1/2
604 RowBuilder rows(m);
605 for (Index t = 0; t < m; ++t) {
606 Index i = tr.site_of[t];
607 rows.Add(t, graph_.EscapeRate(i));
608 for (const ToFGraph::Edge& e : graph_.OutEdges(i)) {
609 if (e.rate > 0.0 && sites[e.to].role != ToFGraph::Role::Sink) {
610 rows.Add(tr.index_of[e.to], -e.rate);
611 }
612 }
613 rows.FinishRow();
614 }
615 SpMatR Q1 = rows.Build(m);
616 SpMatR W = ps.Build(m);
617 GeneratorSolver gs1(Q1, tr.half_log_weight, solver, settings_);
618 Eigen::VectorXd tau_s = gs1.Solve(Eigen::VectorXd::Constant(m, 0.5));
619 tau = gs.Solve(Eigen::VectorXd::Ones(M) + W * tau_s);
620 g = gs.SolveAdjoint(w);
621 // expected number of transitions into each single-carrier state
622 Eigen::VectorXd inflow = W.transpose() * g;
623 g_single = gs1.SolveAdjoint(inflow);
624 single_solver = gs1.name;
625 single_error = gs1.backward_error;
626 result.iterations += gs1.iterations;
627 result.converged = gs1.converged;
628 }
629 result.solver = gs.name;
630 if (transit && single_solver != gs.name) {
631 result.solver += " (remaining carrier: " + single_solver + ")";
632 }
633 result.backward_error = std::max(gs.backward_error, single_error);
634 result.refinement_steps = gs.refinement_steps;
635 result.iterations += gs.iterations;
636 result.converged =
637 result.converged && gs.converged && result.backward_error < 1e-10;
638
639 Index np = Index(pairs.size());
640 result.tau.resize(np);
641 if (!transit) {
642 result.tau_std.resize(np);
643 }
644 double mean2 = 0.0;
645 for (Index k = 0; k < np; ++k) {
646 Index r = PairRank(pairs[k][0], pairs[k][1]);
647 result.sources.push_back(
648 {tr.site_of[pairs[k][0]], tr.site_of[pairs[k][1]]});
649 result.tau[k] = tau[r];
650 if (!transit) {
651 result.tau_std[k] = std::sqrt(std::max(m2[r] - tau[r] * tau[r], 0.0));
652 mean2 += m2[r];
653 }
654 }
655 result.tau_mean = result.tau.mean();
656 if (!transit) {
657 mean2 /= double(np);
658 result.tau_std_injection =
659 std::sqrt(std::max(mean2 - result.tau_mean * result.tau_mean, 0.0));
660 }
661 result.tau_harmonic = double(np) / result.tau.cwiseInverse().sum();
662
663 // time each site is occupied: sum over the pair states containing it,
664 // plus the single-carrier states for Transit
665 result.occupation_time = Eigen::VectorXd::Zero(n);
666 for (Index j = 1; j < m; ++j) {
667 for (Index i = 0; i < j; ++i) {
668 double t = g[PairRank(i, j)];
669 result.occupation_time[tr.site_of[i]] += t;
670 result.occupation_time[tr.site_of[j]] += t;
671 }
672 }
673 if (transit) {
674 for (Index t = 0; t < m; ++t) {
675 result.occupation_time[tr.site_of[t]] += g_single[t];
676 }
677 }
678 return result;
679}
