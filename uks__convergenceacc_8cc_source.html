48void UKSConvergenceAcc::setOverlap(AOOverlap& S, double etol) {
49 // As ConvergenceAcc::setOverlap: removed directions are kept out of the
50 // occupied and low virtual space via removed_projector_.
51 S_ = &S;
52 const SymmetricEigenSystem es = SymmetricEigen(S.Matrix());
53 const Eigen::VectorXd& s_eig = es.values;
54 Eigen::VectorXd inv_sqrt = Eigen::VectorXd::Zero(s_eig.size());
55 Index removed = 0;
56 for (Index i = 0; i < s_eig.size(); ++i) {
57 if (s_eig(i) < etol) {
58 ++removed;
59 } else {
60 inv_sqrt(i) = 1.0 / std::sqrt(s_eig(i));
61 }
62 }
63 Sminusahalf = es.vectors * inv_sqrt.asDiagonal() * es.vectors.transpose();
64 removed_projector_.resize(0, 0);
65 if (removed > 0) {
66 const Eigen::MatrixXd U = es.vectors.leftCols(removed);
67 removed_projector_ = U * U.transpose();
68 }
69 XTP_LOG(Log::error, *log_)
70 << TimeStamp() << " Smallest value of AOOverlap matrix is " << s_eig(0)
71 << std::flush;
72 XTP_LOG(Log::error, *log_)
73 << TimeStamp() << " Removed " << removed
74 << " basisfunction from inverse overlap matrix (threshold " << etol << ")"
75 << std::flush;
76 if (removed == 0 && s_eig(0) < 1e3 * etol) {
77 XTP_LOG(Log::error, *log_)
78 << TimeStamp()
79 << " WARNING: the overlap matrix is nearly singular; if the SCF is "
80 "unstable, raise xtpdft.overlap_tolerance above "
81 << s_eig(0) << "." << std::flush;
82 }
83}

85tools::EigenSystem UKSConvergenceAcc::SolveFockmatrix(
86 const Eigen::MatrixXd& H) const {
87 Eigen::MatrixXd H_ortho = Sminusahalf.transpose() * H * Sminusahalf;
88 if (removed_projector_.size() > 0) {
89 H_ortho += kRemovedShift * removed_projector_;
90 }
91 const SymmetricEigenSystem es = SymmetricEigen(H_ortho);
92 if (!es.values.allFinite()) {
93 throw std::runtime_error("Matrix Diagonalisation failed");
94 }
95
96 tools::EigenSystem result;
97 result.eigenvalues() = es.values;
98 result.eigenvectors() = Sminusahalf * es.vectors;
99 return result;
100}

102Eigen::MatrixXd UKSConvergenceAcc::UnflattenRotation(
103 const Eigen::VectorXd& v_ov, Index nao, Index nocclevels) const {
104 Index nvirt = nao - nocclevels;
105 Eigen::MatrixXd kappa = Eigen::MatrixXd::Zero(nao, nao);
106 for (Index i = 0; i < nocclevels; ++i) {
107 for (Index a = 0; a < nvirt; ++a) {
108 double val = v_ov(i * nvirt + a);
109 kappa(i, nocclevels + a) = val;
110 kappa(nocclevels + a, i) = -val;
111 }
112 }
113 return kappa;
114}

116std::pair<Eigen::MatrixXd, Eigen::MatrixXd>
117 UKSConvergenceAcc::UnflattenCoupledRotation(const Eigen::VectorXd& v,
118 Index nao_alpha,
119 Index nocclevels_alpha,
120 Index nao_beta,
121 Index nocclevels_beta) const {
122 Index n_ov_alpha = nocclevels_alpha * (nao_alpha - nocclevels_alpha);
123 Eigen::VectorXd v_alpha = v.head(n_ov_alpha);
124 Eigen::VectorXd v_beta = v.tail(v.size() - n_ov_alpha);
125 Eigen::MatrixXd kappa_alpha =
126 UnflattenRotation(v_alpha, nao_alpha, nocclevels_alpha);
127 Eigen::MatrixXd kappa_beta =
128 UnflattenRotation(v_beta, nao_beta, nocclevels_beta);
129 return {kappa_alpha, kappa_beta};
130}

132Eigen::VectorXd UKSConvergenceAcc::BuildCoupledSigmaVector(
133 const Eigen::VectorXd& v, const Eigen::MatrixXd& C_alpha,
134 Index nocclevels_alpha, const Eigen::MatrixXd& C_beta,
135 Index nocclevels_beta, const CoupledFockBuilder& coupled_fock_builder,
136 double finite_diff_step) const {
137 Index nao_alpha = C_alpha.rows();
138 Index nvirt_alpha = nao_alpha - nocclevels_alpha;
139 Index nao_beta = C_beta.rows();
140 Index nvirt_beta = nao_beta - nocclevels_beta;
141 Index n_ov_alpha = nocclevels_alpha * nvirt_alpha;
142 Index n_ov_beta = nocclevels_beta * nvirt_beta;
143
144 auto [kappa_alpha_trial, kappa_beta_trial] = UnflattenCoupledRotation(
145 v, nao_alpha, nocclevels_alpha, nao_beta, nocclevels_beta);
146 kappa_alpha_trial *= finite_diff_step;
147 kappa_beta_trial *= finite_diff_step;
148
149 // Same linearized-exponential + Lowdin-reorthonormalization approach
150 // as BuildSigmaVector's own EvaluateGradientAt -- but rotates BOTH
151 // channels together and builds BOTH new Fock matrices together in
152 // ONE coupled_fock_builder call, rather than one channel at a time
153 // with the other held fixed. This is what actually captures the
154 // alpha-beta coupling: the shared Coulomb potential and the XC
155 // kernel's cross-spin terms naturally see both perturbed densities
156 // together inside coupled_fock_builder, rather than needing this
157 // function to construct any cross-coupling term explicitly itself.
158 auto EvaluateBothGradientsAt = [&](const Eigen::MatrixXd& kappa_alpha,
159 const Eigen::MatrixXd& kappa_beta)
160 -> std::pair<Eigen::MatrixXd, Eigen::MatrixXd> {
161 Eigen::MatrixXd C_alpha_rot =
162 C_alpha *
163 (Eigen::MatrixXd::Identity(nao_alpha, nao_alpha) + kappa_alpha);
164 Eigen::MatrixXd nonortho_alpha =
165 C_alpha_rot.transpose() * S_->Matrix() * C_alpha_rot;
166 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es_alpha(nonortho_alpha);
167 C_alpha_rot = C_alpha_rot * es_alpha.operatorInverseSqrt();
168
169 Eigen::MatrixXd C_beta_rot =
170 C_beta * (Eigen::MatrixXd::Identity(nao_beta, nao_beta) + kappa_beta);
171 Eigen::MatrixXd nonortho_beta =
172 C_beta_rot.transpose() * S_->Matrix() * C_beta_rot;
173 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es_beta(nonortho_beta);
174 C_beta_rot = C_beta_rot * es_beta.operatorInverseSqrt();
175
176 Eigen::MatrixXd C_alpha_occ_rot = C_alpha_rot.leftCols(nocclevels_alpha);
177 Eigen::MatrixXd D_alpha_rot = C_alpha_occ_rot * C_alpha_occ_rot.transpose();
178 Eigen::MatrixXd C_beta_occ_rot = C_beta_rot.leftCols(nocclevels_beta);
179 Eigen::MatrixXd D_beta_rot = C_beta_occ_rot * C_beta_occ_rot.transpose();
180
181 SpinFock H_rot = coupled_fock_builder(D_alpha_rot, D_beta_rot);
182 Eigen::MatrixXd F_MO_alpha_rot =
183 C_alpha_rot.transpose() * H_rot.alpha * C_alpha_rot;
184 Eigen::MatrixXd F_MO_beta_rot =
185 C_beta_rot.transpose() * H_rot.beta * C_beta_rot;
186 return {F_MO_alpha_rot, F_MO_beta_rot};
187 };
188
189 auto [F_MO_alpha_plus, F_MO_beta_plus] =
190 EvaluateBothGradientsAt(kappa_alpha_trial, kappa_beta_trial);
191 auto [F_MO_alpha_minus, F_MO_beta_minus] =
192 EvaluateBothGradientsAt(-kappa_alpha_trial, -kappa_beta_trial);
193
194 // The energy gradient is dE/dkappa_ia = -2 F_ia (see
195 // EnergyGradientScale), so the Hessian-vector product is -2 times the
196 // central difference of F_ia along v.
197 const double scale = EnergyGradientScale / (2.0 * finite_diff_step);
198 Eigen::VectorXd sigma(n_ov_alpha + n_ov_beta);
199 for (Index i = 0; i < nocclevels_alpha; ++i) {
200 for (Index a = 0; a < nvirt_alpha; ++a) {
201 double g_plus_ia = F_MO_alpha_plus(i, nocclevels_alpha + a);
202 double g_minus_ia = F_MO_alpha_minus(i, nocclevels_alpha + a);
203 sigma(i * nvirt_alpha + a) = scale * (g_plus_ia - g_minus_ia);
204 }
205 }
206 for (Index i = 0; i < nocclevels_beta; ++i) {
207 for (Index a = 0; a < nvirt_beta; ++a) {
208 double g_plus_ia = F_MO_beta_plus(i, nocclevels_beta + a);
209 double g_minus_ia = F_MO_beta_minus(i, nocclevels_beta + a);
210 sigma(n_ov_alpha + i * nvirt_beta + a) = scale * (g_plus_ia - g_minus_ia);
211 }
212 }
213 return sigma;
214}

216Eigen::MatrixXd UKSConvergenceAcc::DirectMinimizationRotation(
217 const Eigen::MatrixXd& H_AO, const tools::EigenSystem& MOs,
218 Index nocclevels, double& predicted_energy_change) const {
219 Index nao = MOs.eigenvectors().rows();
220 const Eigen::MatrixXd& C = MOs.eigenvectors();
221 const Eigen::VectorXd& eps = MOs.eigenvalues();
222
223 // Orbital gradient: the occ-virt block of the MO-basis Fock matrix,
224 // which vanishes exactly at self-consistency (F_MO would be block-
225 // diagonal, occ-occ and virt-virt only).
226 Eigen::MatrixXd F_MO = C.transpose() * H_AO * C;
227
228 // Antisymmetric rotation generator: nonzero only in the occ-virt (and
229 // virt-occ, by antisymmetry) blocks -- occ-occ/virt-virt rotations
230 // would leave the density matrix (and therefore the energy) entirely
231 // unchanged, so there is nothing to gain from including them.
232 Eigen::MatrixXd kappa = Eigen::MatrixXd::Zero(nao, nao);
233 // g_h_ratio(i,a) below stores kappa_ia's own g_ia/h_ia BEFORE the
234 // whole-matrix trust-radius scaling further down -- needed again
235 // afterward to compute the quadratic model's predicted energy change
236 // using the FINAL (per-element- and trust-radius-clamped) kappa,
237 // since g_ia and h_ia themselves do not change under that later,
238 // uniform rescaling, only kappa does.
239 Eigen::MatrixXd h_matrix = Eigen::MatrixXd::Zero(nao, nao);
240 // Guards near-degenerate occ-virt orbital pairs from producing a
241 // wildly oversized step for THAT pair specifically -- confirmed
242 // directly against an independent ORCA reference run on this exact
243 // system, whose own TRAH implementation repeatedly hit BOTH
244 // genuinely negative gaps (occupied/virtual character swapping
245 // during iteration, since occupation here is defined by column
246 // index, not current energy ordering -- down to -0.635 Ha) and
247 // gaps numerically indistinguishable from zero (as small as
248 // 0.000002 Ha), needing its own explicit warnings and special
249 // handling for both. abs() here is essential, not cosmetic: a
250 // signed gap of -0.6 would otherwise pass std::max(-0.6, kMinGap)
251 // as if it were the tiny POSITIVE value kMinGap, producing an
252 // enormous, wrong-direction step for exactly that pair -- the
253 // Hessian approximation must be treated as positive-definite
254 // (a legitimate descent direction) regardless of the true, possibly
255 // negative or near-zero curvature this diagonal approximation
256 // cannot itself represent.
257 constexpr double kMinGap = 1e-3;
258 // Per-PAIR step cap, independent of and in addition to the overall,
259 // whole-matrix trust-radius bound below -- confirmed necessary
260 // because a single pathological pair (as above) could otherwise
261 // dominate the entire rotation DIRECTION even after the whole
262 // matrix is normalized to the trust radius, silently scaling every
263 // other, well-behaved pair down to near-nothing while that one bad
264 // pair still controls where the step actually points.
265 constexpr double kMaxKappaElement = 0.1;
266 for (Index i = 0; i < nocclevels; ++i) {
267 for (Index a = nocclevels; a < nao; ++a) {
268 double gap = std::max(std::abs(eps(a) - eps(i)), kMinGap);
269 // Energy gradient g_ia = -2 F_ia (EnergyGradientScale) and the
270 // diagonal of the energy Hessian, 2 (eps_a - eps_i): the leading
271 // term of d^2E/dkappa_ia^2 for one spin channel.
272 double g_ia = EnergyGradientScale * F_MO(i, a);
273 double h_ia = 2.0 * gap;
274 double kappa_ia = -g_ia / h_ia;
275 kappa_ia = std::clamp(kappa_ia, -kMaxKappaElement, kMaxKappaElement);
276 kappa(i, a) = kappa_ia;
277 kappa(a, i) = -kappa_ia;
278 h_matrix(i, a) = h_ia;
279 }
280 }
281
282 double knorm = kappa.norm();
283 if (knorm > trust_.Radius()) {
284 kappa *= (trust_.Radius() / knorm);
285 }
286
287 // Quadratic model's predicted energy change, Sum_ia[g_ia*kappa_ia
288 // + 0.5*h_ia*kappa_ia^2], for the kappa actually applied (after the
289 // per-element clamp and the trust-radius scaling), as needed by
290 // Iterate's Fletcher accept/reject check.
291 predicted_energy_change = 0.0;
292 for (Index i = 0; i < nocclevels; ++i) {
293 for (Index a = nocclevels; a < nao; ++a) {
294 double kappa_ia = kappa(i, a);
295 predicted_energy_change += EnergyGradientScale * F_MO(i, a) * kappa_ia +
296 0.5 * h_matrix(i, a) * kappa_ia * kappa_ia;
297 }
298 }
299
300 // exp(kappa) ~= I + kappa is valid specifically because the trust-
301 // radius bound above keeps kappa small -- re-orthonormalize
302 // afterward since I+kappa is only approximately unitary, using the
303 // same Lowdin (symmetric orthogonalization) approach as
304 // DFTEngine::OrthogonalizeGuess.
305 Eigen::MatrixXd C_new = C * (Eigen::MatrixXd::Identity(nao, nao) + kappa);
306 Eigen::MatrixXd nonortho = C_new.transpose() * S_->Matrix() * C_new;
307 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es_ortho(nonortho);
308 return C_new * es_ortho.operatorInverseSqrt();
309}

311std::pair<Eigen::MatrixXd, Eigen::MatrixXd>
312 UKSConvergenceAcc::CoupledAugmentedHessianStep(
313 const Eigen::MatrixXd& H_AO_alpha, const tools::EigenSystem& MOs_alpha,
314 Index nocclevels_alpha, const Eigen::MatrixXd& H_AO_beta,
315 const tools::EigenSystem& MOs_beta, Index nocclevels_beta,
316 const CoupledFockBuilder& coupled_fock_builder, double trust_radius,
317 double& predicted_energy_change) const {
318 Index nao_alpha = MOs_alpha.eigenvectors().rows();
319 Index nvirt_alpha = nao_alpha - nocclevels_alpha;
320 Index n_ov_alpha = nocclevels_alpha * nvirt_alpha;
321 const Eigen::MatrixXd& C_alpha = MOs_alpha.eigenvectors();
322 const Eigen::VectorXd& eps_alpha = MOs_alpha.eigenvalues();
323
324 Index nao_beta = MOs_beta.eigenvectors().rows();
325 Index nvirt_beta = nao_beta - nocclevels_beta;
326 Index n_ov_beta = nocclevels_beta * nvirt_beta;
327 const Eigen::MatrixXd& C_beta = MOs_beta.eigenvectors();
328 const Eigen::VectorXd& eps_beta = MOs_beta.eigenvalues();
329
330 Index n_ov = n_ov_alpha + n_ov_beta;
331
332 Eigen::MatrixXd F_MO_alpha = C_alpha.transpose() * H_AO_alpha * C_alpha;
333 Eigen::MatrixXd F_MO_beta = C_beta.transpose() * H_AO_beta * C_beta;
334
335 // Energy gradient g = -2 F_ia (EnergyGradientScale) and the diagonal
336 // preconditioner 2 (eps_a - eps_i), the leading diagonal term of the
337 // energy Hessian that BuildCoupledSigmaVector applies; alpha block
338 // first, beta block after it, as in UnflattenCoupledRotation. The
339 // preconditioner has no alpha-beta cross terms; only the Davidson
340 // convergence depends on it, not the step.
341 Eigen::VectorXd g(n_ov);
342 Eigen::VectorXd diag_h(n_ov);
343 constexpr double kMinGap = 1e-3;
344 for (Index i = 0; i < nocclevels_alpha; ++i) {
345 for (Index a = 0; a < nvirt_alpha; ++a) {
346 g(i * nvirt_alpha + a) =
347 EnergyGradientScale * F_MO_alpha(i, nocclevels_alpha + a);
348 double gap = std::max(
349 std::abs(eps_alpha(nocclevels_alpha + a) - eps_alpha(i)), kMinGap);
350 diag_h(i * nvirt_alpha + a) = 2.0 * gap;
351 }
352 }
353 for (Index i = 0; i < nocclevels_beta; ++i) {
354 for (Index a = 0; a < nvirt_beta; ++a) {
355 g(n_ov_alpha + i * nvirt_beta + a) =
356 EnergyGradientScale * F_MO_beta(i, nocclevels_beta + a);
357 double gap = std::max(
358 std::abs(eps_beta(nocclevels_beta + a) - eps_beta(i)), kMinGap);
359 diag_h(n_ov_alpha + i * nvirt_beta + a) = 2.0 * gap;
360 }
361 }
362
363 // Hessian products by finite differences of the coupled Fock matrices,
364 // one vector at a time (two coupled Fock builds each).
365 bool analytic = bool(orbital_hessian_) && !analytic_hessian_failed_;
366 trah::HessianProduct hessian = [&](const Eigen::MatrixXd& V) {
367 if (analytic) {
368 try {
369 return orbital_hessian_(MOs_alpha, MOs_beta, H_AO_alpha, H_AO_beta, V);
370 } catch (const std::runtime_error& error) {
371 XTP_LOG(Log::error, *log_)
372 << TimeStamp() << " Analytic UKS Hessian not available ("
373 << error.what() << "); using finite differences" << std::flush;
374 analytic = false;
375 analytic_hessian_failed_ = true;
376 }
377 }
378 Eigen::MatrixXd HV(V.rows(), V.cols());
379 for (Index k = 0; k < V.cols(); ++k) {
380 HV.col(k) =
381 BuildCoupledSigmaVector(V.col(k), C_alpha, nocclevels_alpha, C_beta,
382 nocclevels_beta, coupled_fock_builder);
383 }
384 return HV;
385 };
386 const trah::StepResult step = trah::Step(g, diag_h, hessian, trust_radius,
387 trah_options_, &trah_subspace_);
388 predicted_energy_change = step.predicted;
389 last_step_linear_ = step.linear;
390 last_step_norm_ = step.x.norm();
391
392 XTP_LOG(Log::warning, *log_)
393 << TimeStamp() << " TRAH step: |g|=" << g.norm()
394 << ", |x|=" << step.x.norm() << ", trust radius=" << trust_radius
395 << ", alpha=" << step.alpha << ", micro-iterations=" << step.iterations
396 << ", Hessian products=" << step.products
397 << (analytic ? " (analytic)" : " (finite differences)")
398 << (step.reused ? " (subspace reused)" : "")
399 << ", residual=" << step.residual
400 << (step.converged ? "" : " (not converged)")
401 << ", predicted dE=" << predicted_energy_change << std::flush;
402
403 return {
404 trah::RotateOrbitals(C_alpha, step.x.head(n_ov_alpha), nocclevels_alpha),
405 trah::RotateOrbitals(C_beta, step.x.tail(n_ov_beta), nocclevels_beta)};
406}

408Eigen::MatrixXd UKSConvergenceAcc::DensityMatrixGroundState_unres(
409 const Eigen::MatrixXd& MOs, Index nocclevels) const {
410 if (nocclevels == 0) {

414 return occstates * occstates.transpose();

428void UKSConvergenceAcc::Levelshift(Eigen::MatrixXd& H,
429 const Eigen::MatrixXd& MOs_old,
430 const options& opt, Index nocclevels) const {
431 if (opt.levelshift < 1e-9) {
432 return;
433 }
434 Eigen::VectorXd virt = Eigen::VectorXd::Zero(H.rows());
435 for (Index i = nocclevels; i < H.rows(); ++i) {
436 virt(i) = opt.levelshift;
437 }
438
439 XTP_LOG(Log::error, *log_)
440 << TimeStamp() << " Using levelshift:" << opt.levelshift << " Hartree"
441 << std::flush;
442
443 Eigen::MatrixXd vir = S_->Matrix() * MOs_old * virt.asDiagonal() *
444 MOs_old.transpose() * S_->Matrix();
445 H += vir;
446}

448Eigen::MatrixXd UKSConvergenceAcc::BuildErrorMatrix(
449 const Eigen::MatrixXd& dmat, const Eigen::MatrixXd& H) const {
450 const Eigen::MatrixXd& S = S_->Matrix();
451 return Sminusahalf.transpose() * (H * dmat * S - S * dmat * H) * Sminusahalf;
452}

454double UKSConvergenceAcc::CombinedError(const Eigen::MatrixXd& err_alpha,
455 const Eigen::MatrixXd& err_beta) const {
456 return std::max(err_alpha.cwiseAbs().maxCoeff(),
457 err_beta.cwiseAbs().maxCoeff());

459
460UKSConvergenceAcc::SpinDensity UKSConvergenceAcc::Iterate(
461 const SpinDensity& dmat, SpinFock& H, tools::EigenSystem& MOs_alpha,
462 tools::EigenSystem& MOs_beta, double totE) {
463
464 // Check the second-order step returned by the previous call, now that
465 // the energy of its density (totE) is known: Fletcher's trust-radius
466 // update. A rejected step is undone and, while second-order steps
467 // continue, repeated at once with the smaller radius from the stored
468 // Fock matrices and Davidson subspace (see ConvergenceAcc::Iterate).
469 last_rejected_ = false;
470 if (direct_min_pending_) {
471 direct_min_pending_ = false;
472 const double actual_change = totE - direct_min_pre_energy_;
473 const bool accepted =
474 trust_.Update(direct_min_predicted_change_, actual_change,
475 direct_min_linear_, direct_min_step_norm_);
476 XTP_LOG(Log::warning, *log_)
477 << TimeStamp()
478 << " Second-order step check: actual dE=" << actual_change
479 << ", predicted dE=" << direct_min_predicted_change_
480 << ", r=" << trust_.LastRatio()
481 << (accepted ? " (accepted)" : " (rejected)")
482 << ", new trust radius=" << trust_.Radius() << std::flush;
483 if (!accepted) {
484 if (trust_.Radius() < kMinTrustRadius) {
485 direct_min_floor_hit_ = true;
486 trah_active_ = false;
487 XTP_LOG(Log::warning, *log_)
488 << TimeStamp() << " Trust radius below " << kMinTrustRadius
489 << " without an accepted step; leaving second-order steps for "
490 "this SCF"
491 << std::flush;
492 }
493 MOs_alpha.eigenvectors() = direct_min_pre_MOs_alpha_;
494 MOs_beta.eigenvectors() = direct_min_pre_MOs_beta_;
495 MOs_alpha.eigenvalues() = direct_min_pre_MOs_alpha_energies_;
496 MOs_beta.eigenvalues() = direct_min_pre_MOs_beta_energies_;
497 totE_.push_back(direct_min_pre_energy_);
498 last_rejected_ = true;
499 usedmixing_ = false;
500 if (trah_active_) {
501 const SpinFock H_pre = direct_min_pre_H_;
502 return SecondOrderStep(DensityMatrix(MOs_alpha, MOs_beta), H_pre,
503 MOs_alpha, MOs_beta, direct_min_pre_energy_);
504 }
505 return DensityMatrix(MOs_alpha, MOs_beta);
506 }
507 }
508
509 // History, trimmed at ONE index for the Fock, density and error
510 // histories and for DIIS's own error history: the oldest entry, or with
511 // DIIS_maxout the entry with the largest error. Previously the index was
512 // a stale position (never re-based after an erase, maxerror_ never
513 // reset), and DIIS::Update could trim at a different index than the
514 // Fock history, pairing coefficients with the wrong matrices. totE_ is
515 // not trimmed: it is the chronological energy list getDeltaE reads.
516 Index drop = 0;
517 if (int(mathist_alpha_.size()) == opt_alpha_.histlength) {
518 if (opt_alpha_.maxout) {
519 drop = Index(std::max_element(errhist_.begin(), errhist_.end()) -
520 errhist_.begin());
521 }
522 mathist_alpha_.erase(mathist_alpha_.begin() + drop);
523 mathist_beta_.erase(mathist_beta_.begin() + drop);
524 dmatHist_alpha_.erase(dmatHist_alpha_.begin() + drop);
525 dmatHist_beta_.erase(dmatHist_beta_.begin() + drop);
526 errhist_.erase(errhist_.begin() + drop);
527 }
528
529 totE_.push_back(totE);
530
531 if (nocclevels_alpha_ > 0 &&
532 nocclevels_alpha_ < MOs_alpha.eigenvalues().size()) {
533 double gap_alpha = MOs_alpha.eigenvalues()(nocclevels_alpha_) -
534 MOs_alpha.eigenvalues()(nocclevels_alpha_ - 1);
535 // consecutive_adiis_failures_ < kMaxConsecutiveADIISFailures added
536 // deliberately: a real run showed level shift catastrophically
537 // breaking the Davidson solve this class's own direct-minimization
538 // fallback depends on (residual climbing past 395, far beyond any
539 // prior failure mode) once direct-minimization engaged (originally
540 // discovered against an earlier, decoupled AugmentedHessianStep
541 // implementation; the same mechanism still applies to
542 // CoupledAugmentedHessianStep's own diagonal preconditioner below,
543 // which is built the same way). Root cause: Levelshift() modifies H
544 // in place, BEFORE CoupledAugmentedHessianStep/DirectMinimizationRotation
545 // ever see it -- and that preconditioner (diag_h) is built directly
546 // from eps(a)-eps(i), the raw orbital energy gap, which becomes
547 // systematically inflated by the shift for virtual orbitals.
548 // ADIIS/DIIS do not
549 // have this problem (they operate on the Fock/density matrices
550 // themselves, never deriving a separate quantity like an orbital
551 // gap from the shifted eigenvalues), so level shift stays active
552 // for them -- this guard only disables it once direct-minimization
553 // is about to take over, since level shift's own purpose
554 // (discouraging occ-virt mixing during diagonalization) does not
555 // even apply to a method that never diagonalizes H at all.
556 if ((diiserror_ > opt_alpha_.levelshiftend &&
557 opt_alpha_.levelshift > 0.0) ||
558 gap_alpha < 1e-6) {
559 // "- 1" margin: consecutive_adiis_failures_ is only incremented
560 // LATER in this same Iterate() call (after this check runs), so
561 // without the margin this would still see the PREVIOUS
562 // iteration's count on the exact iteration where the threshold
563 // is reached and direct-minimization actually fires -- letting
564 // level shift through on precisely the iteration it needs to be
565 // blocked, one iteration too late.
566 if (consecutive_adiis_failures_ < kMaxConsecutiveADIISFailures - 1 &&
567 !trah_active_) {
568 Levelshift(H.alpha, MOs_alpha.eigenvectors(), opt_alpha_,
569 nocclevels_alpha_);
570 }
571 }
572 }
573
574 if (nocclevels_beta_ > 0 &&
575 nocclevels_beta_ < MOs_beta.eigenvalues().size()) {
576 double gap_beta = MOs_beta.eigenvalues()(nocclevels_beta_) -
577 MOs_beta.eigenvalues()(nocclevels_beta_ - 1);
578 if ((diiserror_ > opt_beta_.levelshiftend && opt_beta_.levelshift > 0.0) ||
579 gap_beta < 1e-6) {
580 if (consecutive_adiis_failures_ < kMaxConsecutiveADIISFailures - 1 &&
581 !trah_active_) {
582 Levelshift(H.beta, MOs_beta.eigenvectors(), opt_beta_,
583 nocclevels_beta_);
584 }
585 }
586 }
587
588 Eigen::MatrixXd err_alpha = BuildErrorMatrix(dmat.alpha, H.alpha);
589 Eigen::MatrixXd err_beta = BuildErrorMatrix(dmat.beta, H.beta);
590
591 diiserror_ = CombinedError(err_alpha, err_beta);
592
593 // Trailing-average trigger bookkeeping (see this class's own header
594 // comment on diiserror_history_ for the full ORCA-derived reasoning)
595 // -- tracked unconditionally, every iteration, regardless of what
596 // this iteration goes on to do (ADIIS/DIIS/mixing/direct-
597 // minimization), since the whole point is to observe the genuine,
598 // realized trajectory of diiserror_ itself.
599 ++total_iteration_count_;
600 diiserror_history_.push_back(diiserror_);
601 if (Index(diiserror_history_.size()) > kTrailingWindowSize) {
602 diiserror_history_.erase(diiserror_history_.begin());
603 }
604
605 mathist_alpha_.push_back(H.alpha);
606 mathist_beta_.push_back(H.beta);
607 dmatHist_alpha_.push_back(dmat.alpha);
608 dmatHist_beta_.push_back(dmat.beta);
609
610 errhist_.push_back(diiserror_);
611
612 // crucial: one shared error matrix = alpha + beta contribution
613 diis_.Update(drop, err_alpha, err_beta);
614
615 bool diis_error = false;
616 XTP_LOG(Log::error, *log_)
617 << TimeStamp() << " DIIs error " << diiserror_ << std::flush;
618 XTP_LOG(Log::error, *log_)
619 << TimeStamp() << " Delta Etot " << getDeltaE() << std::flush;
620
621 // Once second-order steps have started (second_order "auto" after the
622 // trigger below, or "always" from the start) they continue until the SCF
623 // has converged or the trust radius has collapsed.
624 if (!trah_active_ && !direct_min_floor_hit_ &&
625 opt_alpha_.second_order != "off") {
626 const bool failures =
627 consecutive_adiis_failures_ >= kMaxConsecutiveADIISFailures;
628 const bool stalled = TrailingAverageStalled();
629 if (opt_alpha_.second_order == "always" || failures || stalled) {
630 XTP_LOG(Log::warning, *log_)
631 << TimeStamp() << " Switching to second-order (TRAH) steps"
632 << (failures ? " ((A)DIIS failed repeatedly)" : "")
633 << (stalled ? " (DIIS error stalled)" : "") << std::flush;
634 trah_active_ = true;
635 }
636 }
637 if (trah_active_) {
638 return SecondOrderStep(dmat, H, MOs_alpha, MOs_beta, totE);
639 }
640
641 Eigen::MatrixXd H_guess_alpha = H.alpha;
642 Eigen::MatrixXd H_guess_beta = H.beta;
643
644 // As in ConvergenceAcc::Iterate: below DIIS_start (e.g. after a warm
645 // start) no damping, a plain first step and DIIS from two entries on.
646 const bool near_convergence = diiserror_ < opt_alpha_.diis_start;
647 const std::size_t min_history = near_convergence ? 1 : 2;
648 if ((diiserror_ < opt_alpha_.adiis_start ||
649 diiserror_ < opt_alpha_.diis_start) &&
650 opt_alpha_.usediis && mathist_alpha_.size() > min_history) {
651
652 Eigen::VectorXd coeffs;
653
654 if (diiserror_ > opt_alpha_.diis_start ||
655 totE_.back() - totE_[totE_.size() - 2] > kEnergyRiseForADIIS) {
656 coeffs = adiis_.CalcCoeff(dmatHist_alpha_, dmatHist_beta_, mathist_alpha_,
657 mathist_beta_);
658 diis_error = !adiis_.Info() || coeffs.size() == 0;
659 XTP_LOG(Log::warning, *log_)
660 << TimeStamp() << " Using ADIIS for next UKS guess" << std::flush;
661 } else {
662 coeffs = diis_.CalcCoeff();
663 diis_error = !diis_.Info() || coeffs.size() == 0;
664 XTP_LOG(Log::warning, *log_)
665 << TimeStamp() << " Using DIIS for next UKS guess" << std::flush;
666 }
667
668 if (diis_error) {
669 ++consecutive_adiis_failures_;
670 XTP_LOG(Log::warning, *log_)
671 << TimeStamp() << " (A)DIIS failed using mixing instead"
672 << std::flush;
673 H_guess_alpha = H.alpha;
674 H_guess_beta = H.beta;
675 } else {
676 consecutive_adiis_failures_ = 0;
677 H_guess_alpha.setZero();
678 H_guess_beta.setZero();
679 for (Index i = 0; i < coeffs.size(); ++i) {
680 if (std::abs(coeffs(i)) < 1e-8) {
681 continue;
682 }
683 H_guess_alpha += coeffs(i) * mathist_alpha_[i];
684 H_guess_beta += coeffs(i) * mathist_beta_[i];
685 }
686 }
687 }
688
689 MOs_alpha = SolveFockmatrix(H_guess_alpha);
690 MOs_beta = SolveFockmatrix(H_guess_beta);
691
692 SpinDensity dmatout = DensityMatrix(MOs_alpha, MOs_beta);
693
694 if (diiserror_ > opt_alpha_.mixingend || !opt_alpha_.usediis || diis_error ||
695 (mathist_alpha_.size() <= 2 && !near_convergence)) {
696 // mixingend, not adiis_start -- deliberately decoupled (see the
697 // options struct's own comment in convergenceacc.h): ORCA keeps
698 // DampErr fully independent of DIISStart, and recommends making
699 // DampErr much SMALLER for difficult systems specifically so
700 // damping stays active well past the point where DIIS itself
701 // starts being tried -- reusing adiis_start here could never
702 // represent that independently, since it would tie "when does
703 // mixing turn off" to the same value as "when does ADIIS/DIIS
704 // engage at all", which are conceptually separate questions.
705 usedmixing_ = true;
706 // Adaptive damping (matches ORCA's own DampFac/DampMax design,
707 // confirmed directly from a real ORCA log's own resolved SCF
708 // settings -- see the options struct's own comment in
709 // convergenceacc.h for the full reasoning): ramp LINEARLY from
710 // mixingparameter (the base, e.g. 0.7) toward mixingmax (the
711 // ceiling, e.g. 0.98) as consecutive_adiis_failures_ increases
712 // toward kMaxConsecutiveADIISFailures, rather than applying the
713 // ceiling value for the entire run regardless of whether the SCF
714 // is actually struggling. Ties the ramp to the SAME signal already
715 // driving the direct-minimization trigger itself, rather than
716 // introducing a separate struggle metric -- consecutive_adiis_
717 // failures_ resets to 0 on any successful ADIIS/DIIS step, so the
718 // ramp relaxes back toward the base value just as readily as it
719 // climbed.
720 double ramp_fraction =
721 std::min(1.0, double(consecutive_adiis_failures_) /
722 double(kMaxConsecutiveADIISFailures));
723 double mixingparameter_alpha_current =
724 opt_alpha_.mixingparameter +
725 ramp_fraction * (opt_alpha_.mixingmax - opt_alpha_.mixingparameter);
726 double mixingparameter_beta_current =
727 opt_beta_.mixingparameter +
728 ramp_fraction * (opt_beta_.mixingmax - opt_beta_.mixingparameter);
729 dmatout.alpha = mixingparameter_alpha_current * dmat.alpha +
730 (1.0 - mixingparameter_alpha_current) * dmatout.alpha;
731 dmatout.beta = mixingparameter_beta_current * dmat.beta +
732 (1.0 - mixingparameter_beta_current) * dmatout.beta;
733 XTP_LOG(Log::warning, *log_)
734 << TimeStamp() << " Using coupled UKS mixing with adaptive alpha="
735 << mixingparameter_alpha_current
736 << " (base=" << opt_alpha_.mixingparameter
737 << ", ceiling=" << opt_alpha_.mixingmax
738 << ", ramp fraction=" << ramp_fraction << ")" << std::flush;
739 } else {
740 usedmixing_ = false;
741 }
742
743 return dmatout;
744}
745
746// After kAutoStartIteration iterations: true if the DIIS error has, on
747// average over the last kTrailingWindowSize iterations, shrunk by less than
748// a factor kMeanRatioTolerance per iteration (ORCA's AutoTRAH criterion:
749// "Auto Start mean grad. ratio tolernc. 1.125"). Evaluated every iteration,
750// also when the error stays above ADIIS_start and (A)DIIS is never tried.
751bool UKSConvergenceAcc::TrailingAverageStalled() const {
752 if (total_iteration_count_ < kAutoStartIteration ||
753 Index(diiserror_history_.size()) < kTrailingWindowSize) {
754 return false;
755 }
756 double mean_ratio = 0.0;
757 Index ratio_count = 0;
758 for (Index i = 1; i < Index(diiserror_history_.size()); ++i) {
759 if (diiserror_history_[i - 1] > 1e-12) {
760 mean_ratio += diiserror_history_[i] / diiserror_history_[i - 1];
761 ++ratio_count;
762 }
763 }
764 return ratio_count > 0 &&
765 mean_ratio / double(ratio_count) > 1.0 / kMeanRatioTolerance;
766}

768UKSConvergenceAcc::SpinDensity UKSConvergenceAcc::SecondOrderStep(
769 const SpinDensity& dmat, const SpinFock& H, tools::EigenSystem& MOs_alpha,
770 tools::EigenSystem& MOs_beta, double totE) {
771 // The step rotates MOs_alpha/MOs_beta, and its gradient and the Fletcher
772 // check are taken relative to H and totE, which belong to dmat. If dmat
773 // is not the density of these orbitals (after a damped iteration, or a
774 // guess density), return their density first; the step follows on the
775 // next call, from a consistent starting point.
776 SpinDensity dmat_mos = DensityMatrix(MOs_alpha, MOs_beta);
777 const double mismatch =
778 std::max((dmat_mos.alpha - dmat.alpha).cwiseAbs().maxCoeff(),
779 (dmat_mos.beta - dmat.beta).cwiseAbs().maxCoeff());
780 if (mismatch > kDensityMismatchTolerance) {
781 XTP_LOG(Log::warning, *log_)
782 << TimeStamp()
783 << " Density differs from that of the current orbitals (max "
784 "deviation "
785 << mismatch << "); returning the orbital density first" << std::flush;
786 usedmixing_ = false;
787 return dmat_mos;
788 }
789
790 direct_min_pre_energy_ = totE;
791 direct_min_pre_MOs_alpha_ = MOs_alpha.eigenvectors();
792 direct_min_pre_MOs_beta_ = MOs_beta.eigenvectors();
793 direct_min_pre_MOs_alpha_energies_ = MOs_alpha.eigenvalues();
794 direct_min_pre_MOs_beta_energies_ = MOs_beta.eigenvalues();
795 direct_min_pre_H_ = H;
796
797 Eigen::MatrixXd C_new_alpha;
798 Eigen::MatrixXd C_new_beta;
799 // without the linear term (diagonal path) a rejection shrinks by 0.7
800 direct_min_linear_ = 0.0;
801 direct_min_step_norm_ = 0.0;
802 if (coupled_fock_builder_ || orbital_hessian_) {
803 std::tie(C_new_alpha, C_new_beta) = CoupledAugmentedHessianStep(
804 H.alpha, MOs_alpha, nocclevels_alpha_, H.beta, MOs_beta,
805 nocclevels_beta_, coupled_fock_builder_, trust_.Radius(),
806 direct_min_predicted_change_);
807 direct_min_linear_ = last_step_linear_;
808 direct_min_step_norm_ = last_step_norm_;
809 } else {
810 // no Fock builder (e.g. the atomic UKS guess): diagonal Hessian, one
811 // channel at a time
812 double predicted_alpha = 0.0;
813 double predicted_beta = 0.0;
814 C_new_alpha = DirectMinimizationRotation(
815 H.alpha, MOs_alpha, nocclevels_alpha_, predicted_alpha);
816 C_new_beta = DirectMinimizationRotation(H.beta, MOs_beta, nocclevels_beta_,
817 predicted_beta);
818 direct_min_predicted_change_ = predicted_alpha + predicted_beta;
819 }
820 direct_min_pending_ = true;
821
822 MOs_alpha.eigenvectors() = C_new_alpha;
823 MOs_beta.eigenvectors() = C_new_beta;
824 // diagonal of the old Fock matrix in the rotated orbitals: for the
825 // preconditioner and the gap checks until the next diagonalisation
826 MOs_alpha.eigenvalues() =
827 (C_new_alpha.transpose() * H.alpha * C_new_alpha).diagonal();
828 MOs_beta.eigenvalues() =
829 (C_new_beta.transpose() * H.beta * C_new_beta).diagonal();
830 usedmixing_ = false;
831 return DensityMatrix(MOs_alpha, MOs_beta);
832}
