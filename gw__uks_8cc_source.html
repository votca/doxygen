41void GW_UKS::configure(const options& opt) {
42 opt_ = opt;
43
44 // Normalize legacy and new grid-search settings once at configuration time.
45 // This keeps the UKS path numerically aligned with the RKS path.
46 qp_solver::NormalizeGridSearchOptions(opt_);
47 qptotal_ = opt_.qpmax - opt_.qpmin + 1;
48 rpa_.configure(opt_.homo_alpha, opt_.homo_beta, opt_.rpamin, opt_.rpamax);
49 rpa_.setOutOfWindowShift(opt_.out_of_window_shift);
50
51 sigma_alpha_ = SigmaFactory_UKS().Create(opt_.sigma_integration, Mmn_, rpa_,
52 TCMatrix::SpinChannel::Alpha);
53 sigma_beta_ = SigmaFactory_UKS().Create(opt_.sigma_integration, Mmn_, rpa_,
54 TCMatrix::SpinChannel::Beta);
55
56 if (!sigma_alpha_ || !sigma_beta_) {
57 throw std::runtime_error(
58 "GW_UKS currently supports only implemented unrestricted sigma "
59 "evaluators. At present this is 'ppm', 'cda', and 'exact'.");
60 }
61
62 Sigma_base_UKS::options sigma_opt;
63 sigma_opt.homo = opt_.homo_alpha;
64 sigma_opt.qpmax = opt_.qpmax;
65 sigma_opt.qpmin = opt_.qpmin;
66 sigma_opt.rpamin = opt_.rpamin;
67 sigma_opt.rpamax = opt_.rpamax;
68 sigma_opt.eta = opt_.eta;
69 sigma_opt.alpha = opt_.alpha;
70 sigma_opt.quadrature_scheme = opt_.quadrature_scheme;
71 sigma_opt.order = opt_.order;
72 sigma_alpha_->configure(sigma_opt);
73 sigma_opt.homo = opt_.homo_beta;
74 sigma_beta_->configure(sigma_opt);
75
76 Sigma_x_alpha_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
77 Sigma_x_beta_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
78 Sigma_c_alpha_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
79 Sigma_c_beta_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
80
81 auto print_spin_ranges = [&](Spin spin, Index homo) {
82 const Index lumo = homo + 1;
83
84 const Index n_occ_rpa =
85 (opt_.rpamin <= homo) ? (homo - opt_.rpamin + 1) : 0;
86 const Index n_virt_rpa =
87 (lumo <= opt_.rpamax) ? (opt_.rpamax - lumo + 1) : 0;
88
89 const Index qp_occ_min = opt_.qpmin;
90 const Index qp_occ_max = std::min(opt_.qpmax, homo);
91 const Index n_occ_qp =
92 (qp_occ_min <= qp_occ_max) ? (qp_occ_max - qp_occ_min + 1) : 0;
93
94 const Index qp_virt_min = std::max(opt_.qpmin, lumo);
95 const Index qp_virt_max = opt_.qpmax;
96 const Index n_virt_qp =
97 (qp_virt_min <= qp_virt_max) ? (qp_virt_max - qp_virt_min + 1) : 0;
98
99 XTP_LOG(Log::error, log_)
100 << TimeStamp() << " UKS " << SpinName(spin)
101 << " effective ranges: HOMO=" << homo << " LUMO=" << lumo
102 << " | RPA occ[" << opt_.rpamin << ":" << homo << "]"
103 << " virt[" << lumo << ":" << opt_.rpamax << "]"
104 << " (#occ=" << n_occ_rpa << ", #virt=" << n_virt_rpa << ")"
105 << " | GW occ[" << qp_occ_min << ":" << qp_occ_max << "]"
106 << " virt[" << qp_virt_min << ":" << qp_virt_max << "]"
107 << " (#occ=" << n_occ_qp << ", #virt=" << n_virt_qp << ")"
108 << std::flush;
109 };
110
111 print_spin_ranges(Spin::Alpha, opt_.homo_alpha);
112 print_spin_ranges(Spin::Beta, opt_.homo_beta);
113
114 if (opt_.sigma_integration == "ppm") {
115 auto* ppm_a = dynamic_cast<Sigma_PPM_UKS*>(sigma_alpha_.get());
116 auto* ppm_b = dynamic_cast<Sigma_PPM_UKS*>(sigma_beta_.get());
117
118 if (ppm_a == nullptr || ppm_b == nullptr) {
119 throw std::runtime_error(
120 "GW_UKS: expected Sigma_PPM_UKS evaluators for "
121 "sigma_integration=ppm");
122 }
123
124 ppm_a->SetSharedPPM(ppm_);
125 ppm_b->SetSharedPPM(ppm_);
126 }
127}

182double GW_UKS::CalcSpinHomoLumoShift(const Eigen::VectorXd& frequencies,
183 Spin spin) const {
184 const Index homo = Homo(spin);
185 const Eigen::VectorXd& dft = DftEnergies(spin);
186 const double DFTgap = dft(homo + 1) - dft(homo);
187 const double QPgap =
188 frequencies(homo + 1 - opt_.qpmin) - frequencies(homo - opt_.qpmin);
189 return QPgap - DFTgap;
190}

192void GW_UKS::CalculateGWPerturbation() {
193 Sigma_x_alpha_ = (1 - opt_.ScaHFX) * sigma_alpha_->CalcExchangeMatrix();
194 Sigma_x_beta_ = (1 - opt_.ScaHFX) * sigma_beta_->CalcExchangeMatrix();
195 if (core_exchange_alpha_.size() > 0) {
196 Sigma_x_alpha_ += (1 - opt_.ScaHFX) * core_exchange_alpha_;
197 Sigma_x_beta_ += (1 - opt_.ScaHFX) * core_exchange_beta_;
198 }
199 XTP_LOG(Log::error, log_)
200 << TimeStamp()
201 << " Calculated spin-resolved Hartree exchange contribution"
202 << std::flush;
203
204 Eigen::VectorXd dft_shifted_alpha =
205 ScissorShift_DFTlevel(dft_energies_alpha_, opt_.homo_alpha);
206 Eigen::VectorXd dft_shifted_beta =
207 ScissorShift_DFTlevel(dft_energies_beta_, opt_.homo_beta);
208
209 XTP_LOG(Log::error, log_)
210 << TimeStamp()
211 << " Scissor shifting alpha/beta DFT energies by: " << opt_.shift
212 << " Hrt" << std::flush;
213
214 rpa_.setRPAInputEnergies(
215 dft_shifted_alpha.segment(opt_.rpamin, opt_.rpamax - opt_.rpamin + 1),
216 dft_shifted_beta.segment(opt_.rpamin, opt_.rpamax - opt_.rpamin + 1));
217
218 Eigen::VectorXd frequencies_alpha =
219 dft_shifted_alpha.segment(opt_.qpmin, qptotal_);
220 Eigen::VectorXd frequencies_beta =
221 dft_shifted_beta.segment(opt_.qpmin, qptotal_);
222
223 Anderson mixing_alpha;
224 Anderson mixing_beta;
225 mixing_alpha.Configure(opt_.gw_mixing_order, opt_.gw_mixing_alpha);
226 mixing_beta.Configure(opt_.gw_mixing_order, opt_.gw_mixing_alpha);
227 // The QP energies of the last QP solve and the RPA energies it saw; see
228 // GW::CalculateGWPerturbation.
229 Eigen::VectorXd qp_solution_alpha = frequencies_alpha;
230 Eigen::VectorXd qp_solution_beta = frequencies_beta;
231 Eigen::VectorXd rpa_solved_alpha = rpa_.getRPAInputEnergiesAlpha();
232 Eigen::VectorXd rpa_solved_beta = rpa_.getRPAInputEnergiesBeta();
233
234 // alpha and beta RPA energies as one vector, for the screening update
235 auto rpa_both = [this]() {
236 const Eigen::VectorXd& a = rpa_.getRPAInputEnergiesAlpha();
237 const Eigen::VectorXd& b = rpa_.getRPAInputEnergiesBeta();
238 Eigen::VectorXd both(a.size() + b.size());
239 both << a, b;
240 return both;
241 };
242 ScreeningUpdate screening;
243 screening.Configure(ScreeningUpdate::Parse(opt_.screening_update),
244 opt_.screening_update_ratio,
245 opt_.screening_update_max_inner);
246 bool rebuild_screening = true;
247 Index mixing_iteration = 0; // see GW::CalculateGWPerturbation
248 Index n_iterations = 0;
249 // root tracking per spin, as in GW::CalculateGWPerturbation
250 tracked_root_[0].resize(0);
251 tracked_root_[1].resize(0);
252 tracking_active_ = false;
253 tracked_local_total_ = 0;
254 bool tracking_verified = false;
255 bool verifying = false; // this iteration is the verifying full search
256
257 for (Index i_gw = 0; i_gw < opt_.gw_sc_max_iterations; ++i_gw) {
258 gw_sc_iteration_ = i_gw;
259 tracking_active_ = opt_.qp_root_tracking && i_gw > 0 && !tracking_verified;
260 const Eigen::VectorXd roots_before_alpha = tracked_root_[0];
261 const Eigen::VectorXd roots_before_beta = tracked_root_[1];
262 if (opt_.reset_3c > 0 && i_gw % opt_.reset_3c == 0 && i_gw != 0) {
263 Mmn_.alpha.Rebuild();
264 Mmn_.beta.Rebuild();
265 XTP_LOG(Log::info, log_)
266 << TimeStamp() << " Rebuilding alpha/beta 3c integrals" << std::flush;
267 rebuild_screening = true;
268 }
269
270 if (rebuild_screening) {
271 if (i_gw == 0) {
272 XTP_LOG(Log::error, log_) << TimeStamp() << " Building the screening; "
273 << MemoryBudget::Usage() << std::flush;
274 }
275 if (opt_.sigma_integration == "ppm") {
276 ppm_.PPM_construct_parameters(rpa_);
277 if (ppm_.InvalidModes() > 0) {
278 XTP_LOG(Log::error, log_)
279 << TimeStamp() << " WARNING: " << ppm_.InvalidModes()
280 << " PPM modes with negative squared frequency (|.| used)"
281 << std::flush;
282 }
283 }
284 sigma_alpha_->PrepareScreening();
285 sigma_beta_->PrepareScreening();
286 screening.Built(rpa_both());
287 if (screening.mode() == ScreeningUpdate::Mode::Adaptive &&
288 screening.builds() > 1) {
289 mixing_alpha = Anderson();
290 mixing_beta = Anderson();
291 mixing_alpha.Configure(opt_.gw_mixing_order, opt_.gw_mixing_alpha);
292 mixing_beta.Configure(opt_.gw_mixing_order, opt_.gw_mixing_alpha);
293 mixing_iteration = 0;
294 }
295
296 const Log::Level screening_level =
297 (i_gw == 0) ? Log::Level::error : Log::Level::info;
298 XTP_LOG(screening_level, log_)
299 << TimeStamp() << " Calculated unrestricted screening via RPA; "
300 << MemoryBudget::Usage() << std::flush;
301 }
302
303 const bool mix = opt_.gw_mixing_order > 0 && mixing_iteration > 0;
304 if (mix) {
305 mixing_alpha.UpdateInput(frequencies_alpha);
306 mixing_beta.UpdateInput(frequencies_beta);
307 }
308
309 const Eigen::VectorXd qp_input_alpha = frequencies_alpha;
310 const Eigen::VectorXd qp_input_beta = frequencies_beta;
311 frequencies_alpha = SolveQP(Spin::Alpha, frequencies_alpha);
312 frequencies_beta = SolveQP(Spin::Beta, frequencies_beta);
313 qp_solution_alpha = frequencies_alpha;
314 qp_solution_beta = frequencies_beta;
315 rpa_solved_alpha = rpa_.getRPAInputEnergiesAlpha();
316 rpa_solved_beta = rpa_.getRPAInputEnergiesBeta();
317 ++mixing_iteration;
318 ++n_iterations;
319
320 if (opt_.gw_sc_max_iterations > 1) {
321 Eigen::VectorXd rpa_alpha_old = rpa_.getRPAInputEnergiesAlpha();
322 Eigen::VectorXd rpa_beta_old = rpa_.getRPAInputEnergiesBeta();
323
324 if (mix) {
325 mixing_alpha.UpdateOutput(frequencies_alpha);
326 mixing_beta.UpdateOutput(frequencies_beta);
327 frequencies_alpha = mixing_alpha.MixHistory();
328 frequencies_beta = mixing_beta.MixHistory();
329 }
330
331 rpa_.UpdateRPAInputEnergies(dft_energies_alpha_, dft_energies_beta_,
332 frequencies_alpha, frequencies_beta,
333 opt_.qpmin);
334
335 XTP_LOG(Log::info, log_)
336 << TimeStamp() << " GW_Iteration:" << i_gw << " Shift_alpha[Hrt]:"
337 << CalcSpinHomoLumoShift(frequencies_alpha, Spin::Alpha)
338 << " Shift_beta[Hrt]:"
339 << CalcSpinHomoLumoShift(frequencies_beta, Spin::Beta) << std::flush;
340
341 // log the largest changes per spin; the decision is ScreeningUpdate's
342 Converged(rpa_.getRPAInputEnergiesAlpha(), rpa_alpha_old,
343 opt_.gw_sc_limit);
344 Converged(rpa_.getRPAInputEnergiesBeta(), rpa_beta_old, opt_.gw_sc_limit);
345 const double residual =
346 std::max((qp_solution_alpha - qp_input_alpha).cwiseAbs().maxCoeff(),
347 (qp_solution_beta - qp_input_beta).cwiseAbs().maxCoeff());
348 XTP_LOG(Log::info, log_) << TimeStamp() << " QP residual (solution - "
349 << "input) max=" << residual << std::flush;
350 XTP_LOG(Log::info, log_)
351 << TimeStamp() << " Largest QP residuals alpha (level: input -> "
352 << "solution): "
353 << qp_solver::LargestResiduals(qp_solution_alpha, qp_input_alpha,
354 opt_.qpmin)
355 << std::flush;
356 XTP_LOG(Log::info, log_)
357 << TimeStamp() << " Largest QP residuals beta: "
358 << qp_solver::LargestResiduals(qp_solution_beta, qp_input_beta,
359 opt_.qpmin)
360 << std::flush;
361 double drift = 0.0;
362 const ScreeningUpdate::Next next =
363 screening.Decide(residual, rpa_both(), opt_.gw_sc_limit, &drift);
364 if (screening.mode() == ScreeningUpdate::Mode::Adaptive) {
365 XTP_LOG(Log::info, log_)
366 << TimeStamp() << " Change since the screening was built: " << drift
367 << " Hrt after " << screening.inner() << " iteration(s)"
368 << ((next == ScreeningUpdate::Next::Rebuild) ? "; rebuilding W"
369 : "")
370 << std::flush;
371 }
372 if (verifying && roots_before_alpha.size() == qptotal_ &&
373 roots_before_beta.size() == qptotal_) {
374 // levels the verifying full search put on another root
375 std::vector<Index> moved_alpha;
376 std::vector<Index> moved_beta;
377 for (Index l = 0; l < qptotal_; ++l) {
378 if (std::abs(tracked_root_[0](l) - roots_before_alpha(l)) >
379 10.0 * opt_.gw_sc_limit) {
380 moved_alpha.push_back(l + opt_.qpmin);
381 }
382 if (std::abs(tracked_root_[1](l) - roots_before_beta(l)) >
383 10.0 * opt_.gw_sc_limit) {
384 moved_beta.push_back(l + opt_.qpmin);
385 }
386 }
387 if ((!moved_alpha.empty() || !moved_beta.empty()) &&
388 next != ScreeningUpdate::Next::Converged) {
389 IndexParser rp;
390 XTP_LOG(Log::error, log_)
391 << TimeStamp() << " QP root tracking: the full search chose "
392 << "other roots for levels alpha ["
393 << rp.CreateIndexString(moved_alpha) << "] beta ["
394 << rp.CreateIndexString(moved_beta)
395 << "]; continuing with the full search" << std::flush;
396 }
397 verifying = false;
398 }
399 if (next == ScreeningUpdate::Next::Converged && opt_.qp_root_tracking &&
400 !tracking_verified && i_gw > 0) {
401 // converged on the tracked roots: one QP solve with the full search
402 // (same G and W) checks them; see GW::CalculateGWPerturbation
403 tracking_verified = true;
404 verifying = true;
405 rebuild_screening = false;
406 XTP_LOG(Log::info, log_)
407 << TimeStamp()
408 << " QP root tracking converged; verifying with a full search"
409 << std::flush;
410 if (i_gw < opt_.gw_sc_max_iterations - 1) {
411 continue;
412 }
413 }
414 if (next == ScreeningUpdate::Next::Converged) {
415 XTP_LOG(Log::info, log_)
416 << TimeStamp() << " Converged after " << i_gw + 1
417 << " unrestricted GW iterations." << std::flush;
418 break;
419 } else if (i_gw == opt_.gw_sc_max_iterations - 1) {
420 XTP_LOG(Log::error, log_)
421 << TimeStamp()
422 << " WARNING! UKS GW-self-consistency cycle not converged after "
423 << opt_.gw_sc_max_iterations << " iterations." << std::flush;
424 break;
425 }
426 rebuild_screening = (next == ScreeningUpdate::Next::Rebuild);
427 }
428 }
429 screening_builds_ = screening.builds();
430 qp_iterations_ = n_iterations;
431 tracking_active_ = false;
432 if (opt_.gw_sc_max_iterations > 1) {
433 XTP_LOG(Log::error, log_)
434 << TimeStamp() << " evGW (UKS): " << n_iterations << " QP iterations, "
435 << screening.builds() << " screening builds ("
436 << ScreeningUpdate::Name(screening.mode()) << ")"
437 << (opt_.qp_root_tracking
438 ? "; root tracking solved " +
439 std::to_string(tracked_local_total_) + " of " +
440 std::to_string(2 * n_iterations * qptotal_) +
441 " level solves locally"
442 : "")
443 << std::flush;
444 }
445
446 // Sigma_c of the last QP solve, at its solution and with the energies in
447 // G it saw; the RPA energies updated after the solve stay.
448 PrintCompetingRoots(Spin::Alpha, Log::error);
449 PrintCompetingRoots(Spin::Beta, Log::error);
450 const Eigen::VectorXd rpa_next_alpha = rpa_.getRPAInputEnergiesAlpha();
451 const Eigen::VectorXd rpa_next_beta = rpa_.getRPAInputEnergiesBeta();
452 rpa_.setRPAInputEnergies(rpa_solved_alpha, rpa_solved_beta);
453 Sigma_c_alpha_.diagonal() =
454 sigma_alpha_->CalcCorrelationDiag(qp_solution_alpha);
455 Sigma_c_beta_.diagonal() = sigma_beta_->CalcCorrelationDiag(qp_solution_beta);
456 rpa_.setRPAInputEnergies(rpa_next_alpha, rpa_next_beta);
457 PrintGWA_Energies(Spin::Alpha);
458 PrintGWA_Energies(Spin::Beta);
459}

478void GW_UKS::PrintCompetingRoots(Spin spin, Log::Level level) const {
479 const auto& list = competing_roots_[spin == Spin::Alpha ? 0 : 1];
480 Index n = 0;
481 for (const auto& c : list) {
482 n += c.competing ? 1 : 0;
483 }
484 if (n == 0) {
485 return;
486 }
487 XTP_LOG(level, log_) << TimeStamp() << " " << SpinName(spin) << ": " << n
488 << " QP level(s) with two roots of comparable weight "
489 "(second Z at least half the largest):"
490 << std::flush;
491 for (Index i = 0; i < Index(list.size()); ++i) {
492 const auto& c = list[std::size_t(i)];
493 if (c.competing) {
494 XTP_LOG(level, log_)
495 << (boost::format(" Level = %1$4d chosen %2$+1.6f (Z = %3$.3f), "
496 "other %4$+1.6f (Z = %5$.3f)") %
497 (i + opt_.qpmin) % c.omega % c.Z % c.omega_alt % c.Z_alt)
498 .str()
499 << std::flush;
500 }
501 }

504Eigen::VectorXd GW_UKS::SolveQP(Spin spin,
505 const Eigen::VectorXd& frequencies) const {
506 competing_roots_[spin == Spin::Alpha ? 0 : 1].assign(
507 std::size_t(qptotal_), qp_solver::CompetingRoots{});
508 const Eigen::VectorXd intercepts =
509 DftEnergies(spin).segment(opt_.qpmin, qptotal_) +
510 SigmaX(spin).diagonal() - Vxc(spin).diagonal();
511
512 Eigen::VectorXd frequencies_new = frequencies;
513
514 Index use_threads = qptotal_;
515#ifdef _OPENMP
516 use_threads =
517 OPENMP::getMaxThreads() > qptotal_ ? qptotal_ : OPENMP::getMaxThreads();
518#endif
519
520 const std::size_t si = (spin == Spin::Alpha) ? 0 : 1;
521 const bool track = opt_.qp_root_tracking;
522 const bool use_tracked =
523 track && tracking_active_ && tracked_root_[si].size() == qptotal_;
524 Eigen::VectorXd roots_new = frequencies;
525 Eigen::VectorXd Z_new = Eigen::VectorXd::Zero(qptotal_);
526 Index n_local = 0;
527
528#pragma omp parallel for schedule(dynamic) num_threads(use_threads)
529 for (Index gw_level = 0; gw_level < qptotal_; ++gw_level) {
530 const double initial_f = frequencies[gw_level];
531 const double intercept = intercepts[gw_level];
532
533 boost::optional<double> newf;
534 QPStats fixed_stats;
535 QPStats grid_stats;
536 QPStats lin_stats;
537 double Z_root = 0.0;
538
539 if (use_tracked) {
540 const double center = tracked_root_[si](gw_level);
541 const double moved =
542 std::max(tracked_change_[si](gw_level), std::abs(initial_f - center));
543 const double halfwidth = std::clamp(4.0 * moved, 0.01, 0.1);
544 QPFunc fqp(gw_level, SigmaEvaluator(spin), intercept);
545 qp_solver::SolverOptions solver_opt;
546 solver_opt.g_sc_limit = opt_.g_sc_limit;
547 solver_opt.qp_bisection_max_iter = opt_.g_sc_max_iterations;
548 solver_opt.qp_dense_spacing = opt_.qp_dense_spacing;
549 auto cand = qp_solver::SolveQP_Tracked(
550 fqp, center, halfwidth, tracked_Z_[si](gw_level), 0.5, solver_opt,
551 opt_.qp_root_finder == "brent");
552 fixed_stats = fqp.GetStats();
553 if (cand) {
554 newf = cand->omega;
555 Z_root = cand->Z;
556#pragma omp atomic
557 ++n_local;
558 }
559 }
560
561 if (!newf && opt_.qp_solver == "fixedpoint") {
562 newf = SolveQP_FixedPoint(spin, intercept, initial_f, gw_level,
563 &fixed_stats);
564 }
565
566 if (newf) {
567 frequencies_new[gw_level] = newf.value();
568 } else {
569 newf = SolveQP_Grid(spin, intercept, initial_f, gw_level, &grid_stats);
570 if (newf) {
571 frequencies_new[gw_level] = newf.value();
572 } else {
573 newf = SolveQP_Linearisation(spin, intercept, initial_f, gw_level,
574 &lin_stats);
575 if (newf) {
576 frequencies_new[gw_level] = newf.value();
577 }
578 }
579 }
580 if (track && newf) {
581 roots_new[gw_level] = newf.value();
582 if (Z_root <= 0.0) {
583 QPFunc fqp(gw_level, SigmaEvaluator(spin), intercept);
584 const double d = fqp.deriv(newf.value());
585 Z_root = (std::abs(d) > 1e-14) ? -1.0 / d : 0.0;
586 }
587 Z_new[gw_level] = Z_root;
588 }
589
590 if (Log::current_level > Log::error) {
591#pragma omp critical
592 {
593 QPStats total_stats;
594 total_stats.Add(fixed_stats);
595 total_stats.Add(grid_stats);
596 total_stats.Add(lin_stats);
597
598 const Index mo_level = opt_.qpmin + gw_level;
599
600 XTP_LOG(Log::info, log_)
601 << " QP stats " << LevelLabel(spin, mo_level) << " mo=" << mo_level
602 << " spin=" << SpinName(spin)
603 << " sigma_calls=" << total_stats.TotalSigmaCalls()
604 << " scan=" << total_stats.sigma_scan_calls
605 << " refine=" << total_stats.sigma_refine_calls
606 << " other=" << total_stats.sigma_other_calls
607 << " deriv_calls=" << total_stats.deriv_calls
608 << " unique_omega=" << total_stats.sigma_unique_frequencies
609 << " repeat_omega=" << total_stats.sigma_repeat_calls << std::flush;
610 }
611 }
612 }
613
614 if (track) {
615 tracked_change_[si] =
616 (tracked_root_[si].size() == qptotal_)
617 ? Eigen::VectorXd((roots_new - tracked_root_[si]).cwiseAbs())
618 : Eigen::VectorXd::Zero(qptotal_);
619 tracked_root_[si] = roots_new;
620 tracked_Z_[si] = Z_new;
621 tracked_local_total_ += n_local;
622 }
623 return frequencies_new;
624}

647boost::optional<double> GW_UKS::SolveQP_Grid(Spin spin, double intercept0,
648 double frequency0, Index gw_level,
649 QPStats* stats) const {
650 // The full QP search window is now controlled explicitly and no longer
651 // derived from the dense-grid spacing.
652 const double range = opt_.qp_full_window_half_width;
653
654 const double full_left_limit = frequency0 - range;
655 const double full_right_limit = frequency0 + range;
656
657 double restricted_left_limit = full_left_limit;
658 double restricted_right_limit = full_right_limit;
659
660 bool use_restricted_window = false;
661
662 if (opt_.qp_restrict_search) {
663 const Index mo_level = gw_level + opt_.qpmin;
664 const bool is_occupied = (mo_level <= Homo(spin));
665
666 if (is_occupied) {
667 restricted_right_limit = std::min(full_right_limit, -opt_.qp_zero_margin);
668 } else {
669 restricted_left_limit =
670 std::max(full_left_limit, opt_.qp_virtual_min_energy);
671 }
672
673 const double tol = 1e-12;
674 use_restricted_window =
675 (std::abs(restricted_left_limit - full_left_limit) > tol) ||
676 (std::abs(restricted_right_limit - full_right_limit) > tol);
677 }
678
679 if (use_restricted_window && restricted_left_limit < restricted_right_limit) {
680 auto restricted =
681 SolveQP_Grid_Windowed(spin, intercept0, frequency0, gw_level,
682 restricted_left_limit, restricted_right_limit,
683 false, // accepted roots only
684 stats);
685
686 if (restricted) {
687 return restricted;
688 }
689
690 if (Log::current_level > Log::error) {
691#pragma omp critical
692 {
693 XTP_LOG(Log::info, log_)
694 << " Restricted QP search failed for "
695 << LevelLabel(spin, opt_.qpmin + gw_level) << " in window ["
696 << restricted_left_limit << ", " << restricted_right_limit
697 << "], forcing full dense window [" << full_left_limit << ", "
698 << full_right_limit << "]" << std::flush;
699 }
700 }
701
702 return SolveQP_Grid_Windowed_Dense(spin, intercept0, frequency0, gw_level,
703 full_left_limit, full_right_limit, true,
704 stats);
705 }
706
707 return SolveQP_Grid_Windowed(spin, intercept0, frequency0, gw_level,
708 full_left_limit, full_right_limit, true, stats);
709}

711boost::optional<double> GW_UKS::SolveQP_Grid_Windowed_Adaptive(
712 Spin spin, double intercept0, double frequency0, Index gw_level,
713 double left_limit, double right_limit, bool allow_rejected_return,
714 QPStats* stats) const {
715
716 QPFunc fqp(gw_level, SigmaEvaluator(spin), intercept0);
717
718 qp_solver::SolverOptions solver_opt;
719
720 // Pass only canonical search controls into the shared grid-search utility.
721 // UKS uses the same adaptive/dense logic as RKS; only the spin-resolved
722 // sigma evaluation changes.
723 solver_opt.g_sc_limit = opt_.g_sc_limit;
724 solver_opt.qp_bisection_max_iter = opt_.g_sc_max_iterations;
725 solver_opt.qp_full_window_half_width = opt_.qp_full_window_half_width;
726 solver_opt.qp_dense_spacing = opt_.qp_dense_spacing;
727 solver_opt.qp_adaptive_shell_width = opt_.qp_adaptive_shell_width;
728 solver_opt.qp_adaptive_shell_count = opt_.qp_adaptive_shell_count;
729 solver_opt.prefer_nearest_root =
730 opt_.qp_root_continuity && gw_sc_iteration_ > 0;
731 QPWindowDiagnostics wdiag;
732 std::vector<QPRootCandidate> accepted_roots;
733 std::vector<QPRootCandidate> rejected_roots;
734
735 bool use_brent = false;
736 if (opt_.qp_root_finder == "brent") {
737 use_brent = true;
738 }
739
740 auto result = qp_solver::SolveQP_Grid_Windowed(
741 fqp, frequency0, left_limit, right_limit, gw_sc_iteration_, solver_opt,
742 &wdiag, &accepted_roots, &rejected_roots, use_brent);
743
744 if (Log::current_level > Log::error) {
745#pragma omp critical
746 {
747 if (accepted_roots.empty() && rejected_roots.empty()) {
748 XTP_LOG(Log::info, log_)
749 << " No roots found for " << LevelLabel(spin, opt_.qpmin + gw_level)
750 << " (center="
751 << std::max(left_limit, std::min(right_limit, frequency0))
752 << ", shells=" << wdiag.shells_explored << ")" << std::flush;
753 } else {
754 XTP_LOG(Log::info, log_)
755 << " Adaptive scan " << LevelLabel(spin, opt_.qpmin + gw_level)
756 << " shells=" << wdiag.shells_explored
757 << " first_interval_shell=" << wdiag.first_interval_shell
758 << " first_accepted_shell=" << wdiag.first_accepted_shell
759 << " intervals=" << wdiag.intervals_found << std::flush;
760 }
761 }
762 }
763
764 if (stats != nullptr) {
765 *stats = fqp.GetStats();
766 }
767
768 if (!accepted_roots.empty()) {
769 if (result) {
770 RecordCompetingRoots(spin, gw_level, accepted_roots, result.value());
771 }
772 return result;
773 }
774
775 if (!rejected_roots.empty() && !allow_rejected_return) {
776 if (Log::current_level > Log::error) {
777#pragma omp critical
778 {
779 XTP_LOG(Log::info, log_)
780 << " Adaptive scan " << LevelLabel(spin, opt_.qpmin + gw_level)
781 << " produced only rejected roots in [" << left_limit << ", "
782 << right_limit << "], forcing wider retry" << std::flush;
783 }
784 }
785 return boost::none;
786 }
787
788 return result;
789}

791boost::optional<double> GW_UKS::SolveQP_Grid_Windowed_Dense(
792 Spin spin, double intercept0, double frequency0, Index gw_level,
793 double left_limit, double right_limit, bool allow_rejected_return,
794 QPStats* stats) const {
795
796 QPFunc fqp(gw_level, SigmaEvaluator(spin), intercept0);
797
798 qp_solver::SolverOptions solver_opt;
799
800 // The dense path performs a uniform sign-change scan over the requested
801 // interval. Its mesh is controlled by qp_dense_spacing and is independent
802 // from the adaptive shell spacing.
803 solver_opt.g_sc_limit = opt_.g_sc_limit;
804 solver_opt.qp_bisection_max_iter = opt_.g_sc_max_iterations;
805 solver_opt.qp_full_window_half_width = opt_.qp_full_window_half_width;
806 solver_opt.qp_dense_spacing = opt_.qp_dense_spacing;
807 solver_opt.qp_adaptive_shell_width = opt_.qp_adaptive_shell_width;
808 solver_opt.qp_adaptive_shell_count = opt_.qp_adaptive_shell_count;
809 solver_opt.prefer_nearest_root =
810 opt_.qp_root_continuity && gw_sc_iteration_ > 0;
811
812 const bool use_brent = (opt_.qp_root_finder == "brent");
813
814 std::vector<QPRootCandidate> accepted_roots;
815 std::vector<QPRootCandidate> rejected_roots;
816 std::vector<std::pair<double, double>> roots; // omega : Z
817
818 if (left_limit < right_limit) {
819 double freq_prev = left_limit;
820 double targ_prev = fqp.value(freq_prev, EvalStage::Scan);
821
822 // Dense scanning is the robustness path: it uniformly samples the entire
823 // interval to avoid missing sign changes that a shell-based scan may not
824 // hit directly.
825 const Index n_steps = std::max<Index>(
826 2, static_cast<Index>(
827 std::ceil((right_limit - left_limit) / opt_.qp_dense_spacing)) +
828 1);
829
830 for (Index i_node = 1; i_node < n_steps; ++i_node) {
831 const double freq =
832 (i_node == n_steps - 1)
833 ? right_limit
834 : std::min(right_limit, left_limit + static_cast<double>(i_node) *
835 opt_.qp_dense_spacing);
836
837 const double targ = fqp.value(freq, EvalStage::Scan);
838
839 if (targ_prev * targ < 0.0) {
840 auto cand =
841 qp_solver::RefineQPInterval(freq_prev, targ_prev, freq, targ, fqp,
842 frequency0, solver_opt, use_brent);
843 if (cand) {
844 roots.emplace_back(cand->omega, cand->Z);
845 if (cand->accepted) {
846 accepted_roots.push_back(*cand);
847 } else {
848 rejected_roots.push_back(*cand);
849 }
850 }
851 }
852
853 freq_prev = freq;
854 targ_prev = targ;
855 }
856 }
857
858 if (Log::current_level > Log::error) {
859#pragma omp critical
860 {
861 if (accepted_roots.empty() && rejected_roots.empty()) {
862 XTP_LOG(Log::info, log_)
863 << " Dense scan " << LevelLabel(spin, opt_.qpmin + gw_level)
864 << " found no roots in [" << left_limit << ", " << right_limit
865 << "]" << std::flush;
866 } else {
867 XTP_LOG(Log::info, log_)
868 << " Dense scan " << LevelLabel(spin, opt_.qpmin + gw_level)
869 << " roots (omega:Z)\n\t\t";
870 for (const auto& root : roots) {
871 XTP_LOG(Log::info, log_) << std::setprecision(5) << root.first << ":"
872 << root.second << " ";
873 }
874 XTP_LOG(Log::info, log_) << std::flush;
875 }
876 }
877 }
878
879 if (stats != nullptr) {
880 *stats = fqp.GetStats();
881 }
882
883 if (!accepted_roots.empty()) {
884 const double best = qp_solver::SelectRoot(accepted_roots, solver_opt).omega;
885 RecordCompetingRoots(spin, gw_level, accepted_roots, best);
886 return best;
887 }
888
889 if (!rejected_roots.empty()) {
890 if (!allow_rejected_return) {
891 if (Log::current_level > Log::error) {
892#pragma omp critical
893 {
894 XTP_LOG(Log::info, log_)
895 << " Dense scan " << LevelLabel(spin, opt_.qpmin + gw_level)
896 << " produced only rejected roots in [" << left_limit << ", "
897 << right_limit << "], forcing wider retry" << std::flush;
898 }
899 }
900 return boost::none;
901 }
902 auto least_bad = std::max_element(
903 rejected_roots.begin(), rejected_roots.end(),
904 [](const QPRootCandidate& a, const QPRootCandidate& b) {
905 return qp_solver::ScoreRoot(a) < qp_solver::ScoreRoot(b);
906 });
907 return least_bad->omega;
908 }
909
910 return boost::none;
911}

913boost::optional<double> GW_UKS::SolveQP_Grid_Windowed(
914 Spin spin, double intercept0, double frequency0, Index gw_level,
915 double left_limit, double right_limit, bool allow_rejected_return,
916 QPStats* stats) const {
917
918 if (opt_.qp_grid_search_mode == "adaptive") {
919 return SolveQP_Grid_Windowed_Adaptive(spin, intercept0, frequency0,
920 gw_level, left_limit, right_limit,
921 allow_rejected_return, stats);
922 }
923
924 if (opt_.qp_grid_search_mode == "dense") {
925 return SolveQP_Grid_Windowed_Dense(spin, intercept0, frequency0, gw_level,
926 left_limit, right_limit,
927 allow_rejected_return, stats);
928 }
929
930 if (opt_.qp_grid_search_mode == "adaptive_with_dense_fallback") {
931 QPStats total_stats;
932 auto adaptive = SolveQP_Grid_Windowed_Adaptive(
933 spin, intercept0, frequency0, gw_level, left_limit, right_limit,
934 allow_rejected_return, &total_stats);
935 if (adaptive) {
936 if (stats != nullptr) {
937 *stats = total_stats;
938 }
939 return adaptive;
940 }
941
942 QPStats dense_stats;
943 auto dense = SolveQP_Grid_Windowed_Dense(
944 spin, intercept0, frequency0, gw_level, left_limit, right_limit,
945 allow_rejected_return, &dense_stats);
946 total_stats.Add(dense_stats);

949#pragma omp critical

951 XTP_LOG(Log::info, log_)

953 << LevelLabel(spin, opt_.qpmin + gw_level)
954 << ", retrying dense grid scan in [" << left_limit << ", "
955 << right_limit << "]" << std::flush;
956 }
957 }
958
959 if (stats != nullptr) {
960 *stats = total_stats;
961 }
962 return dense;
963 }
964
965 throw std::runtime_error("Unknown gw.qp_grid_search_mode '" +
966 opt_.qp_grid_search_mode + "'");
967}
