119class EwaldPeriodicDipoleOperator
120 : public Eigen::EigenBase<EwaldPeriodicDipoleOperator> {
121 public:
122 // Required typedefs, constants, and method:
123 using Scalar = double;
124 using RealScalar = double;
125 using StorageIndex = votca::Index;
126 enum {
127 ColsAtCompileTime = Eigen::Dynamic,
128 MaxColsAtCompileTime = Eigen::Dynamic,
129 IsRowMajor = false
130 };
131
132 // ids: the segments solved for; their order defines the vector layout
133 // (see class documentation). Every id must be registered at
134 // EwaldChargeState::Neutral.
135 // alpha_ewald: Ewald splitting parameter, the same as used for
136 // real_sum/recip_sum; needed for the intramolecular interactor.
137 // thole_a: Thole damping parameter, the same as used for real_sum
138 // (polarmethod.aDamp in ewdbgpol.xml); used by
139 // AddIntraSegmentCoupling.
140 // shape: the EwaldShapeCorrection also used for the permanent field.
141 // Applied to the induced dipoles in every product, as legacy does
142 // with FU12_ShapeField_At_By in each induction iteration.
143 EwaldPeriodicDipoleOperator(EwaldRegistry& registry,
144 const EwaldRealSpaceSum& real_sum,
145 const EwaldReciprocalSpaceSum& recip_sum,
146 const EwaldShapeCorrection& shape,
147 std::vector<Index> ids, double alpha_ewald,
148 double thole_a);
149
150 // Accumulated wall-clock time (seconds) spent in RawMultiply, split by
151 // phase, over the lifetime of this object. n_calls is the number of
152 // RawMultiply calls (one per solver iteration plus one for baseline_).
153 struct RawMultiplyTimings {
154 double setup = 0.0; // dipole set + Reset over every site
155 double real_space = 0.0; // EwaldRealSpaceSum::AddFieldAt
156 double reciprocal = 0.0; // EwaldReciprocalSpaceSum::AddFieldAtMany
157 double shape = 0.0; // EwaldShapeCorrection::AddFieldAt
158 double assemble = 0.0; // P^-1*v - V - M*v result assembly
159 double intra = 0.0; // AddIntraSegmentCoupling
160 Index n_calls = 0;
161 double total() const {
162 return setup + real_space + reciprocal + shape + assemble + intra;
163 }
164 };
165 const RawMultiplyTimings& Timings() const { return timings_; }
166
167 class InnerIterator {
168 public:
169 InnerIterator(const EwaldPeriodicDipoleOperator& xpr, const Index& id)
170 : xpr_(xpr), id_(id) {};
171
172 InnerIterator& operator++() {
173 row_++;
174 return *this;
175 }
176 operator bool() const { return row_ < xpr_.size_; }
177 double value() const { return xpr_(row_, id_); }
178 Index row() const { return row_; }
179 Index col() const { return id_; }
180 Index index() const { return row(); }
181
182 private:
183 const EwaldPeriodicDipoleOperator& xpr_;
184 const Index id_;
185 Index row_ = 0;
186 };
187
188 Index rows() const { return size_; }
189 Index cols() const { return size_; }
190 Index outerSize() const { return size_; }
191
192 template <typename Vtype>
193 Eigen::Product<EwaldPeriodicDipoleOperator, Vtype, Eigen::AliasFreeProduct>
194 operator*(const Eigen::MatrixBase<Vtype>& x) const {
195 return Eigen::Product<EwaldPeriodicDipoleOperator, Vtype,
196 Eigen::AliasFreeProduct>(*this, x.derived());
197 }
198
199 // Same-site entries return the getPInv() block; cross-site entries
200 // return 0.0 (only the diagonal is used by the preconditioner).
201 double operator()(Index i, Index j) const;
202
203 // Matrix-vector product A*v (see class documentation).
204 Eigen::VectorXd multiply(const Eigen::VectorXd& v) const;
205
206 private:
207 // Computes getPInv()*v minus the field at every ids_ site from all
208 // registered segments (including those not in ids_). Affine rather
209 // than linear in v; multiply() subtracts baseline_ = RawMultiply(0).
210 Eigen::VectorXd RawMultiply(const Eigen::VectorXd& v) const;
211
212 // Adds the intramolecular Thole-damped induced-dipole field (the part
213 // not already contained in the reciprocal-space sum) to result, for
214 // every segment in ids_ with more than one site. Linear in v.
215 void AddIntraSegmentCoupling(const Eigen::VectorXd& v,
216 Eigen::VectorXd& result) const;
217
218 // Maps a global vector index to (segment index in ids_, site index in
219 // that segment) by binary search on offsets_. Used by operator()(i,j).
220 std::pair<Index, Index> LocateSite(Index i) const;
221
222 EwaldRegistry& registry_;
223 const EwaldRealSpaceSum& real_sum_;
224 const EwaldReciprocalSpaceSum& recip_sum_;
225 const EwaldShapeCorrection& shape_;
226 std::vector<Index> ids_;
227 // Provides ComputeB/ComputeErfB and ComputeThole for the
228 // intramolecular term.
229 EwaldRealSpaceInteractor intra_interactor_;
230 // offsets_[n] is the scalar index where segment ids_[n] starts;
231 // offsets_[n+1]-offsets_[n] = 3 * (number of sites in that segment).
232 // offsets_.size() == ids_.size()+1, with offsets_.back() == size_.
233 std::vector<Index> offsets_;
234 Index size_;
235 Eigen::VectorXd baseline_;
236 mutable RawMultiplyTimings timings_;
237 // 3x3 matrix M such that a dipole v_i produces a spurious
238 // reciprocal-space self-field -M*v_i at its own position (see
239 // EwaldReciprocalSpaceSum::SelfFieldMatrix). Depends only on the
240 // k-vectors, alpha and volume. RawMultiply cancels this self-field for
241 // every site, as legacy does via EwdInteractor::FU12_ERF_At_By with
242 // the same site as both arguments.
243 Eigen::Matrix3d self_field_matrix_;
244};
