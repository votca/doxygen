42// Projection Operator Diabatization, POD2 variant (Ghan et al.), which
43// avoids the global Lowdin orthogonalization of the original POD that
44// makes results basis-set dependent (comparison of POD, FODFT and CDFT:
45// arXiv:1512.00200, "Critical analysis of fragment-orbital DFT schemes
46// for the calculation of electronic coupling values").
47//
48// Input is a single converged, unconstrained ground-state DFT calculation
49// of the intact supermolecule; unlike FODFT, covalent bonds across the
50// fragment boundary are therefore allowed. The donor and acceptor AO
51// blocks of the Fock matrix are diagonalized separately in the original
52// AO basis, giving fragment-localized diabatic orbitals; the coupling is
53// the off-diagonal Fock block transformed into these orbitals.
54class PODCoupling {
55 public:
56 PODCoupling(Orbitals& orbitals, Logger* log,
57 std::vector<Index> fragment_A_atoms,
58 std::vector<Index> fragment_B_atoms);
59
60 // Computes the couplings between all pairs of orbitals in the ranges
61 // of fragments A and B, occupied and virtual, with the conventions of
62 // DFTcoupling. numberofstates = N selects {HOMO-N+1, ..., HOMO, LUMO,
63 // ..., LUMO+N-1} (as DFTcoupling::DetermineRangeOfStates); N=1 gives
64 // {HOMO, LUMO}.
65 void CalculateCouplings(Index numberofstatesA, Index numberofstatesB);
66
67 // Coupling [Hartree] between orbitals levelA of fragment A and levelB
68 // of fragment B. Levels are absolute indices into the fragment's
69 // eigendecomposition (e.g. getFragmentAHomoIndex()), not relative to
70 // the computed range, as in DFTcoupling::getCouplingElement. Throws if
71 // outside the range of the last CalculateCouplings call.
72 double getCouplingElement(Index levelA, Index levelB) const;
73
74 // HOMO/LUMO index of each fragment in the index space of
75 // getCouplingElement. Available after construction, since they depend
76 // only on the fragment's nuclear charge (CountFragmentElectrons in
77 // podcoupling.cc).
78 Index getFragmentAHomoIndex() const { return nocc_A_ - 1; }
79 Index getFragmentALumoIndex() const { return nocc_A_; }
80 Index getFragmentBHomoIndex() const { return nocc_B_ - 1; }
81 Index getFragmentBLumoIndex() const { return nocc_B_; }
82
83 // Orbital level (absolute index, as in getCouplingElement) of fragment
84 // A (fragment_A true) or B, embedded in the full-molecule AO basis with
85 // zero coefficients outside the fragment, e.g. for cube files. Throws
86 // if outside the range of the last CalculateCouplings call.
87 Eigen::VectorXd GetFragmentOrbital(bool fragment_A, Index level) const;
88
89 // Lists the top_n AOs with the largest |Mulliken population| in a
90 // fragment orbital, with atom index, element and shell type. Useful to
91 // identify which AO of the fragment causes density in the region of the
92 // other fragment (diffuse tails; coefficients on the other fragment's
93 // AOs are zero by construction).
94 //
95 // Ranked by Mulliken population c_i*(S*c)_i rather than |c_i|: in a
96 // non-orthogonal basis, strongly overlapping AOs (e.g. several
97 // contracted s shells on nearby atoms) can have large, mutually
98 // cancelling coefficients that contribute little to the orbital.
99 std::string DescribeFragmentOrbitalComposition(bool fragment_A, Index level,
100 Index top_n = 5) const;
101
102 private:
103 Orbitals& orbitals_;
104 Logger* pLog_;
105 std::vector<Index> fragment_A_atoms_;
106 std::vector<Index> fragment_B_atoms_;
107
108 // Estimated number of occupied orbitals per fragment, set in the
109 // constructor (see CountFragmentElectrons in podcoupling.cc).
110 Index nocc_A_ = 0;
111 Index nocc_B_ = 0;
112
113 // (start, size) of the computed orbital range, in absolute fragment
114 // orbital indices; set by CalculateCouplings. As in DFTcoupling.
115 std::pair<Index, Index> Range_orbA_;
116 std::pair<Index, Index> Range_orbB_;
117
118 // Coupling matrix [Hartree], set by CalculateCouplings: JAB_(i,j)
119 // couples orbital Range_orbA_.first + i of A with Range_orbB_.first + j
120 // of B. Unlike DFTcoupling's JAB, only the A-B block is stored.
121 Eigen::MatrixXd JAB_;
122
123 // Set by CalculateCouplings for GetFragmentOrbital: fragment
124 // eigenvectors (all orbitals), fragment AO indices in the full basis,
125 // and the full AO count.
126 Eigen::MatrixXd fragment_A_eigenvectors_;
127 Eigen::MatrixXd fragment_B_eigenvectors_;
128 std::vector<Index> ao_indices_A_;
129 std::vector<Index> ao_indices_B_;
130 Index nao_full_ = 0;
131
132 // Fragment AO overlap blocks, kept for the Mulliken populations in
133 // DescribeFragmentOrbitalComposition.
134 Eigen::MatrixXd S_AA_;
135 Eigen::MatrixXd S_BB_;
136};
