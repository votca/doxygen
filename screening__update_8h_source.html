56class ScreeningUpdate {
57 public:
58 enum class Mode { Every, Adaptive };
59 enum class Next { Continue, Rebuild, Converged };
60
61 static Mode Parse(const std::string& name) {
62 if (name == "every") {
63 return Mode::Every;
64 } else if (name == "adaptive") {
65 return Mode::Adaptive;
66 }
67 throw std::runtime_error("Unknown gw.screening_update '" + name +
68 "'; choices are every, adaptive");
69 }
70 static std::string Name(Mode mode) {
71 return (mode == Mode::Adaptive) ? "adaptive" : "every";
72 }
73
74 void Configure(Mode mode, double ratio, Index max_inner) {
75 mode_ = mode;
76 ratio_ = ratio;
77 max_inner_ = max_inner;
78 }
79 Mode mode() const { return mode_; }
80
82 void Built(const Eigen::VectorXd& energies) {
83 built_from_ = energies;
84 inner_ = 0;
85 ++builds_;
86 }
87 Index builds() const { return builds_; }
89 Index inner() const { return inner_; }
90
95 Next Decide(double residual, const Eigen::VectorXd& new_energies,
96 double limit, double* drift_out = nullptr) {
97 ++inner_;
98 const double step = residual;
99 const double drift = (new_energies - built_from_).cwiseAbs().maxCoeff();
100 if (drift_out != nullptr) {
101 *drift_out = drift;
102 }
103 if (mode_ == Mode::Every) {
104 return (step < limit) ? Next::Converged : Next::Rebuild;
105 }
106 if (step < limit) {
107 return (drift < limit) ? Next::Converged : Next::Rebuild;
108 }
109 if (step < ratio_ * drift || inner_ >= max_inner_) {
110 return Next::Rebuild;
111 }
112 return Next::Continue;
113 }
114
115 private:
116 Mode mode_ = Mode::Every;
117 double ratio_ = 0.25;
118 Index max_inner_ = 10;
119 Eigen::VectorXd built_from_;
120 Index inner_ = 0;
121 Index builds_ = 0;
122};
