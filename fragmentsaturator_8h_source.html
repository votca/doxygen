43class FragmentSaturator {
44 public:
45 // Generic C-H bond length [Angstrom], independent of hybridization.
46 // Only the starting point for RelaxNewAtoms.
47 static constexpr double kDefaultCHBondLengthAngstrom = 1.09;
48
49 // new_atom_parent_ids has size mol.size() and is indexed by atom id:
50 // -1 for original atoms, and for each added H the id of the original
51 // atom it saturates. Callers use it to assign added H atoms to the
52 // fragment of their parent (e.g. IPodCoupling::EvalJob); the saturator
53 // itself has no notion of fragments.
54 struct SaturationResult {
55 QMMolecule mol;
56 std::vector<Index> new_atom_parent_ids;
57 };
58
59 // Returns a copy of mol with one H atom appended for every atom with
60 // hasExternalBond(), placed at bond_length_angstrom along its external
61 // bond direction. Original atoms keep their indices, so atom indices
62 // computed before saturation remain valid.
63 static SaturationResult SaturateExternalBonds(
64 const QMMolecule& mol,
65 double bond_length_angstrom = kDefaultCHBondLengthAngstrom);
66 // Returns a copy of mol after a constrained MMFF94 relaxation (UFF if
67 // MMFF94 setup fails): atoms with index < n_original_atoms are fixed,
68 // the appended H atoms move. Requires the bond connectivity
69 // (QMAtom::getBondedPartnerIds()), from which OpenBabel's
70 // PerceiveBondOrders() derives bond orders.
71 //
72 // n_steps is the maximum number of conjugate-gradient steps; the
73 // relaxation stops earlier once the energy has converged.
74 static QMMolecule RelaxNewAtoms(const QMMolecule& mol, Index n_original_atoms,
75 Index n_steps = 500);
76};
