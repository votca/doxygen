49double DavidsonToleranceValue(const std::string& tol) {
50 if (tol == "loose") {
51 return 1e-3;
52 } else if (tol == "normal") {
53 return 1e-4;
54 } else if (tol == "strict") {
55 return 1e-5;
56 } else if (tol == "lapack") {
57 return 1e-9;
58 }
59 throw std::runtime_error(tol + " is not a valid Davidson tolerance");
60}

64 const Eigen::MatrixXd& Hqp_in) {

77 }
78 SetupDirectInteractionOperator(RPAInputEnergies, 0.0);

168Eigen::MatrixXd BSE::AdjustHqpSize(const Eigen::MatrixXd& Hqp,
169 const Eigen::VectorXd& RPAInputEnergies) {
170
171 Index hqp_size = bse_vtotal_ + bse_ctotal_;
172 Index gwsize = opt_.qpmax - opt_.qpmin + 1;
173 Index RPAoffset = opt_.vmin - opt_.rpamin;
174 Eigen::MatrixXd Hqp_BSE = Eigen::MatrixXd::Zero(hqp_size, hqp_size);
175
176 if (opt_.vmin >= opt_.qpmin) {
177 Index start = opt_.vmin - opt_.qpmin;
178 if (opt_.cmax <= opt_.qpmax) {
179 Hqp_BSE = Hqp.block(start, start, hqp_size, hqp_size);
180 } else {
181 Index virtoffset = gwsize - start;
182 Hqp_BSE.topLeftCorner(virtoffset, virtoffset) =
183 Hqp.block(start, start, virtoffset, virtoffset);
184
185 Index virt_extra = opt_.cmax - opt_.qpmax;
186 Hqp_BSE.diagonal().tail(virt_extra) =
187 RPAInputEnergies.segment(RPAoffset + virtoffset, virt_extra);
188 }
189 }
190
191 if (opt_.vmin < opt_.qpmin) {
192 Index occ_extra = opt_.qpmin - opt_.vmin;
193 Hqp_BSE.diagonal().head(occ_extra) =
194 RPAInputEnergies.segment(RPAoffset, occ_extra);
195
196 Hqp_BSE.block(occ_extra, occ_extra, gwsize, gwsize) = Hqp;
197
198 if (opt_.cmax > opt_.qpmax) {
199 Index virtoffset = occ_extra + gwsize;
200 Index virt_extra = opt_.cmax - opt_.qpmax;
201 Hqp_BSE.diagonal().tail(virt_extra) =
202 RPAInputEnergies.segment(RPAoffset + virtoffset, virt_extra);
203 }
204 }
205
206 return Hqp_BSE;
207}

225void BSE::MatchDressingToRoute() {
226 const bool environment = reaction_field_.size() > 0;
227 const bool dressed_route = environment && include_kreac_;
228 // Dressed: every pairing of two M's is through u = v + v_reac, so eps
229 // below is the environment-screened one and Kx carries K_reac. Bare:
230 // Kx stays v, and W_tot is assembled explicitly below. GW may have left
231 // the integrals either way.
232 if (dressed_route && !Mmn_.Dressed()) {
233 Mmn_.DressAuxIndex(dressing_);
234 } else if (!dressed_route && Mmn_.Dressed()) {
235 Mmn_.UndressAuxIndex();
236 }
237}

239// W = vectors diag(values) vectors^T in the current frame of the
240// integrals, as the operators use it: the inverse of epsilon without its
241// non-positive modes, or W_tot of the bare environment route.
242SymmetricEigenSystem BSE::ScreenedInteraction(
243 const Eigen::VectorXd& RPAInputEnergies, double energy,
244 DFTTimings& timings) const {
245 RPA rpa = RPA(log_, Mmn_);
246 rpa.configure(opt_.homo, opt_.rpamin, opt_.rpamax);
247 rpa.setRPAInputEnergies(RPAInputEnergies);
248 rpa.setTimings(&timings);
249 const Eigen::MatrixXd screening = rpa.calculate_epsilon_r(energy);
250
251 auto t = timings.Measure("eigensolver");
252 if (reaction_field_.size() > 0 && !include_kreac_) {
253 // Operators expect M U with U orthogonal and diag(epsilon_0_inv_) the
254 // screened interaction in that frame: Kd = M U d U^T M^T and
255 // Kx = M U U^T M^T = M M^T = v. So diagonalize W_tot, not eps, and
256 // hand its eigenvalues over as epsilon_0_inv_.
257 const Index n = screening.rows();
258 const Eigen::MatrixXd I = Eigen::MatrixXd::Identity(n, n);
259 // 1 + R and u^-1 + eps - 1 are positive definite (Cholesky, LU if not)
260 const Eigen::MatrixXd u_inv =
261 InverseSPD(I + Mmn_.ToCurrentAuxFrame(reaction_field_));
262 Eigen::MatrixXd W = InverseSPD(u_inv + screening - I);
263 W = 0.5 * (W + W.transpose()).eval();
264 return SymmetricEigen(W);
265 }
266
267 SymmetricEigenSystem es = SymmetricEigen(screening);
268 for (Index i = 0; i < es.values.size(); ++i) {
269 es.values(i) = (es.values(i) > 1e-8) ? 1 / es.values(i) : 0.0;
270 }
271 return es;
272}

274void BSE::SetupDirectInteractionOperator(
275 const Eigen::VectorXd& RPAInputEnergies, double energy) {
276 ReleaseDirectCaches(); // the integrals and the screening change
277 MatchDressingToRoute();
278 DFTTimings timings;
279 SymmetricEigenSystem W =
280 ScreenedInteraction(RPAInputEnergies, energy, timings);
281 {
282 auto t = timings.Measure("rotation");
283 RotateIntoScreeningFrame(W.vectors);
284 }
285 epsilon_0_inv_ = std::move(W.values);
286 XTP_LOG(Log::error, log_) << TimeStamp() << " BSE screening at " << energy
287 << " Ha: " << timings.Format() << flush;
288}

305void BSE::DecideDirectCaches(bool coupling) const {
306 const Index n = bse_size_;
307 const double need = SymmetricTiles::Bytes(n);
308 // what is kept but not built yet does not show in the resident memory
309 double pending = 0.0;
310 for (const auto* c : {&direct_caches_.hd, &direct_caches_.hd2}) {
311 if (*c != nullptr && !(*c)->built()) {
312 pending += need;
313 }
314 }
315 double available = MemoryBudget::AvailableBytes() - pending;
316 std::ostringstream msg;
317 msg.precision(3);
318 msg << " BSE direct terms (lower triangle, " << need * 1e-9 << " GB each):";
319 auto decide = [&](std::shared_ptr<SymmetricTiles>& cache, bool& decided,
320 const char* name) {
321 if (decided) {
322 msg << " " << name << (cache ? " kept (reused)" : " not kept") << ";";
323 return;
324 }
325 decided = true;
326 if (need <= available) {
327 cache = std::make_shared<SymmetricTiles>();
328 available -= need;
329 msg << " " << name << " kept;";
330 } else {
331 msg << " " << name << " not kept;";
332 }
333 };
334 decide(direct_caches_.hd, direct_caches_.hd_decided, "Hd");
335 if (coupling) {
336 decide(direct_caches_.hd2, direct_caches_.hd2_decided, "Hd2");
337 }
338 msg << " " << MemoryBudget::Describe();
339 const Index missing = Index(direct_caches_.hd == nullptr) +
340 Index(coupling && direct_caches_.hd2 == nullptr);
341 if (missing > 0) {
342 // what the process holds now plus the matrices, with the margin
343 const double resident = std::max(0.0, MemoryBudget::ResidentBytes());
344 // the needed terms not built yet, and one kept but unbuilt otherwise
345 auto unbuilt = [](const std::shared_ptr<SymmetricTiles>& c) {
346 return c == nullptr || !c->built();
347 };
348 double extra = unbuilt(direct_caches_.hd) ? need : 0.0;
349 if (coupling ? unbuilt(direct_caches_.hd2)
350 : (direct_caches_.hd2 && !direct_caches_.hd2->built())) {
351 extra += need;
352 }
353 const double wanted = resident + extra;
354 msg << "; --memory " << std::ceil(1.1 * wanted * 1e-9 + 1.0)
355 << " or more would keep all";
356 }
357 XTP_LOG(Log::error, log_) << TimeStamp() << msg.str() << flush;
358}

397Eigen::VectorXd BSE::ReactionFieldExchange(const tools::EigenSystem& es,
398 bool tda) const {
399 // The transition density of each state in the auxiliary basis,
400 // d = sum_vc (X+Y)_vc M_v,c, and <K_reac> = 2 d^T R d: the singlet
401 // spin factor of Kx, which K_reac shares, times the first-order change
402 // (X+Y)^T dK (X+Y) that a perturbation entering A and B alike produces.
403 const Eigen::MatrixXd Rc = Mmn_.ToCurrentAuxFrame(reaction_field_);
404 const Index vmin = opt_.vmin - opt_.rpamin;
405 const Index cmin = bse_cmin_ - opt_.rpamin;
406 const Index nstates = es.eigenvalues().size();
407 Eigen::VectorXd result = Eigen::VectorXd::Zero(nstates);
408#pragma omp parallel for schedule(dynamic)
409 for (Index s = 0; s < nstates; ++s) {
410 Eigen::VectorXd coeffs = es.eigenvectors().col(s);
411 if (!tda) {
412 coeffs += es.eigenvectors2().col(s);
413 }
414 const Eigen::Map<const Eigen::MatrixXd> mat(coeffs.data(), bse_ctotal_,
415 bse_vtotal_);
416 Eigen::VectorXd d = Eigen::VectorXd::Zero(Mmn_.auxsize());
417 for (Index v = 0; v < bse_vtotal_; ++v) {
418 d +=
419 Mmn_[v + vmin].middleRows(cmin, bse_ctotal_).transpose() * mat.col(v);
420 }
421 result(s) = 2.0 * d.dot(Rc * d);
422 }
423 return result;
424}

459template <typename BSE_OPERATOR>
460tools::EigenSystem BSE::solve_hermitian(BSE_OPERATOR& h) const {
461
462 std::chrono::time_point<std::chrono::system_clock> start =
463 std::chrono::system_clock::now();
464
465 tools::EigenSystem result;
466
467 DavidsonSolver DS(log_);
468
469 DS.set_correction(opt_.davidson_correction);
470 DS.set_tolerance(opt_.davidson_tolerance);
471 DS.set_size_update(opt_.davidson_update);
472 DS.set_iter_max(opt_.davidson_maxiter);
473 DS.set_max_search_space(10 * opt_.nmax);
474 DS.solve(h, opt_.nmax);
475 result.eigenvalues() = DS.eigenvalues();
476 result.eigenvectors() = DS.eigenvectors();
477
478 std::chrono::time_point<std::chrono::system_clock> end =
479 std::chrono::system_clock::now();
480 std::chrono::duration<double> elapsed_time = end - start;
481
482 XTP_LOG(Log::info, log_) << TimeStamp() << " Diagonalization done in "
483 << elapsed_time.count() << " secs" << flush;
484
485 return result;
486}

510template <typename BSE_OPERATOR_A, typename BSE_OPERATOR_B>
511tools::EigenSystem BSE::Solve_nonhermitian_Davidson(BSE_OPERATOR_A& Aop,
512 BSE_OPERATOR_B& Bop) const {
513 // Hermitian (A-B)(A+B) form: one application of A and of B per new
514 // vector. Falls back to the general non-Hermitian Davidson if A-B or A+B
515 // is not positive definite.
516 try {
517 FullBSEDavidson::Options fopt;
518 fopt.tolerance = DavidsonToleranceValue(opt_.davidson_tolerance);
519 fopt.max_iterations = opt_.davidson_maxiter;
520 fopt.max_subspace = 10 * opt_.nmax;
521 FullBSEDavidson solver(log_, fopt);
522 return solver.Solve(Aop, Bop, opt_.nmax);
523 } catch (const std::runtime_error& error) {
524 XTP_LOG(Log::error, log_)
525 << TimeStamp() << " " << error.what()
526 << "; using the non-Hermitian Davidson solver" << flush;
527 }
528
529 std::chrono::time_point<std::chrono::system_clock> start =
530 std::chrono::system_clock::now();
531
532 // operator
533 HamiltonianOperator<BSE_OPERATOR_A, BSE_OPERATOR_B> Hop(Aop, Bop);
534
535 // Davidson solver
536 DavidsonSolver DS(log_);
537 DS.set_correction(opt_.davidson_correction);
538 DS.set_tolerance(opt_.davidson_tolerance);
539 DS.set_size_update(opt_.davidson_update);
540 DS.set_iter_max(opt_.davidson_maxiter);
541 DS.set_max_search_space(10 * opt_.nmax);
542 DS.set_matrix_type("HAM");
543 Eigen::MatrixXd initial_guess = BuildFullBSEXRankedInitialGuess(
544 Aop.diagonal(), Bop.diagonal(), opt_.nmax);
545
546 DS.solve(Hop, opt_.nmax, initial_guess);
547
548 // results
549 tools::EigenSystem result;
550 result.eigenvalues() = DS.eigenvalues();
551 Eigen::MatrixXd tmpX = DS.eigenvectors().topRows(Aop.rows());
552 Eigen::MatrixXd tmpY = DS.eigenvectors().bottomRows(Bop.rows());
553
554 // normalization so that X^T X - Y^T Y = 1
555 if (const Index bad = NormalizeExcitationVectors(tmpX, tmpY); bad > 0) {
556 XTP_LOG(Log::error, log_)
557 << TimeStamp() << " WARNING: " << bad
558 << " BSE root(s) without positive X^2 - Y^2 norm (not converged?)"
559 << flush;
560 }
561 result.eigenvectors() = tmpX;
562 result.eigenvectors2() = tmpY;
563
564 std::chrono::time_point<std::chrono::system_clock> end =
565 std::chrono::system_clock::now();
566 std::chrono::duration<double> elapsed_time = end - start;
567
568 XTP_LOG(Log::info, log_) << TimeStamp() << " Diagonalization done in "
569 << elapsed_time.count() << " secs" << flush;
570
571 return result;
572}

574void BSE::printFragInfo(const std::vector<QMFragment<BSE_Population>>& frags,
575 Index state) const {
576 for (const QMFragment<BSE_Population>& frag : frags) {
577 double dq = frag.value().H[state] + frag.value().E[state];
578 double qeff = dq + frag.value().Gs;
579 XTP_LOG(Log::error, log_)
580 << boost::format(
581 " Fragment %1$4d -- hole: %2$5.1f%% electron: "
582 "%3$5.1f%% dQ: %4$+5.2f Qeff: %5$+5.2f") %

717template <typename BSE_OPERATOR>
718BSE::ExpectationValues BSE::ExpectationValue_Operator(
719 const QMStateType& type, const Orbitals& orb, const BSE_OPERATOR& H) const {
720
721 const tools::EigenSystem& BSECoefs = GetBSEEigenSystem(type, orb);
722
723 ExpectationValues expectation_values;
724
725 const Eigen::MatrixXd temp = H * BSECoefs.eigenvectors();
726
727 expectation_values.direct_term = ExpValue(BSECoefs.eigenvectors(), temp);
728 if (!orb.getTDAApprox()) {
729 expectation_values.direct_term +=
730 ExpValue(BSECoefs.eigenvectors2(), H * BSECoefs.eigenvectors2());
731 expectation_values.cross_term =
732 2 * ExpValue(BSECoefs.eigenvectors2(), temp);
733 } else {
734 expectation_values.cross_term = Eigen::VectorXd::Zero(0);
735 }
736 return expectation_values;
737}

739// Composition of the excitation energy in terms of QP, direct (screened),
740// and exchance contributions in the BSE
741// Full BSE:
742//
743// | A* | | H K | | A |
744// | -B* | | -K -H | | B | = A*.H.A + B*.H.B + 2A*.K.B
745//
746// with: H = H_qp + H_d + eta.H_x
747// K = H_d2 + eta.H_x
748//
749// reports composition for FULL BSE as
750// <FT> = A*.H_qp.A + B*.H_qp.B
751// <Kx> = eta.(A*.H_x.A + B*.H_x.B + 2A*.H_x.B)
752// <Kd> = A*.H_d.A + B*.H_d.B + 2A*.H_d2.B
753namespace {
754// A kept direct term for the analysis only if it is built already: building
755// it for the few states of the analysis would cost more than it saves.
756std::shared_ptr<SymmetricTiles> ReusableCache(
757 const std::shared_ptr<SymmetricTiles>& cache) {
758 return (cache != nullptr && cache->built()) ? cache : nullptr;
759}

762BSE::Interaction BSE::Analyze_eh_interaction(const QMStateType& type,
763 const Orbitals& orb) const {
764 Interaction analysis;
765 {
766 HqpOperator hqp(epsilon_0_inv_, Mmn_, Hqp_);
767 configureBSEOperator(hqp);
768 ExpectationValues expectation_values =
769 ExpectationValue_Operator(type, orb, hqp);
770 analysis.qp_contrib = expectation_values.direct_term;
771 }
772 {
773 // the direct terms of the solve, if kept, are not rebuilt here
774 HdOperator hd(epsilon_0_inv_, Mmn_, Hqp_);
775 configureBSEOperator(hd, ReusableCache(direct_caches_.hd));
776 ExpectationValues expectation_values =
777 ExpectationValue_Operator(type, orb, hd);
778 analysis.direct_contrib = expectation_values.direct_term;
779 }
780 if (!orb.getTDAApprox()) {
781 // only the cross term 2 Y.Hd2.X: Hd2 on X alone
782 Hd2Operator hd2(epsilon_0_inv_, Mmn_, Hqp_);
783 configureBSEOperator(hd2, ReusableCache(direct_caches_.hd2));
784 const tools::EigenSystem& coefs = GetBSEEigenSystem(type, orb);
785 analysis.direct_contrib +=
786 2 * ExpValue(coefs.eigenvectors2(), hd2 * coefs.eigenvectors());
787 }
788
789 double xpref = ExchangePrefactor(type);
790 if (xpref != 0.0) {
791 HxOperator hx(epsilon_0_inv_, Mmn_, Hqp_);
792 configureBSEOperator(hx);
793 ExpectationValues expectation_values =
794 ExpectationValue_Operator(type, orb, hx);
795 analysis.exchange_contrib = xpref * expectation_values.direct_term;
796 if (!orb.getTDAApprox()) {
797 analysis.exchange_contrib += xpref * expectation_values.cross_term;
798 }
799 } else {
800 analysis.exchange_contrib =
801 Eigen::VectorXd::Zero(analysis.direct_contrib.size());
802 }
803
804 return analysis;
805}

813// Number of hole levels per pass, so that the stacked intermediates of one
814// pass stay within the given bytes.
815Index HoleChunk(Index vtotal, Index ctotal, Index auxsize, double bytes) {
816 const double per_level =
817 8.0 * double(auxsize) * double(ctotal + 2 * std::max(vtotal, ctotal));
818 return std::clamp<Index>(Index(bytes / per_level), 1, vtotal);
819}

821// sum_PQ W_PQ G_PQ for W = vectors diag(values) vectors^T
822double ContractScreened(const SymmetricEigenSystem& W,
823 const Eigen::MatrixXd& G) {
824 const Eigen::MatrixXd GPhi = G * W.vectors;
825 return (W.vectors.cwiseProduct(GPhi).colwise().sum().transpose().array() *
826 W.values.array())
827 .sum();
828}
829} // namespace

831// G with <A|Hd|B> = -sum_PQ W_PQ G_PQ, summed over the pairs (A, B), for the
832// screened direct term Hd(v1c1, v2c2) = -M[c1](c2,:) W M[v1](v2,:)^T:
833// G = sum_{v1 v2} b_{v1v2} a_{v1v2}^T, a_{v1v2} = M[v1](v2,:),
834// b_{v1v2} = sum_{c1 c2} A(v1,c1) B(v2,c2) M[c1](c2,:).
835Eigen::MatrixXd BSE::DirectTermDensity(
836 const std::vector<std::pair<const Eigen::VectorXd*,
837 const Eigen::VectorXd*>>& pairs) const {
838 const Index vmin = opt_.vmin - opt_.rpamin;
839 const Index cmin = bse_cmin_ - opt_.rpamin;
840 const Index vt = bse_vtotal_;
841 const Index ct = bse_ctotal_;
842 const Index N = Mmn_.auxsize();
843 const Index chunk = HoleChunk(vt, ct, N, state_matrix_bytes_);
844 Eigen::MatrixXd G = Eigen::MatrixXd::Zero(N, N);
845 for (Index v0 = 0; v0 < vt; v0 += chunk) {
846 const Index nc = std::min(chunk, vt - v0);
847 // rows j * vt + v1: b_{v1, v0 + j}
848 Eigen::MatrixXd b = Eigen::MatrixXd::Zero(nc * vt, N);
849 for (const auto& pair : pairs) {
850 // coefficient vectors as (c, v) matrices, index v * ct + c
851 Eigen::Map<const Eigen::MatrixXd> A(pair.first->data(), ct, vt);
852 Eigen::Map<const Eigen::MatrixXd> B(pair.second->data(), ct, vt);
853 const Eigen::MatrixXd Bchunk = B.middleCols(v0, nc).transpose();
854 // T[j](c1, :) = sum_c2 B(v0 + j, c2) M[c1](c2, :)
855 std::vector<RowMajorMatrix> T(std::size_t(nc), RowMajorMatrix(ct, N));
856#pragma omp parallel for schedule(dynamic)
857 for (Index c1 = 0; c1 < ct; ++c1) {
858 const Eigen::MatrixXd t = Bchunk * Mmn_[c1 + cmin].middleRows(cmin, ct);
859 for (Index j = 0; j < nc; ++j) {
860 T[std::size_t(j)].row(c1) = t.row(j);
861 }
862 }
863#pragma omp parallel for schedule(dynamic)
864 for (Index j = 0; j < nc; ++j) {
865 b.middleRows(j * vt, vt).noalias() += A.transpose() * T[std::size_t(j)];
866 }
867 }
868 Eigen::MatrixXd a(nc * vt, N);
869#pragma omp parallel for schedule(dynamic)
870 for (Index v1 = 0; v1 < vt; ++v1) {
871 for (Index j = 0; j < nc; ++j) {
872 a.row(j * vt + v1) = Mmn_[v1 + vmin].row(vmin + v0 + j);
873 }
874 }
875 G.noalias() += b.transpose() * a;
876 }
877 return G;
878}
879
880// G with <A|Hd2|B> = -sum_PQ W_PQ G_PQ for the screened coupling term
881// Hd2(v1c1, v2c2) = -M[v1](c2,:) W M[c1](v2,:)^T:
882// G = sum_{c1 v2} b_{c1v2} a_{c1v2}^T, a_{c1v2} = M[c1](v2,:),
883// b_{c1v2} = sum_{v1 c2} A(v1,c1) B(v2,c2) M[v1](c2,:).
884Eigen::MatrixXd BSE::CouplingTermDensity(const Eigen::VectorXd& Avec,
885 const Eigen::VectorXd& Bvec) const {
886 const Index vmin = opt_.vmin - opt_.rpamin;
887 const Index cmin = bse_cmin_ - opt_.rpamin;
888 const Index vt = bse_vtotal_;
889 const Index ct = bse_ctotal_;
890 const Index N = Mmn_.auxsize();
891 const Index chunk = HoleChunk(vt, ct, N, state_matrix_bytes_);
892 Eigen::Map<const Eigen::MatrixXd> A(Avec.data(), ct, vt);
893 Eigen::Map<const Eigen::MatrixXd> B(Bvec.data(), ct, vt);
894 Eigen::MatrixXd G = Eigen::MatrixXd::Zero(N, N);
895 for (Index v0 = 0; v0 < vt; v0 += chunk) {
896 const Index nc = std::min(chunk, vt - v0);
897 const Eigen::MatrixXd Bchunk = B.middleCols(v0, nc).transpose();
898 // U[j](v1, :) = sum_c2 B(v0 + j, c2) M[v1](c2, :)
899 std::vector<RowMajorMatrix> U(std::size_t(nc), RowMajorMatrix(vt, N));
900#pragma omp parallel for schedule(dynamic)
901 for (Index v1 = 0; v1 < vt; ++v1) {
902 const Eigen::MatrixXd u = Bchunk * Mmn_[v1 + vmin].middleRows(cmin, ct);
903 for (Index j = 0; j < nc; ++j) {
904 U[std::size_t(j)].row(v1) = u.row(j);
905 }
906 }
907 // rows j * ct + c1: b_{c1, v0 + j} and a_{c1, v0 + j}
908 Eigen::MatrixXd b(nc * ct, N);
909#pragma omp parallel for schedule(dynamic)
910 for (Index j = 0; j < nc; ++j) {
911 b.middleRows(j * ct, ct).noalias() = A * U[std::size_t(j)];
912 }
913 Eigen::MatrixXd a(nc * ct, N);
914#pragma omp parallel for schedule(dynamic)
915 for (Index c1 = 0; c1 < ct; ++c1) {
916 for (Index j = 0; j < nc; ++j) {
917 a.row(j * ct + c1) = Mmn_[c1 + cmin].row(vmin + v0 + j);
918 }
919 }
920 G.noalias() += b.transpose() * a;
921 }
922 return G;
923}

925// Dynamical Screening in BSE as perturbation to static excitation energies
926// as in Phys. Rev. B 80, 241405 (2009) for the TDA case
927//
928// The integrals stay in the frame of the static screening set up in
929// configure, where W(0) = diag(epsilon_0_inv_). For each state, the
930// screened terms <X|Hd|X> + <Y|Hd|Y> + 2<Y|Hd2|X> are -sum_PQ W_PQ G_PQ with
931// state matrices G built once, so W(w) at the next frequency costs one
932// dielectric matrix and its eigenvectors: no pass over the BSE Hamiltonian
933// and no rotation of the integrals per state and step.
934void BSE::Perturbative_DynamicalScreening(const QMStateType& type,
935 Orbitals& orb) {
936 // no BSE operators from here on: free the kept direct terms
937 ReleaseDirectCaches();
938
939 const tools::EigenSystem& BSECoefs = GetBSEEigenSystem(type, orb);
940 const Eigen::VectorXd& RPAInputEnergies = orb.RPAInputEnergies();
941 const bool full = !orb.getTDAApprox();
942
943 const Eigen::VectorXd& BSEenergies = BSECoefs.eigenvalues();
944
945 // initial copy of static BSE energies to dynamic
946 Eigen::VectorXd BSEenergies_dynamic = BSEenergies;
947
948 for (Index i_exc = 0; i_exc < BSEenergies.size(); i_exc++) {
949 XTP_LOG(Log::info, log_) << "Dynamical Screening BSE, Excitation " << i_exc
950 << " static " << BSEenergies(i_exc) << flush;
951
952 const Eigen::VectorXd X = BSECoefs.eigenvectors().col(i_exc);
953 Eigen::VectorXd Y;
954 std::vector<std::pair<const Eigen::VectorXd*, const Eigen::VectorXd*>>
955 pairs{{&X, &X}};
956 if (full) {
957 Y = BSECoefs.eigenvectors2().col(i_exc);
958 pairs.emplace_back(&Y, &Y);
959 }
960 const Eigen::MatrixXd Gd = DirectTermDensity(pairs);
961 Eigen::MatrixXd G2;
962 if (full) {
963 G2 = CouplingTermDensity(Y, X);
964 }
965 // screened direct contribution for W(w), or for the static W(0)
966 auto Hd = [&](const SymmetricEigenSystem* W) {
967 double value = (W != nullptr) ? ContractScreened(*W, Gd)
968 : epsilon_0_inv_.dot(Gd.diagonal());
969 if (full) {
970 value += 2 * ((W != nullptr) ? ContractScreened(*W, G2)
971 : epsilon_0_inv_.dot(G2.diagonal()));
972 }
973 return -value;
974 };
975 const double Hd_static_contribution = Hd(nullptr);
976
977 for (Index iter = 0; iter < max_dyn_iter_; iter++) {
978
979 // screened interaction at the last energy as screening frequency
980 double old_energy = BSEenergies_dynamic(i_exc);
981 DFTTimings timings;
982 const SymmetricEigenSystem W =
983 ScreenedInteraction(RPAInputEnergies, old_energy, timings);
984
985 // new energy perturbatively
986 BSEenergies_dynamic(i_exc) =
987 BSEenergies(i_exc) + Hd_static_contribution - Hd(&W);
988
989 XTP_LOG(Log::info, log_)
990 << "Dynamical Screening BSE, excitation " << i_exc << " iteration "
991 << iter << " dynamic " << BSEenergies_dynamic(i_exc) << flush;
992
993 // check tolerance
994 if (std::abs(BSEenergies_dynamic(i_exc) - old_energy) < dyn_tolerance_) {
995 break;
996 }
997 }
998 }
999
1000 double hrt2ev = tools::conv::hrt2ev;
1001
1002 if (type == QMStateType::Singlet) {
1003 orb.BSESinglets_dynamic() = BSEenergies_dynamic;
1004 XTP_LOG(Log::error, log_) << " ====== singlet energies with perturbative "
1005 "dynamical screening (eV) ====== "
1006 << flush;
1007 Eigen::VectorXd oscs = orb.Oscillatorstrengths();
1008 for (Index i = 0; i < opt_.nmax; ++i) {
1009 double osc = oscs[i];
1010 XTP_LOG(Log::error, log_)
1011 << boost::format(
1012 " S(dynamic) = %1$4d Omega = %2$+1.12f eV lamdba = %3$+3.2f "
1013 "nm f = %4$+1.4f") %
1014 (i + 1) % (hrt2ev * BSEenergies_dynamic(i)) %
1015 (1240.0 / (hrt2ev * BSEenergies_dynamic(i))) %
1016 (osc * BSEenergies_dynamic(i) / BSEenergies(i))
1017 << flush;
1018 }
1019
1020 } else if (type == QMStateType::Triplet) {
1021 orb.BSETriplets_dynamic() = BSEenergies_dynamic;
1022 XTP_LOG(Log::error, log_) << " ====== triplet energies with perturbative "
1023 "dynamical screening (eV) ====== "
1024 << flush;
1025 for (Index i = 0; i < opt_.nmax; ++i) {
1026 XTP_LOG(Log::error, log_)
1027 << boost::format(
1028 " T(dynamic) = %1$4d Omega = %2$+1.12f eV lamdba = %3$+3.2f "
1029 "nm ") %
1030 (i + 1) % (hrt2ev * BSEenergies_dynamic(i)) %
1031 (1240.0 / (hrt2ev * BSEenergies_dynamic(i)))
1032 << flush;
1033 }
1034
1035 } else {
1036 throw std::runtime_error(
1037 "Unsupported QMStateType in BSE::Perturbative_DynamicalScreening");
1038 }
1039}
