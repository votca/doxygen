38InactiveSpace InactiveScreening::BuildSpace(const Orbitals& full, Index ncore) {
39 const Eigen::MatrixXd& DB = full.getInactiveDensity();
40 if (DB.size() == 0) {
41 throw std::runtime_error(
42 "InactiveScreening: no inactive density; run the projection-based "
43 "embedding first");
44 }
45 const Index nocc = full.getNumberOfAlphaElectrons();
46 const Eigen::MatrixXd& C = full.MOs().eigenvectors();
47 const Eigen::VectorXd& e = full.MOs().eigenvalues();
48 if (DB.rows() != C.rows()) {
49 throw std::runtime_error(
50 "InactiveScreening: inactive density and MOs have different AO "
51 "dimensions (truncated basis is not supported)");
52 }
53
54 AOOverlap overlap;
55 overlap.Fill(full.getDftBasis());
56 const Eigen::MatrixXd SCocc = overlap.Matrix() * C.leftCols(nocc);
57
58 // Projector of the inactive density onto the occupied space; its
59 // eigenvalues are 1 for the inactive and 0 for the active orbitals.
60 const Eigen::MatrixXd Q = 0.5 * SCocc.transpose() * DB * SCocc;
61 const SymmetricEigenSystem q = SymmetricEigen(Q);
62 const Index nB = (q.values.array() > 0.5).count();
63 const Eigen::MatrixXd UB = q.vectors.rightCols(nB);
64
65 InactiveSpace space;
66 space.idempotency_error = (Q * Q - Q).cwiseAbs().maxCoeff();
67
68 // semicanonical: diagonalize the Fock operator within the inactive space
69 const Eigen::MatrixXd FB = UB.transpose() * e.head(nocc).asDiagonal() * UB;
70 const SymmetricEigenSystem f = SymmetricEigen(FB);
71 if (ncore < 0 || ncore > nB) {
72 throw std::runtime_error("InactiveScreening: " + std::to_string(ncore) +
73 " core levels requested, but the inactive "
74 "region has only " +
75 std::to_string(nB) + " occupied orbitals");
76 }
77 const Index nkeep = nB - ncore;
78 space.occupied = C.leftCols(nocc) * UB * f.vectors.rightCols(nkeep);
79 space.occupied_energies = f.values.tail(nkeep);
80 const Index nvirt = C.cols() - nocc;
81 space.virtuals = C.rightCols(nvirt);
82 space.virtual_energies = e.tail(nvirt);
83 return space;
84}

86std::vector<Eigen::MatrixXd> InactiveScreening::Responses(
87 const AOBasis& auxbasis, const AOBasis& dftbasis,
88 const Eigen::MatrixXd& occupied, const Eigen::VectorXd& occupied_energies,
89 const Eigen::MatrixXd& virtuals, const Eigen::VectorXd& virtual_energies,
90 double chunk_bytes, const std::vector<double>& imaginary_frequencies,
91 const Eigen::MatrixXd& inv_sqrt) {
92 const Index naux = auxbasis.AOBasisSize();
93 const Index nocc = occupied.cols();
94 const Index nvirt = virtuals.cols();
95 std::vector<Eigen::MatrixXd> X(imaginary_frequencies.size(),
96 Eigen::MatrixXd::Zero(naux, naux));
97 if (nocc == 0 || nvirt == 0) {
98 return X;
99 }
100 const double bytes_per_level = 8.0 * double(nvirt) * double(naux);
101 const Index chunk = std::clamp<Index>(
102 static_cast<Index>(chunk_bytes / bytes_per_level), 1, nocc);
103
104 // the metric of the auxiliary basis is the same for every chunk
105 Eigen::MatrixXd metric = inv_sqrt;
106 for (Index first = 0; first < nocc; first += chunk) {
107 const Index nc = std::min(chunk, nocc - first);
108 Eigen::MatrixXd orbitals(occupied.rows(), nc + nvirt);
109 orbitals << occupied.middleCols(first, nc), virtuals;
110 // one integral pass for all frequencies
111 TCMatrix_gwbse Mmn;
112 Mmn.Initialize(naux, 0, nc - 1, nc, nc + nvirt - 1);
113 if (metric.size() == 0) {
114 Mmn.Fill(auxbasis, dftbasis, orbitals);
115 metric = Mmn.InvSqrt();
116 } else {
117 Mmn.Fill(auxbasis, dftbasis, orbitals, metric);
118 }
119 for (std::size_t f = 0; f < imaginary_frequencies.size(); ++f) {
120 const double w2 = imaginary_frequencies[f] * imaginary_frequencies[f];
121 X[f] += WeightedGram::Compute(
122 nc, naux,
123 [&](Index m, Eigen::VectorXd& w) {
124 const Eigen::ArrayXd d =
125 virtual_energies.array() - occupied_energies(first + m);
126 w = 4.0 * d / (d.square() + w2);
127 },
128 [&](Index m, const WeightedGram::RowsVisitor& use) { use(Mmn[m]); });
129 }
130 }
131 return X;
132}

168InactiveSpace InactiveScreening::NearSpace(const Orbitals& full, double radius,
169 Index& n_far) {
170 const Eigen::MatrixXd& L = full.getLMOs();
171 const Index nocc = full.getNumberOfAlphaElectrons();
172 if (L.cols() != nocc || L.rows() != full.MOs().eigenvectors().rows()) {
173 throw std::runtime_error(
174 "inactive_screening_radius needs the localized occupied orbitals of "
175 "the full calculation (localize step)");
176 }
177 AOOverlap overlap;
178 overlap.Fill(full.getDftBasis());
179 const Eigen::MatrixXd& S = overlap.Matrix();
180 const Eigen::MatrixXd SL = S * L;
181 // inactive LMOs: weight ~1 in the inactive density
182 const Eigen::VectorXd q =
183 0.5 * (SL.transpose() * full.getInactiveDensity() * SL).diagonal();
184 // charge centroids from Mulliken atom populations
185 const QMMolecule& atoms = full.QMAtoms();
186 const Eigen::MatrixXd pop = ActiveRegion::AtomPopulations(full, L);
187 Eigen::MatrixXd R(3, atoms.size());
188 for (Index a = 0; a < atoms.size(); ++a) {
189 R.col(a) = atoms[a].getPos();
190 }
191 const Eigen::MatrixXd centroid =
192 R * pop * pop.colwise().sum().cwiseInverse().asDiagonal();
193 std::vector<Index> active;
194 std::vector<Index> inactive;
195 for (Index k = 0; k < nocc; ++k) {
196 (q(k) > 0.5 ? inactive : active).push_back(k);
197 }
198 std::vector<Index> near;
199 for (Index k : inactive) {
200 double dmin = std::numeric_limits<double>::max();
201 for (Index j : active) {
202 dmin = std::min(dmin, (centroid.col(k) - centroid.col(j)).norm());
203 }
204 if (dmin <= radius) {
205 near.push_back(k);
206 }
207 }
208 n_far = Index(inactive.size() - near.size());
209 Eigen::MatrixXd Ln(L.rows(), Index(near.size()));
210 for (Index i = 0; i < Index(near.size()); ++i) {
211 Ln.col(i) = L.col(near[std::size_t(i)]);
212 }
213 // semicanonical among the near ones: F = S C diag(e) C^T S
214 const Index n = full.MOs().eigenvectors().cols();
215 const Eigen::MatrixXd SC = S * full.MOs().eigenvectors();
216 const Eigen::MatrixXd CL = SC.leftCols(nocc).transpose() * Ln; // MO basis
217 const Eigen::MatrixXd FL =
218 CL.transpose() * full.MOs().eigenvalues().head(nocc).asDiagonal() * CL;
219 const SymmetricEigenSystem f = SymmetricEigen(0.5 * (FL + FL.transpose()));
220 InactiveSpace space;
221 space.occupied = Ln * f.vectors;
222 space.occupied_energies = f.values;
223 space.virtuals = full.MOs().eigenvectors().rightCols(n - nocc);
224 space.virtual_energies = full.MOs().eigenvalues().tail(n - nocc);
225 return space;
226}

228InactiveScreeningInput InactiveScreening::Prepare(
229 const Orbitals& full, InactiveScreeningInput::Mode mode, bool complete_g,
230 double radius) {
231 InactiveScreeningInput input;
232 input.mode = mode;
233 input.complete_g = complete_g && mode != InactiveScreeningInput::Mode::None;
234 if (mode == InactiveScreeningInput::Mode::None) {
235 return input;
236 }
237 if (full.getCalculationType() == "Truncated") {
238 throw std::runtime_error(
239 "inactive_screening is not implemented for truncated active "
240 "regions (dft_in_dft.truncate_basis).");
241 }
242 // the inactive space is built in both modes: its size and idempotency
243 // are the checks that the full MOs and the inactive density belong
244 // together
245 InactiveSpace space = BuildSpace(full);
246 input.n_inactive = space.size();
247 input.idempotency_error = space.idempotency_error;
248 if (input.complete_g) {
249 input.inactive_occupied = space.occupied;
250 input.inactive_energies = space.occupied_energies;
251 }
252 if (mode == InactiveScreeningInput::Mode::Block && radius > 0.0) {
253 InactiveSpace near = NearSpace(full, radius, input.n_far);
254 input.radius = radius;
255 input.occupied = std::move(near.occupied);
256 input.occupied_energies = std::move(near.occupied_energies);
257 } else if (mode == InactiveScreeningInput::Mode::Block) {
258 input.occupied = std::move(space.occupied);
259 input.occupied_energies = std::move(space.occupied_energies);
260 } else {
261 if (radius > 0.0) {
262 throw std::runtime_error(
263 "inactive_screening_radius is only available for "
264 "inactive_screening block");
265 }
266 const Index nocc = full.getNumberOfAlphaElectrons();
267 input.occupied = full.MOs().eigenvectors().leftCols(nocc);
268 input.occupied_energies = full.MOs().eigenvalues().head(nocc);
269 }
270 input.virtuals = std::move(space.virtuals);
271 input.virtual_energies = std::move(space.virtual_energies);
272 return input;
273}

275tools::EigenSystem InactiveScreening::CompleteGOrbitals(
276 const tools::EigenSystem& embedded, Index n_active_occupied,
277 const InactiveScreeningInput& input) {
278 const Index nB = input.inactive_occupied.cols();
279 const Index nvirt = input.virtuals.cols();
280 const Index nA = n_active_occupied;
281 if (nA + nvirt > embedded.eigenvectors().cols()) {
282 throw std::runtime_error(
283 "CompleteGOrbitals: the embedded MOs have fewer levels than the "
284 "active occupied plus virtual ones");
285 }
286 tools::EigenSystem result;
287 result.eigenvectors().resize(embedded.eigenvectors().rows(), nB + nA + nvirt);
288 result.eigenvectors() << input.inactive_occupied,
289 embedded.eigenvectors().leftCols(nA + nvirt);
290 result.eigenvalues().resize(nB + nA + nvirt);
291 result.eigenvalues() << input.inactive_energies,
292 embedded.eigenvalues().head(nA + nvirt);
293 return result;
294}

296Eigen::MatrixXd InactiveScreening::Kernel(const AOBasis& auxbasis,
297 const AOBasis& dftbasis,
298 const InactiveScreeningInput& input,
299 const tools::EigenSystem& embedded,
300 Index homo, double chunk_bytes,
301 double imaginary_frequency,
302 Index first_active) {
303 Eigen::MatrixXd Y = Response(
304 auxbasis, dftbasis, input.occupied, input.occupied_energies,
305 input.virtuals, input.virtual_energies, chunk_bytes, imaginary_frequency);
306 if (input.mode == InactiveScreeningInput::Mode::Exact) {
307 const Index nA = homo + 1 - first_active;
308 const Index nvirt = input.virtuals.cols();
309 if (homo + 1 + nvirt > embedded.eigenvectors().cols()) {
310 throw std::runtime_error(
311 "inactive_screening: the embedded MOs have fewer levels than the "
312 "full calculation's occupied and virtual ones");
313 }
314 Y -= Response(auxbasis, dftbasis,
315 embedded.eigenvectors().middleCols(first_active, nA),
316 embedded.eigenvalues().segment(first_active, nA),
317 embedded.eigenvectors().middleCols(homo + 1, nvirt),
318 embedded.eigenvalues().segment(homo + 1, nvirt), chunk_bytes,
319 imaginary_frequency);
320 }
321 return 0.5 * (Y + Y.transpose());
322}

324std::vector<Eigen::MatrixXd> InactiveScreening::Kernels(
325 const AOBasis& auxbasis, const AOBasis& dftbasis,
326 const InactiveScreeningInput& input, const TCMatrix_gwbse& active,
327 const Eigen::VectorXd& rpa_energies, Index homo, Index rpamin, Index rpamax,
328 double chunk_bytes, const std::vector<double>& imaginary_frequencies,
329 double virtual_cutoff) {
330 // the virtuals of the full calculation inside the RPA range; above them
331 // (no complete G) sit the inactive levels the projector shifted up,
332 // which are left out of both terms
333 Index nvirt = std::min(rpamax - homo, input.virtuals.cols());
334 if (virtual_cutoff > 0.0 && nvirt > 0) {
335 // only virtuals up to the cutoff above the LUMO, in both terms
336 const double top = input.virtual_energies(0) + virtual_cutoff;
337 nvirt = std::min(
338 nvirt,
339 Index((input.virtual_energies.head(nvirt).array() <= top).count()));
340 }
341 std::vector<Eigen::MatrixXd> Y = Responses(
342 auxbasis, dftbasis, input.occupied, input.occupied_energies,
343 input.virtuals.leftCols(nvirt), input.virtual_energies.head(nvirt),
344 chunk_bytes, imaginary_frequencies, active.InvSqrt());
345 if (input.mode == InactiveScreeningInput::Mode::Exact) {
346 // X_A from the integrals GW already holds (excluded holes, e.g. the
347 // inactive levels of complete G, are left out as in the RPA)
348 if (active.HasAdditionalResponse() || active.AdditionalPoleCount() > 0) {
349 throw std::runtime_error(
350 "inactive_screening: X_A must be built before the inactive "
351 "response is attached to the integrals");
352 }
353 const Index first = rpamin + active.ExcludedHoles();
354 const Index mmin = active.get_mmin();
355 const Index nmin = active.get_nmin();
356 for (std::size_t f = 0; f < Y.size(); ++f) {
357 const double w2 = imaginary_frequencies[f] * imaginary_frequencies[f];
358 Y[f] -= WeightedGram::Compute(
359 homo + 1 - first, active.auxsize(),
360 [&](Index m, Eigen::VectorXd& w) {
361 const Eigen::ArrayXd d =
362 rpa_energies.segment(homo + 1 - rpamin, nvirt).array() -
363 rpa_energies(first + m - rpamin);
364 w = 4.0 * d / (d.square() + w2);
365 },
366 [&](Index m, const WeightedGram::RowsVisitor& use) {
367 use(active[first + m - mmin].middleRows(homo + 1 - nmin, nvirt));
368 });
369 }
370 }
371 for (Eigen::MatrixXd& y : Y) {
372 y = 0.5 * (y + y.transpose());
373 }
374 return Y;
375}

377InactiveScreening::PoleFit InactiveScreening::FitPoles(
378 const Eigen::MatrixXd& Y0, const Eigen::MatrixXd& Yi, double w_i,
379 double tolerance) {
380 PoleFit fit;
381 const SymmetricEigenSystem y = SymmetricEigen(0.5 * (Y0 + Y0.transpose()));
382 const double cut = tolerance * std::max(y.values.maxCoeff(), 0.0);
383 const Index keep = (y.values.array() > cut).count();
384 fit.dropped = y.values.size() - keep;
385 if (keep == 0) {
386 return fit;
387 }
388 // Y(0) = G G^T on its range, G = U s^1/2; P = U s^-1/2 = G^+T
389 const Eigen::VectorXd s = y.values.tail(keep);
390 const Eigen::MatrixXd U = y.vectors.rightCols(keep);
391 const Eigen::MatrixXd P = U * s.cwiseSqrt().cwiseInverse().asDiagonal();
392 const Eigen::MatrixXd Yp = P.transpose() * Yi * P;
393 const SymmetricEigenSystem q = SymmetricEigen(0.5 * (Yp + Yp.transpose()));
394 // l = W^2/(W^2 + w_i^2): W from 1e-3 Ha (l_min) to 1e3 Ha (l_max)
395 const double l_min = 1e-6 / (1e-6 + w_i * w_i);
396 const double l_max = 1e6 / (1e6 + w_i * w_i);
397 Eigen::VectorXd l = q.values;
398 for (Index k = 0; k < l.size(); ++k) {
399 if (l(k) <= 0.0 || l(k) >= 1.0) {
400 // no real pole: keep the mode's static response, as a pole far above
401 // every excitation (a spurious low-lying pole would be worse for
402 // the exact self-energy than a static contribution)
403 ++fit.clipped;
404 l(k) = l_max;
405 } else {
406 l(k) = std::clamp(l(k), l_min, l_max);
407 }
408 }
409 fit.energies = w_i * (l.array() / (1.0 - l.array())).sqrt();
410 const Eigen::MatrixXd G = U * s.cwiseSqrt().asDiagonal() * q.vectors;
411 fit.vectors = G * (0.25 * fit.energies.array()).sqrt().matrix().asDiagonal();
412 return fit;
413}

415Eigen::MatrixXd InactiveScreening::ReactionField(const Eigen::MatrixXd& Y,
416 const Eigen::MatrixXd& R_env) {
417 const Index n = Y.rows();
418 const Eigen::MatrixXd one = Eigen::MatrixXd::Identity(n, n);
419 Eigen::MatrixXd u_inv = one + Y;
420 if (R_env.size() > 0) {
421 if (R_env.rows() != n) {
422 throw std::runtime_error(
423 "InactiveScreening: environment reaction field and inactive "
424 "response have different auxiliary dimensions");
425 }
426 // (1 + R_env)^-1 + Y; 1 + R_env is positive definite (Check)
427 u_inv = InverseSPD(one + R_env) + Y;
428 }
429 Eigen::MatrixXd R = InverseSPD(u_inv) - one;
430 return 0.5 * (R + R.transpose());
431}
