46inline Eigen::MatrixXd ApplyBSEExchange(
47 const TCMatrix_gwbse& Mout, const BSEWindow& out, const TCMatrix_gwbse& Min,
48 const BSEWindow& in, const Eigen::MatrixXd& x, double prefactor) {
49 const Index auxsize = Min.auxsize();
50 const Index k = x.cols();
51 // B = sum_v2 M_in[v2]_c^T x_v2, accumulated per thread
52 Eigen::MatrixXd B = Eigen::MatrixXd::Zero(auxsize, k);
53#pragma omp parallel
54 {
55 Eigen::MatrixXd B_thread = Eigen::MatrixXd::Zero(auxsize, k);
56#pragma omp for schedule(dynamic)
57 for (Index v2 = 0; v2 < in.vtotal; ++v2) {
58 B_thread.noalias() +=
59 Min[v2 + in.vmin].middleRows(in.cmin, in.ctotal).transpose() *
60 x.middleRows(v2 * in.ctotal, in.ctotal);
61 }
62#pragma omp critical
63 B += B_thread;
64 }
65 Eigen::MatrixXd y(out.size(), k);
66#pragma omp parallel for schedule(dynamic)
67 for (Index v1 = 0; v1 < out.vtotal; ++v1) {
68 y.middleRows(v1 * out.ctotal, out.ctotal).noalias() =
69 prefactor * Mout[v1 + out.vmin].middleRows(out.cmin, out.ctotal) * B;
70 }
71 return y;
72}

79inline Eigen::MatrixXd ApplyBSEQuasiparticle(const Eigen::MatrixXd& Hqp,
80 Index vtotal, Index ctotal,
81 const Eigen::MatrixXd& x,
82 double prefactor) {
83 const Eigen::MatrixXd Hvv = Hqp.topLeftCorner(vtotal, vtotal);
84 const Eigen::MatrixXd Hcc = Hqp.block(vtotal, vtotal, ctotal, ctotal);
85 Eigen::MatrixXd y(x.rows(), x.cols());
86 // all columns at once for Hcc: x as ctotal x (vtotal * k)
87 Eigen::Map<const Eigen::MatrixXd> X_all(x.data(), ctotal, vtotal * x.cols());
88 Eigen::Map<Eigen::MatrixXd> Y_all(y.data(), ctotal, vtotal * x.cols());
89 Y_all.noalias() = prefactor * Hcc * X_all;
90 for (Index col = 0; col < x.cols(); ++col) {
91 Eigen::Map<const Eigen::MatrixXd> X(x.col(col).data(), ctotal, vtotal);
92 Eigen::Map<Eigen::MatrixXd> Y(y.col(col).data(), ctotal, vtotal);
93 Y.noalias() -= prefactor * X * Hvv;
94 }
95 return y;
96}
