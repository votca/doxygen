31double TimeDependentRates::EscapeRate(Index site, double t) const {
32 double sum = 0.0;
33 for (Index e = graph_.EdgeOffset(site); e < graph_.EdgeOffset(site + 1);
34 ++e) {
35 sum += Rate(e, t);
36 }
37 return sum;
38}

41// cumulative sums of per-edge values within each site
42std::vector<double> CumulativePerSite(const ToFGraph& graph,
43 const std::vector<double>& value,
44 std::vector<double>& total) {
45 std::vector<double> cumulative(value.size());
46 total.assign(graph.NumberOfSites(), 0.0);
47 for (Index i = 0; i < graph.NumberOfSites(); ++i) {
48 double sum = 0.0;
49 for (Index e = graph.EdgeOffset(i); e < graph.EdgeOffset(i + 1); ++e) {
50 sum += value[e];
51 cumulative[e] = sum;
52 }
53 total[i] = sum;
54 }
55 return cumulative;
56}
57
58// edge of site i with cumulative value reaching target
59Index FindEdge(const ToFGraph& graph, const std::vector<double>& cumulative,
60 Index site, double target) {
61 Index begin = graph.EdgeOffset(site);
62 Index end = graph.EdgeOffset(site + 1);
63 Index e = Index(std::lower_bound(cumulative.begin() + begin,
64 cumulative.begin() + end, target) -
65 cumulative.begin());
66 return std::min(e, end - 1);
67}

74void ExponentialWaitingTime::Prepare(const TimeDependentRates& rates) {
75 if (!rates.isStatic()) {
76 throw std::runtime_error(
77 "ExponentialWaitingTime needs static rates, use thinning or the "
78 "integrated-rate inversion for time-dependent rates");
79 }
80 graph_ = &rates.Graph();
81 std::vector<double> value(graph_->NumberOfEdges());
82 for (Index e = 0; e < graph_->NumberOfEdges(); ++e) {
83 value[e] = rates.Rate(e, 0.0);
84 }
85 cumulative_ = CumulativePerSite(*graph_, value, escape_);
86}

88double ExponentialWaitingTime::NextTime(Index site, double t,
89 KMCRandom& random) const {
90 double escape = escape_[site];
91 return (escape > 0.0) ? t + random.Exponential(escape) : Infinity();
92}

94Index ExponentialWaitingTime::ChooseHop(Index site, double,
95 KMCRandom& random) const {

101void ThinningWaitingTime::Prepare(const TimeDependentRates& rates) {
102 rates_ = &rates;
103 const ToFGraph& graph = rates.Graph();
104 std::vector<double> bound(graph.NumberOfEdges());
105 for (Index e = 0; e < graph.NumberOfEdges(); ++e) {
106 bound[e] = rates.UpperBound(e);
107 if (!(bound[e] >= 0.0)) {
108 throw std::runtime_error("ThinningWaitingTime: invalid upper bound");
109 }
110 }
111 cumulative_ = CumulativePerSite(graph, bound, bound_escape_);
112}
113
114double ThinningWaitingTime::NextTime(Index site, double t,
115 KMCRandom& random) const {
116 double escape = bound_escape_[site];
117 return (escape > 0.0) ? t + random.Exponential(escape) : Infinity();
118}
119
120Index ThinningWaitingTime::ChooseHop(Index site, double t,
121 KMCRandom& random) const {
122 const ToFGraph& graph = rates_->Graph();
123 Index e = FindEdge(graph, cumulative_, site,
124 random.Uniform() * bound_escape_[site]);
125 double bound = rates_->UpperBound(e);
126 double rate = rates_->Rate(e, t);
127 if (rate > bound * (1.0 + 1e-12)) {
128 throw std::runtime_error(
129 "ThinningWaitingTime: rate above its upper bound for edge " +
130 std::to_string(e));
131 }
132 return (random.Uniform() * bound <= rate) ? e : -1;
133}

166std::unique_ptr<WaitingTimeStrategy> MakeWaitingTimeStrategy(
167 const std::string& name) {
168 if (name == "exponential") {
169 return std::make_unique<ExponentialWaitingTime>();
170 } else if (name == "thinning") {
171 return std::make_unique<ThinningWaitingTime>();
172 } else if (name == "inversion") {
173 return std::make_unique<IntegratedRateInversion>(
174 IntegratedRateInversion::Settings());
175 }
176 throw std::runtime_error("unknown waiting-time strategy '" + name +
177 "', use exponential, thinning or inversion");
178}
