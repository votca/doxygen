30Atom::Atom(Index resnr, std::string md_atom_name, Index atom_id,
31 Eigen::Vector3d pos, std::string type)
32 : id_(atom_id), name_(md_atom_name), resnr_(resnr), pos_(pos) {
33
34 std::string elename = GetElementFromString(md_atom_name);
35 std::string eletype = GetElementFromString(type);
36 tools::Elements ele;
37 bool found_element_name = true;
38 bool found_element_type = true;
39 try {
40 ele.getMass(elename);
41 } catch (std::runtime_error&) {
42 found_element_name = false;
43 }
44
45 try {
46 ele.getMass(eletype);
47 } catch (std::runtime_error&) {
48 found_element_type = false;
49 }
50
51 if (found_element_name && found_element_type) {
52 if (elename != eletype) {
53 throw std::runtime_error("Elements " + elename + " and" + eletype +
54 " from atom name: " + md_atom_name +
55 " and atom type:" + type + " do not match.");
56 }
57 element_ = elename;
58 } else if (found_element_name) {
59 element_ = elename;
60 } else if (found_element_type) {
61 element_ = elename;
62 } else {
63 throw std::runtime_error("Could not get Element from atom name:" +
64 md_atom_name + " or atom type:" + type);
65 }
66}

71std::string Atom::GetElementFromString(const std::string& MDName) {
72 std::string element = MDName.substr(0, 1);
73
74 if (MDName.size() > 1) {
75 if (std::islower(MDName[1])) {
76 element += MDName[1];
77 }
78 }
79 return element;
80}

88void Atom::SetupCptTable(CptTable& table) {
89 table.addCol<Index>("index", HOFFSET(data, id));
90 table.addCol<std::string>("element", HOFFSET(data, element));
91 table.addCol<std::string>("name", HOFFSET(data, name));
92 table.addCol<double>("pos.x", HOFFSET(data, x));
93 table.addCol<double>("pos.y", HOFFSET(data, y));
94 table.addCol<double>("pos.z", HOFFSET(data, z));
95 table.addCol<Index>("resnr", HOFFSET(data, resnr));
96 table.addCol<bool>("has_external_bond", HOFFSET(data, has_external_bond));
97 table.addCol<double>("ext_bond_dir.x", HOFFSET(data, ext_bond_dir_x));
98 table.addCol<double>("ext_bond_dir.y", HOFFSET(data, ext_bond_dir_y));
99 table.addCol<double>("ext_bond_dir.z", HOFFSET(data, ext_bond_dir_z));
100 // Deliberately only the SEGMENT id is persisted here -- the
101 // transient external_bond_partner_atom_id_ (see this class's own
102 // header comment for why) has no corresponding column at all, by
103 // design.
104 table.addCol<Index>("ext_bond_partner_segment_id",
105 HOFFSET(data, ext_bond_partner_segment_id));
106 // CptTable::addCol<U>() only supports fundamental types (confirmed
107 // directly by reading its own static_assert before relying on this
108 // -- no array support at all), so each of the kMaxBondedPartners
109 // slots needs its own, separate column, rather than one, single
110 // "array" column. Uses HOFFSET(data, bonded_partner_ids) + i *
111 // sizeof(Index), the offset of the array member itself plus a
112 // computed byte offset, rather than HOFFSET(data,
113 // bonded_partner_ids[i]) directly -- offsetof() into a specific
114 // array element is a known "gray area" (technically undefined
115 // behavior per the strict C++ standard, even though widely
116 // supported in practice), and this project's own -Wall -Wextra
117 // build flags make it worth avoiding rather than risking a warning.
118 for (Index i = 0; i < kMaxBondedPartners; i++) {
119 table.addCol<Index>(
120 "bonded_partner_id." + std::to_string(i),
121 HOFFSET(data, bonded_partner_ids) + size_t(i) * sizeof(Index));
122 }
123}
