35void GW::configure(const options& opt) {
36 opt_ = opt;
37
38 // Normalize legacy and new grid-search settings once at configuration time.
39 // From this point on, the solver operates only on the decoupled canonical
40 // controls, independent of how the user specified them in XML.
41 qp_solver::NormalizeGridSearchOptions(opt_);
42 qptotal_ = opt_.qpmax - opt_.qpmin + 1;
43 rpa_.configure(opt_.homo, opt_.rpamin, opt_.rpamax);
44 sigma_ = SigmaFactory().Create(opt_.sigma_integration, Mmn_, rpa_);
45 Sigma_base::options sigma_opt;
46 sigma_opt.homo = opt_.homo;
47 sigma_opt.qpmax = opt_.qpmax;
48 sigma_opt.qpmin = opt_.qpmin;
49 sigma_opt.rpamin = opt_.rpamin;
50 sigma_opt.rpamax = opt_.rpamax;
51 sigma_opt.eta = opt_.eta;
52 sigma_opt.alpha = opt_.alpha;
53 sigma_opt.quadrature_scheme = opt_.quadrature_scheme;
54 sigma_opt.order = opt_.order;
55 sigma_->configure(sigma_opt);
56 Sigma_x_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
57 Sigma_c_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
58}

60double GW::CalcHomoLumoShift(Eigen::VectorXd frequencies) const {
61 double DFTgap = dft_energies_(opt_.homo + 1) - dft_energies_(opt_.homo);
62 double QPgap = frequencies(opt_.homo + 1 - opt_.qpmin) -
63 frequencies(opt_.homo - opt_.qpmin);
64 return QPgap - DFTgap;
65}

67Eigen::MatrixXd GW::getHQP() const {
68 return Sigma_x_ + Sigma_c_ - vxc_ +
69 Eigen::MatrixXd(
70 dft_energies_.segment(opt_.qpmin, qptotal_).asDiagonal());
71}

73Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> GW::DiagonalizeQPHamiltonian()

113void GW::PrintQSGW_Composition(double threshold) const {
114 XTP_LOG(Log::error, log_)
115 << TimeStamp()
116 << " ====== QSGW QP state composition (dominant KS contributions) ======"
117 << std::flush;
118
119 for (Index n = 0; n < qptotal_; n++) {
120 const Index abs_n = n + opt_.qpmin;
121 std::string level = " Level";
122 if (abs_n == opt_.homo)
123 level = " HOMO ";
124 else if (abs_n == opt_.homo + 1)
125 level = " LUMO ";
126
127 // Collect contributions above threshold, sorted by weight descending
128 Eigen::VectorXd weights = qsgw_rotation_.col(n).cwiseAbs2();
129 std::vector<std::pair<double, Index>> contribs;
130 for (Index m = 0; m < qptotal_; m++) {
131 if (weights(m) >= threshold) {
132 contribs.push_back({weights(m), m + opt_.qpmin});
133 }
134 }
135 std::sort(contribs.begin(), contribs.end(),
136 [](const auto& a, const auto& b) { return a.first > b.first; });
137
138 std::string line;
139 for (const auto& [w, ks] : contribs) {
140 std::string ks_label;
141 if (ks == opt_.homo)
142 ks_label = "(HOMO)";
143 else if (ks == opt_.homo + 1)
144 ks_label = "(LUMO)";
145
146 line += (boost::format(" %1$5.1f%% KS_%2$d%3$s") % (100.0 * w) % ks %
147 ks_label)
148 .str();
149 }
150 XTP_LOG(Log::error, log_)
151 << level << (boost::format(" = %1$4d:") % abs_n).str() << line
152 << std::flush;
153 }
154}

218void GW::CalculateGWPerturbation() {
219
220 Sigma_x_ = (1 - opt_.ScaHFX) * sigma_->CalcExchangeMatrix();
221 XTP_LOG(Log::error, log_)
222 << TimeStamp() << " Calculated Hartree exchange contribution"
223 << std::flush;
224 // dftenergies has size aobasissize
225 // rpaenergies/Mmn have size rpatotal
226 // gwaenergies/frequencies have size qptotal
227 // homo index is relative to dft_energies
228 XTP_LOG(Log::error, log_)
229 << TimeStamp() << " Scissor shifting DFT energies by: " << opt_.shift
230 << " Hrt" << std::flush;
231 Eigen::VectorXd dft_shifted_energies = ScissorShift_DFTlevel(dft_energies_);
232 rpa_.setRPAInputEnergies(
233 dft_shifted_energies.segment(opt_.rpamin, opt_.rpamax - opt_.rpamin + 1));
234 Eigen::VectorXd frequencies =
235 dft_shifted_energies.segment(opt_.qpmin, qptotal_);
236
237 Anderson mixing_;
238 mixing_.Configure(opt_.gw_mixing_order, opt_.gw_mixing_alpha);
239
240 for (Index i_gw = 0; i_gw < opt_.gw_sc_max_iterations; ++i_gw) {
241 gw_sc_iteration_ = i_gw;
242 if (i_gw % opt_.reset_3c == 0 && i_gw != 0) {
243 Mmn_.Rebuild();
244 XTP_LOG(Log::info, log_)
245 << TimeStamp() << " Rebuilding 3c integrals" << std::flush;
246 }
247 sigma_->PrepareScreening();
248 XTP_LOG(Log::info, log_)
249 << TimeStamp() << " Calculated screening via RPA" << std::flush;
250 XTP_LOG(Log::info, log_)
251 << TimeStamp() << " Solving QP equations " << std::flush;
252 if (opt_.gw_mixing_order > 0 && i_gw > 0) {
253 mixing_.UpdateInput(frequencies);
254 }
255
256 frequencies = SolveQP(frequencies);
257
258 if (opt_.gw_sc_max_iterations > 1) {
259 Eigen::VectorXd rpa_energies_old = rpa_.getRPAInputEnergies();
260 if (opt_.gw_mixing_order > 0 && i_gw > 0) {
261 if (opt_.gw_mixing_order == 1) {
262 XTP_LOG(Log::debug, log_)
263 << "GWSC using linear mixing with alpha: " << opt_.gw_mixing_alpha
264 << std::flush;
265 } else {
266 XTP_LOG(Log::debug, log_)
267 << "GWSC using Anderson mixing with history "
268 << opt_.gw_mixing_order << ", alpha: " << opt_.gw_mixing_alpha
269 << std::flush;
270 }
271 mixing_.UpdateOutput(frequencies);
272 Eigen::VectorXd mixed_frequencies = mixing_.MixHistory();
273 rpa_.UpdateRPAInputEnergies(dft_energies_, mixed_frequencies,
274 opt_.qpmin);
275 frequencies = mixed_frequencies;
276
277 } else {
278 XTP_LOG(Log::debug, log_) << "GWSC using plain update " << std::flush;
279 rpa_.UpdateRPAInputEnergies(dft_energies_, frequencies, opt_.qpmin);
280 }
281
282 for (int i = 0; i < frequencies.size(); i++) {
283 XTP_LOG(Log::debug, log_)
284 << "... GWSC iter " << i_gw << " state " << i << " "
285 << std::setprecision(9) << frequencies(i) << std::flush;
286 }
287
288 XTP_LOG(Log::info, log_)
289 << TimeStamp() << " GW_Iteration:" << i_gw
290 << " Shift[Hrt]:" << CalcHomoLumoShift(frequencies) << std::flush;
291 if (Converged(rpa_.getRPAInputEnergies(), rpa_energies_old,
292 opt_.gw_sc_limit)) {
293 XTP_LOG(Log::info, log_) << TimeStamp() << " Converged after "
294 << i_gw + 1 << " GW iterations." << std::flush;
295 break;
296 } else if (i_gw == opt_.gw_sc_max_iterations - 1) {
297 XTP_LOG(Log::error, log_)
298 << TimeStamp()
299 << " WARNING! GW-self-consistency cycle not converged after "
300 << opt_.gw_sc_max_iterations << " iterations." << std::flush;
301 XTP_LOG(Log::error, log_)
302 << TimeStamp() << " Run continues. Inspect results carefully!"
303 << std::flush;
304 break;
305 }
306 }
307 }
308 Sigma_c_.diagonal() = sigma_->CalcCorrelationDiag(frequencies);
309 PrintGWA_Energies();
310}

312Eigen::VectorXd GW::getGWAResults() const {
313 // When QSGW with virtual trimming has run, return the pre-computed merged
314 // energy vector (QSGW for the trimmed window, seed for excluded virtuals).
315 // This avoids recomputing from sigma matrices which were zeroed post-loop.
316 if (qsgw_final_energies_.size() > 0) {
317 return qsgw_final_energies_;
318 }
319 return Sigma_x_.diagonal() + Sigma_c_.diagonal() - vxc_.diagonal() +
320 dft_energies_.segment(opt_.qpmin, qptotal_);
321}

323Eigen::VectorXd GW::SolveQP(const Eigen::VectorXd& frequencies) const {
324 sigma_->ResetDiagEvalCounter();
325 Eigen::VectorXd env = Eigen::VectorXd::Zero(qptotal_);
326
327 const Eigen::VectorXd intercepts =
328 dft_energies_.segment(opt_.qpmin, qptotal_) + Sigma_x_.diagonal() -
329 vxc_.diagonal();
330
331 Eigen::VectorXd frequencies_new = frequencies;
332 Eigen::Array<bool, Eigen::Dynamic, 1> converged =
333 Eigen::Array<bool, Eigen::Dynamic, 1>::Zero(qptotal_);
334
335 QPStats total_stats;
336
337#ifdef _OPENMP
338 Index use_threads =
339 OPENMP::getMaxThreads() > qptotal_ ? qptotal_ : OPENMP::getMaxThreads();
340#else
341 Index use_threads = 1;
342#endif
343
344#pragma omp parallel for schedule(dynamic) num_threads(use_threads)
345 for (Index gw_level = 0; gw_level < qptotal_; ++gw_level) {
346
347 double initial_f = frequencies[gw_level];
348 double intercept = intercepts[gw_level];
349 boost::optional<double> newf;
350 QPStats local_stats;
351
352 if (opt_.qp_solver == "fixedpoint") {
353 newf = SolveQP_FixedPoint(intercept, initial_f, gw_level, &local_stats);
354 }
355 if (newf) {
356 frequencies_new[gw_level] = newf.value();
357 converged[gw_level] = true;
358 } else {
359 newf = SolveQP_Grid(intercept, initial_f, gw_level, &local_stats);
360 if (newf) {
361 frequencies_new[gw_level] = newf.value();
362 converged[gw_level] = true;
363 } else {
364 newf =
365 SolveQP_Linearisation(intercept, initial_f, gw_level, &local_stats);
366 if (newf) {
367 frequencies_new[gw_level] = newf.value();
368 }
369 }
370 }
371
372#pragma omp critical
373 {
374 total_stats.Add(local_stats);
375 }
376 }
377
378 if (!converged.all()) {
379 std::vector<Index> states;
380 for (Index s = 0; s < converged.size(); s++) {
381 if (!converged[s]) {
382 states.push_back(s);
383 }
384 }
385 IndexParser rp;
386 XTP_LOG(Log::error, log_) << TimeStamp() << " Not converged PQP states are:"
387 << rp.CreateIndexString(states) << std::flush;
388 XTP_LOG(Log::error, log_)
389 << TimeStamp() << " Increase the grid search interval" << std::flush;
390 }
391
392 XTP_LOG(Log::info, log_) << TimeStamp()
393 << " Sigma diagonal evaluations in SolveQP: "
394 << sigma_->GetDiagEvalCounter() << std::flush;
395
396 XTP_LOG(Log::info, log_) << TimeStamp() << " QP diagnostics: "
397 << "scan=" << total_stats.sigma_scan_calls
398 << " refine=" << total_stats.sigma_refine_calls
399 << " deriv_sigma="
400 << total_stats.sigma_derivative_calls
401 << " other=" << total_stats.sigma_other_calls
402 << " total_sigma=" << total_stats.TotalSigmaCalls()
403 << " unique_omega="
404 << total_stats.sigma_unique_frequencies
405 << " repeat_sigma=" << total_stats.sigma_repeat_calls
406 << " deriv_calls=" << total_stats.deriv_calls
407 << std::flush;
408
409 return frequencies_new;
410}

412boost::optional<double> GW::SolveQP_Linearisation(double intercept0,
413 double frequency0,
414 Index gw_level,
415 QPStats* stats) const {
416 boost::optional<double> newf = boost::none;
417
418 QPFunc fqp(gw_level, *sigma_.get(), intercept0);
419
420 double sigma = fqp.sigma(frequency0, EvalStage::Other);
421 double dsigma_domega = fqp.deriv(frequency0);
422 double Z = 1.0 - dsigma_domega;
423 if (std::abs(Z) > 1e-9) {
424 newf = frequency0 + (intercept0 - frequency0 + sigma) / Z;
425 }
426
427 if (stats != nullptr) {
428 *stats = fqp.GetStats();
429 }
430 return newf;
431}

433boost::optional<double> GW::SolveQP_Grid_Windowed_Adaptive(
434 double intercept0, double frequency0, Index gw_level, double left_limit,
435 double right_limit, bool allow_rejected_return, QPStats* stats) const {
436
437 QPFunc fqp(gw_level, *sigma_.get(), intercept0);
438
439 qp_solver::SolverOptions solver_opt;
440
441 // Pass only canonical search controls into the shared grid-search utility.
442 // RKS and UKS intentionally use the same search semantics; only the sigma
443 // evaluator differs between the two implementations.
444 solver_opt.g_sc_limit = opt_.g_sc_limit;
445 solver_opt.qp_bisection_max_iter = opt_.g_sc_max_iterations;
446 solver_opt.qp_full_window_half_width = opt_.qp_full_window_half_width;
447 solver_opt.qp_dense_spacing = opt_.qp_dense_spacing;
448 solver_opt.qp_adaptive_shell_width = opt_.qp_adaptive_shell_width;
449 solver_opt.qp_adaptive_shell_count = opt_.qp_adaptive_shell_count;
450
451 QPWindowDiagnostics wdiag;
452 std::vector<QPRootCandidate> accepted_roots;
453 std::vector<QPRootCandidate> rejected_roots;
454
455 bool use_brent = false;
456 if (opt_.qp_root_finder == "brent") {
457 use_brent = true;
458 }
459
460 auto result = qp_solver::SolveQP_Grid_Windowed(
461 fqp, frequency0, left_limit, right_limit, gw_sc_iteration_, solver_opt,
462 &wdiag, &accepted_roots, &rejected_roots, use_brent);
463
464 if (Log::current_level > Log::error) {
465#pragma omp critical
466 {
467 if (!accepted_roots.empty() || !rejected_roots.empty()) {
468 XTP_LOG(Log::info, log_)
469 << " Adaptive scan qplevel:" << gw_level << " center="
470 << std::max(left_limit, std::min(right_limit, frequency0))
471 << " shells=" << wdiag.shells_explored
472 << " first_interval_shell=" << wdiag.first_interval_shell
473 << " first_accepted_shell=" << wdiag.first_accepted_shell
474 << " intervals=" << wdiag.intervals_found << std::flush;
475 }
476 }
477 }
478
479 if (stats != nullptr) {
480 *stats = fqp.GetStats();
481 }
482
483 if (!accepted_roots.empty()) {
484 return result;
485 }
486
487 if (!rejected_roots.empty() && !allow_rejected_return) {
488 if (Log::current_level > Log::error) {
489#pragma omp critical
490 {
491 XTP_LOG(Log::info, log_)
492 << " Adaptive scan qplevel:" << gw_level
493 << " produced only rejected roots in [" << left_limit << ", "
494 << right_limit << "], forcing wider retry" << std::flush;
495 }
496 }
497 return boost::none;
498 }
499
500 return result;
501}

503boost::optional<double> GW::SolveQP_Grid_Windowed_Dense(
504 double intercept0, double frequency0, Index gw_level, double left_limit,
505 double right_limit, bool allow_rejected_return, QPStats* stats) const {
506
507 QPFunc fqp(gw_level, *sigma_.get(), intercept0);
508
509 qp_solver::SolverOptions solver_opt;
510
511 // The dense path performs a uniform sign-change scan over the requested
512 // interval. Its spacing is qp_dense_spacing, not the adaptive shell width.
513 solver_opt.g_sc_limit = opt_.g_sc_limit;
514 solver_opt.qp_bisection_max_iter = opt_.g_sc_max_iterations;
515 solver_opt.qp_full_window_half_width = opt_.qp_full_window_half_width;
516 solver_opt.qp_dense_spacing = opt_.qp_dense_spacing;
517 solver_opt.qp_adaptive_shell_width = opt_.qp_adaptive_shell_width;
518 solver_opt.qp_adaptive_shell_count = opt_.qp_adaptive_shell_count;
519
520 const bool use_brent = (opt_.qp_root_finder == "brent");
521
522 std::vector<QPRootCandidate> accepted_roots;
523 std::vector<QPRootCandidate> rejected_roots;
524 std::vector<std::pair<double, double>> roots; // omega : Z
525
526 if (left_limit < right_limit) {
527 double freq_prev = left_limit;
528 double targ_prev = fqp.value(freq_prev, EvalStage::Scan);
529
530 // Dense scanning is the robustness path: it uniformly samples the entire
531 // interval to avoid missing sign changes that a coarser adaptive shell
532 // search might step over.
533 const Index n_steps = std::max<Index>(
534 2, static_cast<Index>(
535 std::ceil((right_limit - left_limit) / opt_.qp_dense_spacing)) +
536 1);
537
538 for (Index i_node = 1; i_node < n_steps; ++i_node) {
539 const double freq =
540 (i_node == n_steps - 1)
541 ? right_limit
542 : std::min(right_limit, left_limit + static_cast<double>(i_node) *
543 opt_.qp_dense_spacing);
544
545 const double targ = fqp.value(freq, EvalStage::Scan);
546
547 if (targ_prev * targ < 0.0) {
548 auto cand =
549 qp_solver::RefineQPInterval(freq_prev, targ_prev, freq, targ, fqp,
550 frequency0, solver_opt, use_brent);
551 if (cand) {
552 roots.emplace_back(cand->omega, cand->Z);
553 if (cand->accepted) {
554 accepted_roots.push_back(*cand);
555 } else {
556 rejected_roots.push_back(*cand);
557 }
558 }
559 }
560
561 freq_prev = freq;
562 targ_prev = targ;
563 }
564 }
565
566 if (Log::current_level > Log::error) {
567#pragma omp critical
568 {
569 if (accepted_roots.empty() && rejected_roots.empty()) {
570 XTP_LOG(Log::info, log_)
571 << " Dense scan qplevel:" << gw_level << " found no roots in ["
572 << left_limit << ", " << right_limit << "]" << std::flush;
573 } else {
574 XTP_LOG(Log::info, log_)
575 << " Dense scan qplevel:" << gw_level << " roots (omega:Z)\n\t\t";
576 for (const auto& root : roots) {
577 XTP_LOG(Log::info, log_) << std::setprecision(5) << root.first << ":"
578 << root.second << " ";
579 }
580 XTP_LOG(Log::info, log_) << std::flush;
581 }
582 }
583 }
584
585 if (stats != nullptr) {
586 *stats = fqp.GetStats();
587 }
588
589 if (!accepted_roots.empty()) {
590 auto best = std::max_element(
591 accepted_roots.begin(), accepted_roots.end(),
592 [](const QPRootCandidate& a, const QPRootCandidate& b) {
593 return qp_solver::ScoreRoot(a) < qp_solver::ScoreRoot(b);
594 });
595 return best->omega;
596 }
597
598 if (!rejected_roots.empty()) {
599 if (!allow_rejected_return) {
600 if (Log::current_level > Log::error) {
601#pragma omp critical
602 {
603 XTP_LOG(Log::info, log_)
604 << " Dense scan qplevel:" << gw_level
605 << " produced only rejected roots in [" << left_limit << ", "
606 << right_limit << "], forcing wider retry" << std::flush;
607 }
608 }
609 return boost::none;
610 }
611
612 auto least_bad = std::max_element(
613 rejected_roots.begin(), rejected_roots.end(),
614 [](const QPRootCandidate& a, const QPRootCandidate& b) {
615 return qp_solver::ScoreRoot(a) < qp_solver::ScoreRoot(b);
616 });
617 return least_bad->omega;
618 }
619
620 return boost::none;
621}

631 }

637 }
638

674 opt_.qp_grid_search_mode + "'");

677boost::optional<double> GW::SolveQP_Grid(double intercept0, double frequency0,
678 Index gw_level, QPStats* stats) const {
679 // The full QP search window is now controlled explicitly and no longer
680 // inferred from the dense-grid spacing.
681 const double range = opt_.qp_full_window_half_width;
682
683 const double full_left_limit = frequency0 - range;
684 const double full_right_limit = frequency0 + range;
685
686 double restricted_left_limit = full_left_limit;
687 double restricted_right_limit = full_right_limit;
688
689 bool use_restricted_window = false;
690
691 if (opt_.qp_restrict_search) {
692 const Index mo_level = gw_level + opt_.qpmin;
693 const bool is_occupied = (mo_level <= opt_.homo);
694
695 if (is_occupied) {
696 restricted_right_limit = std::min(full_right_limit, -opt_.qp_zero_margin);
697 } else {
698 restricted_left_limit =
699 std::max(full_left_limit, opt_.qp_virtual_min_energy);
700 }
701
702 const double tol = 1e-12;
703 use_restricted_window =
704 (std::abs(restricted_left_limit - full_left_limit) > tol) ||
705 (std::abs(restricted_right_limit - full_right_limit) > tol);
706 }
707
708 if (use_restricted_window && restricted_left_limit < restricted_right_limit) {
709 auto restricted =
710 SolveQP_Grid_Windowed(intercept0, frequency0, gw_level,
711 restricted_left_limit, restricted_right_limit,
712 false, // accepted roots only
713 stats);
714 if (restricted) {
715 return restricted;
716 }
717
718 if (Log::current_level > Log::error) {
719#pragma omp critical
720 {
721 XTP_LOG(Log::info, log_)
722 << " Restricted QP search failed for qplevel:" << gw_level
723 << " in window [" << restricted_left_limit << ", "
724 << restricted_right_limit << "], forcing full dense window ["
725 << full_left_limit << ", " << full_right_limit << "]" << std::flush;
726 }
727 }
728
729 // Important: do NOT go back to adaptive here.
730 // The restricted window already failed to produce an accepted root.
731 // Force the old robust full-window dense search.
732 return SolveQP_Grid_Windowed_Dense(intercept0, frequency0, gw_level,
733 full_left_limit, full_right_limit, true,
734 stats);
735 }
736
737 return SolveQP_Grid_Windowed(intercept0, frequency0, gw_level,
738 full_left_limit, full_right_limit, true, stats);
739}

778// =============================================================================
779// GW::CalculateQSGW
780//
781// Quasiparticle self-consistent GW: iterates the rotation of the
782// three-centre integrals until the QP energies are converged.
783//
784// Index conventions (same as evGW throughout):
785// gw_level : 0-based index within the QP window [0, qptotal_)
786// MO level : absolute MO index = gw_level + opt_.qpmin
787//
788// H_QSGW[m,n] = (e_DFT[m] - v_xc[m]) * delta_{mn} + tilde_Sigma[m,n]
789//
790// where tilde_Sigma = 0.5 * Re( Sigma(e_m) + Sigma(e_n) )
791// = 0.5 * (Sigma_x + Sigma_x^T) <- exchange (already sym)
792// + 0.5 * (Sigma_c(e_m,e_n) + Sigma_c(e_n,e_m)^T) <- corr
793//
794// Note: CalcCorrelationOffDiag(freqs) evaluates element (m,n) at
795// frequency freqs[m] for the row index. To form the symmetrised
796// average we call it twice with swapped frequency vectors.
797// =============================================================================
798void GW::CalculateQSGW() {
799 XTP_LOG(Log::error, log_)
800 << TimeStamp() << " Starting QSGW self-consistency loop " << std::flush;
801
802 XTP_LOG(Log::error, log_)
803 << TimeStamp() << " QSGW start: qptotal_=" << qptotal_
804 << " vxc rows=" << vxc_.rows() << " cols=" << vxc_.cols()
805 << " Sigma_x rows=" << Sigma_x_.rows() << std::flush;
806
807 // Initialise: start from evGW/G0W0 seed energies.
808 const Eigen::VectorXd e_qp_full = getGWAResults(); // full [qpmin,qpmax]
809 qsgw_seed_energies_ = e_qp_full;
810
811 // ── Virtual-level threshold: auto-trim qpmax ─────────────────────────────
812 // Scan virtual levels from LUMO upward. Exclude the first virtual whose
813 // perturbative correction |e_QP - e_DFT| exceeds qsgw_max_virt_correction.
814 // Occupied states are never excluded — large core corrections are physical.
815 Index qsgw_qpmax = opt_.qpmax;
816 {
817 const Index lumo_local = opt_.homo - opt_.qpmin + 1;
818 bool trimmed = false;
819 for (Index n = lumo_local; n < qptotal_; n++) {
820 const double corr =
821 std::abs(e_qp_full(n) - dft_energies_(opt_.qpmin + n));
822 if (corr > opt_.qsgw_max_virt_correction) {
823 qsgw_qpmax = opt_.qpmin + n - 1;
824 trimmed = true;
825 XTP_LOG(Log::error, log_)
826 << TimeStamp() << " QSGW virtual threshold: level "
827 << (opt_.qpmin + n)
828 << " correction = " << corr * votca::tools::conv::hrt2ev
829 << " eV exceeds "
830 << opt_.qsgw_max_virt_correction * votca::tools::conv::hrt2ev
831 << " eV. Trimming QSGW window to [" << opt_.qpmin << ","
832 << qsgw_qpmax << "]." << std::flush;
833 break;
834 }
835 }
836 if (!trimmed) {
837 XTP_LOG(Log::error, log_)
838 << TimeStamp() << " QSGW virtual threshold: all corrections within "
839 << opt_.qsgw_max_virt_correction * votca::tools::conv::hrt2ev
840 << " eV. Using full window [" << opt_.qpmin << "," << qsgw_qpmax
841 << "]." << std::flush;
842 }
843 }
844
845 const Index qsgw_qptotal = qsgw_qpmax - opt_.qpmin + 1;
846 const bool window_trimmed = (qsgw_qpmax < opt_.qpmax);
847
848 // If trimmed, reconfigure sigma for the reduced window
849 if (window_trimmed) {
850 Sigma_base::options sigma_opt;
851 sigma_opt.homo = opt_.homo;
852 sigma_opt.qpmin = opt_.qpmin;
853 sigma_opt.qpmax = qsgw_qpmax;
854 sigma_opt.rpamin = opt_.rpamin;
855 sigma_opt.rpamax = opt_.rpamax;
856 sigma_opt.eta = opt_.eta;
857 sigma_opt.quadrature_scheme = opt_.quadrature_scheme;
858 sigma_opt.order = opt_.order;
859 sigma_opt.alpha = opt_.alpha;
860 sigma_->configure(sigma_opt);
861 Sigma_x_ = Eigen::MatrixXd::Zero(qsgw_qptotal, qsgw_qptotal);
862 Sigma_c_ = Eigen::MatrixXd::Zero(qsgw_qptotal, qsgw_qptotal);
863 }
864
865 Eigen::VectorXd e_qp = e_qp_full.head(qsgw_qptotal);
866 qsgw_rotation_ = Eigen::MatrixXd::Identity(qsgw_qptotal, qsgw_qptotal);
867
868 // QSGW requires a local (GGA/LDA) DFT starting point.
869 // With a hybrid functional, the HF exchange embedded in the DFT eigenvalues
870 // introduces a starting-point dependence that is not removed by the QSGW
871 // self-consistency loop. See Faleev, van Schilfgaarde & Kotani PRL 2004
872 // and Betzinger, Friedrich & Blugel PRB 2010 for discussion.
873 if (opt_.ScaHFX > 0.0) {
874 throw std::runtime_error(
875 "GW::CalculateQSGW: QSGW is not compatible with hybrid DFT starting "
876 "points (ScaHFX = " +
877 std::to_string(opt_.ScaHFX) +
878 "). Use a pure GGA or LDA functional as the DFT starting point.");
879 }
880
881 if (opt_.sigma_integration == "cda") {
882 throw std::runtime_error(
883 "GW::CalculateQSGW: QSGW is not supported with the CDA sigma "
884 "integration method. The Gaussian quadrature grid used by CDA becomes "
885 "ill-conditioned as the QP wavefunctions rotate away from the DFT-MO "
886 "basis, causing numerical divergence. Use sigma_integration=ppm or "
887 "sigma_integration=exact instead.");
888 }
889
890 // H0 = diag(e_DFT - v_xc): diagonal approximation for the non-interacting
891 // part of H_QSGW. The off-diagonal V_xc elements are neglected here.
892 // Their significance is printed above for diagnostics.
893 const Eigen::VectorXd e_dft_minus_vxc =
894 dft_energies_.segment(opt_.qpmin, qsgw_qptotal) -
895 vxc_.diagonal().head(qsgw_qptotal);
896
897 Eigen::MatrixXd H0 = -vxc_.topLeftCorner(qsgw_qptotal, qsgw_qptotal);
898 H0.diagonal() += dft_energies_.segment(opt_.qpmin, qsgw_qptotal);
899
900 // Store a copy of the original (unrotated) Mmn_ QP-window slices.
901 // At each iteration we restore Mmn_ to the DFT-MO state and apply the full
902 // rotation qsgw_rotation_ in one shot.
903 // This is necessary because dU (eigenvectors of H_QSGW, always in DFT-MO
904 // basis) gives the total rotation from DFT-MOs directly, not an incremental
905 // rotation relative to the previous QP basis.
906 // Store ALL m-slices in the full RPA window, not just the QP window.
907 // PrepareScreening (PPM, CDA) calls MultiplyRightWithAuxMatrix which
908 // overwrites all slices including those outside the QP window (e.g. the
909 // core level when qpmin > rpamin). If we only restore QP-window slices,
910 // out-of-window slices accumulate PPM-basis corruption across iterations,
911 // causing the persistent oscillation seen when qpmin > rpamin.
912 const Index mtotal = Mmn_.msize();
913 std::vector<Eigen::MatrixXd> Mmn_orig(mtotal);
914 for (Index m = 0; m < mtotal; m++) {
915 Mmn_orig[m] = Mmn_[m];
916 }
917
918 // Anderson/DIIS mixer for tilde_Sigma.
919 // Reuses gw_mixing_order and gw_mixing_alpha options.
920 Anderson qsgw_mixer;
921 qsgw_mixer.Configure(opt_.gw_mixing_order, opt_.gw_mixing_alpha);
922 XTP_LOG(Log::error, log_)
923 << TimeStamp() << " QSGW mixer: order=" << opt_.gw_mixing_order
924 << " alpha=" << opt_.gw_mixing_alpha << std::flush;
925
926 // Register the QSGW rotation with sigma and rpa so that PrepareScreening
927 // and the RPA functions apply the m-rotation to QP-window hole slices.
928 // At iteration 0 qsgw_rotation_ is identity so no actual rotation is done
929 // (the pointer is set; the rotation only matters when iter > 0).
930 sigma_->setQSGWRotation(&qsgw_rotation_, opt_.qpmin, opt_.homo);
931 rpa_.setQSGWRotation(&qsgw_rotation_, opt_.qpmin, opt_.homo);
932
933 double diff_max_prev = std::numeric_limits<double>::max();
934
935 for (Index iter = 0; iter < opt_.qsgw_max_iterations; iter++) {
936
937 // ── Step 1: restore Mmn_ to DFT-MO basis and apply full rotation ─────────
938 for (Index m = 0; m < mtotal; m++) {
939 Mmn_[m] = Mmn_orig[m];
940 }
941 if (iter > 0) {
942 Mmn_.Rotate(qsgw_rotation_, opt_.qpmin, qsgw_qpmax);
943 }
944
945 // Recompute screening W and Sigma in current QP basis.
946 sigma_->PrepareScreening();
947
948 // Exchange: symmetric, frequency-independent.
949 Sigma_x_ = sigma_->CalcExchangeMatrix();
950
951 // Symmetrised static self-energy (full matrix including diagonal)
952 Eigen::MatrixXd Sc_row = sigma_->CalcCorrelationOffDiag(e_qp);
953 Eigen::MatrixXd Sc_col = Sc_row.transpose();
954 Eigen::MatrixXd tilde_Sigma = Sigma_x_ + 0.5 * (Sc_row + Sc_col);
955 tilde_Sigma.diagonal() += sigma_->CalcCorrelationDiag(e_qp);
956
957 // ── Step 2: mix tilde_Sigma with Anderson/DIIS ───────────────────────────
958 // Follows the evGW mixer pattern:
959 // UpdateInput is called at the END of each iteration with the raw
960 // tilde_Sigma (what "went in" to that step).
961 // UpdateOutput is called at the START of the next iteration with the
962 // new raw tilde_Sigma (what "came out").
963 // MixHistory then returns the Anderson-mixed tilde_Sigma.
964 // This ensures output_.size() == input_.size() at all times, giving the
965 // mixer a consistent residual (output - input) to minimise.
966 Eigen::VectorXd S_flat = Eigen::Map<const Eigen::VectorXd>(
967 tilde_Sigma.data(), qsgw_qptotal * qsgw_qptotal);
968 if (iter > 0) {
969 qsgw_mixer.UpdateOutput(S_flat);
970 S_flat = qsgw_mixer.MixHistory();
971 }
972
973 // ── Step 3: diagonalise H_QSGW ───────────────────────────────────────────
974 // Two diagonalisations:
975 //
976 // (a) Mixed H_QSGW -> dU for rotating Mmn_ next iteration.
977 // S_flat is the Anderson-mixed tilde_Sigma — this is where the damping
978 // actually takes effect. Different alpha must give different dU here.
979 //
980 // (b) Unmixed H_QSGW -> e_new for convergence check.
981 // The convergence criterion must track the true fixed point, not the
982 // damped approximation, so we use the raw tilde_Sigma for eigenvalues.
983
984 // (a) mixed: rotation
985 Eigen::MatrixXd tilde_Sigma_mixed = Eigen::Map<const Eigen::MatrixXd>(
986 S_flat.data(), qsgw_qptotal, qsgw_qptotal);
987 Eigen::MatrixXd H_qsgw_mixed = H0 + tilde_Sigma_mixed;
988 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es_mixed(H_qsgw_mixed);
989 if (es_mixed.info() != Eigen::ComputationInfo::Success) {
990 throw std::runtime_error(
991 "GW::CalculateQSGW: diagonalisation of mixed H_QSGW failed at iter " +
992 std::to_string(iter));
993 }
994 Eigen::MatrixXd dU = es_mixed.eigenvectors();
995
996 // (b) unmixed: convergence check
997 Eigen::MatrixXd H_qsgw_new = H0 + tilde_Sigma;
998 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es_new(H_qsgw_new);
999 if (es_new.info() != Eigen::ComputationInfo::Success) {
1000 throw std::runtime_error(
1001 "GW::CalculateQSGW: diagonalisation of H_QSGW failed at iter " +
1002 std::to_string(iter));
1003 }
1004 Eigen::VectorXd e_new = es_new.eigenvalues();
1005
1006 double diff_max = (e_new - e_qp).cwiseAbs().maxCoeff();
1007 XTP_LOG(Log::error, log_)
1008 << TimeStamp() << " QSGW iter " << iter
1009 << " max|dE_QP| = " << diff_max * votca::tools::conv::hrt2ev << " eV"
1010 << std::flush;
1011
1012 // Reset Anderson history when residual increases significantly.
1013 // This prevents accumulation of bad history when the mixer overshoots.
1014 // Anderson::Configure() does not clear the history vectors, so we
1015 // reconstruct the mixer object entirely to get a true reset.
1016 if (iter > 1 && diff_max > 2.0 * diff_max_prev) {
1017 qsgw_mixer = Anderson();
1018 qsgw_mixer.Configure(opt_.gw_mixing_order, opt_.gw_mixing_alpha);
1019 }
1020 diff_max_prev = diff_max;
1021
1022 if (diff_max < opt_.qsgw_sc_limit) {
1023 XTP_LOG(Log::error, log_) << TimeStamp() << " QSGW converged in "
1024 << iter + 1 << " iterations." << std::flush;
1025 e_qp = e_new;
1026 qsgw_rotation_ = dU;
1027 // Update RPA with the final converged e_qp before breaking so that
1028 // RPAInputEnergies() reflects the converged state, not the previous iter.
1029 rpa_.UpdateRPAInputEnergies(dft_energies_, e_qp, opt_.qpmin);
1030 break;
1031 }
1032
1033 if (iter == opt_.qsgw_max_iterations - 1) {
1034 XTP_LOG(Log::error, log_)
1035 << TimeStamp() << " WARNING: QSGW did not converge in "
1036 << opt_.qsgw_max_iterations
1037 << " iterations. Inspect results carefully." << std::flush;
1038 }
1039
1040 // ── Step 5: update QP energies and total rotation
1041 // ─────────────────────────
1042 e_qp = e_new;
1043 qsgw_rotation_ = dU;
1044
1045 // Store the raw (unmixed) tilde_Sigma as Anderson input for the next
1046 // iteration. Must be the unmixed version so that MixHistory computes
1047 // the correct residual: new_tilde_Sigma (output) - old_tilde_Sigma (input).
1048 qsgw_mixer.UpdateInput(Eigen::Map<const Eigen::VectorXd>(
1049 tilde_Sigma.data(), qsgw_qptotal * qsgw_qptotal));
1050
1051 // Update the RPA input energies to the new QP energies.
1052 rpa_.UpdateRPAInputEnergies(dft_energies_, e_qp, opt_.qpmin);
1053 }
1054
1055 // Store converged results in the standard output fields.
1056 // Sigma_c_ diagonal is set to the converged on-diagonal correlation
1057 // (needed by getGWAResults / PrintGWA_Energies).
1058 Sigma_c_.diagonal() = sigma_->CalcCorrelationDiag(e_qp);
1059
1060 // Store QP energies and eigenvectors (= accumulated rotation from DFT MOs)
1061 // in Sigma_c_ off-diagonal is left as the last iteration's off-diagonal.
1062 // The calling code in gwbse.cc reads QPdiag from DiagonalizeQPHamiltonian,
1063 // but for QSGW the "Hamiltonian" IS already diagonal in the converged basis.
1064 // We set Sigma_c_ so that getHQP() returns the correct H_QSGW:
1065 // H_QSGW = diag(e_DFT - v_xc) + tilde_Sigma
1066 // which is what Sigma_x_ + Sigma_c_ - vxc_.diagonal() + e_DFT gives
1067 // when Sigma_c_ is set appropriately. The cleanest approach is to
1068 // store the full off-diagonal tilde_Sigma in Sigma_c_ and set Sigma_x_
1069 // to zero for the final iteration -- but that would break getGWAResults.
1070 // Instead we leave Sigma_x_ and Sigma_c_ as-is from the last iteration
1071 // and note that DiagonalizeQPHamiltonian() called by gwbse.cc will
1072 // produce the correct converged eigensystem from getHQP().
1073
1074 // Clear QSGW rotation from sigma and rpa so subsequent G0W0/evGW/BSE calls
1075 // (if any) operate in the standard DFT-MO basis.
1076 sigma_->setQSGWRotation(nullptr, 0, 0);
1077 rpa_.setQSGWRotation(nullptr, 0, 0);
1078
1079 // If the virtual window was trimmed, merge converged QSGW energies for
1080 // [qpmin, qsgw_qpmax] with perturbative seed energies for (qsgw_qpmax,
1081 // qpmax]. Pad rotation matrix with identity for excluded virtuals so that
1082 // getQSGWRotation() returns a qptotal_ x qptotal_ matrix for BSE hookup.
1083 if (window_trimmed) {
1084 const Index n_excluded = qptotal_ - qsgw_qptotal;
1085 XTP_LOG(Log::error, log_)
1086 << TimeStamp() << " QSGW: merging converged energies [" << opt_.qpmin
1087 << "," << qsgw_qpmax << "] with perturbative seed energies for "
1088 << n_excluded << " excluded virtual(s) [" << qsgw_qpmax + 1 << ","
1089 << opt_.qpmax << "]" << std::flush;
1090
1091 // Pad rotation with identity for excluded virtual levels
1092 Eigen::MatrixXd U_full = Eigen::MatrixXd::Identity(qptotal_, qptotal_);
1093 U_full.topLeftCorner(qsgw_qptotal, qsgw_qptotal) = qsgw_rotation_;
1094 qsgw_rotation_ = U_full;
1095
1096 // Build merged energy vector: QSGW for trimmed window, seed for rest.
1097 // Store in qsgw_final_energies_ so getGWAResults() returns the correct
1098 // full-window result without needing to recompute from sigma matrices.
1099 Eigen::VectorXd e_merged = e_qp_full;
1100 e_merged.head(qsgw_qptotal) = e_qp;
1101 qsgw_final_energies_ = e_merged;
1102
1103 // Update RPA with merged energies so RPAInputEnergies() is consistent.
1104 rpa_.UpdateRPAInputEnergies(dft_energies_, e_merged, opt_.qpmin);
1105
1106 // Restore Sigma_x_ and Sigma_c_ to full qptotal_ x qptotal_ size so that
1107 // any downstream code doesn't encounter a size mismatch with
1108 // vxc_.diagonal() (which has qptotal_ elements). getGWAResults() bypasses
1109 // these matrices via qsgw_final_energies_, so zeroing them is safe.
1110 Sigma_x_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
1111 Sigma_c_ = Eigen::MatrixXd::Zero(qptotal_, qptotal_);
1112
1113 // Restore full-size sigma configuration so CalcCorrelationDiag (called
1114 // after the loop for diagonal Sigma_c_ output) works correctly.
1115 Sigma_base::options sigma_opt_full;
1116 sigma_opt_full.homo = opt_.homo;
1117 sigma_opt_full.qpmin = opt_.qpmin;
1118 sigma_opt_full.qpmax = opt_.qpmax;
1119 sigma_opt_full.rpamin = opt_.rpamin;
1120 sigma_opt_full.rpamax = opt_.rpamax;
1121 sigma_opt_full.eta = opt_.eta;
1122 sigma_opt_full.quadrature_scheme = opt_.quadrature_scheme;
1123 sigma_opt_full.order = opt_.order;
1124 sigma_opt_full.alpha = opt_.alpha;
1125 sigma_->configure(sigma_opt_full);
1126 }
1127
1128 XTP_LOG(Log::error, log_)
1129 << TimeStamp() << " QSGW loop complete." << std::flush;
1130}

1135 Eigen::VectorXd frequencies =
1136 rpa_.getRPAInputEnergies().segment(opt_.qpmin - opt_.rpamin, qptotal_);
