94 template <typename Rhs, typename Dest>
95 void _solve_impl(const Rhs& b, Dest& x) const {
96 // Segments own disjoint slices of x, so they are independent.
97#pragma omp parallel for schedule(dynamic, 4)
98 for (Index nn = 0; nn < Index(ids_.size()); ++nn) {
99 const std::size_t n = std::size_t(nn);
100 const Index base = offsets_[n];
101 const Index width = offsets_[n + 1] - base;
102 // Single-site segments hold a 3x3 LDLT of getPInv(), so no special
103 // case is needed.
104 x.segment(base, width) = factorizations_[n].solve(b.segment(base, width));
105 }
106 }

123 void BuildOffsets() {
124 offsets_.reserve(ids_.size() + 1);
125 offsets_.push_back(0);
126 for (Index id : ids_) {
127 if (!registry_->Has(id, EwaldChargeState::Neutral)) {
128 throw std::runtime_error(
129 "EwaldBlockJacobiPreconditioner: segment id not registered at "
130 "EwaldChargeState::Neutral");
131 }
132 Index n_sites = registry_->Get(id, EwaldChargeState::Neutral).size();
133 offsets_.push_back(offsets_.back() + 3 * n_sites);
134 }
135 size_ = offsets_.back();
136 }
137
138 // Builds and factorizes the local block of every segment. The
139 // intramolecular coupling enters with the sign it has in the operator
140 // A = P^-1 - C (subtracted), i.e. opposite to the field accumulated by
141 // AddIntraSegmentCoupling.
142 void FactorizeBlocks() {
143 // Blocks are independent and factorized in parallel.
144 factorizations_.resize(ids_.size());
145#pragma omp parallel for schedule(dynamic, 4)
146 for (Index nn = 0; nn < Index(ids_.size()); ++nn) {
147 const std::size_t n = std::size_t(nn);
148 const PolarSegment& segment =
149 registry_->Get(ids_[n], EwaldChargeState::Neutral);
150 const Index n_sites = segment.size();
151 const Index width = 3 * n_sites;
152 Eigen::MatrixXd block = Eigen::MatrixXd::Zero(width, width);
153
154 for (Index s = 0; s < n_sites; ++s) {
155 block.block<3, 3>(3 * s, 3 * s) = segment[s].getPInv();
156 }
157
158 if (n_sites >= 2) {
159 for (Index i = 0; i < n_sites; ++i) {
160 const PolarSite& site_i = segment[i];
161 for (Index j = i + 1; j < n_sites; ++j) {
162 const PolarSite& site_j = segment[j];
163 const Eigen::Vector3d r_vec = site_i.getPos() - site_j.getPos();
164 const double r = r_vec.norm();
165 // The operator couples an intramolecular pair through its
166 // explicit term l*B_erfc + (l-1)*B_erf plus the B_erf part of
167 // the reciprocal sum, in total the Thole-damped bare coupling
168 // l*(B_erfc + B_erf), which is used here.
169 const EwaldRealSpaceInteractor::BFunctions b =
170 intra_interactor_->ComputeB(r);
171 const EwaldRealSpaceInteractor::BFunctions berf =
172 intra_interactor_->ComputeErfB(r);
173 const EwaldRealSpaceInteractor::TholeFactors t =
174 intra_interactor_->ComputeThole(r, site_j, site_i);
175 const Eigen::Matrix3d coupling =
176 t.l5 * (b.B2 + berf.B2) * (r_vec * r_vec.transpose()) -
177 t.l3 * (b.B1 + berf.B1) * Eigen::Matrix3d::Identity();
178 // Subtracted, as in A = P^-1 - C. With the wrong sign
179 // (P^-1 + C_intra) a 5000-site solve needed 29 instead of 16
180 // PCG iterations.
181 block.block<3, 3>(3 * i, 3 * j) -= coupling;
182 block.block<3, 3>(3 * j, 3 * i) -= coupling.transpose();
183 }
184 }
185 }
186
187 factorizations_[n].compute(block);
188 }
189 }
