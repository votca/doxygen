160 template <typename Rhs, typename Dest>
161 void _solve_impl(const Rhs& b, Dest& x) const {
162 // Segments own disjoint slices of x, so they are independent.
163#pragma omp parallel for schedule(dynamic, 4)
164 for (Index nn = 0; nn < Index(ids_.size()); ++nn) {
165 const std::size_t n = std::size_t(nn);
166 const Index base = offsets_[n];
167 const Index width = offsets_[n + 1] - base;
168 // LDLT::solve works identically regardless of block size -- no
169 // special case needed for single-site (width==3) segments; their
170 // own factorizations_[n] already holds a plain 3x3 LDLT of
171 // getPInv() alone (see FactorizeBlocks), the same result
172 // DiagonalPreconditioner itself would give them.
173 x.segment(base, width) = factorizations_[n].solve(b.segment(base, width));
174 }
175 }

192 void BuildOffsets() {
193 offsets_.reserve(ids_.size() + 1);
194 offsets_.push_back(0);
195 for (Index id : ids_) {
196 if (!registry_->Has(id, EwaldChargeState::Neutral)) {
197 throw std::runtime_error(
198 "EwaldBlockJacobiPreconditioner: segment id not registered at "
199 "EwaldChargeState::Neutral");
200 }
201 Index n_sites = registry_->Get(id, EwaldChargeState::Neutral).size();
202 offsets_.push_back(offsets_.back() + 3 * n_sites);
203 }
204 size_ = offsets_.back();
205 }

207 // Builds and factorizes every segment's own local block, matching
208 // EwaldPeriodicDipoleOperator's own intramolecular term: the same
209 // B-function/ComputeThole calls, and the same sign it enters the
210 // OPERATOR with -- i.e. subtracted, since RawMultiply forms
211 // A = P^-1 - C. Note this is the opposite sign to
212 // AddIntraSegmentCoupling's own internal accumulation, which builds
213 // the coupling field itself (a positive quantity) that RawMultiply
214 // then subtracts. An earlier version of this comment claimed to
215 // mirror that method's sign convention "exactly", and the code did --
216 // which is precisely why this was wrong, and why it survived the
217 // operator's own sign fix without being noticed. Compare against the
218 // operator's assembled block, not against AddIntraSegmentCoupling in
219 // isolation.
220 void FactorizeBlocks() {
221 // One independent block per segment: built and factorized in
222 // parallel, each into its own slot.
223 factorizations_.resize(ids_.size());
224#pragma omp parallel for schedule(dynamic, 4)
225 for (Index nn = 0; nn < Index(ids_.size()); ++nn) {
226 const std::size_t n = std::size_t(nn);
227 const PolarSegment& segment =
228 registry_->Get(ids_[n], EwaldChargeState::Neutral);
229 const Index n_sites = segment.size();
230 const Index width = 3 * n_sites;
231 Eigen::MatrixXd block = Eigen::MatrixXd::Zero(width, width);
232
233 for (Index s = 0; s < n_sites; ++s) {
234 block.block<3, 3>(3 * s, 3 * s) = segment[s].getPInv();
235 }
236
237 if (n_sites >= 2) {
238 for (Index i = 0; i < n_sites; ++i) {
239 const PolarSite& site_i = segment[i];
240 for (Index j = i + 1; j < n_sites; ++j) {
241 const PolarSite& site_j = segment[j];
242 const Eigen::Vector3d r_vec = site_i.getPos() - site_j.getPos();
243 const double r = r_vec.norm();
244 const EwaldRealSpaceInteractor::BFunctions b =
245 intra_interactor_->ComputeB(r);
246 const EwaldRealSpaceInteractor::TholeFactors t =
247 intra_interactor_->ComputeThole(r, site_j, site_i);
248 const Eigen::Matrix3d coupling =
249 t.l5 * b.B2 * (r_vec * r_vec.transpose()) -
250 t.l3 * b.B1 * Eigen::Matrix3d::Identity();
251 // BUG FIX (this session): MINUS, not plus. The operator
252 // being preconditioned is A = P^-1 - C: RawMultiply builds
253 // the intramolecular term with exactly the `coupling`
254 // expression above and then subtracts it (result -= intra).
255 // This class assembled it with a plus, so it was
256 // factorizing P^-1 + C_intra -- the operator as it stood
257 // BEFORE the coupling-sign fix, which this file was written
258 // against and which never propagated here.
259 //
260 // The consequence was not a wrong answer (a preconditioner
261 // cannot change the fixed point, only the path to it) but a
262 // markedly worse one: on a real 5000-site solve this took
263 // PCG from 16 iterations to 29, with a visibly
264 // non-monotonic p.A.p curvature trace, versus a smoothly
265 // decreasing one unpreconditioned.
266 block.block<3, 3>(3 * i, 3 * j) -= coupling;
267 block.block<3, 3>(3 * j, 3 * i) -= coupling.transpose();
268 }
269 }
270 }
271
272 factorizations_[n].compute(block);
273 }
274 }
