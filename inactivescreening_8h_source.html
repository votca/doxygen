50struct InactiveSpace {
51 Eigen::MatrixXd occupied; // AO x n_inactive
52 Eigen::VectorXd occupied_energies;
53 Eigen::MatrixXd virtuals; // AO x n_virtual
54 Eigen::VectorXd virtual_energies;
55 // max |Q^2 - Q| of the projector of the inactive density onto the
56 // occupied space; 0 if the inactive density is made of occupied orbitals
57 double idempotency_error = 0.0;
58
59 Index size() const { return occupied.cols(); }
60 bool empty() const { return occupied.cols() == 0; }
61};

77struct InactiveScreeningInput {
78 enum class Mode { None, Block, Exact };
79 Mode mode = Mode::None;

82 Eigen::MatrixXd virtuals;
83 Eigen::VectorXd virtual_energies;
84 Index n_inactive = 0;
85 double idempotency_error = 0.0;
86 // block mode with a spatial cutoff: inactive orbitals left out
87 Index n_far = 0;
88 double radius = 0.0; // bohr; 0: all
89 // complete G: the inactive occupied orbitals (semicanonical) also enter
90 // the self-energy; they are put in front of the active ones
91 // (InactiveScreening::CompleteGOrbitals) and excluded from the RPA
92 bool complete_g = false;
93 Eigen::MatrixXd inactive_occupied;
94 Eigen::VectorXd inactive_energies;
95
96 bool empty() const { return mode == Mode::None; }
97 static Mode Parse(const std::string& name);
98 static std::string Name(Mode mode);
99};
