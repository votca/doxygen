66 public:

69 // Registers `segment` under (id, state). Overwrites any existing entry
70 // for the same (id, state) pair. `id` is expected to match the stable
71 // segment id used elsewhere in the topology (Segment::getId()); it is
72 // not required to match segment.getId() and is kept as an explicit
73 // argument so the same PolarSegment content could in principle be
74 // registered under a different id if that is ever useful, though in
75 // normal use the two will agree.
76 void Register(Index id, EwaldChargeState state, PolarSegment segment);
77
78 // True if an entry exists for (id, state).
79 bool Has(Index id, EwaldChargeState state) const;
80
81 // Returns the stored segment for (id, state). Throws std::out_of_range
82 // if no such entry exists; callers are expected to check Has() first
83 // when the presence of a given state is not already guaranteed by
84 // construction.
85 const PolarSegment& Get(Index id, EwaldChargeState state) const;
86 PolarSegment& Get(Index id, EwaldChargeState state);
87
88 // Removes the entry for (id, state), if present. No-op otherwise.
89 void Erase(Index id, EwaldChargeState state);
90
91 // Every segment id that has at least one registered state.
92 std::vector<Index> AllIds() const;
93
94 // Every state currently registered for `id`. Empty if `id` is unknown.
95 std::vector<EwaldChargeState> StatesFor(Index id) const;
96
97 Index size() const { return static_cast<Index>(store_.size()); }
98
99 void WriteToCpt(CheckpointWriter& w) const;
100 void ReadFromCpt(CheckpointReader& r);
101
102 private:
103 using Key = std::pair<Index, EwaldChargeState>;
104
105 struct KeyLess {
106 bool operator()(const Key& a, const Key& b) const {
107 if (a.first != b.first) {
108 return a.first < b.first;
109 }
110 return static_cast<int>(a.second) < static_cast<int>(b.second);
111 }
112 };
113
114 std::map<Key, PolarSegment, KeyLess> store_;
115};
