348 QPFunc(Index gw_level, const Sigma_base& sigma, double offset)

351 std::pair<double, double> operator()(double frequency) const {
352 std::pair<double, double> result;
353 result.first = value(frequency, EvalStage::Other);
354 result.second = deriv(frequency);
355 return result;
356 }
357
358 double sigma(double frequency, EvalStage stage = EvalStage::Other) const {
359 const std::uint64_t key = FrequencyKey(frequency);
360
361 auto insert_result = seen_frequencies_.insert(key);
362 if (!insert_result.second) {
363 ++stats_.sigma_repeat_calls;
364 } else {
365 ++stats_.sigma_unique_frequencies;
366 }
367
368 CountSigmaStage(stage);
369 auto cached = cache_.find(key);
370 if (cached != cache_.end()) {
371 return cached->second;
372 }
373 return sigma_c_func_.CalcCorrelationDiagElement(gw_level_, frequency);
374 }
375
378 void Prefetch(const std::vector<double>& frequencies) const {
379 std::vector<double> todo;
380 for (double f : frequencies) {
381 if (cache_.find(FrequencyKey(f)) == cache_.end()) {
382 todo.push_back(f);
383 }
384 }
385 if (todo.empty()) {
386 return;
387 }
388 const Eigen::VectorXd values = sigma_c_func_.CalcCorrelationDiagElements(
389 gw_level_,
390 Eigen::Map<const Eigen::VectorXd>(todo.data(), Index(todo.size())));
391 for (std::size_t i = 0; i < todo.size(); ++i) {
392 cache_[FrequencyKey(todo[i])] = values(Index(i));
393 }
394 }
395
396 double value(double frequency, EvalStage stage = EvalStage::Other) const {
397 return sigma(frequency, stage) + offset_ - frequency;
398 }
399
400 double deriv(double frequency) const {
401 ++stats_.deriv_calls;
402 return sigma_c_func_.CalcCorrelationDiagElementDerivative(gw_level_,
403 frequency) -
404 1.0;
405 }
406
407 const QPStats& GetStats() const { return stats_; }
408
409 private:
410 static std::uint64_t FrequencyKey(double x) {
411 std::uint64_t key = 0;
412 static_assert(sizeof(double) == sizeof(std::uint64_t),
413 "Unexpected double size");
414 std::memcpy(&key, &x, sizeof(double));
415 return key;
416 }
417
418 void CountSigmaStage(EvalStage stage) const {
419 switch (stage) {
420 case EvalStage::Scan:
421 ++stats_.sigma_scan_calls;
422 break;
423 case EvalStage::Refine:
424 ++stats_.sigma_refine_calls;
425 break;
426 case EvalStage::Derivative:
427 ++stats_.sigma_derivative_calls;
428 break;
429 case EvalStage::Other:
430 default:
431 ++stats_.sigma_other_calls;
432 break;
433 }
434 }
