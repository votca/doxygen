56int dsyevd(char jobz, char uplo, long n, double* a, long lda, double* w) {
57 if (__builtin_available(macOS 13.3, *)) {
58 const __LAPACK_int N = static_cast<__LAPACK_int>(n);
59 const __LAPACK_int LDA = static_cast<__LAPACK_int>(std::max(1L, lda));
60 __LAPACK_int info = 0;
61 // workspace query
62 double work_query = 0.0;
63 __LAPACK_int iwork_query = 0;
64 __LAPACK_int lwork = -1;
65 __LAPACK_int liwork = -1;
66 dsyevd_(&jobz, &uplo, &N, a, &LDA, w, &work_query, &lwork, &iwork_query,
67 &liwork, &info);
68 if (info != 0) {
69 return static_cast<int>(info);
70 }
71 lwork = static_cast<__LAPACK_int>(work_query);
72 liwork = iwork_query;
73 std::vector<double> work(
74 static_cast<std::size_t>(std::max<__LAPACK_int>(1, lwork)));
75 std::vector<__LAPACK_int> iwork(
76 static_cast<std::size_t>(std::max<__LAPACK_int>(1, liwork)));
77 dsyevd_(&jobz, &uplo, &N, a, &LDA, w, work.data(), &lwork, iwork.data(),
78 &liwork, &info);
79 return static_cast<int>(info);
80 }
81 return kAccelerateUnavailable;
82}
