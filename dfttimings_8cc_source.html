29void DFTTimings::Report(Logger& log, Log::Level level) const {
30 const double total = Elapsed();
31 double accounted = 0.0;
32 XTP_LOG(level, log) << " DFT timing summary (wall clock)" << std::flush;
33 XTP_LOG(level, log) << (boost::format(" %-38s %7s %11s %10s %6s") % "part" %
34 "calls" % "time [s]" % "per call" % "%")
35 .str()
36 << std::flush;
37 for (const Entry& e : entries_) {
38 accounted += e.seconds;
39 XTP_LOG(level, log) << (boost::format(" %-38s %7d %11.2f %10.3f %6.1f") %
40 e.name % e.calls % e.seconds %
41 (e.seconds / double(e.calls)) %
42 (total > 0 ? 100.0 * e.seconds / total : 0.0))
43 .str()
44 << std::flush;
45 }
46 XTP_LOG(level, log) << (boost::format(" %-38s %7s %11.2f %10s %6.1f") %
47 "not itemised" % "" % (total - accounted) % "" %
48 (total > 0 ? 100.0 * (total - accounted) / total
49 : 0.0))
50 .str()
51 << std::flush;
52 XTP_LOG(level, log)
53 << (boost::format(" %-38s %7s %11.2f") % "total" % "" % total).str()
54 << std::flush;
55}

57double DFTTimings::ResidentMemoryGB(bool peak) {
58 std::ifstream status("/proc/self/status");
59 if (!status.is_open()) {
60 return -1.0;
61 }
62 const std::string key = peak ? "VmHWM:" : "VmRSS:";
63 std::string word;
64 while (status >> word) {
65 if (word == key) {
66 double kb = 0.0;
67 status >> kb;
68 return kb / (1024.0 * 1024.0);
69 }
70 }
71 return -1.0;
72}
