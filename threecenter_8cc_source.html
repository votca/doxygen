35void TCMatrix_gwbse::Initialize(Index basissize, Index mmin, Index mmax,
36 Index nmin, Index nmax) {
37
38 // here as storage indices starting from zero
39 nmin_ = nmin;
40 nmax_ = nmax;
41 ntotal_ = nmax - nmin + 1;
42 mmin_ = mmin;
43 mmax_ = mmax;
44 mtotal_ = mmax - mmin + 1;
45 auxbasissize_ = basissize;
46
47 // vector has mtotal elements
48 // largest object should be allocated in multithread fashion
49 matrix_ = std::vector<Eigen::MatrixXd>(mtotal_);
50#pragma omp parallel for schedule(dynamic, 4)
51 for (Index i = 0; i < mtotal_; i++) {
52 matrix_[i] = Eigen::MatrixXd::Zero(ntotal_, auxbasissize_);
53 }
54}

56/*
57 * Modify 3-center matrix elements consistent with use of symmetrized
58 * Coulomb interaction using either CUDA or Openmp.
59 */
60void TCMatrix_gwbse::MultiplyRightWithAuxMatrix(const Eigen::MatrixXd& matrix) {
61 OpenMP_CUDA gemm;
62 gemm.setOperators(matrix_, matrix);
63#pragma omp parallel
64 {
65 Index threadid = OPENMP::getThreadId();
66#pragma omp for schedule(dynamic)
67 for (Index i = 0; i < msize(); i++) {
68 gemm.MultiplyRight(matrix_[i], threadid);
69 }
70 }
71 if (matrix.rows() != matrix.cols()) {
72 aux_frame_known_ = false; // a projection, not a change of frame
73 aux_frame_.resize(0, 0);
74 } else if (aux_frame_known_) {
75 aux_frame_ = (aux_frame_.size() == 0) ? matrix : aux_frame_ * matrix;
76 }
77}

79bool TCMatrix_gwbse::AuxFrameIsOrthogonal() const {
80 if (!aux_frame_known_) {
81 return false;
82 }
83 if (aux_frame_.size() == 0) {
84 return true;
85 }
86 const Index n = aux_frame_.rows();
87 return (aux_frame_.transpose() * aux_frame_ - Eigen::MatrixXd::Identity(n, n))
88 .cwiseAbs()
89 .maxCoeff() < 1e-10;
90}

92void TCMatrix_gwbse::MultiplyInFillFrame(const Eigen::MatrixXd& A) {
93 if (!aux_frame_known_) {
94 throw std::runtime_error(
95 "TCMatrix_gwbse: cannot dress the auxiliary index after a non-square "
96 "MultiplyRightWithAuxMatrix. Rebuild first.");
97 }
98 if (aux_frame_.size() == 0) {
99 MultiplyRightWithAuxMatrix(A);
100 return;
101 }
102 // M_fill F X = M_fill A F => X = F^-1 A F. The frame becomes A F.
103 const Eigen::PartialPivLU<Eigen::MatrixXd> lu(aux_frame_);
104 MultiplyRightWithAuxMatrix(lu.solve(A * aux_frame_));
105}

107void TCMatrix_gwbse::DressAuxIndex(const Eigen::MatrixXd& S) {
108 if (Dressed()) {
109 throw std::runtime_error(
110 "TCMatrix_gwbse::DressAuxIndex: already dressed. Undress first.");
111 }
112 if (S.rows() != auxbasissize_ || S.cols() != auxbasissize_) {
113 throw std::runtime_error(
114 "TCMatrix_gwbse::DressAuxIndex: S is " + std::to_string(S.rows()) +
115 "x" + std::to_string(S.cols()) + ", auxiliary basis has " +
116 std::to_string(auxbasissize_) + " functions.");
117 }
118 MultiplyInFillFrame(S);
119 dressing_ = S;
120}

122void TCMatrix_gwbse::UndressAuxIndex() {
123 if (!Dressed()) {
124 return;
125 }
126 MultiplyInFillFrame(dressing_.inverse());
127 dressing_.resize(0, 0);
128}

130Eigen::MatrixXd TCMatrix_gwbse::ToCurrentAuxFrame(
131 const Eigen::MatrixXd& K) const {
132 if (!aux_frame_known_) {
133 throw std::runtime_error(
134 "TCMatrix_gwbse::ToCurrentAuxFrame: the auxiliary index of the "
135 "three-centre integrals was multiplied by a non-square matrix, so "
136 "there is no frame to bring the kernel into. Apply it before that, "
137 "or Rebuild.");
138 }
139 if (aux_frame_.size() == 0) {
140 return K;
141 }
142 if (K.rows() != aux_frame_.rows() || K.cols() != aux_frame_.rows()) {
143 throw std::runtime_error(
144 "TCMatrix_gwbse::ToCurrentAuxFrame: kernel is " +
145 std::to_string(K.rows()) + "x" + std::to_string(K.cols()) +
146 ", auxiliary frame is " + std::to_string(aux_frame_.rows()) + "x" +
147 std::to_string(aux_frame_.cols()) + ".");
148 }
149 const Index n = aux_frame_.rows();
150 const double off_orthogonal =
151 (aux_frame_.transpose() * aux_frame_ - Eigen::MatrixXd::Identity(n, n))
152 .cwiseAbs()
153 .maxCoeff();
154 Eigen::MatrixXd result;
155 if (off_orthogonal < 1e-10) {
156 result = aux_frame_.transpose() * K * aux_frame_;
157 } else {
158 const Eigen::PartialPivLU<Eigen::MatrixXd> lu(aux_frame_);
159 const Eigen::MatrixXd X = lu.solve(K); // F^-1 K
160 result = lu.solve(Eigen::MatrixXd(X.transpose())).transpose();
161 }
162 return result;
163}
164/*
165 * Fill the 3-center object by looping over shells of GW basis set and
166 * calling FillBlock, which calculates all 3-center overlap integrals
167 * associated to a particular shell, convoluted with the DFT orbital
168 * coefficients
169 */
170void TCMatrix_gwbse::Fill(const AOBasis& auxbasis, const AOBasis& dftbasis,
171 const Eigen::MatrixXd& dft_orbitals) {
172 // needed for Rebuild())
173 auxbasis_ = &auxbasis;
174 dftbasis_ = &dftbasis;
175 dft_orbitals_ = &dft_orbitals;
176
177 Fill3cMO(auxbasis, dftbasis, dft_orbitals);
178
179 AOOverlap auxoverlap;
180 auxoverlap.Fill(auxbasis);
181 AOCoulomb auxcoulomb;
182 auxcoulomb.Fill(auxbasis);
183 inv_sqrt_ = auxcoulomb.Pseudo_InvSqrt_GWBSE(auxoverlap, metric_tolerance);
184 removedfunctions_ = auxcoulomb.Removedfunctions();
185 MultiplyRightWithAuxMatrix(inv_sqrt_);
186 // What Fill leaves behind is the reference frame by definition.
187 aux_frame_.resize(0, 0);
188 aux_frame_known_ = true;
189 dressing_.resize(0, 0);
190
191 return;
192}

194// =============================================================================
195// TCMatrix_gwbse::Rotate
196//
197// Rotates only the n-index (inner rows) of ALL m-slices for the QP window.
198//
199// In QSGW the self-energy matrix element indices (outer m) stay in the
200// DFT-MO basis. Only the construction sum indices (inner n-rows) need to be
201// in the QP wavefunction basis. This method applies:
202//
203// new_M[m].middleRows(qp_offset_n, qptotal) =
204// U^T * old_M[m].middleRows(qp_offset_n, qptotal)
205//
206// for ALL m-slices in the full RPA range (because every sigma calculation
207// uses every m-slice as a matrix element index and needs updated n-rows).
208// Rows outside the QP window remain as DFT-MOs.
209// =============================================================================
210void TCMatrix_gwbse::Rotate(const Eigen::MatrixXd& U, Index qpmin,
211 Index qpmax) {
212 const Index qptotal = qpmax - qpmin + 1;
213 const Index qp_offset_n = qpmin - nmin_; // row offset in n-storage
214 const Index qp_offset_m = qpmin - mmin_; // slice offset in m-storage
215
216 assert(qpmin >= nmin_ && qpmax <= nmax_);
217 assert(qpmin >= mmin_ && qpmax <= mmax_);
218 assert(U.rows() == qptotal && U.cols() == qptotal);
219
220 // Rotate the n-rows of ONLY the QP-window m-slices [qpmin, qpmax].
221 // Slices outside this range (e.g. core levels below qpmin, or high virtuals
222 // above qpmax) are always DFT-MOs and must NOT be touched -- they are not
223 // saved/restored by Mmn_orig in gw.cc and would accumulate drift if rotated.
224 // Only the QP-window n-row block [qp_offset_n, qp_offset_n+qptotal) is
225 // rotated; rows outside remain as DFT-MOs (consistent with evGW treatment).
226 // new_rows = U^T * old_rows
227#pragma omp parallel for schedule(dynamic)
228 for (Index m = 0; m < qptotal; m++) {
229 matrix_[m + qp_offset_m].middleRows(qp_offset_n, qptotal) =
230 U.transpose() *
231 matrix_[m + qp_offset_m].middleRows(qp_offset_n, qptotal);
232 }
233}

235void TCMatrix_dft::SetupLayout(const AOBasis& dftbasis,
236 const std::vector<std::vector<Index>>& kept) {
237 basissize_ = dftbasis.AOBasisSize();
238 const std::vector<Index> shell2bf = dftbasis.getMapToBasisFunctions();
239 runs_.clear();
240 col_offset_.assign(basissize_, 0);
241 Index offset = 0;
242 for (Index a = 0; a < Index(kept.size()); ++a) {
243 const Index start = shell2bf[a];
244 const Index size = dftbasis.getShell(a).getNumFunc();
245 for (Index i = 0; i < size; ++i) {
246 const Index mu = start + i;
247 col_offset_[mu] = offset;
248 // merge neighbouring kept shells into one run of rows
249 Index k = 0;
250 while (k < Index(kept[a].size())) {
251 Index b_first = kept[a][k];
252 Index b_last = b_first;
253 while (k + 1 < Index(kept[a].size()) && kept[a][k + 1] == b_last + 1) {
254 ++k;
255 b_last = kept[a][k];
256 }
257 ++k;
258 const Index row = shell2bf[b_first];
259 const Index end = std::min(
260 shell2bf[b_last] + dftbasis.getShell(b_last).getNumFunc(), mu + 1);
261 runs_.push_back({mu, row, end - row, offset});
262 offset += end - row;
263 }
264 }
265 }

268void TCMatrix_dft::FillFullMatrix(Index P, Eigen::MatrixXd& full) const {
269 assert(full.rows() == basissize_ && full.cols() == basissize_);
270 const double* src = data_.col(P).data();
271 for (const Run& r : runs_) {
272 std::copy(src + r.offset, src + r.offset + r.length,
273 full.data() + r.col * basissize_ + r.row);
274 }
275 full.triangularView<Eigen::StrictlyLower>() = full.transpose();
276}

278Eigen::MatrixXd TCMatrix_dft::FullMatrix(Index P) const {
279 Eigen::MatrixXd full = Eigen::MatrixXd::Zero(basissize_, basissize_);
280 FillFullMatrix(P, full);
281 return full;
282}

284Eigen::VectorXd TCMatrix_dft::PackWeighted(const Eigen::MatrixXd& full) const {
285 assert(full.rows() == basissize_ && full.cols() == basissize_);
286 Eigen::VectorXd packed(data_.rows());
287 for (const Run& r : runs_) {
288 for (Index k = 0; k < r.length; ++k) {
289 const Index row = r.row + k;
290 packed(r.offset + k) =
291 (row == r.col) ? full(r.col, row) : 2.0 * full(r.col, row);
292 }
293 }
294 return packed;
295}

297Eigen::MatrixXd TCMatrix_dft::Unpack(const Eigen::VectorXd& packed) const {
298 assert(packed.size() == data_.rows());
299 Eigen::MatrixXd full = Eigen::MatrixXd::Zero(basissize_, basissize_);
300 for (const Run& r : runs_) {
301 full.block(r.row, r.col, r.length, 1) = packed.segment(r.offset, r.length);
302 }
303 full.triangularView<Eigen::StrictlyLower>() = full.transpose();
304 return full;
305}
