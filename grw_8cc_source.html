38void GRW::ParseOptions(const tools::Property& options) {
39 carriertype_ = QMStateType(options.get(".carriertype").as<std::string>());
40 if (!carriertype_.isKMCState()) {
41 throw std::runtime_error("grw cannot be run for state: " +
42 carriertype_.ToLongString());
43 }
44 temperature_ = options.get(".temperature").as<double>() * tools::conv::kB *
45 tools::conv::ev2hrt;
46 field_Vm_ = options.get(".field").as<Eigen::Vector3d>();
47 // V/m -> Hartree/bohr
48 field_ = field_Vm_ * (tools::conv::ev2hrt / (1e9 * tools::conv::nm2bohr));
49 tof_ = ToFGraph::OptionsFromProperty(options.get(".tof"));
50 carriers_ = options.get(".numberofcarriers").as<Index>();
51 observable_ = GraphRandomWalk::ObservableFromString(
52 options.get(".observable").as<std::string>());
53 settings_.max_states = options.get(".maxstates").as<Index>();
54 solver_ = GraphRandomWalk::SolverFromString(
55 options.get(".solver").as<std::string>());
56 settings_.direct_limit = options.get(".directlimit").as<Index>();
57 settings_.tolerance = options.get(".tolerance").as<double>();
58 settings_.max_iterations = options.get(".maxiterations").as<Index>();
59 sourcefile_ = options.get(".sourcefile").as<std::string>();
60 occupationfile_ = options.get(".occupationfile").as<std::string>();
61 resultfile_ = options.get(".resultfile").as<std::string>();
62 log_.setReportLevel(Log::current_level);
63 log_.setMultithreading(true);
64 log_.setCommonPreface("\n ...");
65}

67bool GRW::Evaluate(Topology& top) {
68 XTP_LOG(Log::error, log_) << "\n-----------------------------------"
69 "\n GRAPH RANDOM WALK TIME OF FLIGHT"
70 "\n-----------------------------------"
71 << std::flush;
72 if (carriers_ != 1 && carriers_ != 2) {
73 throw std::runtime_error(
74 "grw: numberofcarriers must be 1 or 2 (the state space grows as "
75 "n^N/N!; use kinetic Monte Carlo for more carriers)");
76 }
77 ToFGraph graph(top, tof_, carriertype_, temperature_, field_);
78 XTP_LOG(Log::error, log_) << graph.Summary() << std::flush;
79 for (Index s : graph.DisconnectedSources()) {
80 XTP_LOG(Log::info, log_)
81 << "source segment " << s << " has no path to a sink, excluded"
82 << std::flush;
83 }
84
85 GraphRandomWalk grw(graph);
86 grw.setSettings(settings_);
87 GraphRandomWalk::Result result = (carriers_ == 1)
88 ? grw.SingleCarrier(solver_)
89 : grw.TwoCarriers(observable_, solver_);
90 bool transit =
91 carriers_ == 2 && observable_ == GraphRandomWalk::Observable::Transit;
92 if (carriers_ == 2) {
93 XTP_LOG(Log::error, log_)
94 << "Two carriers with site exclusion, observable: "
95 << (transit ? "mean transit time per carrier (carriers removed at "
96 "the sink)"
97 : "time until the first carrier reaches the sink")
98 << std::flush;
99 }
100
101 XTP_LOG(Log::error, log_)
102 << "Solver: " << result.solver << ", " << result.unknowns << " unknowns, "
103 << result.iterations << " iterations, backward error "
104 << result.backward_error << " after " << result.refinement_steps
105 << " refinement steps" << std::flush;
106 if (!result.converged) {
107 XTP_LOG(Log::error, log_)
108 << "Warning: not converged (iteration limit or backward error above "
109 "1e-10)"
110 << std::flush;
111 }
112 XTP_LOG(Log::error, log_)
113 << std::setprecision(6) << "Time of flight (uniform injection over "
114 << result.sources.size() << " source "
115 << (carriers_ == 1 ? "sites" : "pairs") << "): mean " << result.tau_mean
116 << " s" << std::flush;
117 if (!transit) {
118 XTP_LOG(Log::error, log_) << "Standard deviation of the hitting time: "
119 << result.tau_std_injection << " s" << std::flush;
120 }
121 XTP_LOG(Log::error, log_)
122 << "Harmonic mean over the injection states: " << result.tau_harmonic
123 << " s" << std::flush;
124
125 double Fn = field_Vm_.dot(graph.Normal()); // V/m along the normal
126 double L = graph.Length() * tools::conv::bohr2nm * 1e-9; // m
127 if (Fn != 0.0) {
128 double charge = 0.0;
129 if (carriertype_ == QMStateType::Hole) {
130 charge = 1.0;
131 } else if (carriertype_ == QMStateType::Electron) {
132 charge = -1.0;
133 }
134 if (charge * Fn <= 0.0) {
135 XTP_LOG(Log::error, log_)
136 << "Warning: the field does not drive the carriers towards the sink"
137 << std::flush;
138 }
139 double factor = 1e4 / std::abs(Fn); // m^2/Vs -> cm^2/Vs
140 XTP_LOG(Log::error, log_)
141 << "Mobility L/(tau F): " << L / result.tau_mean * factor
142 << " cm^2/Vs (mean), " << L / result.tau_harmonic * factor
143 << " cm^2/Vs (harmonic), with L = " << L * 1e9 << " nm and F = " << Fn
144 << " V/m along the normal" << std::flush;
145 }
146
147 if (!resultfile_.empty()) {
148 ResultFile out("grw: mean hitting times of the sink (graph random walk)");
149 out.Add("carriers", result.carriers);
150 out.Add("observable",
151 transit ? std::string("transit") : std::string("first"));
152 out.Add("unknowns", result.unknowns);
153 out.Add("sources", Index(result.sources.size()));
154 out.Add("tau_mean", result.tau_mean, "s");
155 out.Add("tau_harmonic", result.tau_harmonic, "s");
156 if (!transit) {
157 out.Add("tau_std", result.tau_std_injection, "s");
158 }
159 out.Add("length", L, "m");
160 out.Add("field_normal", Fn, "V/m");
161 if (Fn != 0.0) {
162 out.Add("mobility_mean", L / result.tau_mean * 1e4 / std::abs(Fn),
163 "cm^2/Vs");
164 out.Add("mobility_harmonic", L / result.tau_harmonic * 1e4 / std::abs(Fn),
165 "cm^2/Vs");
166 }
167 out.Add("backward_error", result.backward_error, "-");
168 out.Add("converged", Index(result.converged));
169 out.Write(resultfile_);
170 }
171 if (!sourcefile_.empty()) {
172 WriteSources(graph, result);
173 }
174 if (!occupationfile_.empty()) {
175 WriteOccupation(graph, result);
176 }
177 std::cout << log_;
178 return true;
179}

181void GRW::WriteSources(const ToFGraph& graph,
182 const GraphRandomWalk::Result& result) const {
183 std::ofstream out(sourcefile_);
184 out << "# per injection state: for each carrier segment id, x along the "
185 "normal [nm], site energy [eV]; then the mean time [s] and, if "
186 "available, the standard deviation of the hitting time [s]\n";
187 out << std::setprecision(10);
188 for (Index k = 0; k < Index(result.sources.size()); ++k) {
189 for (Index s : result.sources[k]) {
190 const ToFGraph::Site& site = graph.getSite(s);
191 out << site.segment << " " << site.x * tools::conv::bohr2nm << " "
192 << site.energy * tools::conv::hrt2ev << " ";
193 }
194 out << result.tau[k];
195 if (result.tau_std.size() == result.tau.size()) {
196 out << " " << result.tau_std[k];
197 }
198 out << "\n";
199 }
200}

202void GRW::WriteOccupation(const ToFGraph& graph,
203 const GraphRandomWalk::Result& result) const {
204 std::ofstream out(occupationfile_);
205 out << "# expected time each segment is occupied before the carriers reach "
206 "the sink, uniform injection\n";
207 out << "# segment id, x along the normal [nm], site energy [eV], role, "
208 "time [s], fraction of the total occupation time\n";
209 out << std::setprecision(10);
210 const char* roles[] = {"bulk", "source", "sink"};
211 for (const ToFGraph::Site& site : graph.Sites()) {
212 double t = result.occupation_time[site.segment];
213 out << site.segment << " " << site.x * tools::conv::bohr2nm << " "
214 << site.energy * tools::conv::hrt2ev << " "
215 << roles[static_cast<int>(site.role)] << " " << t << " "
216 << t / (double(result.carriers) * result.tau_mean) << "\n";
217 }
218}
