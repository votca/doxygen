173 void Fill(const AOBasis& auxbasis, const AOBasis& dftbasis,
174 const Eigen::MatrixXd& dft_orbitals);
175 // Rebuilds ThreeCenterIntegrals, only works if the original basisobjects
176 // still exist
177 void Rebuild() { Fill(*auxbasis_, *dftbasis_, *dft_orbitals_); }
178
179 void MultiplyRightWithAuxMatrix(const Eigen::MatrixXd& matrix);
180
181 // The metric that was folded into the stored integrals at Fill time:
182 // T = Pseudo_InvSqrt_GWBSE, with T T^T = V^-1 (pseudo-inverse) and
183 // T^T V T the projector onto the retained auxiliary functions. After
184 // Fill, operator[] holds M = (mn|Q) T, so the bare Coulomb operator
185 // exists only implicitly, as M M^T.
186 //
187 // Kept because anything that must act on the Coulomb interaction in
188 // the same space as M needs it. The environment reaction field is the
189 // case in point: a kernel B between auxiliary functions, taken as charge
190 // densities, enters as M (T^T B T) M^T -- see EnvironmentScreening.
191 //
192 // IN THE FILL-TIME AUXILIARY BASIS. MultiplyRightWithAuxMatrix, which the
193 // PPM and the BSE both call with an eigenvector matrix U, rotates the
194 // auxiliary index of M. This matrix is not rotated with it: a kernel
195 // built from it has to be brought into the current frame with
196 // ToCurrentAuxFrame before it meets operator[]. Refilled, and so reset,
197 // by every Fill and Rebuild.
198 const Eigen::MatrixXd& InvSqrt() const { return inv_sqrt_; }
199
200 // The auxiliary frame operator[] is currently in. Every
201 // MultiplyRightWithAuxMatrix since the last Fill or Rebuild is recorded
202 // here, so that operator[] holds M_fill * AuxFrame() with M_fill the
203 // integrals as Fill left them. Empty means the identity: nothing has
204 // rotated them yet.
205 //
206 // The bare Coulomb interaction does not care -- it is the identity in
207 // every orthonormal frame, which is why nothing tracked this before. A
208 // kernel K between the columns of M_fill does care: M_fill K M_fill^T is
209 // M K' M^T only with K' = ToCurrentAuxFrame(K).
210 const Eigen::MatrixXd& AuxFrame() const { return aux_frame_; }
211
212 // For code that snapshots slices through operator[] and later writes
213 // them back, which bypasses the bookkeeping above: restore the frame
214 // that was current when the snapshot was taken, alongside the slices.
215 void RestoreAuxFrame(const Eigen::MatrixXd& frame) {
216 aux_frame_ = frame;
217 aux_frame_known_ = true;
218 }
219
220 // K, a kernel between the columns of M_fill, as it must be applied to
221 // operator[] now: F^-1 K F^-T with F = AuxFrame(), which is F^T K F for
222 // the orthogonal eigenvector matrices the PPM and the BSE rotate with.
223 // Throws if the frame is not known -- after a non-square
224 // MultiplyRightWithAuxMatrix, which is not a change of frame.
225 Eigen::MatrixXd ToCurrentAuxFrame(const Eigen::MatrixXd& K) const;
226
227 // Whether AuxFrame() is orthogonal, so that the bare interaction M M^T is
228 // what it was at Fill time. False once the index has been dressed.
229 bool AuxFrameIsOrthogonal() const;
230
249 void DressAuxIndex(const Eigen::MatrixXd& S);
250 // Back to M_fill (times whatever rotations happened since), bare v.
251 void UndressAuxIndex();
252 bool Dressed() const { return dressing_.size() > 0; }
253
272 void Rotate(const Eigen::MatrixXd& U, Index qpmin, Index qpmax);
273
274 private:
275 // store vector of matrices
276 std::vector<Eigen::MatrixXd> matrix_;

278 // band summation indices
279 Index mmin_;
280 Index mmax_;
281 Index nmin_;
282 Index nmax_;
