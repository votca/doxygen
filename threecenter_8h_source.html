71 public:

101 const Eigen::MatrixXd& dft_orbitals);
102 // Rebuilds ThreeCenterIntegrals, only works if the original basisobjects
103 // still exist
104 void Rebuild() { Fill(*auxbasis_, *dftbasis_, *dft_orbitals_); }

106 void MultiplyRightWithAuxMatrix(const Eigen::MatrixXd& matrix);
107
108 // The metric that was folded into the stored integrals at Fill time:
109 // T = Pseudo_InvSqrt_GWBSE, with T T^T = V^-1 (pseudo-inverse) and
110 // T^T V T the projector onto the retained auxiliary functions. After
111 // Fill, operator[] holds M = (mn|Q) T, so the bare Coulomb operator
112 // exists only implicitly, as M M^T.
113 //
114 // Kept because anything that must act on the Coulomb interaction in
115 // the same space as M needs it. The environment reaction field is the
116 // case in point: a kernel B between auxiliary functions, taken as charge
117 // densities, enters as M (T^T B T) M^T -- see EnvironmentScreening.
118 //
119 // IN THE FILL-TIME AUXILIARY BASIS. MultiplyRightWithAuxMatrix, which the
120 // PPM and the BSE both call with an eigenvector matrix U, rotates the
121 // auxiliary index of M. This matrix is not rotated with it: a kernel
122 // built from it has to be brought into the current frame with
123 // ToCurrentAuxFrame before it meets operator[]. Refilled, and so reset,
124 // by every Fill and Rebuild.
125 const Eigen::MatrixXd& InvSqrt() const { return inv_sqrt_; }
126
127 // The auxiliary frame operator[] is currently in. Every
128 // MultiplyRightWithAuxMatrix since the last Fill or Rebuild is recorded
129 // here, so that operator[] holds M_fill * AuxFrame() with M_fill the
130 // integrals as Fill left them. Empty means the identity: nothing has
131 // rotated them yet.
132 //
133 // The bare Coulomb interaction does not care -- it is the identity in
134 // every orthonormal frame, which is why nothing tracked this before. A
135 // kernel K between the columns of M_fill does care: M_fill K M_fill^T is
136 // M K' M^T only with K' = ToCurrentAuxFrame(K).
137 const Eigen::MatrixXd& AuxFrame() const { return aux_frame_; }
138
139 // For code that snapshots slices through operator[] and later writes
140 // them back, which bypasses the bookkeeping above: restore the frame
141 // that was current when the snapshot was taken, alongside the slices.
142 void RestoreAuxFrame(const Eigen::MatrixXd& frame) {
143 aux_frame_ = frame;
144 aux_frame_known_ = true;
145 }
146
147 // K, a kernel between the columns of M_fill, as it must be applied to
148 // operator[] now: F^-1 K F^-T with F = AuxFrame(), which is F^T K F for
149 // the orthogonal eigenvector matrices the PPM and the BSE rotate with.
150 // Throws if the frame is not known -- after a non-square
151 // MultiplyRightWithAuxMatrix, which is not a change of frame.
152 Eigen::MatrixXd ToCurrentAuxFrame(const Eigen::MatrixXd& K) const;
153
154 // Whether AuxFrame() is orthogonal, so that the bare interaction M M^T is
155 // what it was at Fill time. False once the index has been dressed.
156 bool AuxFrameIsOrthogonal() const;
157
176 void DressAuxIndex(const Eigen::MatrixXd& S);
177 // Back to M_fill (times whatever rotations happened since), bare v.
178 void UndressAuxIndex();
179 bool Dressed() const { return dressing_.size() > 0; }
180
199 void Rotate(const Eigen::MatrixXd& U, Index qpmin, Index qpmax);
200
201 private:
202 // store vector of matrices
203 std::vector<Eigen::MatrixXd> matrix_;
204
205 // band summation indices
206 Index mmin_;
207 Index mmax_;
208 Index nmin_;
209 Index nmax_;
210 Index ntotal_;
211 Index mtotal_;

213

215 bool aux_frame_known_ = true;
216 Eigen::MatrixXd dressing_; // S of DressAuxIndex; empty: bare
217
218 // M_fill -> M_fill A, whatever frame the integrals are in.
219 void MultiplyInFillFrame(const Eigen::MatrixXd& A);

223 const Eigen::MatrixXd* dft_orbitals_ = nullptr;

225 void Fill3cMO(const AOBasis& auxbasis, const AOBasis& dftbasis,
226 const Eigen::MatrixXd& dft_orbitals);
