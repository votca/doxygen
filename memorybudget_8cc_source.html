37// A "<key> <n> kB" entry of /proc/self/status in bytes, -1 if not there
38double ProcStatusBytes(const std::string& key) {
39 std::ifstream status("/proc/self/status");
40 std::string word;
41 while (status >> word) {
42 if (word == key) {
43 double kb = 0.0;
44 status >> kb;
45 return kb * 1024.0;
46 }
47 }
48 return -1.0;
49}

69double MemoryBudget::ResidentBytes() {
70#if defined(__APPLE__)
71 mach_task_basic_info_data_t info;
72 mach_msg_type_number_t count = MACH_TASK_BASIC_INFO_COUNT;
73 if (task_info(mach_task_self(), MACH_TASK_BASIC_INFO,
74 reinterpret_cast<task_info_t>(&info), &count) == KERN_SUCCESS) {
75 return double(info.resident_size);
76 }
77 return -1.0;
78#else
79 return ProcStatusBytes("VmRSS:");
80#endif
81}

97std::string MemoryBudget::Usage() {
98 const double resident = ResidentBytes();
99 if (resident < 0.0) {
100 return "";
101 }
102 std::ostringstream out;
103 out.precision(4);
104 out << "memory in use " << resident * 1e-9 << " GB";
105 const double peak = PeakResidentBytes();
106 if (peak >= 0.0) {
107 out << ", peak " << peak * 1e-9 << " GB";
108 }
109 return out.str();
110}

112double MemoryBudget::AvailableBytes() {
113 if (!Given()) {
114 return DefaultAllowanceBytes();
115 }
116 const double share = JobBudgetBytes();
117 const double resident = std::max(0.0, ResidentBytes());
118 const double margin = std::max(1e9, 0.05 * share);
119 return std::max(0.0, share - resident / double(concurrent_jobs_) - margin);
120}

122std::string MemoryBudget::Describe() {
123 std::ostringstream out;
124 out.precision(3);
125 if (!Given()) {
126 out << "no --memory given, " << DefaultAllowanceBytes() * 1e-9
127 << " GB for optional buffers";
128 return out.str();
129 }
130 out << "budget " << JobBudgetBytes() * 1e-9 << " GB";
131 if (concurrent_jobs_ > 1) {
132 out << " per job (" << total_bytes_ * 1e-9 << " GB for " << concurrent_jobs_
133 << " jobs)";
134 }
135 const double resident = ResidentBytes();
136 if (resident >= 0.0) {
137 out << ", " << resident * 1e-9 << " GB in use";
138 }
139 return out.str();
140}
