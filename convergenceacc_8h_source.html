61 // Independent from adiis_start deliberately -- see
62 // UKSConvergenceAcc::Iterate's own comment on where this is used
63 // for the full reasoning: ORCA's own DampErr is kept fully
64 // separate from its own DIISStart, and their own guidance for
65 // difficult systems is to make DampErr much SMALLER than default
66 // (keeping damping active LONGER), independent of when DIIS
67 // itself engages. Reusing adiis_start for both purposes could not
68 // represent that independently.
69 double mixingend;
70 Index numberofelectrons;
71 double mixingparameter;
72 // Ceiling mixingparameter can adaptively ramp up toward as the SCF
73 // struggles, rather than staying fixed at mixingparameter (the
74 // BASE/starting value) for the whole run -- matches ORCA's own
75 // static-damping design directly (confirmed from a real ORCA log's
76 // own resolved SCF settings, not the manual's generic defaults):
77 // DampFac (the base, 0.7 by default) and DampMax (the ceiling, 0.98
78 // by default) are two separate parameters there, not one fixed
79 // value. Notably, 0.98 is exactly the value this session's own
80 // hand-tuning independently landed on for a difficult water dimer
81 // case -- this adaptive design is a more principled way to obtain
82 // that same benefit only when actually needed, rather than paying
83 // its cost (slower convergence on easy iterations) for an entire
84 // run regardless of whether the system is struggling at all.
85 double mixingmax;
86 double Econverged;
87 double error_converged;
88 Index number_alpha_electrons = 0;
89 Index number_beta_electrons = 0;
90 // Maximum iterations for the Davidson eigensolver used by
91 // CoupledAugmentedHessianStep's own direct-minimization fallback --
92 // NOT CDFT-specific, since that fallback can engage for any
93 // sufficiently difficult UKS SCF (see this field's own XML help
94 // text, dftpackage.xml, for the real case that motivated exposing
95 // this at all: a strong CDFT constraint over a large fragment left
96 // the solver still short of its own convergence tolerance at the
97 // previous, hardcoded default of 50).
98 Index davidson_max_iter = 50;
99 // Second-order (TRAH) SCF steps: "auto" switches to them when (A)DIIS
100 // stalls, "always" uses them from the start, "off" never.
101 std::string second_order = "auto";
102 // Energy rise (Hartree) above the lowest energy reached so far at which
103 // the SCF discards its extrapolation history and restarts from the
104 // lowest-energy density. 0 disables.

122 void Configure(const ConvergenceAcc::options& opt) {
123 opt_ = opt;
124 if (opt_.mode == KSmode::closed) {
125 nocclevels_ = opt_.numberofelectrons / 2;
126 } else if (opt_.mode == KSmode::open) {
127 nocclevels_ = opt_.numberofelectrons;
128 } else if (opt_.mode == KSmode::fractional) {
129 nocclevels_ = 0;
130 } else if (opt_.mode == KSmode::restricted_open) {
131 nocclevels_ =
132 std::max(opt_.number_alpha_electrons, opt_.number_beta_electrons);
133 }
134 diis_.setHistLength(opt_.histlength);
135 trah_options_.max_iterations = opt_.davidson_max_iter;
136 StartNewSCF();
137 }
138
139 // Forget the lowest-energy point used by the energy-rise reset. Called
140 // at the start of every SCF: successive SCFs on one accelerator (e.g.
141 // the stages of DFT-in-DFT embedding) need not share an energy scale.
142 // The extrapolation history itself is left as it was.
143 void StartNewSCF() {
144 have_best_ = false;
145 energy_resets_ = 0;
146 iterations_since_reset_ = kResetCooldown;
147 trah_active_ = false;
148 trah_pending_ = false;
149 trah_disabled_ = false;
150 last_rejected_ = false;
151 trah_subspace_.Clear();
152 trust_ = trah::TrustRadius(kInitialTrustRadius);
153 consecutive_failures_ = 0;
154 error_history_.clear();
155 scf_iterations_ = 0;
156 }

180 bool isConverged() const {
181 if (totE_.size() < 2) {
182 return false;
183 } else {
184 // not on the call that undid a rejected second-order step: totE and
185 // the DIIS error then belong to the restored, not the returned, state
186 return !last_rejected_ && std::abs(getDeltaE()) < opt_.Econverged &&
187 getDIIsError() < opt_.error_converged;
188 }
189 }

192 double getDeltaE() const {
193 if (totE_.size() < 2) {
194 return 0;
195 } else {
196 return totE_.back() - totE_[totE_.size() - 2];
197 }
