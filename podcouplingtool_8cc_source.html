29void PodCouplingTool::ParseOptions(const tools::Property& user_options) {
30
31 log_.setReportLevel(Log::current_level);
32 log_.setMultithreading(true);
33 log_.setCommonPreface("\n...");
34
35 tools::Property options = user_options;
36
37 orb_file_ = options.get(".orb_file").as<std::string>();
38
39 // Atom index lists in IndexParser syntax (e.g. "1 3 13:17"), as for
40 // cdft.indices and the fragments in diabatization.xml.
41 std::string fragment_A_str = options.get(".fragment_A").as<std::string>();
42 std::string fragment_B_str = options.get(".fragment_B").as<std::string>();
43 if (fragment_A_str.empty() || fragment_B_str.empty()) {
44 throw std::runtime_error(
45 "PodCouplingTool: fragment_A and fragment_B must both be given, "
46 "e.g. '0:5' and '6:11' (same syntax as cdft.indices).");
47 }
48 fragment_A_atoms_ = IndexParser().CreateIndexVector(fragment_A_str);
49 fragment_B_atoms_ = IndexParser().CreateIndexVector(fragment_B_str);
50
51 // Number of occupied and virtual levels per fragment; option names and
52 // meaning as in DFTcoupling (levA/levB).
53 numberofstatesA_ = options.ifExistsReturnElseReturnDefault<Index>("levA", 1);
54 numberofstatesB_ = options.ifExistsReturnElseReturnDefault<Index>("levB", 1);
55
56 XTP_LOG(Log::error, log_) << "Fragment A: " << fragment_A_atoms_.size()
57 << " atoms, levA=" << numberofstatesA_ << flush;
58 XTP_LOG(Log::error, log_) << "Fragment B: " << fragment_B_atoms_.size()
59 << " atoms, levB=" << numberofstatesB_ << flush;
60
61 // Optional cube-file export; grid options as in gencube.xml.
62 write_cube_files_ =
63 options.ifExistsReturnElseReturnDefault<bool>("write_cube_files", false);
64 if (write_cube_files_) {
65 cube_steps_.x() =
66 options.ifExistsReturnElseReturnDefault<Index>("xsteps", 25);
67 cube_steps_.y() =
68 options.ifExistsReturnElseReturnDefault<Index>("ysteps", 25);
69 cube_steps_.z() =
70 options.ifExistsReturnElseReturnDefault<Index>("zsteps", 25);
71 cube_padding_ =
72 options.ifExistsReturnElseReturnDefault<double>("padding", 6.5);
73 }
74}

76bool PodCouplingTool::Run() {
77
78 OPENMP::setMaxThreads(nThreads_);
79
80 log_.setReportLevel(Log::current_level);
81 log_.setMultithreading(true);
82 log_.setCommonPreface("\n...");
83
84 XTP_LOG(Log::error, log_)
85 << TimeStamp() << " Reading orbitals from: " << orb_file_ << flush;
86
87 Orbitals orbitals;
88 orbitals.ReadFromCpt(orb_file_);
89
90 PODCoupling pod(orbitals, &log_, fragment_A_atoms_, fragment_B_atoms_);
91 pod.CalculateCouplings(numberofstatesA_, numberofstatesB_);
92
93 Index homoA = pod.getFragmentAHomoIndex();
94 Index lumoA = pod.getFragmentALumoIndex();
95 Index homoB = pod.getFragmentBHomoIndex();
96 Index lumoB = pod.getFragmentBLumoIndex();
97
98 // Full coupling matrix over the requested range, as in DFTcoupling,
99 // labeled by absolute fragment orbital indices.
100 XTP_LOG(Log::error, log_) << TimeStamp()
101 << " Full coupling matrix [eV] (fragment A orbital "
102 "index -> fragment B orbital index):"
103 << flush;
104 for (Index levelA = homoA - numberofstatesA_ + 1;
105 levelA <= lumoA + numberofstatesA_ - 1; ++levelA) {
106 std::ostringstream row;
107 row << " A[" << levelA << (levelA == homoA ? "=HOMO" : "")
108 << (levelA == lumoA ? "=LUMO" : "") << "]:";
109 for (Index levelB = homoB - numberofstatesB_ + 1;
110 levelB <= lumoB + numberofstatesB_ - 1; ++levelB) {
111 double coupling_ev =
112 pod.getCouplingElement(levelA, levelB) * votca::tools::conv::hrt2ev;
113 row << boost::format(" B[%1%]=%2$+1.6f") % levelB % coupling_ev;
114 }
115 XTP_LOG(Log::error, log_) << row.str() << flush;
116 }
117
118 // HOMO-HOMO (hole) and LUMO-LUMO (electron) couplings separately.
119 double homo_homo_ev =
120 pod.getCouplingElement(homoA, homoB) * votca::tools::conv::hrt2ev;
121 double lumo_lumo_ev =
122 pod.getCouplingElement(lumoA, lumoB) * votca::tools::conv::hrt2ev;
123 XTP_LOG(Log::error, log_)
124 << boost::format("POD2 Coupling, HOMO-HOMO (hole): %1$+1.12f eV") %
125 homo_homo_ev
126 << flush;
127 XTP_LOG(Log::error, log_)
128 << boost::format("POD2 Coupling, LUMO-LUMO (electron): %1$+1.12f eV") %
129 lumo_lumo_ev
130 << flush;
131
132 // Cube files of the computed fragment orbitals. CubeFile_Writer only
133 // reads orb.MOs().eigenvectors().col(index), so each orbital, embedded
134 // in the full AO basis (PODCoupling::GetFragmentOrbital), is stored as
135 // the single MO of a copy of orbitals and written as in gencube.cc.
136 if (write_cube_files_) {
137 XTP_LOG(Log::error, log_)
138 << TimeStamp()
139 << " Writing cube files for the computed fragment "
140 "orbitals..."
141 << flush;
142 CubeFile_Writer writer(cube_steps_, cube_padding_, log_);
143 auto WriteFragmentOrbitalCube = [&](bool fragment_A, Index level,
144 const std::string& label) {
145 Eigen::VectorXd orbital = pod.GetFragmentOrbital(fragment_A, level);
146 Orbitals orbitals_copy = orbitals;
147 orbitals_copy.MOs().eigenvectors() = orbital;
148 orbitals_copy.MOs().eigenvalues() = Eigen::VectorXd::Zero(1);
149 std::string filename =
150 "podcoupling_" + label + "_orb" + std::to_string(level) + ".cube";
151 writer.WriteFile(filename, orbitals_copy,
152 QMState(QMStateType::KSstate, 0, false), false);
153 XTP_LOG(Log::error, log_) << " wrote " << filename << flush;
154 // Dominant AOs of the orbital, e.g. to identify which AO's tail
155 // causes density in the region of the other fragment.
156 XTP_LOG(Log::error, log_)
157 << pod.DescribeFragmentOrbitalComposition(fragment_A, level) << flush;
158 };
159 for (Index levelA = homoA - numberofstatesA_ + 1;
160 levelA <= lumoA + numberofstatesA_ - 1; ++levelA) {
161 WriteFragmentOrbitalCube(true, levelA, "fragA");
162 }
163 for (Index levelB = homoB - numberofstatesB_ + 1;
164 levelB <= lumoB + numberofstatesB_ - 1; ++levelB) {
165 WriteFragmentOrbitalCube(false, levelB, "fragB");
166 }
167 }
168
169 std::cout << log_;
170 // Trailing newline: xtp_tools does not print one after Evaluate(), so
171 // the shell prompt would follow the last output line directly.
172 std::cout << std::endl;
173
174 return true;
175}
