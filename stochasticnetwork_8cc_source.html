44ModelRandom::ModelRandom(std::uint64_t seed, std::uint64_t stream) {
45 // std::seed_seq and mt19937_64 are fully specified by the standard
46 std::seed_seq seq{
47 std::uint32_t(seed & 0xffffffffu), std::uint32_t(seed >> 32),
48 std::uint32_t(stream & 0xffffffffu), std::uint32_t(stream >> 32)};
49 engine_.seed(seq);
50}

52double ModelRandom::Normal() {
53 if (has_spare_) {
54 has_spare_ = false;
55 return spare_;
56 }
57 double r = std::sqrt(-2.0 * std::log(Uniform()));
58 double phi = 2.0 * tools::conv::Pi * Uniform();
59 spare_ = r * std::sin(phi);
60 has_spare_ = true;
61 return r * std::cos(phi);
62}

64Index ModelRandom::Poisson(double mean) {
65 if (!(mean >= 0.0) || !std::isfinite(mean)) {
66 throw std::runtime_error("ModelRandom::Poisson: mean must be >= 0");
67 }
68 // sum of Poisson variables with means <= 500 (Knuth's product of uniforms)
69 Index n = 0;
70 while (mean > 0.0) {
71 double m = std::min(mean, 500.0);
72 mean -= m;
73 const double limit = std::exp(-m);
74 double p = Uniform();
75 while (p > limit) {
76 ++n;
77 p *= Uniform();
78 }
79 }
80 return n;
81}

83Index ModelRandom::Integer(Index n) {
84 if (n <= 0) {
85 throw std::runtime_error("ModelRandom::Integer: n must be positive");
86 }
87 const std::uint64_t un = std::uint64_t(n);
88 const std::uint64_t limit = std::numeric_limits<std::uint64_t>::max() -
89 std::numeric_limits<std::uint64_t>::max() % un;
90 std::uint64_t x;
91 do {
92 x = engine_();
93 } while (x >= limit);
94 return Index(x % un);
95}

99PeriodicCells::PeriodicCells(const Eigen::Vector3d& box, double radius,
100 const std::vector<Eigen::Vector3d>& points)
101 : box_(box), points_(points) {
102 if (!(box.minCoeff() > 0.0)) {
103 throw std::runtime_error("PeriodicCells: box lengths must be positive");
104 }
105 if (!(radius > 0.0) || !(2.0 * radius < box.minCoeff())) {
106 throw std::runtime_error(
107 "PeriodicCells: the search radius must be positive and smaller than "
108 "half the box");
109 }
110 for (Index d = 0; d < 3; ++d) {
111 n_[d] = std::max(Index(1), Index(std::floor(box[d] / radius)));
112 }
113 // not many more cells than points (larger cells are fine for Within)
114 while (double(n_[0]) * double(n_[1]) * double(n_[2]) >
115 2.0 * double(points.size()) + 27.0) {
116 Index d = Index(std::max_element(n_.begin(), n_.end()) - n_.begin());
117 n_[d] = std::max(Index(1), n_[d] / 2);
118 }
119 for (Index d = 0; d < 3; ++d) {
120 cell_[d] = box[d] / double(n_[d]);
121 }
122 Index ncells = n_[0] * n_[1] * n_[2];
123 std::vector<Index> cell_of(points.size());
124 start_.assign(ncells + 1, 0);
125 std::array<Index, 3> c;
126 for (Index i = 0; i < Index(points.size()); ++i) {
127 cell_of[i] = CellIndex(points[i], c);
128 ++start_[cell_of[i] + 1];
129 }
130 for (Index k = 0; k < ncells; ++k) {
131 start_[k + 1] += start_[k];
132 }
133 members_.resize(points.size());
134 std::vector<Index> fill(start_.begin(), start_.end() - 1);
135 for (Index i = 0; i < Index(points.size()); ++i) {
136 members_[fill[cell_of[i]]++] = i;
137 }
138}
139
140Index PeriodicCells::CellIndex(const Eigen::Vector3d& p,
141 std::array<Index, 3>& c) const {
142 for (Index d = 0; d < 3; ++d) {
143 double x = p[d] - box_[d] * std::floor(p[d] / box_[d]);
144 Index k = Index(std::floor(x / cell_[d]));
145 c[d] = std::min(std::max(k, Index(0)), n_[d] - 1);
146 }
147 return (c[0] * n_[1] + c[1]) * n_[2] + c[2];
148}

150template <class F>
151void PeriodicCells::Shell(const std::array<Index, 3>& c, Index s, F&& f) const {
152 // offsets -lo..hi cover every cell exactly once also for few cells
153 std::array<Index, 3> lo, hi;
154 for (Index d = 0; d < 3; ++d) {
155 lo[d] = std::min(s, (n_[d] - 1) / 2);
156 hi[d] = std::min(s, n_[d] - 1 - (n_[d] - 1) / 2);
157 }
158 for (Index o0 = -lo[0]; o0 <= hi[0]; ++o0) {
159 for (Index o1 = -lo[1]; o1 <= hi[1]; ++o1) {
160 for (Index o2 = -lo[2]; o2 <= hi[2]; ++o2) {
161 if (std::max({std::abs(o0), std::abs(o1), std::abs(o2)}) != s) {
162 continue;
163 }
164 Index k0 = (c[0] + o0 + n_[0]) % n_[0];
165 Index k1 = (c[1] + o1 + n_[1]) % n_[1];
166 Index k2 = (c[2] + o2 + n_[2]) % n_[2];
167 f((k0 * n_[1] + k1) * n_[2] + k2);
168 }
169 }
170 }

173Eigen::Vector3d PeriodicCells::Connection(const Eigen::Vector3d& r1,
174 const Eigen::Vector3d& r2) const {
175 Eigen::Vector3d d = r2 - r1;
176 for (Index k = 0; k < 3; ++k) {
177 // points in the box: one image shift at most
178 if (d[k] > 0.5 * box_[k]) {
179 d[k] -= box_[k];
180 } else if (d[k] < -0.5 * box_[k]) {
181 d[k] += box_[k];
182 }
183 if (std::abs(d[k]) > 0.5 * box_[k]) {
184 d[k] -= box_[k] * std::round(d[k] / box_[k]);
185 }
186 }
187 return d;
188}

190void PeriodicCells::Within(
191 const Eigen::Vector3d& p, double radius,
192 const std::function<void(Index, const Eigen::Vector3d&, double)>& f) const {
193 if (radius > cell_.minCoeff() * (1.0 + 1e-12)) {
194 throw std::runtime_error(
195 "PeriodicCells::Within: radius larger than the cell list radius");
196 }
197 std::array<Index, 3> c;
198 CellIndex(p, c);
199 for (Index s = 0; s <= 1; ++s) {
200 Shell(c, s, [&](Index cell) {
201 for (Index k = start_[cell]; k < start_[cell + 1]; ++k) {
202 Index j = members_[k];
203 Eigen::Vector3d conn = Connection(p, points_[j]);
204 double dist = conn.norm();
205 if (dist < radius) {
206 f(j, conn, dist);
207 }
208 }
209 });
210 }
211}

217std::vector<Index> PeriodicCells::NearestTo(const Eigen::Vector3d& p,
218 Index l) const {
219 const Index n = Index(points_.size());
220 if (l > n) {
221 throw std::runtime_error("PeriodicCells::Nearest: more neighbours (" +
222 std::to_string(l) + ") than points (" +
223 std::to_string(n) + ")");
224 }
225 if (l <= 0) {
226 return {};
227 }
228 std::array<Index, 3> c;
229 CellIndex(p, c);
230 std::vector<std::pair<double, Index>> cand;
231 const double inf = std::numeric_limits<double>::infinity();
232 for (Index s = 0;; ++s) {
233 Shell(c, s, [&](Index cell) {
234 for (Index k = start_[cell]; k < start_[cell + 1]; ++k) {
235 Index j = members_[k];
236 cand.emplace_back(Connection(p, points_[j]).norm(), j);
237 }
238 });
239 // every point outside the shells 0..s is farther than `guaranteed`
240 double guaranteed = inf;
241 bool all = true;
242 for (Index d = 0; d < 3; ++d) {
243 Index full = std::max((n_[d] - 1) / 2, n_[d] - 1 - (n_[d] - 1) / 2);
244 if (s < full) {
245 all = false;
246 guaranteed = std::min(guaranteed, double(s) * cell_[d]);
247 }
248 }
249 if (all) {
250 break;
251 }
252 if (Index(cand.size()) >= l &&
253 std::count_if(cand.begin(), cand.end(),
254 [guaranteed](const std::pair<double, Index>& x) {
255 return x.first <= guaranteed;
256 }) >= l) {
257 break;
258 }
259 }
260 std::partial_sort(cand.begin(), cand.begin() + l, cand.end());
261 std::vector<Index> result(l);
262 for (Index k = 0; k < l; ++k) {
263 result[k] = cand[k].second;
264 }
265 return result;
266}

270StochasticNetwork::StochasticNetwork(const Options& opt) : opt_(opt) {
271 Check();
272 ModelRandom points(opt_.seed, 0);
273 iterations_ = SamplePoints(opt_.points, opt_.box, opt_.sites, points,
274 positions_, radii_);
275 ModelRandom energies(opt_.seed, 1);
276 GenerateEnergies(energies);
277 ModelRandom traps(opt_.seed, 3);
278 InsertTraps(traps);
279 ModelRandom pairs(opt_.seed, 2);
280 GeneratePairs(pairs);
281}

283void StochasticNetwork::Check() const {
284 auto fail = [](const std::string& m) {
285 throw std::runtime_error("StochasticNetwork: " + m);
286 };
287 if (!(opt_.box.minCoeff() > 0.0)) {
288 fail("box lengths must be positive");
289 }
290 if (opt_.sites <= 0) {
291 fail("need at least one site");
292 }
293 const PointProcess& pp = opt_.points;
294 if (!(pp.hard_core >= 0.0) || !(pp.radius_mean >= 0.0) ||
295 !(pp.radius_variance >= 0.0) || !(pp.candidates > 0.0) ||
296 pp.max_iterations <= 0) {
297 fail(
298 "point process: hard core, radius mean and variance must be >= 0, "
299 "candidates and iterations > 0");
300 }
301 if (pp.radius_variance > 0.0 && pp.radius_mean == 0.0) {
302 fail("point process: a radius variance needs a positive radius mean");
303 }
304 if (opt_.connection_r.empty() ||
305 opt_.connection_r.size() != opt_.connection_p.size()) {
306 fail("connection table: r and p need the same, nonzero length");
307 }
308 for (std::size_t k = 0; k < opt_.connection_r.size(); ++k) {
309 if (!(opt_.connection_p[k] >= 0.0 && opt_.connection_p[k] <= 1.0)) {
310 fail("connection probabilities must be in [0, 1]");
311 }
312 if (!(opt_.connection_r[k] >= 0.0) ||
313 (k > 0 && !(opt_.connection_r[k] > opt_.connection_r[k - 1]))) {
314 fail("connection distances must be >= 0 and ascending");
315 }
316 }
317 if (!(2.0 * opt_.connection_r.back() < opt_.box.minCoeff())) {
318 fail("the largest connection distance must be smaller than half the box");
319 }
320 if (opt_.carriers.empty()) {
321 fail("need at least one carrier");
322 }
323 for (std::size_t c = 0; c < opt_.carriers.size(); ++c) {
324 const Carrier& car = opt_.carriers[c];
325 if (!car.state.isKMCState()) {
326 fail(car.state.ToString() + " is not a hopping state (e, h, s, t)");
327 }
328 for (std::size_t d = 0; d < c; ++d) {
329 if (opt_.carriers[d].state == car.state) {
330 fail("carrier " + car.state.ToString() + " given twice");
331 }
332 }
333 const EnergyModel& e = car.energy;
334 double w_c = 1.0 - e.w_a - e.w_b;
335 if (!(e.sigma >= 0.0) || !(e.w_a >= 0.0) || !(e.w_b >= 0.0) ||
336 w_c < -1e-12) {
337 fail("energy model: sigma, w_a, w_b >= 0 and w_a + w_b <= 1");
338 }
339 if (e.l_b <= 0 || e.l_c <= 0 || (e.w_b > 0.0 && e.l_b > opt_.sites) ||
340 (w_c > 1e-12 && e.l_c > opt_.sites)) {
341 fail("energy model: l_b and l_c must be in [1, number of sites]");
342 }
343 const CouplingModel& m = car.coupling;
344 for (std::size_t k = 1; k < m.residual_quantiles.size(); ++k) {
345 if (m.residual_quantiles[k] < m.residual_quantiles[k - 1]) {
346 fail("coupling model: residual quantiles must be ascending");
347 }
348 }
349 }
350 if (!(opt_.trap_factor >= 0.0)) {
351 fail("the trap factor must be >= 0");
352 }
353 for (const Trap& t : opt_.traps) {
354 if (t.carrier < 0 || t.carrier >= Index(opt_.carriers.size()) ||
355 !(t.density >= 0.0) || t.offsets.empty()) {
356 fail("traps need a carrier, a density >= 0 and at least one site");
357 }
358 for (const auto& o : t.offsets) {
359 if (o.size() != opt_.carriers.size()) {
360 fail("trap offsets need one value per carrier");
361 }
362 }
363 }
364 if (!(std::abs(opt_.energy_correlation) <= 1.0) ||
365 !(std::abs(opt_.coupling_correlation) <= 1.0)) {
366 fail("correlations must be in [-1, 1]");
367 }
368}

370Index StochasticNetwork::SamplePoints(const PointProcess& pp,
371 const Eigen::Vector3d& box, Index n,
372 ModelRandom& random,
373 std::vector<Eigen::Vector3d>& positions,
374 std::vector<double>& radii) {
375 // radii by inversion of the Gamma distribution function: continuous in the
376 // parameters for the same random numbers (common random numbers in fits);
377 // tabulated quantiles, exact in the tails
378 const bool gamma = pp.radius_variance > 0.0;
379 const double shape =
380 gamma ? pp.radius_mean * pp.radius_mean / pp.radius_variance : 0.0;
381 const double scale = gamma ? pp.radius_variance / pp.radius_mean : 0.0;
382 constexpr Index M = 512;
383 std::vector<double> table;
384 if (gamma) {
385 table.resize(M + 1);
386 for (Index k = 1; k < M; ++k) {
387 table[k] = boost::math::gamma_p_inv(shape, double(k) / double(M));
388 }
389 }
390 auto radius = [&]() {
391 if (!gamma) {
392 return pp.hard_core + pp.radius_mean;
393 }
394 double u = random.Uniform();
395 double t = u * double(M);
396 Index k = Index(t);
397 double x;
398 if (k < 1 || k >= M - 1) {
399 x = boost::math::gamma_p_inv(shape, std::min(u, 1.0 - 1e-16));
400 } else {
401 x = table[k] + (t - double(k)) * (table[k + 1] - table[k]);
402 }
403 return pp.hard_core + x * scale;
404 };
405 const double half_box = 0.5 * box.minCoeff();
406 positions.clear();
407 radii.clear();
408 positions.reserve(n);
409 radii.reserve(n);
410 double max_accepted = 0.0;
411 std::vector<Eigen::Vector3d> cand;
412 std::vector<double> cand_r;
413 std::vector<char> keep;
414 std::vector<Index> survivors;
415 for (Index it = 1; it <= pp.max_iterations; ++it) {
416 Index m = random.Poisson(pp.candidates * double(n));
417 cand.resize(m);
418 cand_r.resize(m);
419 double max_cand = 0.0;
420 for (Index k = 0; k < m; ++k) {
421 for (Index d = 0; d < 3; ++d) {
422 cand[k][d] = (1.0 - random.Uniform()) * box[d];
423 }
424 cand_r[k] = radius();
425 max_cand = std::max(max_cand, cand_r[k]);
426 }
427 double search = std::max(max_cand, max_accepted);
428 if (!(search < half_box)) {
429 throw std::runtime_error("StochasticNetwork: point radii (" +
430 std::to_string(search * tools::conv::bohr2nm) +
431 " nm) must be smaller than half the box");
432 }
433 // a zero search radius means no exclusion
434 search = std::max(search, 1e-9 * half_box);
435 keep.assign(m, 1);
436 // against accepted points: distance >= max(R_candidate, R_accepted)
437 if (!positions.empty()) {
438 PeriodicCells acc(box, search, positions);
439 for (Index k = 0; k < m; ++k) {
440 acc.Within(cand[k], std::max(cand_r[k], max_accepted),
441 [&](Index j, const Eigen::Vector3d&, double dist) {
442 if (dist < std::max(cand_r[k], radii[j])) {
443 keep[k] = 0;
444 }
445 });
446 }
447 }
448 // dominance among all candidates of the iteration: inside the sphere of
449 // a larger point (equal radii: the earlier point wins)
450 {
451 PeriodicCells cells(box, std::max(max_cand, 1e-9 * half_box), cand);
452 for (Index k = 0; k < m; ++k) {
453 if (!keep[k]) {
454 continue; // removed already; it still dominates others
455 }
456 cells.Within(cand[k], std::max(max_cand, 1e-9 * half_box),
457 [&](Index j, const Eigen::Vector3d&, double dist) {
458 if (j == k) {
459 return;
460 }
461 bool larger = cand_r[j] > cand_r[k] ||
462 (j < k && cand_r[j] == cand_r[k]);
463 if (larger && dist < cand_r[j]) {
464 keep[k] = 0;
465 }
466 });
467 }
468 }
469 survivors.clear();
470 for (Index k = 0; k < m; ++k) {
471 if (keep[k]) {
472 survivors.push_back(k);
473 }
474 }
475 Index need = n - Index(positions.size());
476 if (Index(survivors.size()) > need) {
477 // random subset (partial Fisher-Yates)
478 for (Index k = 0; k < need; ++k) {
479 Index r = k + random.Integer(Index(survivors.size()) - k);
480 std::swap(survivors[k], survivors[r]);
481 }
482 survivors.resize(need);
483 }
484 for (Index k : survivors) {
485 positions.push_back(cand[k]);
486 radii.push_back(cand_r[k]);
487 max_accepted = std::max(max_accepted, cand_r[k]);
488 }
489 if (Index(positions.size()) == n) {
490 return it;
491 }
492 }
493 throw std::runtime_error(
494 "StochasticNetwork: the point process placed only " +
495 std::to_string(positions.size()) + " of " + std::to_string(n) +
496 " sites in " + std::to_string(pp.max_iterations) +
497 " iterations; the box is too small for the radii (increase the box or "
498 "max_iterations)");
499}

501std::vector<double> StochasticNetwork::MovingAverage(
502 const PeriodicCells& cells, const EnergyModel& m, Index sites,
503 const std::vector<double>& A, const std::vector<double>& Bf,
504 const std::vector<double>& C) {
505 if (Index(A.size()) != sites || Index(Bf.size()) != sites ||
506 Index(C.size()) != sites) {
507 throw std::runtime_error(
508 "StochasticNetwork::MovingAverage: need one field value per site");
509 }
510 const double w_c = std::max(0.0, 1.0 - m.w_a - m.w_b);
511 const double fa = std::sqrt(m.w_a);
512 const double fb = std::sqrt(m.w_b / double(m.l_b));
513 const double fc = std::sqrt(w_c / double(m.l_c));
514 const Index lb = m.w_b > 0.0 ? m.l_b : 0;
515 const Index lc = w_c > 0.0 ? m.l_c : 0;
516 const Index lmax = std::max(lb, lc);
517 std::vector<double> z(sites);
518#pragma omp parallel for schedule(dynamic, 256)
519 for (Index i = 0; i < sites; ++i) {
520 std::vector<Index> nb = cells.Nearest(i, lmax);
521 double b = 0.0;
522 for (Index k = 0; k < lb; ++k) {
523 b += Bf[nb[k]];
524 }
525 double c = 0.0;
526 for (Index k = 0; k < lc; ++k) {
527 c += C[nb[k]];
528 }
529 z[i] = fa * A[i] + fb * b + fc * c;
530 }
531 return z;
532}

534void StochasticNetwork::GenerateEnergies(ModelRandom& random) {
535 const Index n = NumberOfSites();
536 // cells of about eight points for the nearest-neighbour searches
537 double volume = opt_.box.prod();
538 double radius =
539 std::min(0.49 * opt_.box.minCoeff(), 2.0 * std::cbrt(volume / double(n)));
540 PeriodicCells cells(opt_.box, radius, positions_);
541 energies_.clear();
542 std::vector<double> first;
543 std::vector<double> A(n), B(n), C(n);
544 for (std::size_t c = 0; c < opt_.carriers.size(); ++c) {
545 const EnergyModel& m = opt_.carriers[c].energy;
546 // the fields are always drawn, so that the energies of one carrier do
547 // not depend on the model of another
548 for (std::vector<double>* f : {&A, &B, &C}) {
549 for (double& x : *f) {
550 x = random.Normal();
551 }
552 }
553 std::vector<double> z = MovingAverage(cells, m, n, A, B, C);
554 if (c == 0) {
555 first = z;
556 } else if (c == 1 && opt_.energy_correlation != 0.0) {
557 double rho = opt_.energy_correlation;
558 double s = std::sqrt(std::max(0.0, 1.0 - rho * rho));
559 for (Index i = 0; i < n; ++i) {
560 z[i] = rho * first[i] + s * z[i];
561 }
562 }
563 for (double& x : z) {
564 x = m.mean + m.sigma * x;
565 }
566 energies_.push_back(std::move(z));
567 }
568}

570double StochasticNetwork::Connection(double r) const {
571 const std::vector<double>& rs = opt_.connection_r;
572 const std::vector<double>& ps = opt_.connection_p;
573 if (r <= rs.front()) {
574 return ps.front();
575 }
576 if (r >= rs.back()) {
577 return 0.0;
578 }
579 std::size_t k = std::size_t(std::upper_bound(rs.begin(), rs.end(), r) -
580 rs.begin()); // rs[k-1] < r < rs[k]
581 double t = (r - rs[k - 1]) / (rs[k] - rs[k - 1]);
582 return ps[k - 1] + t * (ps[k] - ps[k - 1]);
583}

587double NormalQuantile(double u) {
588 return -std::sqrt(2.0) * boost::math::erfc_inv(2.0 * u);

592double StochasticNetwork::Residual(const CouplingModel& m, double g) {
593 const std::vector<double>& q = m.residual_quantiles;
594 const Index K = Index(q.size());
595 if (K == 0) {
596 return g;
597 }
598 if (K == 1) {
599 return q[0] + g;
600 }
601 // q[k] belongs to the normal quantile g_k = Phi^-1((k + 1/2) / K);
602 // linear interpolation in g, beyond the table like a normal distribution
603 auto node = [K](Index k) {
604 return NormalQuantile((double(k) + 0.5) / double(K));
605 };
606 double t = NormalCdf(g) * double(K) - 0.5;
607 if (t <= 0.0) {
608 return q[0] + (g - node(0));
609 }
610 if (t >= double(K - 1)) {
611 return q[K - 1] + (g - node(K - 1));
612 }
613 Index k = std::min(Index(std::floor(t)), K - 2);
614 double g0 = node(k);
615 double g1 = node(k + 1);
616 double f = std::min(1.0, std::max(0.0, (g - g0) / (g1 - g0)));
617 return q[k] + f * (q[k + 1] - q[k]);
618}
619
620double StochasticNetwork::Log10J2(const CouplingModel& m, double r, double g) {
621 double rnm = r * tools::conv::bohr2nm;
622 double mean = m.mean_offset + m.mean_slope * rnm;
623 double var = std::max(0.0, m.var_offset + m.var_slope * rnm);
624 return mean + std::sqrt(var) * Residual(m, g);
625}

627void StochasticNetwork::InsertTraps(ModelRandom& random) {
628 inserted_.clear();
629 if (opt_.traps.empty() || opt_.trap_factor == 0.0) {
630 return;
631 }
632 const Index n = NumberOfSites();
633 double volume = opt_.box.prod();
634 double radius =
635 std::min(0.49 * opt_.box.minCoeff(), 2.0 * std::cbrt(volume / double(n)));
636 PeriodicCells cells(opt_.box, radius, positions_);
637 std::vector<char> used(n, 0);
638 for (std::size_t t = 0; t < opt_.traps.size(); ++t) {
639 const Trap& trap = opt_.traps[t];
640 double expected = trap.density * opt_.trap_factor * double(n);
641 Index count = Index(std::floor(expected));
642 if (random.Uniform() <= expected - double(count)) {
643 ++count;
644 }
645 const Index size = Index(trap.offsets.size());
646 for (Index k = 0; k < count; ++k) {
647 // a random free site whose nearest sites are free as well
648 std::vector<Index> sites;
649 for (Index attempt = 0; attempt < 1000 && sites.empty(); ++attempt) {
650 Index centre = random.Integer(n);
651 std::vector<Index> nb = cells.Nearest(centre, size);
652 if (std::none_of(nb.begin(), nb.end(),
653 [&used](Index i) { return used[i] != 0; })) {
654 sites = nb;
655 }
656 }
657 if (sites.empty()) {
658 throw std::runtime_error(
659 "StochasticNetwork: no free place for a trap; the trap density "
660 "is too high");
661 }
662 for (Index q = 0; q < size; ++q) {
663 used[sites[q]] = 1;
664 for (std::size_t c = 0; c < opt_.carriers.size(); ++c) {
665 energies_[c][sites[q]] =
666 opt_.carriers[c].energy.mean + trap.offsets[q][c];
667 }
668 }
669 inserted_.push_back({Index(t), sites});
670 }
671 }
672}

674void StochasticNetwork::GeneratePairs(ModelRandom& random) {
675 const Index n = NumberOfSites();
676 const double rmax = opt_.connection_r.back();
677 pairs_.clear();
678 if (!(rmax > 0.0)) {
679 return;
680 }
681 PeriodicCells cells(opt_.box, rmax, positions_);
682 const double ev2 = tools::conv::ev2hrt * tools::conv::ev2hrt;
683 const std::size_t ncar = opt_.carriers.size();
684 const double rho = opt_.coupling_correlation;
685 const double s = std::sqrt(std::max(0.0, 1.0 - rho * rho));
686 struct Partner {
687 Index j;
688 Eigen::Vector3d R;
689 double dist;
690 };
691 std::vector<Partner> partners;
692 std::vector<double> g(ncar);
693 for (Index i = 0; i < n; ++i) {
694 partners.clear();
695 cells.Within(positions_[i], rmax,
696 [&](Index j, const Eigen::Vector3d& conn, double dist) {
697 if (j > i) {
698 partners.push_back({j, conn, dist});
699 }
700 });
701 std::sort(partners.begin(), partners.end(),
702 [](const Partner& a, const Partner& b) { return a.j < b.j; });
703 for (const Partner& p : partners) {
704 double f = Connection(p.dist);
705 if (f <= 0.0 || (f < 1.0 && random.Uniform() > f)) {
706 continue;
707 }
708 for (std::size_t c = 0; c < ncar; ++c) {
709 g[c] = random.Normal();
710 }
711 if (ncar > 1) {
712 g[1] = rho * g[0] + s * g[1];
713 }
714 Pair pair{i, p.j, p.R, std::vector<double>(ncar)};
715 for (std::size_t c = 0; c < ncar; ++c) {
716 pair.J2[c] =
717 std::pow(10.0, Log10J2(opt_.carriers[c].coupling, p.dist, g[c])) *
718 ev2;
719 }
720 pairs_.push_back(std::move(pair));
721 }
722 }

725void StochasticNetwork::BuildTopology(Topology& top) const {
726 if (!top.Segments().empty() || top.NBList().size() > 0) {
727 throw std::runtime_error("StochasticNetwork: topology must be empty");
728 }
729 Eigen::Matrix3d box = Eigen::Matrix3d::Zero();
730 box.diagonal() = opt_.box;
731 top.setBox(box);
732 top.setStep(0);
733 top.setTime(0.0);
734 for (Index i = 0; i < NumberOfSites(); ++i) {
735 Segment& seg = top.AddSegment(opt_.segment_name);
736 seg.push_back(Atom(0, "C", positions_[i]));
737 for (std::size_t c = 0; c < opt_.carriers.size(); ++c) {
738 const Carrier& car = opt_.carriers[c];
739 seg.setEMpoles(car.state, energies_[c][i]);
740 seg.setU_xX_nN(car.U_xX_nN, car.state);
741 seg.setU_nX_nN(car.U_nX_nN, car.state);
742 seg.setU_xN_xX(car.U_xN_xX, car.state);
743 }
744 }
745 const std::vector<Segment>& segs = top.Segments();
746 QMNBList& nblist = top.NBList();
747 for (const Pair& p : pairs_) {
748 QMPair& pair = nblist.Add(segs[p.i], segs[p.j], p.R);
749 for (std::size_t c = 0; c < opt_.carriers.size(); ++c) {
750 pair.setJeff2(p.J2[c], opt_.carriers[c].state);
751 }
752 }
753}

757StochasticNetwork::Options StochasticNetwork::FromProperty(
758 const tools::Property& p) {
759 const double nm = tools::conv::nm2bohr;
760 const double ev = tools::conv::ev2hrt;
761 Options opt;
762 opt.sites = p.ifExistsReturnElseReturnDefault<Index>("sites", 0);
763 if (p.exists("box")) {
764 std::vector<double> box = p.get("box").as<std::vector<double>>();
765 if (box.size() != 3) {
766 throw std::runtime_error("stochasticnetwork: box needs three lengths");
767 }
768 opt.box = Eigen::Vector3d(box[0], box[1], box[2]) * nm;
769 }
770 opt.seed = p.ifExistsReturnElseReturnDefault<std::uint64_t>("seed", 1);
771 opt.segment_name =
772 p.ifExistsReturnElseReturnDefault<std::string>("segmentname", "site");
773 const tools::Property& pp = p.get("points");
774 opt.points.hard_core = pp.get("hardcore").as<double>() * nm;
775 opt.points.radius_mean = pp.get("radiusmean").as<double>() * nm;
776 opt.points.radius_variance = pp.get("radiusvariance").as<double>() * nm * nm;
777 opt.points.candidates =
778 pp.ifExistsReturnElseReturnDefault<double>("candidates", 3.0);
779 opt.points.max_iterations =
780 pp.ifExistsReturnElseReturnDefault<Index>("maxiterations", 200);
781 opt.connection_r = p.get("connection.r").as<std::vector<double>>();
782 for (double& r : opt.connection_r) {
783 r *= nm;
784 }
785 opt.connection_p = p.get("connection.p").as<std::vector<double>>();
786 opt.energy_correlation =
787 p.ifExistsReturnElseReturnDefault<double>("energycorrelation", 0.0);
788 opt.coupling_correlation =
789 p.ifExistsReturnElseReturnDefault<double>("couplingcorrelation", 0.0);
790 for (const tools::Property* c : p.Select("carrier")) {
791 Carrier car;
792 car.state = QMStateType(c->get("state").as<std::string>());
793 const tools::Property& e = c->get("energy");
794 car.energy.mean = e.get("mean").as<double>() * ev;
795 car.energy.sigma = e.get("sigma").as<double>() * ev;
796 car.energy.w_a = e.get("w_a").as<double>();
797 car.energy.w_b = e.get("w_b").as<double>();
798 car.energy.l_b = e.get("l_b").as<Index>();
799 car.energy.l_c = e.get("l_c").as<Index>();
800 const tools::Property& j = c->get("coupling");
801 car.coupling.mean_offset = j.get("meanoffset").as<double>();
802 car.coupling.mean_slope = j.get("meanslope").as<double>();
803 car.coupling.var_offset = j.get("varoffset").as<double>();
804 car.coupling.var_slope = j.get("varslope").as<double>();
805 if (j.exists("quantiles")) {
806 car.coupling.residual_quantiles =
807 j.get("quantiles").as<std::vector<double>>();
808 }
809 car.U_xX_nN =
810 c->ifExistsReturnElseReturnDefault<double>("U_xX_nN", 0.0) * ev;
811 car.U_nX_nN =
812 c->ifExistsReturnElseReturnDefault<double>("U_nX_nN", 0.0) * ev;
813 car.U_xN_xX =
814 c->ifExistsReturnElseReturnDefault<double>("U_xN_xX", 0.0) * ev;
815 opt.carriers.push_back(car);
816 }
817 for (const tools::Property* t : p.Select("trap")) {
818 Trap trap;
819 QMStateType state(t->get("carrier").as<std::string>());
820 trap.carrier = -1;
821 for (std::size_t c = 0; c < opt.carriers.size(); ++c) {
822 if (opt.carriers[c].state == state) {
823 trap.carrier = Index(c);
824 }
825 }
826 if (trap.carrier < 0) {
827 throw std::runtime_error("stochasticnetwork: trap for carrier " +
828 state.ToString() + " without a carrier block");
829 }
830 trap.density = t->get("density").as<double>();
831 trap.release_rate =
832 t->ifExistsReturnElseReturnDefault<double>("releaserate", 0.0);
833 for (const tools::Property* site : t->Select("site")) {
834 std::vector<double> o = site->as<std::vector<double>>();
835 for (double& x : o) {
836 x *= ev;
837 }
838 trap.offsets.push_back(o);
839 }
840 opt.traps.push_back(trap);
841 }
842 return opt;
843}

924 }
925 out << "</stochasticnetwork>\n";
