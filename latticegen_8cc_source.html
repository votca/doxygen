38void LatticeGen::ParseOptions(const tools::Property& options) {
39 statefile_ = options.ifExistsReturnElseReturnDefault<std::string>(
40 ".statefile", job_name_ + ".hdf5");
41 energyfile_ =
42 options.ifExistsReturnElseReturnDefault<std::string>(".energyfile", "");
43
44 std::vector<Index> dims = options.get(".dims").as<std::vector<Index>>();
45 if (dims.size() != 3) {
46 throw std::runtime_error("latticegen: dims needs three numbers");
47 }
48 lattice_.dims = {dims[0], dims[1], dims[2]};
49 lattice_.spacing =
50 options.get(".spacing").as<double>() * tools::conv::nm2bohr;
51 lattice_.cutoff = options.get(".cutoff").as<double>() * tools::conv::nm2bohr;
52 double ev2 = tools::conv::ev2hrt * tools::conv::ev2hrt;
53 lattice_.J2 = options.get(".J2").as<double>() * ev2;
54 lattice_.decay_length =
55 options.get(".decaylength").as<double>() * tools::conv::nm2bohr;
56 lattice_.reorganization =
57 options.get(".reorganization").as<double>() * tools::conv::ev2hrt;
58 lattice_.mean_energy =
59 options.get(".energy.mean").as<double>() * tools::conv::ev2hrt;
60 lattice_.sigma =
61 options.get(".energy.sigma").as<double>() * tools::conv::ev2hrt;
62 lattice_.disorder = LatticeModel::DisorderFromString(
63 options.get(".energy.disorder").as<std::string>());
64 lattice_.eta_a = options.get(".energy.eta_a").as<double>();
65 lattice_.eta_b = options.get(".energy.eta_b").as<double>();
66 lattice_.l_b = options.get(".energy.l_b").as<Index>();
67 lattice_.l_c = options.get(".energy.l_c").as<Index>();
68 lattice_.include_self = options.get(".energy.includeself").as<bool>();
69 lattice_.seed = options.get(".energy.seed").as<std::uint64_t>();
70
71 std::vector<std::string> states =
72 options.get(".states").as<std::vector<std::string>>();
73 lattice_.states.clear();
74 for (const std::string& s : states) {
75 QMStateType type(s);
76 if (!type.isKMCState()) {
77 throw std::runtime_error("latticegen: " + s +
78 " is not a hopping state (e, h, s, t)");
79 }
80 lattice_.states.push_back(type);
81 }
82 lattice_.segment_name = options.get(".segmentname").as<std::string>();
83}

85bool LatticeGen::Run() {
86 log_.setReportLevel(Log::current_level);
87 log_.setMultithreading(true);
88 log_.setCommonPreface("\n... ...");
89
90 if (tools::filesystem::FileExists(statefile_)) {
91 throw std::runtime_error("latticegen: " + statefile_ +
92 " exists already, remove it or choose another "
93 "statefile name");
94 }
95
96 LatticeModel model(lattice_);
97 Topology top;
98 model.BuildTopology(top);
99
100 const std::vector<double>& e = model.SiteEnergies();
101 double mean = std::accumulate(e.begin(), e.end(), 0.0) / double(e.size());
102 double var = std::accumulate(e.begin(), e.end(), 0.0,
103 [mean](double acc, double x) {
104 return acc + (x - mean) * (x - mean);
105 }) /
106 double(e.size());
107 XTP_LOG(Log::error, log_)
108 << "Lattice " << lattice_.dims[0] << "x" << lattice_.dims[1] << "x"
109 << lattice_.dims[2] << ": " << model.NumberOfSites() << " sites, "
110 << top.NBList().size() << " pairs" << std::flush;
111 XTP_LOG(Log::error, log_)
112 << "Site energies: mean " << mean * tools::conv::hrt2ev
113 << " eV, standard deviation " << std::sqrt(var) * tools::conv::hrt2ev
114 << " eV" << std::flush;
115
116 StateSaver saver(statefile_);
117 saver.WriteFrame(top);
118 XTP_LOG(Log::error, log_) << "Wrote " << statefile_ << std::flush;
119
120 if (!energyfile_.empty()) {
121 std::ofstream out(energyfile_);
122 out << "# segment id, x, y, z [nm], site energy [eV]\n";
123 out << std::setprecision(12);
124 const std::vector<Eigen::Vector3d>& pos = model.Positions();
125 for (Index i = 0; i < model.NumberOfSites(); ++i) {
126 Eigen::Vector3d p = pos[i] * tools::conv::bohr2nm;
127 out << i << " " << p.x() << " " << p.y() << " " << p.z() << " "
128 << e[i] * tools::conv::hrt2ev << "\n";
129 }
130 XTP_LOG(Log::error, log_)
131 << "Wrote site energies to " << energyfile_ << std::flush;
132 }
133 return true;
134}
