142class StochasticNetwork {
143 public:
144 struct PointProcess {
145 double hard_core = 0.0; // r_h
146 double radius_mean = 0.0; // mean of X
147 double radius_variance = 0.0; // variance of X (0: X = mean)
149 double candidates = 3.0;
150 Index max_iterations = 200;
151 };
152
153 struct EnergyModel {
154 double mean = 0.0;
155 double sigma = 0.0; // 0: all sites at mean
156 double w_a = 1.0;
157 double w_b = 0.0;
158 Index l_b = 1;
159 Index l_c = 1;
160 };
161
162 struct CouplingModel {
163 // log10(J^2/eV^2) mean = mean_offset + mean_slope r[nm]
164 double mean_offset = -6.0;
165 double mean_slope = 0.0;
166 // variance = var_offset + var_slope r[nm] (clipped at 0)
167 double var_offset = 1.0;
168 double var_slope = 0.0;
171 std::vector<double> residual_quantiles;
172 };
173
174 struct Carrier {
175 QMStateType state;
176 EnergyModel energy;
177 CouplingModel coupling;
178 double U_xX_nN = 0.0; // internal energy
179 double U_nX_nN = 0.0; // reorganization parts
180 double U_xN_xX = 0.0;
181 };
182
185 struct Trap {
186 Index carrier = 0; // the carrier it traps (index into carriers)
187 double density = 0.0; // traps of this kind per site
189 std::vector<std::vector<double>> offsets;
190 double release_rate = 0.0; // microscopic release rate (1/s, information)
191 };
192
193 struct Options {
194 Eigen::Vector3d box = Eigen::Vector3d::Zero(); // orthorhombic, bohr
195 Index sites = 0;
196 PointProcess points;
198 std::vector<double> connection_r;
199 std::vector<double> connection_p;
200 std::vector<Carrier> carriers;
203 double energy_correlation = 0.0;
204 double coupling_correlation = 0.0;
205 std::uint64_t seed = 1;
206 std::string segment_name = "site";
208 std::vector<Trap> traps;
210 double trap_factor = 1.0;
211 };
212
213 explicit StochasticNetwork(const Options& opt);
214
215 Index NumberOfSites() const { return Index(positions_.size()); }
216 const std::vector<Eigen::Vector3d>& Positions() const { return positions_; }
218 const std::vector<double>& Radii() const { return radii_; }
220 struct InsertedTrap {
221 Index type;
222 std::vector<Index> sites;
223 };
224 const std::vector<InsertedTrap>& Traps() const { return inserted_; }
226 const std::vector<double>& SiteEnergies(Index c) const {
227 return energies_[c];
228 }
229 struct Pair {
230 Index i;
231 Index j;
232 Eigen::Vector3d R; // shortest connection from i to j
233 std::vector<double> J2; // per carrier, Hartree^2
234 };
235 const std::vector<Pair>& Pairs() const { return pairs_; }
237 Index Iterations() const { return iterations_; }
238
240 void BuildTopology(Topology& top) const;
241
244 static Options FromProperty(const tools::Property& p);
246 static std::string ToXML(const Options& opt);
247
250 static std::vector<double> MovingAverage(const PeriodicCells& cells,
251 const EnergyModel& m, Index sites,
252 const std::vector<double>& A,
253 const std::vector<double>& Bf,
254 const std::vector<double>& C);
257 static Index SamplePoints(const PointProcess& pp, const Eigen::Vector3d& box,
258 Index sites, ModelRandom& random,
259 std::vector<Eigen::Vector3d>& positions,
260 std::vector<double>& radii);
262 double Connection(double r) const;
264 static double Residual(const CouplingModel& m, double g);
266 static double Log10J2(const CouplingModel& m, double r, double g);
267
268 private:
269 Options opt_;
270 std::vector<Eigen::Vector3d> positions_;
271 std::vector<double> radii_;
272 std::vector<std::vector<double>> energies_;
273 std::vector<Pair> pairs_;
274 Index iterations_ = 0;
275 std::vector<InsertedTrap> inserted_;
276
277 void Check() const;
278 void GenerateEnergies(ModelRandom& random);
279 void GeneratePairs(ModelRandom& random);
280 void InsertTraps(ModelRandom& random);
281};
