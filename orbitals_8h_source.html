213 Index NumberofStates(QMStateType type) const {
214 switch (type.Type()) {
215 case QMStateType::Singlet:
216 return Index(BSE_singlet_.eigenvalues().size());
217 break;
218 case QMStateType::Triplet:
219 return Index(BSE_triplet_.eigenvalues().size());
220 break;
221 case QMStateType::KSstate:
222 return Index(mos_.eigenvalues().size());
223 break;
224 case QMStateType::PQPstate:
225 return Index(QPpert_energies_.size());
226 break;
227 case QMStateType::DQPstate:
228 return Index(QPdiag_.eigenvalues().size());
229 break;
230 case QMStateType::ExcitonUKS:
231 return Index(BSE_uks_.eigenvalues().size());
232 break;
233 default:
234 return 1;
235 break;
236 }
