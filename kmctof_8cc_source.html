43void KMCToF::ParseOptions(const tools::Property& options) {
44 carriertype_ = QMStateType(options.get(".carriertype").as<std::string>());
45 if (!carriertype_.isKMCState()) {
46 throw std::runtime_error("kmctof cannot be run for state: " +
47 carriertype_.ToLongString());
48 }
49 temperature_ = options.get(".temperature").as<double>() * tools::conv::kB *
50 tools::conv::ev2hrt;
51 field_Vm_ = options.get(".field").as<Eigen::Vector3d>();
52 field_ = field_Vm_ * (tools::conv::ev2hrt / (1e9 * tools::conv::nm2bohr));
53 tof_ = ToFGraph::OptionsFromProperty(options.get(".tof"));
54 seed_ = options.get(".seed").as<std::uint64_t>();
55 runs_ = options.get(".runs").as<Index>();
56
57 const tools::Property& inj = options.get(".injection");
58 KMCEngine::Pulse pulse;
59 pulse.carriers = inj.get("carriers").as<Index>();
60 pulse.shape = KMCEngine::ShapeFromString(inj.get("shape").as<std::string>());
61 pulse.width = inj.get("width").as<double>();
62 Index npulses = inj.get("pulses").as<Index>();
63 double period = inj.get("period").as<double>();
64 if (npulses < 1 || pulse.carriers < 1) {
65 throw std::runtime_error("kmctof: pulses and carriers must be positive");
66 }
67 if (npulses > 1 && !(period > 0.0)) {
68 throw std::runtime_error("kmctof: a pulse train needs period > 0");
69 }
70 pulses_.clear();
71 for (Index p = 0; p < npulses; ++p) {
72 pulse.start = double(p) * period;
73 pulses_.push_back(pulse);
74 }
75
76 waitingtime_ = options.get(".waitingtime").as<std::string>();
77 limits_.max_time = options.get(".maxtime").as<double>();
78 limits_.max_events = options.get(".maxevents").as<Index>();
79 runfile_ = options.get(".runfile").as<std::string>();
80 currentfile_ = options.get(".current.file").as<std::string>();
81 resultfile_ = options.get(".resultfile").as<std::string>();
82 tmin_ = options.get(".current.tmin").as<double>();
83 tmax_ = options.get(".current.tmax").as<double>();
84 bins_per_decade_ = options.get(".current.binsperdecade").as<Index>();
85 log_.setReportLevel(Log::current_level);
86 log_.setMultithreading(true);
87 log_.setCommonPreface("\n ...");
88}

90bool KMCToF::Evaluate(Topology& top) {
91 XTP_LOG(Log::error, log_) << "\n-----------------------------------"
92 "\n KINETIC MONTE CARLO TIME OF FLIGHT"
93 "\n-----------------------------------"
94 << std::flush;
95 ToFGraph graph(top, tof_, carriertype_, temperature_, field_);
96 XTP_LOG(Log::error, log_) << graph.Summary() << std::flush;
97
98 ToFSimulation::Settings settings;
99 settings.seed = seed_;
100 settings.runs = runs_;
101 settings.pulses = pulses_;
102 settings.limits = limits_;
103 settings.tmin = tmin_;
104 settings.tmax = tmax_;
105 settings.bins_per_decade = bins_per_decade_;
106 // rates and waiting times: the static Marcus rates of the graph; a
107 // time-dependent TimeDependentRates (e.g. oscillating field) plugs in here
108 StaticRates rates(graph);
109 std::unique_ptr<WaitingTimeStrategy> strategy =
110 MakeWaitingTimeStrategy(waitingtime_);
111 ToFSimulation simulation(graph, settings, rates, *strategy);
112 XTP_LOG(Log::error, log_)
113 << "Waiting times: " << simulation.StrategyName() << std::flush;
114 XTP_LOG(Log::error, log_)
115 << runs_ << " runs, " << pulses_.size() << " pulse(s) of "
116 << pulses_[0].carriers << " carrier(s), " << OPENMP::getMaxThreads()
117 << " thread(s)" << std::flush;
118 auto start = std::chrono::steady_clock::now();
119 ToFSimulation::Result result = simulation.Simulate();
120 double seconds =
121 std::chrono::duration<double>(std::chrono::steady_clock::now() - start)
122 .count();
123
124 XTP_LOG(Log::error, log_)
125 << std::setprecision(6) << result.hops << " hops, " << result.null_events
126 << " null events (occupied site or rejected), " << seconds << " s"
127 << std::flush;
128 if (result.censored > 0) {
129 XTP_LOG(Log::error, log_)
130 << "Warning: " << result.censored
131 << " runs stopped by maxtime/maxevents before all carriers arrived; "
132 "they are left out, which biases the times low"
133 << std::flush;
134 }
135 if (result.blocked_injections > 0) {
136 XTP_LOG(Log::error, log_)
137 << "Warning: " << result.blocked_injections
138 << " injections dropped (all source sites occupied)" << std::flush;
139 }
140 auto report = [this](const std::string& name,
141 const ToFSimulation::Statistic& s) {
142 XTP_LOG(Log::error, log_)
143 << name << ": " << s.mean << " +- " << s.standard_error
144 << " s (standard error, " << s.samples << " runs), standard deviation "
145 << s.standard_deviation
146 << " s, largest 1 % of the runs: " << 100.0 * s.top_percent_share
147 << " % of the sum" << std::flush;
148 };
149 report("First arrival", result.first);
150 report("Mean transit time per carrier", result.transit);
151 if (result.transit.top_percent_share > 0.2) {
152 XTP_LOG(Log::error, log_)
153 << "Warning: heavy tail, the mean is dominated by rare long runs; "
154 "compare with grw"
155 << std::flush;
156 }
157
158 double Fn = field_Vm_.dot(graph.Normal());
159 if (Fn != 0.0 && result.transit.samples > 0) {
160 double L = graph.Length() * tools::conv::bohr2nm * 1e-9; // m
161 XTP_LOG(Log::error, log_)
162 << "Mobility L/(tau F) from the mean transit time: "
163 << L / result.transit.mean * 1e4 / std::abs(Fn) << " cm^2/Vs"
164 << std::flush;
165 }
166
167 if (!resultfile_.empty()) {
168 ResultFile out("kmctof: time-of-flight kinetic Monte Carlo");
169 out.Add("runs", runs_);
170 out.Add("injected", result.injected);
171 out.Add("censored", result.censored);
172 out.Add("blocked_injections", result.blocked_injections);
173 out.Add("hops", result.hops);
174 out.Add("null_events", result.null_events);
175 out.Add("first_mean", result.first.mean, "s");
176 out.Add("first_error", result.first.standard_error, "s");
177 out.Add("first_std", result.first.standard_deviation, "s");
178 out.Add("transit_mean", result.transit.mean, "s");
179 out.Add("transit_error", result.transit.standard_error, "s");
180 out.Add("transit_std", result.transit.standard_deviation, "s");
181 out.Add("transit_top_percent_share", result.transit.top_percent_share, "-");
182 double L = graph.Length() * tools::conv::bohr2nm * 1e-9; // m
183 out.Add("length", L, "m");
184 out.Add("field_normal", Fn, "V/m");
185 if (Fn != 0.0 && result.transit.samples > 0) {
186 out.Add("mobility_transit", L / result.transit.mean * 1e4 / std::abs(Fn),
187 "cm^2/Vs");
188 }
189 out.Write(resultfile_);
190 }
191 if (!runfile_.empty()) {
192 std::ofstream out(runfile_);
193 out << "# run, injected, absorbed, censored, first arrival [s], mean "
194 "transit [s], hops, null events, end time [s]\n";
195 out << std::setprecision(10);
196 for (Index r = 0; r < Index(result.runs.size()); ++r) {
197 const ToFSimulation::RunData& d = result.runs[r];
198 out << r << " " << d.status.injected << " " << d.status.absorbed << " "
199 << d.status.censored << " " << d.first_arrival << " "
200 << d.mean_transit << " " << d.status.hops << " "
201 << d.status.null_events << " " << d.status.end_time << "\n";
202 }
203 }
204 if (!currentfile_.empty()) {
205 std::ofstream out(currentfile_);
206 out << "# transient current per injected carrier (Ramo-Shockley): "
207 "I = (sum of the hop displacements along the transport normal "
208 "in the bin) / (L x injected carriers x bin width), in 1/s; the "
209 "circuit current is q x injected carriers x I, and the integral "
210 "of I is the mean displacement per injected carrier in units of "
211 "L\n";
212 out << "# collected fraction: carriers absorbed up to t_high / injected "
213 "carriers\n";
214 out << "# t_low [s], t_high [s], t_mid (geometric) [s], current [1/s], "
215 "collected fraction at t_high\n";
216 out << "# displacement before t_low of the first bin: "
217 << result.displacement_before
218 << ", after the last bin: " << result.displacement_after << "\n";
219 out << std::setprecision(10);
220 for (Index b = 0; b < Index(result.current.size()); ++b) {
221 double lo = result.bin_edges[b];
222 double hi = result.bin_edges[b + 1];
223 out << lo << " " << hi << " " << std::sqrt(lo * hi) << " "
224 << result.current[b] << " " << result.collected[b] << "\n";
225 }
226 }
227 std::cout << log_;
228 return true;
229}
