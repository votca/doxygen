45void QMRegion::Initialize(const tools::Property& prop) {
46 if (this->id_ != 0) {
47 throw std::runtime_error(
48 this->identify() +
49 " must always be region 0. Currently only one qm region is possible.");
50 }
51
52 initstate_ = prop.get("state").as<QMState>();
53 /*if (initstate_.Type() == QMStateType::Hole ||
54 initstate_.Type() == QMStateType::Electron) {
55 throw std::runtime_error(
56 "Charged QM Regions are not implemented currently");
57 }*/
58 if (initstate_.Type().isExciton() || initstate_.Type().isGWState()) {
59
60 do_gwbse_ = true;
61 gwbseoptions_ = prop.get("gwbse");
62 if (prop.exists("statetracker")) {
63 tools::Property filter = prop.get("statetracker");
64 statetracker_.setLogger(&log_);
65 statetracker_.Initialize(filter);
66 statetracker_.setInitialState(initstate_);
67 statetracker_.PrintInfo();
68 } else {
69 throw std::runtime_error(
70 "No statetracker options for excited states found");
71 }
72 }
73
74 if (initstate_.Type().isKSState()) {
75
76 if (prop.exists("statetracker")) {
77 tools::Property filter = prop.get("statetracker");
78 statetracker_.setLogger(&log_);
79 statetracker_.Initialize(filter);
80 statetracker_.setInitialState(initstate_);
81 statetracker_.PrintInfo();
82 } else {
83 throw std::runtime_error("No statetracker options for KS states found");
84 }
85 }
86
87 grid_accuracy_for_ext_interaction_ =
88 prop.get("grid_for_potential").as<std::string>();
89 DeltaE_ = prop.get("tolerance_energy").as<double>();
90 DeltaD_ = prop.get("tolerance_density").as<double>();
91 DeltaDmax_ = prop.get("tolerance_density_max").as<double>();
92
93 dftoptions_ = prop.get("dftpackage");
94 localize_options_ = prop.get("localize");
95
96 if (prop.get("dftpackage.xtpdft.dft_in_dft.activeatoms")
97 .as<std::string>()
98 .size() != 0) {
99 do_localize_ = true;
100 do_dft_in_dft_ = true;
101 }
102
103 if (prop.exists("environment_screening")) {
104 const tools::Property env = prop.get("environment_screening");
105 screening_ = env.ifExistsReturnElseReturnDefault<bool>("enabled", true);
106 }
107 if (screening_) {
108 const tools::Property env = prop.get("environment_screening");
109 // Excitations are what the screening is for. A charged DFT run would
110 // put the extra charge's polarization into the ground-state loop and
111 // again into W; a plain ground-state or KS-level run has no GW-BSE to
112 // screen.
113 if (initstate_.Type() == QMStateType::Electron ||
114 initstate_.Type() == QMStateType::Hole) {
115 throw std::runtime_error(
116 "environment_screening with state " + initstate_.ToString() +
117 ": the charged DFT state would be polarized in the QM/MM loop and "
118 "screened again in GW. Run the neutral molecule and ask for the "
119 "quasiparticle level instead: state pqpN (or dqpN) with N the "
120 "HOMO or LUMO index.");
121 }
122 if (initstate_.Type() == QMStateType::ExcitonUKS) {
123 throw std::runtime_error(
124 "environment_screening is implemented for closed-shell GW-BSE "
125 "only; the unrestricted exciton state " +
126 initstate_.ToString() + " is not supported.");
127 }
128 if (!do_gwbse_) {
129 throw std::runtime_error(
130 "environment_screening needs a GW-BSE state (an exciton or a "
131 "quasiparticle level); " +
132 initstate_.ToString() + " has no GW-BSE calculation to screen.");
133 }
134 screening_include_kreac_ =
135 env.ifExistsReturnElseReturnDefault<bool>("include_kreac", true);
136 screening_shell_dielectric_ =
137 env.ifExistsReturnElseReturnDefault<double>("shell_dielectric", 4.0);
138 screening_site_width_ =
139 env.ifExistsReturnElseReturnDefault<double>("site_width", 0.5);
140 if (!(screening_site_width_ >= 0.0)) {
141 throw std::runtime_error(
142 "environment_screening: site_width must be >= 0 (0 for point "
143 "sites), got " +
144 std::to_string(screening_site_width_) + ".");
145 }
146 std::string shells =
147 env.ifExistsReturnElseReturnDefault<std::string>("shell_regions", "");
148 std::replace(shells.begin(), shells.end(), ',', ' ');
149 std::istringstream in(shells);
150 Index id;
151 while (in >> id) {
152 screening_shell_regions_.push_back(id);
153 }
154 XTP_LOG(Log::error, log_)
155 << " Environment screening on: ground-state QM/MM loop, then one "
156 "screened GW-BSE run. K_reac "
157 << (screening_include_kreac_ ? "included" : "excluded")
158 << ", shell dielectric " << screening_shell_dielectric_
159 << ", site width " << screening_site_width_ << " alpha^(1/3)"
160 << std::flush;
161 }
162}

164void QMRegion::PrepareOrbitalsForGWBSE() {
165 if (orb_.isOpenShell()) {
166 throw std::runtime_error("GWBSE not implemented for open-shell systems");
167 }
168 if (do_dft_in_dft_) {
169 Index active_electrons = orb_.getNumOfActiveElectrons();
170
171 if (active_electrons % 2 != 0) {
172 throw std::runtime_error(
173 "DFT-in-DFT embedded GWBSE currently supports only closed-shell "
174 "active spaces with an even number of electrons");
175 }
176
177 orb_.MOs() = orb_.getEmbeddedMOs();
178
179 const Index active_occ = active_electrons / 2;
180
181 // Restricted closed-shell rewrite of the embedded orbital object
182 orb_.setNumberOfOccupiedLevels(active_occ);
183 orb_.setNumberOfOccupiedLevelsBeta(active_occ);
184 orb_.setNumberOfAlphaElectrons(active_occ);
185 orb_.setNumberOfBetaElectrons(active_occ);
186 orb_.setChargeAndSpin(orb_.getCharge(), 1);
187 }

190double QMRegion::StateEnergy(const QMState& state) const {
191 if (state.Type().isExciton()) {
192 return orb_.getExcitedStateEnergy(state);
193 }
194 // quasiparticle: adding an electron to an unoccupied level, removing one
195 // from an occupied level
196 if (state.StateIdx() > orb_.getHomo()) {
197 return orb_.getExcitedStateEnergy(state);
198 }
199 return -orb_.getExcitedStateEnergy(state);
200}

202ScreeningEnvironment QMRegion::BuildScreeningEnvironment(
203 const std::vector<std::unique_ptr<Region> >& regions) const {
204 ScreeningEnvironment env;
205 env.shell_dielectric = screening_shell_dielectric_;
206 env.include_kreac = screening_include_kreac_;
207 env.site_width = screening_site_width_;
208 bool have_damp = false;
209 PolarRegion Polardummy(0, log_);
210 for (const std::unique_ptr<Region>& reg : regions) {
211 if (reg->identify() != Polardummy.identify()) {
212 continue;
213 }
214 const PolarRegion& polar = dynamic_cast<const PolarRegion&>(*reg);
215 const bool shell =
216 std::find(screening_shell_regions_.begin(),
217 screening_shell_regions_.end(),
218 polar.getId()) != screening_shell_regions_.end();
219 if (shell) {
220 env.shell_segments.insert(env.shell_segments.end(), polar.begin(),
221 polar.end());
222 continue;
223 }
224 if (have_damp && polar.ExpDamp() != env.exp_damp) {
225 throw std::runtime_error(
226 "environment_screening: the explicit polar regions respond as one "
227 "coupled Thole system and must share exp_damp; found " +
228 std::to_string(env.exp_damp) + " and " +
229 std::to_string(polar.ExpDamp()) + ".");
230 }
231 env.exp_damp = polar.ExpDamp();
232 have_damp = true;
233 env.explicit_segments.insert(env.explicit_segments.end(), polar.begin(),
234 polar.end());
235 }
236 for (Index id : screening_shell_regions_) {
237 const bool is_polar = std::any_of(
238 regions.begin(), regions.end(),
239 [&](const std::unique_ptr<Region>& reg) {
240 return reg->getId() == id && reg->identify() == Polardummy.identify();
241 });
242 if (!is_polar) {
243 throw std::runtime_error("environment_screening: shell region " +
244 std::to_string(id) +
245 " is not a polar region of this job.");
246 }
247 }
248 if (env.empty()) {
249 XTP_LOG(Log::error, log_)
250 << TimeStamp()
251 << " WARNING: environment_screening is on but there is no polar "
252 "region; GW-BSE runs unscreened."
253 << std::flush;
254 }
255 return env;
256}

258std::string QMRegion::ScreeningAuxBasisName() const {
259 // The auxiliary basis GWBSE::Initialize will pick: the DFT run's own
260 // (dftpackage.auxbasisset), else gwbse.auxbasisset, else aux-<basis>.
261 if (dftoptions_.exists("auxbasisset") &&
262 !dftoptions_.get("auxbasisset").as<std::string>().empty()) {
263 return dftoptions_.get("auxbasisset").as<std::string>();
264 }
265 if (gwbseoptions_.exists("auxbasisset") &&
266 !gwbseoptions_.get("auxbasisset").as<std::string>().empty()) {
267 return gwbseoptions_.get("auxbasisset").as<std::string>();
268 }
269 return "aux-" + dftoptions_.get("basisset").as<std::string>();
270}

272void QMRegion::CheckScreeningEnvironment(
273 const std::vector<std::unique_ptr<Region> >& regions) const {
274 if (!screening_) {
275 return;
276 }
277 // R depends on the QM atoms, the auxiliary basis on them and the polar
278 // sites -- none of which the QM/MM loop changes -- so it is checked
279 // here, before the loop, rather than after the ground state has
280 // converged. Failing now costs seconds; failing in EvaluateScreenedGWBSE
281 // costs the whole loop.
282 const ScreeningEnvironment env = BuildScreeningEnvironment(regions);
283 if (env.empty()) {
284 return;
285 }
286 const std::string auxname = ScreeningAuxBasisName();
287 BasisSet bs;
288 bs.Load(auxname);
289 AOBasis auxbasis;
290 auxbasis.Fill(bs, orb_.QMAtoms());
291 XTP_LOG(Log::error, log_)
292 << TimeStamp() << " Checking the screening environment (" << auxname
293 << ", " << auxbasis.AOBasisSize() << " functions)" << std::flush;
294 const ScreeningCheck check = EnvironmentScreening::Check(
295 auxbasis, orb_.QMAtoms(), env, EnvironmentScreening::Metric(auxbasis));
296 if (!check.ok()) {
297 throw std::runtime_error(
298 "environment_screening: 1 + R is not positive definite (lowest "
299 "eigenvalue of R " +
300 std::to_string(check.lowest) +
301 "): the environment would screen some charge fluctuation of the QM "
302 "region by more than the fluctuation itself. Diagnosis:\n" +
303 check.report);
304 }
305 XTP_LOG(Log::error, log_)
306 << TimeStamp() << " Screening environment is stable: lowest eigenvalue "
307 << "of R " << check.lowest << " (must be > -1)" << std::flush;
308 XTP_LOG(Log::info, log_) << check.report << std::flush;
309}

311void QMRegion::EvaluateScreenedGWBSE(
312 std::vector<std::unique_ptr<Region> >& regions) {
313 if (!screening_) {
314 return;
315 }
316 if (orb_.getCalculationType() == "Truncated") {
317 throw std::runtime_error(
318 "environment_screening is not implemented for truncated active "
319 "regions.");
320 }
321
322 const ScreeningEnvironment env = BuildScreeningEnvironment(regions);
323
324 XTP_LOG(Log::error, log_)
325 << TimeStamp() << " Running environment-screened GW-BSE" << std::flush;
326 const double e_dft = orb_.getDFTTotalEnergy();
327 PrepareOrbitalsForGWBSE();
328 GWBSE gwbse(orb_);
329 gwbse.setLogger(&log_);
330 gwbse.Initialize(gwbseoptions_);
331 gwbse.setScreeningEnvironment(env);
332 gwbse.Evaluate();
333 const QMState state = statetracker_.CalcStateAndUpdate(orb_);
334 const double energy = e_dft + StateEnergy(state);
335 XTP_LOG(Log::error, log_)
336 << TimeStamp() << " Screened state " << state.ToString() << ": "
337 << std::setprecision(10) << StateEnergy(state) * tools::conv::hrt2ev
338 << " eV, region energy " << energy << " Hartree" << std::flush;
339 E_hist_.push_back(energy);
340 screened_state_ = state;
341 screened_done_ = true;
342}

344void QMRegion::AppendScreenedSiteEnergies(tools::Property& prop) const {
345 if (!screened_done_) {
346 return;
347 }
348 // Site energies as -j read stores them: the energy of the state relative
349 // to the neutral ground state of the same environment, in eV. One
350 // screened job provides all of them, from one GW(-BSE) run on top of
351 // the converged ground state:
352 // h: E_h - E_n = -eps_HOMO e: E_e - E_n = eps_LUMO
353 // s/t: Omega of the tracked exciton, if the job's state is one.
354 // Quasiparticle energies are the diagonalized ones (DQP) if available.
355 const Index homo = orb_.getHomo();
356 auto qp = [&](Index level) {
357 const std::string kind = orb_.hasQPdiag() ? "dqp" : "pqp";
358 return orb_.getExcitedStateEnergy(QMState(kind + std::to_string(level)));
359 };
360 tools::Property& out = prop.add("screened_site_energies", "");
361 out.add("h", std::to_string(-qp(homo) * tools::conv::hrt2ev));
362 out.add("e", std::to_string(qp(homo + 1) * tools::conv::hrt2ev));
363 if (screened_state_.Type() == QMStateType::Singlet) {
364 out.add("s",
365 std::to_string(StateEnergy(screened_state_) * tools::conv::hrt2ev));
366 } else if (screened_state_.Type() == QMStateType::Triplet) {
367 out.add("t",
368 std::to_string(StateEnergy(screened_state_) * tools::conv::hrt2ev));
369 }
370 out.add("state", screened_state_.ToString());
371}

376 std::vector<const Eigen::Vector3d*> src = ewaldgrid_.getGridpoints();

383void QMRegion::PrepareEwaldPotentialGrid() {
384
385 // purpose: prepare a grid integration object that can
386 // - hand over the grid to Ewald
387 // - after Ewald calculates the potential at the grid points, be put into
388 // xtpdft
389 //
390 // The grid name is xtpdft's own integration_grid, NOT grid_for_potential.
391 // That is a constraint, not a preference: what comes back from Ewald is a
392 // potential VALUE at each quadrature point, and DFTEngine forms <rho|phi>
393 // on the grid it rebuilds from grid_name_, which is this same key. A
394 // different quality here gives a different number of boxes and a
395 // different number of points per box, and the size checks on the
396 // DFTEngine side would throw. grid_for_potential governs the opposite
397 // direction -- the QM density's influence outward on classical regions,
398 // in ApplyQMFieldToPolarSegments -- and has no business here.
399 if (orb_.QMAtoms().size() == 0) {
400 throw std::runtime_error(
401 "QMRegion::PrepareEwaldPotentialGrid: the region holds no atoms. The "
402 "grid is built over the QM molecule, so this must run after the "
403 "region has been filled.");
404 }
405 std::string dftbasis_name = dftoptions_.get("basisset").as<std::string>();
406 std::string grid_name =
407 dftoptions_.get("xtpdft.integration_grid").as<std::string>();
408 QMMolecule mol = orb_.QMAtoms();
409 BasisSet bs;
410 bs.Load(dftbasis_name);
411 AOBasis dftbasis;
412 dftbasis.Fill(bs, mol);
413 ewaldgrid_.GridSetup(grid_name, mol, dftbasis);
414 XTP_LOG(Log::error, log_)
415 << TimeStamp() << " Constructed Grid for Integration of Ewald Potential"
416 << std::flush;
417 // Vxc_Potential<Vxc_Grid> vxc(grid);
418 ewald_grid_ready_ = true;
419 return;
420}

422bool QMRegion::Converged() const {
423 if (!E_hist_.filled()) {
424 return false;
425 }
426
427 const double Echange = E_hist_.getDiff();
428 const Eigen::MatrixXd diff_ao = Dmat_hist_.getDiff();
429 const double n = double(diff_ao.cols());
430 // The density change is measured in the Loewdin-orthonormalised basis,
431 // S^1/2 dD S^1/2. In the AO basis, a near-linearly dependent basis lets
432 // density-matrix elements move a lot along near-redundant directions while
433 // the density itself hardly changes (C60/def2-TZVP: max change ~170 times
434 // the RMS), which made the max criterion decide the convergence alone.
435 const bool orthonormal = overlap_sqrt_.rows() == diff_ao.rows();
436 const Eigen::MatrixXd diff =
437 orthonormal ? Eigen::MatrixXd(overlap_sqrt_ * diff_ao * overlap_sqrt_)
438 : diff_ao;
439 const double Dchange = diff.norm() / n;
440 const double Dmax = diff.cwiseAbs().maxCoeff();
441 std::string info = "not converged";
442 bool converged = false;
443 if (Dchange < DeltaD_ && Dmax < DeltaDmax_ && std::abs(Echange) < DeltaE_) {
444 info = "converged";
445 converged = true;
446 }
447 std::ostringstream measure;
448 if (orthonormal) {
449 measure << " (orthonormal basis; AO basis: RMS " << diff_ao.norm() / n
450 << " Max " << diff_ao.cwiseAbs().maxCoeff() << ")";
451 } else {
452 measure << " (AO basis)";
453 }
454 XTP_LOG(Log::error, log_)
455 << " Region:" << this->identify() << " " << this->getId() << " is "
456 << info << " deltaE=" << Echange << " RMS Dmat=" << Dchange
457 << " MaxDmat=" << Dmax << measure.str() << std::flush;
458 return converged;
459}

461void QMRegion::Evaluate(std::vector<std::unique_ptr<Region> >& regions) {
462
463 std::vector<double> interact_energies = ApplyInfluenceOfOtherRegions(regions);
464 double e_ext =
465 std::accumulate(interact_energies.begin(), interact_energies.end(), 0.0);
466 XTP_LOG(Log::info, log_)
467 << TimeStamp()
468 << " Calculated interaction potentials with other regions. E[hrt]= "
469 << e_ext << std::flush;
470 XTP_LOG(Log::info, log_) << "Writing inputs" << std::flush;
471
472 Index crg = 0;
473 if (initstate_.Type() == QMStateType::Electron) {
474 crg = -1;
475 } else if (initstate_.Type() == QMStateType::Hole) {
476 crg = +1;
477 }
478 qmpackage_->setCharge(crg);
479 qmpackage_->setRunDir(workdir_);
480 qmpackage_->WriteInputFile(orb_);
481
482 // Attach the periodic background to xtpdft as a potential sampled on
483 // the integration grid. This used to be one of two routes -- the other
484 // handed over multipole moments and k-vectors for the engine to build
485 // its own AO matrices from -- but nothing ever set that one up, so it
486 // has been removed. See the git history of this file if it is ever
487 // wanted back.
488 if (ewald_grid_ready_) {
489 if (qmpackage_->getPackageName() != "xtp") {
490 throw std::runtime_error("QMEwald can only run with XTP as qmpackage.");
491 }
492 qmpackage_->setEwaldgrid(ewaldgrid_);
493 qmpackage_->setEwaldNuclearEnergy(ewald_nuclear_energy_);
494 }
495
496 // From the second inter-region iteration on, only the environment has
497 // changed: start the SCF from the previous iteration's orbitals.
498 const bool warm_start = E_hist_.filled() && orb_.hasMOs();
499 qmpackage_->setWarmStart(warm_start);
500 // From the second inter-region iteration on, the inner SCF only needs to
501 // be converged a factor 10 beyond what the outer loop resolves: its
502 // energy error is second order in the density error, and its density
503 // error stays below the outer density tolerance. Not with GW-BSE, which
504 // takes the orbitals as they are; capped at 1e-5 in both.
505 if (warm_start && !do_gwbse_) {
506 qmpackage_->setSCFToleranceFloor(std::min(0.1 * DeltaE_, 1e-5),
507 std::min(0.1 * DeltaD_, 1e-5));
508 }
509 // Basis sets and geometry stay the same over the inter-region iterations:
510 // keep the RI integrals and the Ewald potential matrix for the next one.
511 // Not with GW-BSE, whose own three-centre integrals would come on top of
512 // the cached ones in memory. The cache lives here because Reset()
513 // recreates the QM package every iteration.
514 if (!do_gwbse_) {
515 if (!setup_cache_) {
516 setup_cache_ = std::make_shared<DFTSetupCache>();
517 }
518 qmpackage_->setSetupCache(setup_cache_);
519 }
520
521 XTP_LOG(Log::error, log_) << "Running DFT calculation" << std::flush;
522 bool run_success = qmpackage_->Run();
523 if (!run_success) {
524 info_ = false;
525 errormsg_ = "DFT-run failed";
526 return;
527 }
528
529 bool Logfile_parse = qmpackage_->ParseLogFile(orb_);
530 if (!Logfile_parse) {
531 info_ = false;
532 errormsg_ = "Parsing DFT logfile failed.";
533 return;
534 }
535 bool Orbfile_parse = qmpackage_->ParseMOsFile(orb_);
536 if (!Orbfile_parse) {
537 info_ = false;
538 errormsg_ = "Parsing DFT orbfile failed.";
539 return;
540 }
541 if (do_localize_) {
542 if (orb_.isOpenShell()) {
543 throw std::runtime_error(
544 "MO localization not implemented for open-shell systems");
545 }
546 PMLocalization pml(log_, localize_options_);
547 pml.computePML(orb_);
548 }
549 QMMolecule originalmol = orb_.QMAtoms();
550 if (do_dft_in_dft_) {
551 if (orb_.isOpenShell()) {
552 throw std::runtime_error(
553 "DFT-in-DFT not implemented for open-shell systems");
554 }
555 // this only works with XTPDFT, so locally override global qmpackage_
556 std::unique_ptr<QMPackage> xtpdft =
557 std::unique_ptr<QMPackage>(QMPackageFactory().Create("xtp"));
558 xtpdft->setLog(&log_);
559 xtpdft->Initialize(dftoptions_);
560 xtpdft->setRunDir(workdir_);
561 Index charge = 0;
562 if (initstate_.Type() == QMStateType::Electron) {
563 charge = -1;
564 } else if (initstate_.Type() == QMStateType::Hole) {
565 charge = +1;
566 }
567 xtpdft->setCharge(charge);
568
569 std::vector<double> energies = std::vector<double>(regions.size(), 0.0);
570 for (std::unique_ptr<Region>& reg : regions) {
571 Index id = reg->getId();
572 if (id == this->getId()) {
573 continue;
574 }
575
576 StaticRegion Staticdummy(0, log_);
577 PolarRegion Polardummy(0, log_);
578 XTP_LOG(Log::error, log_)
579 << TimeStamp() << " Evaluating interaction between "
580 << this->identify() << " " << this->getId() << " and "
581 << reg->identify() << " " << reg->getId() << std::flush;
582 if (reg->identify() == Staticdummy.identify()) {
583 StaticRegion* staticregion = dynamic_cast<StaticRegion*>(reg.get());
584 xtpdft->AddRegion(*staticregion);
585 } else if (reg->identify() == Polardummy.identify()) {
586 PolarRegion* polarregion = dynamic_cast<PolarRegion*>(reg.get());
587 xtpdft->AddRegion(*polarregion);
588 } else {
589 throw std::runtime_error(
590 "Interaction of regions with types:" + this->identify() + " and " +
591 reg->identify() + " not implemented");
592 }
593 }
594
595 e_ext = std::accumulate(interact_energies.begin(), interact_energies.end(),
596 0.0);
597 XTP_LOG(Log::info, log_)
598 << TimeStamp()
599 << " Calculated interaction potentials with other regions. E[hrt]= "
600 << e_ext << std::flush;
601 XTP_LOG(Log::info, log_) << "Writing inputs" << std::flush;
602 xtpdft->WriteInputFile(orb_);
603 bool active_run = xtpdft->RunActiveRegion();
604 if (!active_run) {
605 throw std::runtime_error("\n DFT in DFT embedding failed. Stopping!");
606 }
607 Logfile_parse = xtpdft->ParseLogFile(orb_);
608 if (!Logfile_parse) {
609 throw std::runtime_error("\n Parsing DFT logfile failed. Stopping!");
610 }
611 }
612
613 QMState state = QMState("groundstate");
614 double energy = orb_.getDFTTotalEnergy();
615
616 if (initstate_.Type().isKSState()) {
617 state = statetracker_.CalcStateAndUpdate(orb_);
618 // if unoccupied, add QP level energy
619 if (state.StateIdx() > orb_.getHomo()) {
620 energy += orb_.getExcitedStateEnergy(state);
621 } else {
622 // if unoccupied, subtract QP level energy
623 energy -= orb_.getExcitedStateEnergy(state);
624 }
625 }
626
627 // With environment screening the loop converges the ground state only;
628 // GW-BSE runs once afterwards (EvaluateScreenedGWBSE).
629 if (do_gwbse_ && !screening_) {
630 PrepareOrbitalsForGWBSE();
631 GWBSE gwbse(orb_);
632 gwbse.setLogger(&log_);
633 gwbse.Initialize(gwbseoptions_);
634 gwbse.Evaluate();
635 state = statetracker_.CalcStateAndUpdate(orb_);
636 energy += StateEnergy(state);
637 }
638 // If truncation was enabled then rewrite full basis/aux-basis, MOs in full
639 // basis and full QMAtoms
640 if (orb_.getCalculationType() == "Truncated") {
641 orb_.MOs().eigenvectors() = orb_.getTruncMOsFullBasis();
642 orb_.QMAtoms().clearAtoms();
643 orb_.QMAtoms() = originalmol;
644 orb_.SetupDftBasis(orb_.getDftBasis().Name());
645 orb_.SetupAuxBasis(orb_.getAuxBasis().Name());
646 }
647 E_hist_.push_back(energy);
648
649 Dmat_hist_.push_back(orb_.DensityMatrixFull(state));
650 UpdateOverlapSqrt();
651 return;
652}

654void QMRegion::UpdateOverlapSqrt() {
655 if (overlap_sqrt_.rows() == Dmat_hist_.back().rows()) {
656 return;
657 }
658 overlap_sqrt_.resize(0, 0);
659 if (orb_.getBasisSetSize() != Dmat_hist_.back().rows() ||
660 orb_.getDFTbasisName().empty()) {
661 return; // no matching basis to orthonormalise with: AO measure
662 }
663 AOOverlap overlap;
664 try {
665 overlap.Fill(orb_.getDftBasis());
666 } catch (const std::runtime_error&) {
667 return;
668 }
669 if (overlap.Matrix().rows() != Dmat_hist_.back().rows()) {
670 return;
671 }
672 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es(overlap.Matrix());
673 overlap_sqrt_ = es.eigenvectors() *
674 es.eigenvalues().cwiseMax(0.0).cwiseSqrt().asDiagonal() *
675 es.eigenvectors().transpose();
676}

687double QMRegion::charge() const {
688 double charge = 0.0;
689 if (!do_gwbse_) {
690 Index nuccharge = 0;
691 for (const QMAtom& a : orb_.QMAtoms()) {
692 nuccharge += a.getNuccharge();
693 }
694
695 Index electrons = orb_.getNumberOfAlphaElectrons();
696 if (orb_.isOpenShell()) {
697 electrons += orb_.getNumberOfBetaElectrons();
698 } else {
699 electrons *= 2;
700 }
701
702 charge = double(nuccharge - electrons);
703 } else {
704 QMState state = statetracker_.InitialState();
705 if (state.Type().isExciton()) {
706 charge = 0.0;
707 } else if (state.Type().isSingleParticleState() ||
708 state.Type().isKSState()) {
709 if (state.StateIdx() <= orb_.getHomo()) {
710 charge = +1.0;
711 } else {
712 charge = -1.0;
713 }
714 }
715 }
716 return charge;
717}

770void QMRegion::ApplyQMFieldToPolarSegments(
771 std::vector<PolarSegment>& segments) const {
772
773 Vxc_Grid grid;
774 AOBasis basis =
775 orb_.getDftBasis(); // grid needs a basis in scope all the time
776 grid.GridSetup(grid_accuracy_for_ext_interaction_, orb_.QMAtoms(), basis);
777 DensityIntegration<Vxc_Grid> numint(grid);
778
779 // With environment screening the environment is polarized by the
780 // ground state only: the excitation's own polarization is in W.
781 QMState state = QMState("groundstate");
782 if ((do_gwbse_ && !screening_) || initstate_.Type().isKSState()) {
783 state = statetracker_.CalcState(orb_);
784 }
785 Eigen::MatrixXd dmat = orb_.DensityMatrixFull(state);
786 double Ngrid = numint.IntegrateDensity(dmat);
787 AOOverlap overlap;
788 overlap.Fill(basis);
789 double N_comp = dmat.cwiseProduct(overlap.Matrix()).sum();
790 if (std::abs(Ngrid - N_comp) > 0.001) {
791 XTP_LOG(Log::error, log_) << "=======================" << std::flush;
792 XTP_LOG(Log::error, log_)
793 << "WARNING: Calculated Densities at Numerical Grid, Number of "
794 "electrons "
795 << Ngrid << " is far away from the the real value " << N_comp
796 << ", you should increase the accuracy of the integration grid."
797 << std::flush;
798 }
799#pragma omp parallel for
800 for (Index i = 0; i < Index(segments.size()); ++i) {
801 PolarSegment& seg = segments[i];
802 for (PolarSite& site : seg) {
803 site.V_noE() += numint.IntegrateField(site.getPos());
804 }
805 }
806
807 StaticSegment seg(orb_.QMAtoms().getType(), orb_.QMAtoms().getId());
808 for (const QMAtom& atom : orb_.QMAtoms()) {
809 seg.push_back(StaticSite(atom, double(atom.getNuccharge())));
810 }
811 AddNucleiFields(segments, seg);
812}

859double QMRegion::InteractwithEwaldRegion(const EwaldRegion& region) {
860 // First contact with an EwaldRegion is what decides this region takes its
861 // environment as a potential on a grid, so the grid is built here rather
862 // than in Initialize. See the declaration for why eagerly would be wrong.
863 if (!ewald_grid_ready_) {
864 PrepareEwaldPotentialGrid();
865 }
866
867 // ONCE PER JOB, not once per iteration. phi is the potential of a FROZEN
868 // background evaluated at FIXED points: the registry is converged before
869 // the job starts and nothing re-polarizes it, the foreground registration
870 // is made once by JobTopology, and the grid follows the QM geometry,
871 // which does not move inside the inter-region SCF loop. The QM density
872 // enters nowhere in it. So re-evaluating would give bit-identical numbers
873 // at the cost of a full lattice sum over every quadrature point, every
874 // iteration. The values stay where they were written -- in ewaldgrid_'s
875 // own boxes and in ewald_nuclear_energy_, both members of this region
876 // that Reset() does not touch.
877 //
878 // This stops being true the moment the background is allowed to respond
879 // to the foreground. If that ever changes, this is the guard to remove.
880 if (ewald_potential_evaluated_) {
881 return 0.0;
882 }
883
884 // ELECTRONS. The potential on the integration grid, which the DFT
885 // engine integrates against the density into H0. The nuclei (below) are
886 // appended to the same call: every call recomputes the reciprocal-space
887 // structure factors of the whole background, whatever the point count.
888 std::vector<Eigen::Vector3d> points = copyEwaldGrid();
889 const Index n_grid = Index(points.size());
890 for (const QMAtom& atom : orb_.QMAtoms()) {
891 points.push_back(atom.getPos());
892 }
893 const Eigen::VectorXd phi_all = region.PotentialAt(points);
894 const Eigen::VectorXd phi = phi_all.head(n_grid);
895 const Eigen::VectorXd phi_nuclei = phi_all.tail(phi_all.size() - n_grid);
896
897 // Vxc_Grid::getGridpoints() walks grid_boxes_ in order and each box's
898 // own points in order, so the flat vector maps back by giving each box
899 // its share in turn. Checked rather than assumed: a mispairing here is
900 // a wrong Hamiltonian with nothing to notice it by.
901 Index offset = 0;
902 for (Index i = 0; i < ewaldgrid_.getBoxesSize(); ++i) {
903 GridBox& box = ewaldgrid_[i];
904 const Index n_box = Index(box.getGridPoints().size());
905 std::vector<double>& values = box.getPotentialValues();
906 values.resize(std::size_t(n_box));
907 for (Index p = 0; p < n_box; ++p) {
908 values[std::size_t(p)] = phi[offset + p];
909 }
910 offset += n_box;
911 }
912 if (offset != phi.size()) {
913 throw std::runtime_error(
914 "QMRegion::InteractwithEwaldRegion: the grid holds " +
915 std::to_string(offset) + " points across its boxes but " +
916 std::to_string(phi.size()) +
917 " potential values were evaluated. getGridpoints() and the box "
918 "walk above disagree about the point ordering.");
919 }
920
921 // NUCLEI. The grid potential reaches the electron density only; the
922 // nuclei sit in the same potential and have no other route in. Handed
923 // to the engine as a scalar to add to E0, NOT returned from here --
924 // QMRegion::Evaluate accumulates the Interactwith* return values into
925 // e_ext, logs it, and drops it on the floor.
926 ewald_nuclear_energy_ = 0.0;
927 Index a = 0;
928 for (const QMAtom& atom : orb_.QMAtoms()) {
929 ewald_nuclear_energy_ += double(atom.getNuccharge()) * phi_nuclei[a];
930 ++a;
931 }
932
933 ewald_potential_evaluated_ = true;
934
935 // 0.0, matching this class's own InteractwithPolarRegion and
936 // InteractwithStaticRegion: the environment enters the Hamiltonian, and
937 // its energy comes out in the DFT total rather than here.
938 return 0.0;
939}
