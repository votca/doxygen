382 struct CDFTConstraintSpec {
383 std::vector<Index> atom_indices;
384 // Relative to the fragment's own neutral reference state (the sum
385 // of its atoms' nuclear charges) -- e.g. +1.0 means one electron
386 // REMOVED (a cation). Converted to an absolute target electron
387 // count once, inside BuildCDFTConstraint, matching CP2K's own
388 // internal (absolute) TARGET convention exactly -- only the
389 // user-facing options syntax is charge-relative, per the earlier
390 // design discussion on this.
391 double target_charge = 0.0;
392 double initial_lambda = 0.0;
393 // "warmstart" (default) or "fresh" -- see this field's own XML
394 // help text (dftpackage.xml) for the full reasoning. Stored as
395 // the raw string, not a bool, so an invalid value (a typo, say)
396 // is caught by the XML schema's own choices="..." validation
397 // rather than silently defaulting to one behavior or the other.
398 std::string guess_strategy = "warmstart";
399 };
