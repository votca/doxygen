49 class Scope {
50 public:
51 Scope(DFTTimings& timings, std::string name)
52 : timings_(timings), name_(std::move(name)), start_(Clock::now()) {
53 timings_.nested_.push_back(0.0);
54 }
55 ~Scope() {
56 double elapsed =
57 std::chrono::duration<double>(Clock::now() - start_).count();
58 double nested = timings_.nested_.back();
59 timings_.nested_.pop_back();
60 timings_.AddEntry(name_, elapsed - nested);
61 if (!timings_.nested_.empty()) {
62 timings_.nested_.back() += elapsed;
63 }
64 }
65 Scope(const Scope&) = delete;
66 Scope& operator=(const Scope&) = delete;
67
68 private:
69 DFTTimings& timings_;
70 std::string name_;
71 Clock::time_point start_;
72 };
73
74 Scope Measure(const std::string& name) { return Scope(*this, name); }
75
78 void Add(const std::string& name, double seconds) {
79 AddEntry(name, seconds);
80 if (!nested_.empty()) {
81 nested_.back() += seconds;
82 }
83 }
84
85 void Reset() {
86 entries_.clear();
87 nested_.clear();
88 start_ = Clock::now();
89 }
90
91 double Elapsed() const {
92 return std::chrono::duration<double>(Clock::now() - start_).count();
93 }
94
96 void Report(Logger& log, Log::Level level) const;
97
101 static double ResidentMemoryGB(bool peak);
102
103 private:
104 void AddEntry(const std::string& name, double seconds) {
105 for (Entry& e : entries_) {
106 if (e.name == name) {
107 e.seconds += seconds;
108 ++e.calls;
109 return;
110 }
111 }
112 entries_.push_back({name, seconds, 1});
113 }
114
115 struct Entry {
116 std::string name;
117 double seconds;
118 long calls;
119 };
120 std::vector<Entry> entries_;
121 std::vector<double> nested_; // time of finished inner scopes, per level
122 Clock::time_point start_ = Clock::now();
123};
