35// BUG FIX (this session): PolarSegment::getPos() (via AtomContainer<T>::
36// calcPos()) is a MASS-weighted center of mass. Legacy's own equivalent
37// (PolarSeg::CalcPos()) is a plain, UNWEIGHTED arithmetic mean of site
38// positions instead -- confirmed by direct comparison of the two
39// implementations. Since this position feeds directly into the
40// minimum-image PBC wrap below (raw_offset = target - source's own
41// representative position), even a small difference between the two
42// definitions can flip which periodic image is chosen as the true
43// minimum for a given pair -- a discrete, not continuous, effect,
44// consistent with the scattered (not uniformly scaled) per-site
45// mismatch pattern that motivated this fix, rather than a smooth
46// site-by-site drift.
47//
48// Deliberately NOT fixed by changing calcPos() itself: that's a shared
49// AtomContainer<T> method used well beyond this file (md2qmengine,
50// segmentmapper), where a genuine physical mass-weighted center of mass
51// may be exactly what's wanted. This local helper instead gives THIS
52// file its own, legacy-matching definition, without touching shared
53// code any other caller depends on.
54Eigen::Vector3d UnweightedCentroid(const PolarSegment& seg) {
55 Eigen::Vector3d pos = Eigen::Vector3d::Zero();
56 Index n = 0;
57 for (const PolarSite& site : seg) {
58 pos += site.getPos();
59 ++n;
60 }
61 if (n > 0) {
62 pos /= double(n);
63 }
64 return pos;
65}

68EwaldRealSpaceSum::EwaldRealSpaceSum(
69 const Eigen::Matrix3d& box, const EwaldRegistry& registry, double alpha,
70 double thole_a, double r_min, double field_tol, double shell_width,
71 Index n_max, double screening_factor,
72 const std::vector<std::pair<Index, Eigen::Vector3d>>& foreground)
73 : real_space_cutoff_(screening_factor / alpha),
74 segment_radius_(0.0),
75 box_(box),
76 registry_(registry),
77 interactor_(alpha, thole_a),
78 r_min_(r_min),
79 field_tol_(field_tol),
80 shell_width_(shell_width),
81 n_max_(n_max) {
82 translations_ = GenerateSortedTranslations();
83
84 for (const auto& entry : foreground) {
85 foreground_[entry.first].push_back(entry.second);
86 }
87
88 // Largest site-to-centroid distance anywhere in the registry -- the
89 // margin the segment-granular distance cull needs (see AddFieldAt).
90 // Computed once here rather than per call: the registry's geometry is
91 // fixed for this object's lifetime, the same assumption the neighbor
92 // cache already relies on.
93 for (Index id : registry_.AllIds()) {
94 for (EwaldChargeState state :
95 {EwaldChargeState::Neutral, EwaldChargeState::Electron,
96 EwaldChargeState::Hole}) {
97 if (!registry_.Has(id, state)) {
98 continue;
99 }
100 const PolarSegment& segment = registry_.Get(id, state);
101 const Eigen::Vector3d centroid = UnweightedCentroid(segment);
102 for (const PolarSite& site : segment) {
103 segment_radius_ =
104 std::max(segment_radius_, (site.getPos() - centroid).norm());
105 }
106 }
107 }
108}

110std::vector<EwaldRealSpaceSum::Translation>
111 EwaldRealSpaceSum::GenerateSortedTranslations() const {
112 std::vector<Translation> translations;
113 const Eigen::Vector3d a = box_.col(0);
114 const Eigen::Vector3d b = box_.col(1);
115 const Eigen::Vector3d c = box_.col(2);
116
117 for (Index na = -n_max_; na <= n_max_; ++na) {
118 for (Index nb = -n_max_; nb <= n_max_; ++nb) {
119 for (Index nc = -n_max_; nc <= n_max_; ++nc) {
120 Eigen::Vector3d t = double(na) * a + double(nb) * b + double(nc) * c;
121 translations.push_back({t, t.norm()});
122 }
123 }
124 }
125
126 std::sort(
127 translations.begin(), translations.end(),
128 [](const Translation& x, const Translation& y) { return x.r < y.r; });
129 return translations;
130}

132template <enum Estatic CE>
133void EwaldRealSpaceSum::ApplyNeighborList(const NeighborList& list,
134 PolarSite& target,
135 bool include_static) const {
136 for (const auto& entry : list) {
137 // Stored by pointer rather than by id -- see neighbor_cache_'s own
138 // declaration for why. The Has()/Get() guard the id-based version
139 // needed is gone with it: the cache is keyed on source_state, so
140 // every entry in THIS list was registered at THIS state when the
141 // list was built, and segment addresses are stable for the
142 // registry's lifetime.
143 const PolarSegment& source_segment = *std::get<0>(entry);
144 const Index translation_idx = std::get<1>(entry);
145 const Eigen::Vector3d& baseline_shift = std::get<2>(entry);
146 const Eigen::Vector3d t = baseline_shift + translations_[translation_idx].t;
147 for (const PolarSite& source_site : source_segment) {
148 // Shift passed through rather than applied to a copy of the
149 // source site -- see ApplyStaticField's own source_shift note.
150 if (include_static) {
151 interactor_.ApplyStaticField<PolarSite, CE>(source_site, target, t);
152 }
153 interactor_.ApplyInducedField<CE>(source_site, target, t);
154 }
155 }
156}

160 const SearchCounts& counts) const {

168template <enum Estatic CE>
169void EwaldRealSpaceSum::AddFieldAt(Index target_segment_id, PolarSite& target,
170 EwaldChargeState source_state,
171 bool include_static) const {
172 const std::pair<const PolarSite*, EwaldChargeState> cache_key(&target,
173 source_state);
174 auto cached = neighbor_cache_.find(cache_key);
175 if (cached == neighbor_cache_.end()) {
176 // First call for this target: run the convergence search on a
177 // scratch copy, store the list, then fall through to the same path
178 // every later call takes. Applying the list to target adds exactly
179 // the contributions the search added to the copy, in the same order
180 // and starting from the same V, so the field is bit-identical to
181 // accumulating it during the search.
182 SearchCounts counts;
183 PolarSite scratch = target;
184 NeighborList list = SearchNeighbors<CE>(
185 target_segment_id, scratch, source_state, include_static, counts);
186 InsertNeighborList(cache_key, std::move(list), counts);
187 cached = neighbor_cache_.find(cache_key);
188 }
189 // Fast path: the real, geometry-determined neighbor set for this
190 // target (see this class's own documentation for why reusing it across
191 // calls for the same target is safe). Just apply every cached (source,
192 // translation) pair directly -- no shell-by-shell search, no scan of
193 // registry_.AllIds() members that turned out not to matter.
194 ApplyNeighborList<CE>(cached->second, target, include_static);
195}

197template <enum Estatic CE>
198EwaldRealSpaceSum::NeighborList EwaldRealSpaceSum::SearchNeighbors(
199 Index target_segment_id, PolarSite& target, EwaldChargeState source_state,
200 bool include_static, SearchCounts& counts) const {
201 // The full convergence search. Records every (source, translation)
202 // pair actually visited; the field it accumulates into target (a
203 // scratch copy, see the declaration) is only there for the
204 // shell-convergence test. Nothing in this object is written.
205 NeighborList visited_pairs;
206
207 // See the cull inside the shell loop below for what this is and why.
208 const double cutoff_with_margin = real_space_cutoff_ + 2.0 * segment_radius_;
209
210 Index shell_start = 0;
211 double shell_edge = 0.0;
212 bool converged = false;
213
214 while (shell_start < Index(translations_.size())) {
215 // Advance shell_edge by shell_width_ until it covers at least the next
216 // not-yet-processed translation, then collect every translation up to
217 // that edge into this shell. Since translations_ is sorted by
218 // distance, this always yields a contiguous, radially-ordered shell.
219 shell_edge =
220 translations_[shell_start].r +
221 (shell_edge > translations_[shell_start].r ? 0.0 : shell_width_);
222 Index shell_end = shell_start;
223 while (shell_end < Index(translations_.size()) &&
224 translations_[shell_end].r <= shell_edge) {
225 ++shell_end;
226 }
227
228 // Field is written into target.V()/V_noE() by ApplyStaticField/
229 // ApplyInducedField as a side effect; recover this shell's own
230 // contribution by differencing target's accumulator before and after
231 // processing the whole shell.
232 const Eigen::Vector3d before_shell =
233 (CE == Estatic::noE_V) ? target.V_noE() : target.V();
234
235 for (Index source_id : registry_.AllIds()) {
236 // No per-segment skip: a target's own segment is treated like any
237 // other. Its COINCIDENT copy is dropped by the foreground
238 // suppression below; its other lattice images are ordinary
239 // background molecules and stay. Legacy agrees -- SetupMidground
240 // excludes by (segment id, na, nb, nc), not by segment id.
241 if (!registry_.Has(source_id, source_state)) {
242 continue;
243 }
244 const PolarSegment& source_segment =
245 registry_.Get(source_id, source_state);
246
247 // Per-source baseline wrap (computed once per source_id, not per
248 // shell-translation index, since it depends only on the target
249 // and source_segment's own representative positions, not on
250 // which small shell-translation t is currently being applied).
251 //
252 // BUG FIX (this session, found via a real production comparison
253 // against legacy): translations_ is sorted and shell-converged
254 // by |t| alone (the raw lattice-translation magnitude), but the
255 // quantity that actually determines a pair's true separation is
256 // |raw_offset + t|, not |t| by itself. A translation with a
257 // LARGE |t| can still be the one that gives the SMALLEST actual
258 // distance, if it happens to cancel a large raw target-source
259 // offset -- and such a translation is exactly what the
260 // shell-convergence criterion (stop once shell radius >= r_min_)
261 // can miss entirely, since it never even considers translations
262 // whose OWN magnitude exceeds r_min_, regardless of what they'd
263 // produce once combined with the raw offset. Confirmed on a real
264 // pair: legacy's own minimum-image search (round-based,
265 // independent of any shell-magnitude cutoff) found r=13.6 bohr
266 // via t=(0,-1,-1)*box (|t|~97 bohr, comfortably past r_min_ in
267 // this run's own ~75.6 bohr setting) for a pair this class's own
268 // unfixed search reported no closer than r=55.4 bohr on, because
269 // it never got there.
270 //
271 // Fixed by mirroring legacy's own two-step structure exactly:
272 // wrap the raw (large, unbounded) target-source separation into
273 // its true minimum image FIRST (fractional-coordinate rounding,
274 // correct for any box_ shape, not just orthorhombic), THEN run
275 // the existing small, bounded shell search of translations_ on
276 // top of that already-small baseline -- exactly as legacy's own
277 // dr12_pbc + L structure does. The shell search only ever needs
278 // to explore the neighborhood right around the true minimum
279 // image, not the raw, unbounded separation, so its own
280 // shell_radius >= r_min_ stopping criterion is now sound (it was
281 // never wrong in isolation -- it was being applied to the wrong
282 // starting point).
283 const Eigen::Vector3d source_centroid =
284 UnweightedCentroid(source_segment);
285 const Eigen::Vector3d raw_offset = target.getPos() - source_centroid;
286 const Eigen::Vector3d frac = box_.inverse() * raw_offset;
287 const Eigen::Vector3d wrapped_frac = frac - frac.array().round().matrix();
288 const Eigen::Vector3d min_image_offset = box_ * wrapped_frac;
289 const Eigen::Vector3d baseline_shift = raw_offset - min_image_offset;
290
291 for (Index idx = shell_start; idx < shell_end; ++idx) {
292 // Distance cull. The erfc(alpha*r) screening means a pair's
293 // contribution falls off as fast as erfc does: at alpha*r = 6 a
294 // single pair contributes ~1e-20 of the total field, and the
295 // whole remaining tail sums to far below field_tol_. Without
296 // this test every registered segment is evaluated at every
297 // accepted translation, however far away it is -- measured on a
298 // real 1000-segment system as 14985 (segment, translation)
299 // entries per target, of which only ~22% lie inside the cutoff.
300 // That factor of ~4.6, not any per-pair cost, was the whole
301 // remaining real-space gap against legacy (whose own
302 // PolarNbs list is distance-built, so it never had the excess).
303 //
304 // r_min_ does NOT already do this: it bounds the shell search by
305 // TRANSLATION magnitude |t|, which is unrelated to how far a
306 // given source segment is once that translation is applied.
307 //
308 // The true separation for this (segment, translation) pair is
309 // |min_image_offset - translations_[idx].t| -- the same quantity
310 // the interactor will form, since it evaluates
311 // target - (source + t) with t = baseline_shift +
312 // translations_[idx].t and baseline_shift = raw_offset -
313 // min_image_offset. Compared at segment granularity, so
314 // segment_radius_ (the largest site-to-centroid distance in the
315 // registry) is added twice as a margin: no individual site pair
316 // can then be closer than the cutoff while its segment pair is
317 // culled.
318 const double pair_distance =
319 (min_image_offset - translations_[idx].t).norm();
320 if (pair_distance > cutoff_with_margin) {
321 ++counts.culled;
322 continue;
323 }
324 const Eigen::Vector3d t = baseline_shift + translations_[idx].t;
325
326 // Foreground suppression. This one periodic copy of this segment
327 // is handled explicitly elsewhere (a polar region), so it must
328 // not also appear in the periodic background -- see the
329 // constructor's own foreground documentation. Only the copy
330 // sitting at the recorded position is dropped; the segment's
331 // other lattice images stay in the sum.
332 if (!foreground_.empty()) {
333 auto fg = foreground_.find(source_id);
334 if (fg != foreground_.end()) {
335 const Eigen::Vector3d shifted_centroid = source_centroid + t;
336 bool suppressed = false;
337 for (const Eigen::Vector3d& fg_pos : fg->second) {
338 if ((shifted_centroid - fg_pos).norm() < kForegroundMatchTol) {
339 suppressed = true;
340 break;
341 }
342 }
343 if (suppressed) {
344 ++counts.foreground;
345 continue;
346 }
347 }
348 }
349
350 // The target's own segment at ZERO translation is the segment
351 // itself: intramolecular pairs plus the r = 0 self-pair, never
352 // part of this sum (AddIntraSegmentCoupling owns the induced
353 // side, the permanent side has its own compensation pass).
354 //
355 // Only where the segment has no recorded foreground copy. Where
356 // it does, the suppression above already decides which copy is
357 // carved out, and that need not be the t = 0 one: a foreground
358 // segment sitting at a nonzero image has its t = 0 copy as a
359 // genuine neighbour. Omitting this guard made the field infinite
360 // in every background solve.
361 const bool source_has_foreground =
362 !foreground_.empty() && foreground_.count(source_id) > 0;
363 if (!source_has_foreground && source_id == target_segment_id &&
364 t.squaredNorm() < kSelfTranslationTol * kSelfTranslationTol) {
365 continue;
366 }
367
368 visited_pairs.emplace_back(&source_segment, idx, baseline_shift);
369 for (const PolarSite& source_site : source_segment) {
370 if (include_static) {
371 interactor_.ApplyStaticField<PolarSite, CE>(source_site, target, t);
372 }
373 interactor_.ApplyInducedField<CE>(source_site, target, t);
374 }
375 }
376 }
377
378 const Eigen::Vector3d after_shell =
379 (CE == Estatic::noE_V) ? target.V_noE() : target.V();
380 const Eigen::Vector3d shell_field = after_shell - before_shell;
381
382 const double shell_radius =
383 translations_[shell_end > shell_start ? shell_end - 1 : shell_start].r;
384 if (shell_radius >= r_min_ && shell_field.norm() < field_tol_) {
385 converged = true;
386 break;
387 }
388
389 shell_start = shell_end;
390 }
391
392 if (!converged) {
393 std::stringstream message;
394 message << "EwaldRealSpaceSum: real-space sum for segment "
395 << target_segment_id
396 << " did not converge within the n_max lattice-translation "
397 "search box; increase n_max or field_tol.";
398 throw std::runtime_error(message.str());
399 }
400
401 return visited_pairs;
402}

404double EwaldRealSpaceSum::CalcStaticEnergyAt(
405 const PolarSite& target, EwaldChargeState source_state) const {
406 // See this method's own declaration for what is and is not included.
407 const std::pair<const PolarSite*, EwaldChargeState> cache_key(&target,
408 source_state);
409 auto cached = neighbor_cache_.find(cache_key);
410 if (cached == neighbor_cache_.end()) {
411 throw std::runtime_error(
412 "EwaldRealSpaceSum::CalcStaticEnergyAt: no neighbour list for this "
413 "target. AddFieldAt or PrepareNeighborCache must run first -- this "
414 "method deliberately does not build one, so that an energy query "
415 "cannot silently become the expensive shell search.");
416 }
417
418 double energy = 0.0;
419 for (const auto& entry : cached->second) {
420 const PolarSegment& source_segment = *std::get<0>(entry);
421 const Index translation_idx = std::get<1>(entry);
422 const Eigen::Vector3d& baseline_shift = std::get<2>(entry);
423 const Eigen::Vector3d t = baseline_shift + translations_[translation_idx].t;
424 for (const PolarSite& source_site : source_segment) {
425 energy += interactor_.CalcStaticEnergy<PolarSite, PolarSite>(source_site,
426 target, t);
427 }
428 }
429 return energy;
430}

432Eigen::VectorXd EwaldRealSpaceSum::PotentialAtMany(
433 Index target_segment_id, const std::vector<Eigen::Vector3d>& points,
434 EwaldChargeState source_state) const {
435 // Sources flattened once: AllIds() returns by value and the centroids
436 // are fixed, so recomputing either per point would cost more than the
437 // sum itself on a grid.
438 std::vector<const PolarSegment*> segments;
439 std::vector<Index> segment_ids;
440 std::vector<Eigen::Vector3d> centroids;
441 for (Index source_id : registry_.AllIds()) {
442 if (!registry_.Has(source_id, source_state)) {
443 continue;
444 }
445 const PolarSegment& segment = registry_.Get(source_id, source_state);
446 segments.push_back(&segment);
447 segment_ids.push_back(source_id);
448 centroids.push_back(UnweightedCentroid(segment));
449 }
450
451 const double cutoff_with_margin = real_space_cutoff_ + 2.0 * segment_radius_;
452 const Index n_points = Index(points.size());
453 const Index n_sources = Index(segments.size());
454 Eigen::VectorXd phi = Eigen::VectorXd::Zero(n_points);
455
456 // Hoisted out of both loops below: the box is fixed for this object's
457 // lifetime, and inverting it per (point, source) is n_points * n_sources
458 // 3x3 inversions for one matrix.
459 const Eigen::Matrix3d box_inv = box_.inverse();
460
461#pragma omp parallel for schedule(dynamic, 8)
462 for (Index p = 0; p < n_points; ++p) {
463 const Eigen::Vector3d& point = points[std::size_t(p)];
464
465 // Unit test charge: every energy routine here reduces to
466 // q*phi - mu.E, so with q = 1 and mu = 0 what comes back is phi.
467 // Built per point on the stack, which is safe only because nothing
468 // below keys anything on its address.
469 PolarSite probe(0, "H", point);
470 probe.Reset();
471 probe.setCharge(1.0);
472 probe.setStaticDipole(Eigen::Vector3d::Zero());
473 probe.setInduced_Dipole(Eigen::Vector3d::Zero());
474
475 double acc = 0.0;
476 for (Index s_i = 0; s_i < n_sources; ++s_i) {
477 const PolarSegment& source_segment = *segments[std::size_t(s_i)];
478 const Index source_id = segment_ids[std::size_t(s_i)];
479 const Eigen::Vector3d& source_centroid = centroids[std::size_t(s_i)];
480
481 // Minimum image first, shell search on top of it -- see AddFieldAt's
482 // own account of why the raw separation cannot be handed to a
483 // search bounded by |t|.
484 const Eigen::Vector3d raw_offset = point - source_centroid;
485 const Eigen::Vector3d frac = box_inv * raw_offset;
486 const Eigen::Vector3d wrapped_frac = frac - frac.array().round().matrix();
487 const Eigen::Vector3d min_image_offset = box_ * wrapped_frac;
488 const Eigen::Vector3d baseline_shift = raw_offset - min_image_offset;
489
490 const auto fg = foreground_.find(source_id);
491 const bool source_has_foreground = fg != foreground_.end();
492
493 // STOP, don't skip. translations_ is sorted by |t| ascending, and
494 // |min_image_offset - t| >= |t| - |min_image_offset|, so once |t|
495 // passes this limit every remaining translation is culled as well.
496 // The bound is exact, not a heuristic: nothing inside the cutoff can
497 // sit beyond it.
498 //
499 // This matters far more here than in the site-based paths above,
500 // which never walk the whole list -- their shell-convergence search
501 // hands them a [shell_start, shell_end) window. Without the break
502 // this loop runs (2*n_max+1)^3 translations, 29791 at the default
503 // n_max of 15, for EVERY (point, source) pair. On a DFT integration
504 // grid against a 1000-segment registry that is of order 1e12
505 // distance evaluations and hours of wall time; with it the surviving
506 // range is a couple of lattice shells.
507 const double t_limit = cutoff_with_margin + min_image_offset.norm();
508
509 for (std::size_t idx = 0; idx < translations_.size(); ++idx) {
510 if (translations_[idx].r > t_limit) {
511 break;
512 }
513 const double pair_distance =
514 (min_image_offset - translations_[idx].t).norm();
515 if (pair_distance > cutoff_with_margin) {
516 continue;
517 }
518 const Eigen::Vector3d t = baseline_shift + translations_[idx].t;
519
520 // Foreground suppression: this one periodic copy is handled
521 // explicitly elsewhere, so it must not also appear here.
522 if (source_has_foreground) {
523 const Eigen::Vector3d shifted_centroid = source_centroid + t;
524 bool suppressed = false;
525 for (const Eigen::Vector3d& fg_pos : fg->second) {
526 if ((shifted_centroid - fg_pos).norm() < kForegroundMatchTol) {
527 suppressed = true;
528 break;
529 }
530 }
531 if (suppressed) {
532 continue;
533 }
534 }
535
536 if (!source_has_foreground && source_id == target_segment_id &&
537 t.squaredNorm() < kSelfTranslationTol * kSelfTranslationTol) {
538 continue;
539 }
540
541 for (const PolarSite& source_site : source_segment) {
542 acc += interactor_.CalcStaticEnergy<PolarSite, PolarSite>(source_site,
543 probe, t);
544 // UNDAMPED: the target is a point in space, not a
545 // point-polarizable site, so there is no overlap for Thole to
546 // correct -- and the rest of the package already treats
547 // [induced dipole] x [QM density] undamped (AOMultipole and
548 // DFTEngine::ExternalRepulsion apply no Thole at all). See
549 // CalcInducedSourceEnergy's own declaration, including why
550 // this cannot be said by zeroing the probe's polarizability.
551 acc += interactor_.CalcInducedSourceEnergy(source_site, probe, t,
552 /*damp=*/false);
553 }
554 }
555 }
556 phi[p] = acc;
557 }
558 return phi;
559}

561double EwaldRealSpaceSum::CalcInducedSourceEnergyAt(
562 const PolarSite& target, EwaldChargeState source_state) const {
563 // Deliberately a near-copy of CalcStaticEnergyAt above rather than a
564 // shared template over the interactor call: the two differ only in
565 // which moments they contract, but they are validated separately and
566 // against different legacy channels (_pp and half of _pu), and a
567 // shared body would let a change to one silently move the other.
568 const std::pair<const PolarSite*, EwaldChargeState> cache_key(&target,
569 source_state);
570 auto cached = neighbor_cache_.find(cache_key);
571 if (cached == neighbor_cache_.end()) {
572 throw std::runtime_error(
573 "EwaldRealSpaceSum::CalcInducedSourceEnergyAt: no neighbour list for "
574 "this target. AddFieldAt or PrepareNeighborCache must run first -- "
575 "this method deliberately does not build one, so that an energy "
576 "query cannot silently become the expensive shell search.");
577 }
578
579 double energy = 0.0;
580 for (const auto& entry : cached->second) {
581 const PolarSegment& source_segment = *std::get<0>(entry);
582 const Index translation_idx = std::get<1>(entry);
583 const Eigen::Vector3d& baseline_shift = std::get<2>(entry);
584 const Eigen::Vector3d t = baseline_shift + translations_[translation_idx].t;
585 for (const PolarSite& source_site : source_segment) {
586 energy += interactor_.CalcInducedSourceEnergy(source_site, target, t);
587 }
588 }
589 return energy;
590}

592void EwaldRealSpaceSum::PrepareNeighborCache(
593 const std::vector<std::pair<Index, PolarSite*>>& targets,
594 EwaldChargeState source_state, bool converge_static_field) const {
595 // The list AddFieldAt<V>(..., converge_static_field) would build.
596 PrepareNeighborCacheImpl<Estatic::V>(targets, source_state,
597 converge_static_field);

600template <enum Estatic CE>
601void EwaldRealSpaceSum::PrepareNeighborCacheImpl(
602 const std::vector<std::pair<Index, PolarSite*>>& targets,
603 EwaldChargeState source_state, bool include_static) const {
604 // See PrepareNeighborCache's own declaration for what this is for. The search
605 // is the one AddFieldAt runs on a miss (SearchNeighbors), so the two
606 // paths cannot drift apart.
607 //
608 // Serial selection: skip targets that already have a list, and
609 // repeated entries, so each missing list is searched exactly once.
610 std::vector<std::size_t> todo;
611 {
612 std::unordered_map<std::pair<const PolarSite*, EwaldChargeState>, bool,
613 PairHash>
614 seen;
615 for (std::size_t i = 0; i < targets.size(); ++i) {
616 const std::pair<const PolarSite*, EwaldChargeState> key(targets[i].second,
617 source_state);
618 if (neighbor_cache_.count(key) > 0 || !seen.emplace(key, true).second) {
619 continue;
620 }
621 todo.push_back(i);
622 }
623 }
624 if (todo.empty()) {
625 return;
626 }
627
628 // Parallel search. Each iteration reads this object and writes only
629 // its own slots below. An exception cannot leave an omp region, so
630 // the first one is carried out and rethrown afterwards.
631 std::vector<NeighborList> lists(todo.size());
632 std::vector<SearchCounts> counts(todo.size());
633 std::exception_ptr failure = nullptr;
634 const Index n_todo = Index(todo.size());
635#pragma omp parallel for schedule(dynamic, 16)
636 for (Index k = 0; k < n_todo; ++k) {
637 try {
638 const auto& entry = targets[todo[std::size_t(k)]];
639 PolarSite scratch = *entry.second;
640 lists[std::size_t(k)] =
641 SearchNeighbors<CE>(entry.first, scratch, source_state,
642 include_static, counts[std::size_t(k)]);
643 } catch (...) {
644#pragma omp critical(ewald_real_prepare_failure)
645 {
646 if (!failure) {
647 failure = std::current_exception();
648 }
649 }
650 }
651 }
652 if (failure) {
653 std::rethrow_exception(failure);
654 }
655
656 // Serial insertion, in target order.
657 for (std::size_t k = 0; k < todo.size(); ++k) {
658 const std::pair<const PolarSite*, EwaldChargeState> key(
659 targets[todo[k]].second, source_state);
660 InsertNeighborList(key, std::move(lists[k]), counts[k]);
661 }
662}

664template <enum Estatic CE>
665void EwaldRealSpaceSum::AddFieldAtMany(
666 const std::vector<std::pair<Index, PolarSite*>>& targets,
667 EwaldChargeState source_state, bool include_static) const {
668 // Lists first: afterwards every AddFieldAt below takes the read-only
669 // fast path. The list is converged on the same field AddFieldAt's own
670 // on-demand search would have used, so results do not depend on
671 // whether a target was prepared here or found on demand.
672 PrepareNeighborCacheImpl<CE>(targets, source_state, include_static);
673 const Index n_targets = Index(targets.size());
674#pragma omp parallel for schedule(dynamic, 16)
675 for (Index i = 0; i < n_targets; ++i) {
676 const auto& entry = targets[std::size_t(i)];
677 const auto cached = neighbor_cache_.find({entry.second, source_state});
678 ApplyNeighborList<CE>(cached->second, *entry.second, include_static);
679 }
680}
