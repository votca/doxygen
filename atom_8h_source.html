80 // Unit vector toward a covalently bonded MD atom that lies outside the
81 // segment, i.e. a bond cut by the segment definition. Md2QmEngine sets it
82 // from the MD geometry; SegmentMapper rotates it with the fragment's
83 // MD->template transform for the mapped QMAtom. Only the first external
84 // bond of an atom is stored; check hasExternalBond() before use.
85 //
86 // partner_atom_id is the MD atom ID of the partner. It is transient (not
87 // written to the checkpoint file) and only used within

89 // (setExternalBondPartnerSegmentId()) once all segments exist.

115 // IDs (Atom::getId(), not indices) of all covalently bonded atoms, set by
116 // Md2QmEngine from the MD topology for every atom. The fragment relaxation
117 // in FragmentSaturator needs this full internal connectivity. The IDs are
118 // MD-level here and translated to QM-level IDs in SegmentMapper. Unused
119 // slots hold -1.
120 const Index* getBondedPartnerIds() const { return bonded_partner_ids_; }
121 void AddBondedPartner(Index partner_id) {
122 for (Index& slot : bonded_partner_ids_) {
123 if (slot == -1) {
124 slot = partner_id;
125 return;
126 }
127 }
128 // All slots full: partner is dropped (see kMaxBondedPartners).
129 }
