39void DftGwBse::ParseOptions(const tools::Property& options) {
40
41 // molecule coordinates
42 xyzfile_ = job_name_ + ".xyz";
43
44 // options for dft package
45 package_options_ = options.get(".dftpackage");
46
47 // GWBSEENGINE options
48 gwbseengine_options_ = options;
49
50 // lets get the archive file name from the xyz file name
51 archive_file_ = job_name_ + ".orb";
52
53 // XML OUTPUT
54 xml_output_ = job_name_ + "_summary.xml";
55
56 if (options.exists(".excited_state_forces")) {
57 excited_state_forces_ =
58 options.get(".excited_state_forces").as<std::string>();
59 }
60 {
61 const std::string tasks = options.get(".tasks").as<std::string>();
62 tools::Tokenizer tok(tasks, " ,");
63 const std::vector<std::string> list = tok.ToVector();
64 const bool has =
65 std::find(list.begin(), list.end(), "gradients") != list.end();
66 if (has && list.size() > 1) {
67 throw std::runtime_error(
68 "tasks: gradients cannot be combined with other tasks (with dft "
69 "and gwbse the gradients are computed in the same run anyway)");
70 }
71 gradients_only_ = has;
72 }
73 if (options.exists(".qp_gradients")) {
74 qp_gradients_ = options.get(".qp_gradients").as<std::string>();
75 }
76 if (options.exists(".excited_state_forces_follow")) {
77 excited_state_forces_follow_ =
78 options.get(".excited_state_forces_follow").as<std::string>();
79 }
80 if (options.exists(".excited_state_forces_qp_response")) {
81 excited_state_forces_qp_response_ =
82 options.get(".excited_state_forces_qp_response").as<bool>();
83 }
84
85 if (options.exists(".mpsfile")) {
86 mpsfile_ = options.get(".mpsfile").as<std::string>();
87 }
88
89 // check if guess is requested
90 if (options.exists(".guess")) {
91 guess_file_ = options.get(".guess").as<std::string>();
92 }
93
94 // check if a separate MO-only guess is requested (see this class's
95 // own header comment on moguess_file_ for why this is a distinct
96 // option from .guess, not a variant of it)
97 if (options.exists(".moguess")) {
98 moguess_file_ = options.get(".moguess").as<std::string>();
99 }
100
101 // register all QM packages
102 QMPackageFactory{};
103}

105bool DftGwBse::Run() {
106
107 log_.setReportLevel(Log::current_level);
108
109 log_.setMultithreading(true);
110 log_.setCommonPreface("\n... ...");
111
112 // Get orbitals object
113 Orbitals orbitals;
114
115 // tasks = gradients: no DFT or GW-BSE, the requested gradients
116 // (qp_gradients, excited_state_forces) are computed from the data of an
117 // earlier run in the orb file and added to it
118 if (gradients_only_) {
119 if (excited_state_forces_.empty() && qp_gradients_.empty()) {
120 throw std::runtime_error(
121 "tasks gradients: set qp_gradients and/or excited_state_forces");
122 }
123 XTP_LOG(Log::error, log_)
124 << "Gradients from the GW-BSE data in " << archive_file_ << std::flush;
125 orbitals.ReadFromCpt(archive_file_);
126 CheckStoredRun(orbitals);
127 if (!excited_state_forces_.empty()) {
128 ComputeExcitedStateForces(orbitals);
129 }
130 if (!qp_gradients_.empty()) {
131 ComputeQPGradients(orbitals);
132 }
133 XTP_LOG(Log::error, log_)
134 << "Saving data to " << archive_file_ << std::flush;
135 orbitals.WriteToCpt(archive_file_);
136 return true;
137 }
138
139 if (!guess_file_.empty()) {
140 XTP_LOG(Log::error, log_)
141 << "Reading guess from " << guess_file_ << std::flush;
142 orbitals.ReadFromCpt(guess_file_);
143 } else {
144 XTP_LOG(Log::error, log_)
145 << "Reading structure from " << xyzfile_ << std::flush;
146 orbitals.QMAtoms().LoadFromFile(xyzfile_);
147 }
148
149 // Warm-start the MOs from a PREVIOUS geometry step's own, separate
150 // .orb file, while keeping the geometry just loaded above (the
151 // CURRENT step's own xyzfile_, or guess_file_'s geometry if that
152 // path was used instead) -- deliberately independent of guess_file_
153 // itself, which ties geometry and MOs together from the same file
154 // and therefore cannot represent this "new geometry, warm-started
155 // MOs" case a geometry optimization needs between steps. Loaded
156 // into a separate, temporary Orbitals object so only its own
157 // MOs()/MOs_beta() get copied in -- moguess_orbitals's own geometry
158 // (the PREVIOUS step's) is deliberately discarded, never touching
159 // orbitals.QMAtoms() itself.
160 if (!moguess_file_.empty()) {
161 XTP_LOG(Log::error, log_) << "Reading MO guess (geometry unchanged) from "
162 << moguess_file_ << std::flush;
163 Orbitals moguess_orbitals;
164 moguess_orbitals.ReadFromCpt(moguess_file_);
165 orbitals.MOs() = moguess_orbitals.MOs();
166 if (moguess_orbitals.hasBetaMOs()) {
167 orbitals.MOs_beta() = moguess_orbitals.MOs_beta();
168 }
169 // Defensive, likely-redundant given charge/spin (and therefore the
170 // electron counts) should not change between geometry-optimization
171 // steps -- DFTEngine::Evaluate() itself may set these independently
172 // from package_options_ regardless of what orbitals already holds
173 // here. Kept as a safety net rather than removed outright, since
174 // this has not been confirmed either way by tracing through
175 // DFTEngine's own guess-path logic.
176 orbitals.setNumberOfAlphaElectrons(
177 moguess_orbitals.getNumberOfAlphaElectrons());
178 orbitals.setNumberOfBetaElectrons(
179 moguess_orbitals.getNumberOfBetaElectrons());
180 }
181
182 std::unique_ptr<QMPackage> qmpackage =
183 std::unique_ptr<QMPackage>(QMPackageFactory().Create(
184 package_options_.get("name").as<std::string>()));
185 qmpackage->setLog(&log_);
186 qmpackage->Initialize(package_options_);
187 qmpackage->setRunDir(".");
188
189 if (!mpsfile_.empty()) {
190 StaticRegion region(0, log_);
191 StaticSegment seg = StaticSegment("", 0);
192 seg.LoadFromFile(mpsfile_);
193 region.push_back(seg);
194 qmpackage->AddRegion(region);
195 }
196
197 GWBSEEngine gwbse_engine;
198 gwbse_engine.setLog(&log_);
199 gwbse_engine.setQMPackage(qmpackage.get());
200 gwbse_engine.Initialize(gwbseengine_options_, archive_file_);
201
202 QMMolecule fullMol = orbitals.QMAtoms();
203 gwbse_engine.ExcitationEnergies(orbitals);
204 // If truncation was enabled then rewrite full basis/aux-basis, MOs in full
205 // basis and full QMAtoms
206 if (orbitals.getCalculationType() == "Truncated") {
207 orbitals.QMAtoms().clearAtoms();
208 orbitals.QMAtoms() = fullMol;
209 orbitals.MOs().eigenvectors() = orbitals.getTruncMOsFullBasis();
210 orbitals.SetupDftBasis(orbitals.getDftBasis().Name());
211 if (orbitals.hasAuxbasisName()) {
212 orbitals.SetupAuxBasis(orbitals.getAuxBasis().Name());
213 }
214 }
215
216 if (!excited_state_forces_.empty()) {
217 ComputeExcitedStateForces(orbitals);
218 }
219 if (!qp_gradients_.empty()) {
220 ComputeQPGradients(orbitals);
221 }
222
223 XTP_LOG(Log::error, log_) << "Saving data to " << archive_file_ << std::flush;
224 orbitals.WriteToCpt(archive_file_);
225
226 tools::Property summary = gwbse_engine.ReportSummary();
227 if (summary.exists("output")) { // only do gwbse summary output if we
228 // actually did gwbse
229 XTP_LOG(Log::error, log_)
230 << "Writing output to " << xml_output_ << std::flush;
231 std::ofstream ofout(xml_output_, std::ofstream::out);
232 ofout << (summary.get("output"));
233 ofout.close();
234 }
235 return true;
236}

238// Root following: the state of the requested type with the largest exciton
239// overlap (ExcitonOverlaps) with the state whose forces are stored in the
240// orb file of an earlier geometry (excited_state_forces_follow).
241QMState DftGwBse::FollowState(const Orbitals& orbitals) {
242 Orbitals ref;
243 ref.ReadFromCpt(excited_state_forces_follow_);
244 if (ref.getExcitedStateForcesState().empty()) {
245 throw std::runtime_error(
246 "excited_state_forces_follow: " + excited_state_forces_follow_ +
247 " holds no excited-state forces");
248 }
249 const QMState ref_state(ref.getExcitedStateForcesState());
250 const Eigen::VectorXd ov =
251 ExcitonOverlaps(ref, ref_state, orbitals, ref_state.Type());
252 Index best = 0;
253 ov.maxCoeff(&best);
254 double second = 0.0;
255 for (Index i = 0; i < ov.size(); ++i) {
256 if (i != best) {
257 second = std::max(second, ov(i));
258 }
259 }
260 const QMState state(ref_state.Type(), best, false);
261 XTP_LOG(Log::error, log_)
262 << TimeStamp() << " Root following: " << ref_state.ToString() << " of "
263 << excited_state_forces_follow_ << " is " << state.ToString()
264 << " here (overlap " << ov(best) << ", next largest " << second << ")"
265 << std::flush;
266 if (ov(best) < 0.5 || second > 0.8 * ov(best)) {
267 XTP_LOG(Log::error, log_)
268 << TimeStamp()
269 << " WARNING: root following is ambiguous (states mix strongly or "
270 "the geometry step is large)"
271 << std::flush;
272 }
273 return state;
274}

276// Forces -dE/dR of one BSE state: the ground-state forces of the SCF
277// (xtpdft.compute_forces) minus the analytic gradient of the excitation
278// energy with fixed QP corrections (DFTEngine::BSEExcitationGradient).
279// Response of the QP corrections in the gradients (BSE states and QP
280// levels): the options of the GW run that the gradient has to reproduce.
281void DftGwBse::SetupQPResponse(DFTEngine::BSEGradientInput& input,
282 const Orbitals& orbitals) {
283 // response of the QP corrections: implemented for G0W0 and evGW with
284 // the plasmon-pole and the exact self-energy; otherwise (cda) the
285 // corrections are kept fixed
286 const std::string mode =
287 gwbseengine_options_.get("gwbse.gw.mode").as<std::string>();
288 const std::string integrator =
289 gwbseengine_options_.get("gwbse.gw.sigma_integrator").as<std::string>();
290 if (excited_state_forces_qp_response_ && (mode == "G0W0" || mode == "evGW") &&
291 (integrator == "ppm" || integrator == "exact")) {
292 input.qp_response = true;
293 input.sigma_integrator = integrator;
294 input.qp_selfconsistent = (mode == "evGW");
295 if (input.qp_selfconsistent) {
296 input.out_of_window_shift =
297 gwbseengine_options_.get("gwbse.gw.out_of_window_shift")
298 .as<std::string>();
299 if (input.out_of_window_shift == "max" &&
300 (orbitals.getGWAmax() < orbitals.getRPAmax() ||
301 orbitals.getGWAmin() > orbitals.getRPAmin())) {
302 // max of the window corrections: the energy has kinks where the
303 // level with the largest correction changes; the forces are those
304 // of the branch at the geometry, and optimizations can stall at
305 // such a kink (formaldehyde S1, PBE0/def2-TZVPP)
306 XTP_LOG(Log::error, log_)
307 << TimeStamp()
308 << " WARNING: evGW with gw.out_of_window_shift max: the energy "
309 "is not differentiable where the level with the largest "
310 "correction changes; use nearest for geometry optimizations"
311 << std::flush;
312 }
313 input.out_of_window_fit_fraction =
314 gwbseengine_options_.get("gwbse.gw.out_of_window_fit_fraction")
315 .as<double>();
316 }
317 input.qpmin = orbitals.getGWAmin();
318 input.qpmax = orbitals.getGWAmax();
319 input.qp_energies = orbitals.QPpertEnergies();
320 input.sigma_eta = gwbseengine_options_.get("gwbse.gw.eta").as<double>();
321 XTP_LOG(Log::error, log_)
322 << TimeStamp() << " Including the response of the QP corrections"
323 << std::flush;
324 } else {
325 XTP_LOG(Log::error, log_)
326 << TimeStamp()
327 << " QP corrections kept fixed (excited_state_forces_qp_response "
328 "false, or not G0W0/evGW with sigma_integrator ppm or exact)"
329 << std::flush;
330 }
331}

333// "homo-2:lumo+3", "lumo", "12:20" (absolute, 0-based MO indices)
334std::vector<Index> DftGwBse::ParseLevels(const std::string& text, Index homo) {
335 auto parse_one = [homo](std::string token) {
336 token.erase(std::remove_if(token.begin(), token.end(), ::isspace),
337 token.end());
338 std::transform(token.begin(), token.end(), token.begin(), ::tolower);
339 Index base = 0;
340 std::string rest = token;
341 if (token.rfind("homo", 0) == 0) {
342 base = homo;
343 rest = token.substr(4);
344 } else if (token.rfind("lumo", 0) == 0) {
345 base = homo + 1;
346 rest = token.substr(4);
347 }
348 if (rest.empty() && rest.size() != token.size()) {
349 return base;
350 }
351 std::size_t used = 0;
352 Index offset = 0;
353 try {
354 offset = Index(std::stol(rest, &used));
355 } catch (std::exception&) {
356 used = std::string::npos;
357 }
358 if (used != rest.size() || rest.empty() ||
359 (rest.size() != token.size() && rest[0] != '+' && rest[0] != '-')) {
360 throw std::runtime_error("qp_gradients: cannot read level '" + token +
361 "' (e.g. homo-2, lumo+3 or an MO index)");
362 }
363 return base + offset;
364 };
365 std::vector<Index> levels;
366 std::stringstream list(text);
367 std::string item;
368 while (std::getline(list, item, ',')) {
369 const std::size_t colon = item.find(':');
370 const Index first = parse_one(item.substr(0, colon));
371 const Index last = (colon == std::string::npos)
372 ? first
373 : parse_one(item.substr(colon + 1));
374 if (last < first) {
375 throw std::runtime_error("qp_gradients: empty range '" + item + "'");
376 }
377 for (Index level = first; level <= last; ++level) {
378 if (std::find(levels.begin(), levels.end(), level) == levels.end()) {
379 levels.push_back(level);
380 }
381 }
382 }
383 std::sort(levels.begin(), levels.end());
384 return levels;
385}

387void DftGwBse::ComputeQPGradients(Orbitals& orbitals) {
388 if (package_options_.get("name").as<std::string>() != "xtp") {
389 throw std::runtime_error("qp_gradients: needs the xtp DFT package");
390 }
391 if (!mpsfile_.empty() || orbitals.getCalculationType() == "Truncated") {
392 throw std::runtime_error(
393 "qp_gradients: embedding and truncated bases are not supported");
394 }
395 const std::vector<Index> requested =
396 ParseLevels(qp_gradients_, orbitals.getHomo());
397 // levels whose gradients are already in the orb file (tasks gradients on
398 // a file of an earlier run) are kept and not recomputed
399 const std::vector<Index> stored = orbitals.getQPGradientLevels();
400 const Eigen::MatrixXd stored_gradients = orbitals.getQPGradients();
401 std::vector<Index> levels;
402 for (Index level : requested) {
403 if (std::find(stored.begin(), stored.end(), level) == stored.end()) {
404 levels.push_back(level);
405 }
406 }
407 if (levels.empty()) {
408 XTP_LOG(Log::error, log_)
409 << TimeStamp() << " QP gradients of all requested levels are "
410 << "already stored" << std::flush;
411 return;
412 }
413 if (levels.size() < requested.size()) {
414 XTP_LOG(Log::error, log_)
415 << TimeStamp() << " QP gradients of "
416 << requested.size() - levels.size()
417 << " requested level(s) are already stored" << std::flush;
418 }
419 XTP_LOG(Log::error, log_)
420 << TimeStamp() << " Analytic gradients of the QP energies of "
421 << levels.size() << " level(s), " << levels.front() << " to "
422 << levels.back() << " (HOMO is " << orbitals.getHomo() << ")"
423 << std::flush;
424 DFTEngine::BSEGradientInput input =
425 DFTEngine::QPGradientInputFromOrbitals(orbitals, levels);
426 SetupQPResponse(input, orbitals);
427 DFTEngine dft;
428 dft.setLogger(&log_);
429 dft.Initialize(package_options_);
430 const DFTEngine::BSEGradientResult result =
431 dft.BSEExcitationGradient(orbitals, input);
432 if (input.qp_response) {
433 XTP_LOG(Log::error, log_)
434 << TimeStamp() << " QP energies rebuilt for the gradient: max "
435 << "deviation " << result.qp_rebuild_error << " Ha" << std::flush;
436 if (result.qp_rebuild_error > 1e-4) {
437 throw std::runtime_error(
438 "qp_gradients: the QP energies rebuilt for the gradient differ "
439 "from the GW ones by " +
440 std::to_string(result.qp_rebuild_error) + " Ha");
441 }
442 }
443 const Index natoms = orbitals.QMAtoms().size();
444 // stored and new levels, sorted
445 std::vector<std::pair<Index, Eigen::MatrixXd>> merged;
446 for (std::size_t l = 0; l < stored.size(); ++l) {
447 merged.emplace_back(stored[l],
448 stored_gradients.middleRows(Index(l) * natoms, natoms));
449 }
450 for (std::size_t l = 0; l < levels.size(); ++l) {
451 merged.emplace_back(levels[l], result.qp_gradients[l]);
452 }
453 std::sort(merged.begin(), merged.end(),
454 [](const auto& a, const auto& b) { return a.first < b.first; });
455 Eigen::MatrixXd all(Index(merged.size()) * natoms, 3);
456 std::vector<Index> all_levels;
457 for (std::size_t l = 0; l < merged.size(); ++l) {
458 all.middleRows(Index(l) * natoms, natoms) = merged[l].second;
459 all_levels.push_back(merged[l].first);
460 }
461 orbitals.setQPGradients(all, all_levels);
462}

464// tasks = gradients: the options must be those of the run that wrote the
465// orb file. What is stored there is compared; a different GW setup shows
466// up in the rebuilt QP and excitation energies of the gradient.
467void DftGwBse::CheckStoredRun(const Orbitals& orbitals) const {
468 auto lower = [](std::string text) {
469 std::transform(text.begin(), text.end(), text.begin(), ::tolower);
470 return text;
471 };
472 auto check = [&](const std::string& what, const std::string& stored,
473 const std::string& option) {
474 if (lower(stored) != lower(option)) {
475 throw std::runtime_error("tasks gradients: " + what + " of " +
476 archive_file_ + " is " + stored +
477 ", the options say " + option +
478 "; use the options of the run that wrote "
479 "the file");
480 }
481 };
482 if (!orbitals.hasQMAtoms() || orbitals.MOs().eigenvalues().size() == 0 ||
483 orbitals.RPAInputEnergies().size() == 0) {
484 throw std::runtime_error("tasks gradients: " + archive_file_ +
485 " has no GW data");
486 }
487 check("the basis set", orbitals.getDftBasis().Name(),
488 package_options_.get("basisset").as<std::string>());
489 check("the functional", orbitals.getXCFunctionalName(),
490 package_options_.get("functional").as<std::string>());
491 if (orbitals.hasAuxbasisName()) {
492 check("the auxiliary basis set", orbitals.getAuxBasis().Name(),
493 package_options_.get("auxbasisset").as<std::string>());
494 }
495}

497void DftGwBse::ComputeExcitedStateForces(Orbitals& orbitals) {
498 QMState state(excited_state_forces_);
499 if (!excited_state_forces_follow_.empty()) {
500 state = FollowState(orbitals);
501 }
502 const bool singlet = state.Type() == QMStateType::Singlet;
503 if (!singlet && state.Type() != QMStateType::Triplet) {
504 throw std::runtime_error(
505 "excited_state_forces: only singlet or triplet BSE states (e.g. s1)");
506 }
507 if (package_options_.get("name").as<std::string>() != "xtp") {
508 throw std::runtime_error("excited_state_forces: needs the xtp DFT package");
509 }
510 if (!mpsfile_.empty() || orbitals.getCalculationType() == "Truncated") {
511 throw std::runtime_error(
512 "excited_state_forces: embedding and truncated bases are not "
513 "supported");
514 }
515 if (!orbitals.hasForces()) {
516 throw std::runtime_error(
517 "excited_state_forces: needs the ground-state forces, set "
518 "dftpackage.xtpdft.compute_forces to true");
519 }
520 XTP_LOG(Log::error, log_)
521 << TimeStamp() << " Analytic forces of BSE state " << state.ToString()
522 << (orbitals.getTDAApprox() ? " (TDA)" : " (full BSE)") << std::flush;
523
524 DFTEngine::BSEGradientInput input = DFTEngine::BSEGradientInputFromOrbitals(
525 orbitals, singlet, state.StateIdx());
526 SetupQPResponse(input, orbitals);
527 DFTEngine dft;
528 dft.setLogger(&log_);
529 dft.Initialize(package_options_);
530 const DFTEngine::BSEGradientResult result =
531 dft.BSEExcitationGradient(orbitals, input);
532
533 const double omega =
534 (singlet ? orbitals.BSESinglets() : orbitals.BSETriplets())
535 .eigenvalues()(state.StateIdx());
536 // the gradient belongs to the Rayleigh quotient of the stored exciton,
537 // which differs from the Davidson eigenvalue at the level of the
538 // Davidson residual (bse.davidson.tolerance; for full BSE to first
539 // order); a different GW-BSE setup gives deviations of 1e-3 Ha and more
540 const double omega_error = std::abs(result.omega - omega);
541 XTP_LOG(Log::error, log_)
542 << TimeStamp() << " Excitation energy rebuilt for the gradient: "
543 << "deviation from the BSE eigenvalue " << omega_error << " Ha"
544 << std::flush;
545 if (omega_error > 1e-4) {
546 throw std::runtime_error(
547 "excited_state_forces: the excitation energy rebuilt for the "
548 "gradient (" +
549 std::to_string(result.omega) +
550 " Ha) differs from the BSE eigenvalue (" + std::to_string(omega) +
551 " Ha); the GW-BSE setup (e.g. environment screening, dynamical "
552 "screening of the BSE kernel) is not covered by the gradient");
553 }
554 if (omega_error > 1e-6) {
555 XTP_LOG(Log::error, log_)
556 << TimeStamp()
557 << " WARNING: the BSE eigenvalue is converged only to about "
558 << omega_error
559 << " Ha; tighten bse.davidson.tolerance (strict or lapack) for "
560 "energies consistent with the forces"
561 << std::flush;
562 }
563 if (input.qp_response) {
564 XTP_LOG(Log::error, log_)
565 << TimeStamp() << " QP energies rebuilt for the gradient: max "
566 << "deviation " << result.qp_rebuild_error << " Ha" << std::flush;
567 }
568 if (input.qp_selfconsistent) {
569 XTP_LOG(Log::error, log_)
570 << TimeStamp() << " RPA energies rebuilt for the gradient: max "
571 << "deviation " << result.rpa_rebuild_error << " Ha" << std::flush;
572 }
573 // the QP solver and the interpolated pole sums of GW are converged to
574 // about 1e-6 Ha; a larger deviation means a different setup
575 if (input.qp_response && result.qp_rebuild_error > 1e-4) {
576 throw std::runtime_error(
577 "excited_state_forces: the QP energies rebuilt for the gradient "
578 "differ from the GW ones by " +
579 std::to_string(result.qp_rebuild_error) + " Ha");
580 }
581 const Eigen::MatrixXd forces = orbitals.getForces() - result.gradient;
582 orbitals.setExcitedStateForces(forces, state.ToString());
583 XTP_LOG(Log::error, log_)
584 << TimeStamp() << " Excited-state forces (Hartree/Bohr) of "
