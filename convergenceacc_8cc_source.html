37// Build the symmetric orthogonalization matrix X = S^{-1/2}. All Fock-like
38// matrices are diagonalized in the orthogonal AO basis X^T F X.
39void ConvergenceAcc::setOverlap(AOOverlap& S, double etol) {
40 S_ = &S;
41 // Own eigendecomposition rather than AOOverlap::Pseudo_InvSqrt: the
42 // removed directions are needed as well, see removed_projector_.
43 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es(S.Matrix());
44 const Eigen::VectorXd& s_eig = es.eigenvalues();
45 Eigen::VectorXd inv_sqrt = Eigen::VectorXd::Zero(s_eig.size());
46 Index removed = 0;
47 for (Index i = 0; i < s_eig.size(); ++i) {
48 if (s_eig(i) < etol) {
49 ++removed;
50 } else {
51 inv_sqrt(i) = 1.0 / std::sqrt(s_eig(i));
52 }
53 }
54 Sminusahalf =
55 es.eigenvectors() * inv_sqrt.asDiagonal() * es.eigenvectors().transpose();
56 removed_projector_.resize(0, 0);
57 if (removed > 0) {
58 // In SolveFockmatrix, H_ortho = X^T H X vanishes on these directions,
59 // so they would come out as eigenvalue-0 "orbitals" in the middle of
60 // the spectrum. Shifting them far up keeps them out of the occupied
61 // and low virtual space.
62 const Eigen::MatrixXd U = es.eigenvectors().leftCols(removed);
63 removed_projector_ = U * U.transpose();
64 }
65 XTP_LOG(Log::error, *log_)
66 << TimeStamp() << " Smallest value of AOOverlap matrix is " << s_eig(0)
67 << std::flush;
68 XTP_LOG(Log::error, *log_)
69 << TimeStamp() << " Removed " << removed
70 << " basisfunction from inverse overlap matrix (threshold " << etol << ")"
71 << std::flush;
72 if (removed > 0) {
73 XTP_LOG(Log::error, *log_)
74 << TimeStamp() << " The " << removed
75 << " removed direction(s) appear as zero orbitals at +" << kRemovedShift
76 << " Hartree." << std::flush;
77 } else if (s_eig(0) < 1e3 * etol) {
78 XTP_LOG(Log::error, *log_)
79 << TimeStamp()
80 << " WARNING: the overlap matrix is nearly singular; if the SCF is "
81 "unstable, raise xtpdft.overlap_tolerance above "
82 << s_eig(0) << "." << std::flush;
83 }
84 return;
85}

87// Perform one SCF acceleration step.
88//
89// The commutator residual
90//
91// R = X^T (F P S - S P F) X
92//
93// vanishes at self-consistency in a non-orthogonal AO basis. Its maximum
94// element is used as the DIIS error metric, while the history of Fock and
95// density matrices is passed to either DIIS or ADIIS to construct the next
96// extrapolated Fock matrix. When the extrapolation is deemed unsafe, linear
97// density mixing is used instead.
98Eigen::MatrixXd ConvergenceAcc::Iterate(const Eigen::MatrixXd& dmat,
99 Eigen::MatrixXd& H,
100 tools::EigenSystem& MOs, double totE) {
101 totE_.push_back(totE);
102 // The MOs of the previous step, for the level shift.
103 const Eigen::MatrixXd MOs_old = MOs.eigenvectors();
104 const Eigen::VectorXd MOs_old_energies = MOs.eigenvalues();
105
106 const Eigen::MatrixXd& S = S_->Matrix();
107 const Eigen::MatrixXd errormatrix =
108 Sminusahalf.transpose() * (H * dmat * S - S * dmat * H) * Sminusahalf;
109 diiserror_ = errormatrix.cwiseAbs().maxCoeff();
110
111 XTP_LOG(Log::error, *log_)
112 << TimeStamp() << " DIIs error " << getDIIsError() << std::flush;
113 XTP_LOG(Log::error, *log_)
114 << TimeStamp() << " Delta Etot " << getDeltaE() << std::flush;
115
116 // Energy-rise reset. An extrapolated Fock matrix that sends the energy
117 // far above anything seen before is not a step to build on: the history
118 // that produced it is discarded and the SCF continues, damped, from the
119 // lowest-energy density so far.
120 ++iterations_since_reset_;
121 if (opt_.energy_reset > 0.0 && have_best_ &&
122 totE > best_energy_ + opt_.energy_reset &&
123 iterations_since_reset_ > kResetCooldown &&
124 energy_resets_ < kMaxEnergyResets) {
125 ++energy_resets_;
126 iterations_since_reset_ = 0;
127 XTP_LOG(Log::error, *log_)
128 << TimeStamp() << " WARNING: energy rose by " << totE - best_energy_
129 << " Ha above the lowest so far (" << best_energy_
130 << " Ha); discarding the (A)DIIS history and restarting from that "
131 "density (reset "
132 << energy_resets_ << ")" << std::flush;
133 mathist_.clear();
134 dmatHist_.clear();
135 errhist_.clear();
136 diis_.Clear();
137 Eigen::MatrixXd H_restart = best_H_;
138 if (opt_.levelshift > 0.0) {
139 Levelshift(H_restart, MOs_old);
140 }
141 MOs = SolveFockmatrix(H_restart);
142 usedmixing_ = true;
143 return opt_.mixingparameter * best_dmat_ +
144 (1.0 - opt_.mixingparameter) * DensityMatrix(MOs);
145 }
146 if (!have_best_ || totE < best_energy_) {
147 have_best_ = true;
148 best_energy_ = totE;
149 best_dmat_ = dmat;
150 best_H_ = H;
151 }
152
153 // History, trimmed at one index for all of mathist_, dmatHist_, errhist_
154 // and DIIS's own error history: the oldest entry, or with DIIS_maxout the
155 // one with the largest error. All four are always the same length, so
156 // DIIS::Update trims at the same moment and the same index.
157 Index drop = 0;
158 if (Index(mathist_.size()) == opt_.histlength) {
159 if (opt_.maxout) {
160 drop = Index(std::max_element(errhist_.begin(), errhist_.end()) -
161 errhist_.begin());
162 }
163 mathist_.erase(mathist_.begin() + drop);
164 dmatHist_.erase(dmatHist_.begin() + drop);
165 errhist_.erase(errhist_.begin() + drop);
166 }
167 // The UNSHIFTED Fock matrix goes into the history: the level shift is a
168 // device for the next diagonalization only, and would otherwise enter
169 // the DIIS errors and the ADIIS energy model.
170 mathist_.push_back(H);
171 dmatHist_.push_back(dmat);
172 errhist_.push_back(diiserror_);
173 diis_.Update(drop, errormatrix);
174
175 bool diis_error = false;
176 Eigen::MatrixXd H_guess = H;
177 if ((diiserror_ < opt_.adiis_start || diiserror_ < opt_.diis_start) &&
178 opt_.usediis && mathist_.size() > 2) {
179 Eigen::VectorXd coeffs;
180 // ADIIS above DIIS_start, and also below it whenever the energy went
181 // up: plain DIIS does not minimize the energy and can run away from a
182 // bad step.
183 const bool energy_rose = getDeltaE() > kEnergyRiseForADIIS;
184 if (diiserror_ > opt_.diis_start || energy_rose) {
185 coeffs = adiis_.CalcCoeff(dmatHist_, mathist_);
186 diis_error = !adiis_.Info();
187 XTP_LOG(Log::warning, *log_)
188 << TimeStamp() << " Using ADIIS for next guess" << std::flush;
189 } else {
190 coeffs = diis_.CalcCoeff();
191 diis_error = !diis_.Info();
192 XTP_LOG(Log::warning, *log_)
193 << TimeStamp() << " Using DIIS for next guess" << std::flush;
194 }
195 if (diis_error) {
196 XTP_LOG(Log::warning, *log_)
197 << TimeStamp() << " (A)DIIS failed using mixing instead"
198 << std::flush;
199 } else {
200 H_guess.setZero();
201 for (Index i = 0; i < coeffs.size(); i++) {
202 if (std::abs(coeffs(i)) < 1e-8) {
203 continue;
204 }
205 H_guess += coeffs(i) * mathist_[i];
206 }
207 }
208 }
209
210 if (opt_.mode != KSmode::fractional && nocclevels_ > 0 &&
211 nocclevels_ < MOs_old_energies.size()) {
212 const double gap =
213 MOs_old_energies(nocclevels_) - MOs_old_energies(nocclevels_ - 1);
214 if ((diiserror_ > opt_.levelshiftend && opt_.levelshift > 0.0) ||
215 gap < 1e-6) {
216 Levelshift(H_guess, MOs_old);
217 }
218 }
219
220 MOs = SolveFockmatrix(H_guess);
221
222 Eigen::MatrixXd dmatout = DensityMatrix(MOs);
223 // mixing_end, not ADIIS_start, decides on damping (as in the UKS path):
224 // the two are separate options.
225 if (diiserror_ > opt_.mixingend || !opt_.usediis || diis_error ||
226 mathist_.size() <= 2) {
227 usedmixing_ = true;
228 dmatout =
229 opt_.mixingparameter * dmat + (1.0 - opt_.mixingparameter) * dmatout;
230 XTP_LOG(Log::warning, *log_)
231 << TimeStamp() << " Using Mixing with alpha=" << opt_.mixingparameter
232 << std::flush;
233 } else {
234 usedmixing_ = false;
235 }
236 return dmatout;
237}

239bool ConvergenceAcc::HistoryIsAligned() const {
240 const std::vector<Eigen::MatrixXd>& errors = diis_.ErrorHistory();
241 if (errors.size() != mathist_.size() || dmatHist_.size() != mathist_.size() ||
242 errhist_.size() != mathist_.size()) {
243 return false;
244 }
245 const Eigen::MatrixXd& S = S_->Matrix();
246 for (std::size_t k = 0; k < mathist_.size(); ++k) {
247 const Eigen::MatrixXd e =
248 Sminusahalf.transpose() *
249 (mathist_[k] * dmatHist_[k] * S - S * dmatHist_[k] * mathist_[k]) *
250 Sminusahalf;
251 if ((e - errors[k]).cwiseAbs().maxCoeff() > 1e-12 ||
252 std::abs(e.cwiseAbs().maxCoeff() - errhist_[k]) > 1e-12) {
253 return false;
254 }
255 }
256 return true;
257}

293// Solve the generalized Roothaan-Hall problem
294//
295// F C = S C eps
296//
297// by symmetric orthogonalization: diagonalize X^T F X with X = S^{-1/2} and
298// back-transform the eigenvectors as C = X C' .
299tools::EigenSystem ConvergenceAcc::SolveFockmatrix(
300 const Eigen::MatrixXd& H) const {
301 // transform to orthogonal for
302 Eigen::MatrixXd H_ortho = Sminusahalf.transpose() * H * Sminusahalf;
303 if (removed_projector_.size() > 0) {
304 H_ortho += kRemovedShift * removed_projector_;
305 }
306 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es(H_ortho);
307
308 if (es.info() != Eigen::ComputationInfo::Success) {
309 throw std::runtime_error("Matrix Diagonalisation failed. DiagInfo" +
310 std::to_string(es.info()));
311 }
312
313 tools::EigenSystem result;
314 result.eigenvalues() = es.eigenvalues();
315
316 result.eigenvectors() = Sminusahalf * es.eigenvectors();
317 return result;
318}

320// Add a virtual-space level shift
321//
322// F <- F + S C_virt Delta C_virt^T S,
323//
324// implemented by placing the scalar shift on the virtual diagonal in the MO
325// basis built from the previous iteration. This leaves occupied orbitals
326// unchanged while opening the HOMO-LUMO gap during difficult SCF phases.
327void ConvergenceAcc::Levelshift(Eigen::MatrixXd& H,
328 const Eigen::MatrixXd& MOs_old) const {
329 if (opt_.levelshift < 1e-9) {
330 return;
331 }
332 Eigen::VectorXd virt = Eigen::VectorXd::Zero(H.rows());
333 for (Index i = nocclevels_; i < H.rows(); i++) {
334 virt(i) = opt_.levelshift;
335 }
336
337 XTP_LOG(Log::error, *log_)
338 << TimeStamp() << " Using levelshift:" << opt_.levelshift << " Hartree"
339 << std::flush;
340 Eigen::MatrixXd vir = S_->Matrix() * MOs_old * virt.asDiagonal() *
341 MOs_old.transpose() * S_->Matrix();
342 H += vir;
343 return;
344}

360// Closed-shell AO density matrix
361//
362// P = 2 C_occ C_occ^T,
363//
364// where each occupied spatial orbital contributes one alpha and one beta
365// electron.
366Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState(
367 const Eigen::MatrixXd& MOs) const {
368 const Eigen::MatrixXd occstates = MOs.leftCols(nocclevels_);
369 Eigen::MatrixXd dmatGS = 2.0 * occstates * occstates.transpose();
370 return dmatGS;
371}
372
373// Spin-resolved unrestricted AO density matrix for one spin channel,
374//
375// P^sigma = C_occ^sigma (C_occ^sigma)^T,
376//
377// with no factor of two because a single spin channel is represented.
378Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState_unres(
379 const Eigen::MatrixXd& MOs) const {
380 if (nocclevels_ == 0) {
381 return Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
382 }
383 Eigen::MatrixXd occstates = MOs.leftCols(nocclevels_);
384 Eigen::MatrixXd dmatGS = occstates * occstates.transpose();
385 return dmatGS;
386}

388// Fractionally occupied AO density matrix
389//
390// P = C f C^T,
391//
392// where f is a diagonal matrix of orbital occupations assembled from the
393// configured electron count.
394// Fractional-occupation AO density matrix
395//
396// P = C n C^T,
397//
398// where n is the diagonal matrix of orbital occupations provided in the
399// EigenSystem container.
400Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState_frac(
401 const tools::EigenSystem& MOs) const {
402 if (opt_.numberofelectrons == 0) {
403 return Eigen::MatrixXd::Zero(MOs.eigenvectors().rows(),
404 MOs.eigenvectors().rows());
405 }
406
407 Eigen::VectorXd occupation = Eigen::VectorXd::Zero(MOs.eigenvalues().size());
408 std::vector<std::vector<Index> > degeneracies;
409 double buffer = 1e-4;
410 degeneracies.push_back(std::vector<Index>{0});
411 for (Index i = 1; i < occupation.size(); i++) {
412 if (MOs.eigenvalues()(i) <
413 MOs.eigenvalues()(degeneracies[degeneracies.size() - 1][0]) + buffer) {
414 degeneracies[degeneracies.size() - 1].push_back(i);
415 } else {
416 degeneracies.push_back(std::vector<Index>{i});
417 }
418 }
419 Index numofelec = opt_.numberofelectrons;
420 for (const std::vector<Index>& deglevel : degeneracies) {
421 Index numofpossibleelectrons = 2 * Index(deglevel.size());
422 if (numofpossibleelectrons <= numofelec) {
423 for (Index i : deglevel) {
424 occupation(i) = 2;
425 }
426 numofelec -= numofpossibleelectrons;
427 } else {
428 double occ = double(numofelec) / double(deglevel.size());
429 for (Index i : deglevel) {
430 occupation(i) = occ;
431 }
432 break;
433 }
434 }
435 Eigen::MatrixXd dmatGS = MOs.eigenvectors() * occupation.asDiagonal() *
436 MOs.eigenvectors().transpose();
437 return dmatGS;
438}

444ConvergenceAcc::SpinDensity
445 ConvergenceAcc::DensityMatrixGroundState_restricted_open(
446 const Eigen::MatrixXd& MOs) const {
447
448 const Index n_docc =
449 std::min(opt_.number_alpha_electrons, opt_.number_beta_electrons);
450 const Index n_socc_alpha = opt_.number_alpha_electrons - n_docc;
451
452 SpinDensity result;
453 result.alpha = Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
454 result.beta = Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
455
456 if (n_docc > 0) {
457 const Eigen::MatrixXd docc = MOs.leftCols(n_docc);
458 const Eigen::MatrixXd d_docc = docc * docc.transpose();
459 result.alpha += d_docc;
460 result.beta += d_docc;
461 }
462
463 if (n_socc_alpha > 0) {
464 const Eigen::MatrixXd socc = MOs.middleCols(n_docc, n_socc_alpha);
465 result.alpha += socc * socc.transpose();
466 }
467
468 return result;
469}

471// Construct spin-resolved densities according to the configured occupation
472// model. For restricted open-shell cases the same spatial orbitals are split
473// into doubly and singly occupied subsets using the stored alpha/beta counts.
474ConvergenceAcc::SpinDensity ConvergenceAcc::DensityMatrixSpinResolved(
475 const tools::EigenSystem& MOs) const {
476
477 if (opt_.mode == KSmode::restricted_open) {
478 return DensityMatrixGroundState_restricted_open(MOs.eigenvectors());
479 } else if (opt_.mode == KSmode::closed) {
480 Eigen::MatrixXd d = DensityMatrixGroundState(MOs.eigenvectors());
481 return {0.5 * d, 0.5 * d};
482 } else if (opt_.mode == KSmode::open) {
483 Eigen::MatrixXd d = DensityMatrixGroundState_unres(MOs.eigenvectors());
484 return {d, Eigen::MatrixXd::Zero(d.rows(), d.cols())};
485 } else {
486 Eigen::MatrixXd d = DensityMatrixGroundState_frac(MOs);
487 return {0.5 * d, 0.5 * d};
488 }
489}
490
491Eigen::MatrixXd ConvergenceAcc::DensityMatrix(
492 const tools::EigenSystem& MOs) const {
493 SpinDensity spin_dmat = DensityMatrixSpinResolved(MOs);
494 return spin_dmat.total();
495}
