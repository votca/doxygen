38// Build the symmetric orthogonalization matrix X = S^{-1/2}. All Fock-like
39// matrices are diagonalized in the orthogonal AO basis X^T F X.
40void ConvergenceAcc::setOverlap(AOOverlap& S, double etol) {
41 S_ = &S;
42 // Own eigendecomposition rather than AOOverlap::Pseudo_InvSqrt: the
43 // removed directions are needed as well, see removed_projector_.
44 const SymmetricEigenSystem es = SymmetricEigen(S.Matrix());
45 const Eigen::VectorXd& s_eig = es.values;
46 Eigen::VectorXd inv_sqrt = Eigen::VectorXd::Zero(s_eig.size());
47 Index removed = 0;
48 for (Index i = 0; i < s_eig.size(); ++i) {
49 if (s_eig(i) < etol) {
50 ++removed;
51 } else {
52 inv_sqrt(i) = 1.0 / std::sqrt(s_eig(i));
53 }
54 }
55 Sminusahalf = es.vectors * inv_sqrt.asDiagonal() * es.vectors.transpose();
56 removed_projector_.resize(0, 0);
57 if (removed > 0) {
58 // In SolveFockmatrix, H_ortho = X^T H X vanishes on these directions,
59 // so they would come out as eigenvalue-0 "orbitals" in the middle of
60 // the spectrum. Shifting them far up keeps them out of the occupied
61 // and low virtual space.
62 const Eigen::MatrixXd U = es.vectors.leftCols(removed);
63 removed_projector_ = U * U.transpose();
64 }
65 XTP_LOG(Log::error, *log_)
66 << TimeStamp() << " Smallest value of AOOverlap matrix is " << s_eig(0)
67 << std::flush;
68 XTP_LOG(Log::error, *log_)
69 << TimeStamp() << " Removed " << removed
70 << " basisfunction from inverse overlap matrix (threshold " << etol << ")"
71 << std::flush;
72 if (removed > 0) {
73 XTP_LOG(Log::error, *log_)
74 << TimeStamp() << " The " << removed
75 << " removed direction(s) appear as zero orbitals at +" << kRemovedShift
76 << " Hartree." << std::flush;
77 } else if (s_eig(0) < 1e3 * etol) {
78 XTP_LOG(Log::error, *log_)
79 << TimeStamp()
80 << " WARNING: the overlap matrix is nearly singular; if the SCF is "
81 "unstable, raise xtpdft.overlap_tolerance above "
82 << s_eig(0) << "." << std::flush;
83 }
84 return;
85}

87// Perform one SCF acceleration step.
88//
89// The commutator residual
90//
91// R = X^T (F P S - S P F) X
92//
93// vanishes at self-consistency in a non-orthogonal AO basis. Its maximum
94// element is used as the DIIS error metric, while the history of Fock and
95// density matrices is passed to either DIIS or ADIIS to construct the next
96// extrapolated Fock matrix. When the extrapolation is deemed unsafe, linear
97// density mixing is used instead.
98Eigen::MatrixXd ConvergenceAcc::Iterate(const Eigen::MatrixXd& dmat,
99 Eigen::MatrixXd& H,
100 tools::EigenSystem& MOs, double totE) {
101 // Check the second-order step returned by the previous call, now that
102 // the energy of its density (totE) is known: Fletcher's trust-radius
103 // update. A rejected step is undone and, while second-order steps
104 // continue, repeated at once with the smaller radius from the stored
105 // Fock matrix and Davidson subspace: no Fock build, no SCF iteration and
106 // usually no Hessian products.
107 last_rejected_ = false;
108 if (trah_pending_) {
109 trah_pending_ = false;
110 const double actual = totE - trah_pre_energy_;
111 const bool accepted =
112 trust_.Update(trah_predicted_, actual, trah_linear_, trah_step_norm_);
113 XTP_LOG(Log::warning, *log_)
114 << TimeStamp() << " Second-order step check: actual dE=" << actual
115 << ", predicted dE=" << trah_predicted_ << ", r=" << trust_.LastRatio()
116 << (accepted ? " (accepted)" : " (rejected)")
117 << ", new trust radius=" << trust_.Radius() << std::flush;
118 if (!accepted) {
119 if (trust_.Radius() < kMinTrustRadius) {
120 trah_active_ = false;
121 trah_disabled_ = true;
122 XTP_LOG(Log::warning, *log_)
123 << TimeStamp() << " Trust radius below " << kMinTrustRadius
124 << " without an accepted step; leaving second-order steps for "
125 "this SCF"
126 << std::flush;
127 }
128 MOs = trah_pre_MOs_;
129 totE_.push_back(trah_pre_energy_);
130 last_rejected_ = true;
131 usedmixing_ = false;
132 if (trah_active_) {
133 const Eigen::MatrixXd H_pre = trah_pre_H_;
134 return SecondOrderStep(DensityMatrix(MOs), H_pre, MOs,
135 trah_pre_energy_);
136 }
137 return DensityMatrix(MOs);
138 }
139 }
140
141 totE_.push_back(totE);
142 ++scf_iterations_;
143 // The MOs of the previous step, for the level shift.
144 const Eigen::MatrixXd MOs_old = MOs.eigenvectors();
145 const Eigen::VectorXd MOs_old_energies = MOs.eigenvalues();
146
147 const Eigen::MatrixXd& S = S_->Matrix();
148 const Eigen::MatrixXd errormatrix =
149 Sminusahalf.transpose() * (H * dmat * S - S * dmat * H) * Sminusahalf;
150 diiserror_ = errormatrix.cwiseAbs().maxCoeff();
151
152 XTP_LOG(Log::error, *log_)
153 << TimeStamp() << " DIIs error " << getDIIsError() << std::flush;
154 XTP_LOG(Log::error, *log_)
155 << TimeStamp() << " Delta Etot " << getDeltaE() << std::flush;
156
157 // Energy-rise reset. An extrapolated Fock matrix that sends the energy
158 // far above anything seen before is not a step to build on: the history
159 // that produced it is discarded and the SCF continues, damped, from the
160 // lowest-energy density so far.
161 ++iterations_since_reset_;
162 if (opt_.energy_reset > 0.0 && have_best_ && !trah_active_ &&
163 totE > best_energy_ + opt_.energy_reset &&
164 iterations_since_reset_ > kResetCooldown &&
165 energy_resets_ < kMaxEnergyResets) {
166 ++energy_resets_;
167 iterations_since_reset_ = 0;
168 XTP_LOG(Log::error, *log_)
169 << TimeStamp() << " WARNING: energy rose by " << totE - best_energy_
170 << " Ha above the lowest so far (" << best_energy_
171 << " Ha); discarding the (A)DIIS history and restarting from that "
172 "density (reset "
173 << energy_resets_ << ")" << std::flush;
174 mathist_.clear();
175 dmatHist_.clear();
176 errhist_.clear();
177 diis_.Clear();
178 Eigen::MatrixXd H_restart = best_H_;
179 if (opt_.levelshift > 0.0) {
180 Levelshift(H_restart, MOs_old);
181 }
182 MOs = SolveFockmatrix(H_restart);
183 usedmixing_ = true;
184 return opt_.mixingparameter * best_dmat_ +
185 (1.0 - opt_.mixingparameter) * DensityMatrix(MOs);
186 }
187 if (!have_best_ || totE < best_energy_) {
188 have_best_ = true;
189 best_energy_ = totE;
190 best_dmat_ = dmat;
191 best_H_ = H;
192 }
193
194 // History, trimmed at one index for all of mathist_, dmatHist_, errhist_
195 // and DIIS's own error history: the oldest entry, or with DIIS_maxout the
196 // one with the largest error. All four are always the same length, so
197 // DIIS::Update trims at the same moment and the same index.
198 Index drop = 0;
199 if (Index(mathist_.size()) == opt_.histlength) {
200 if (opt_.maxout) {
201 drop = Index(std::max_element(errhist_.begin(), errhist_.end()) -
202 errhist_.begin());
203 }
204 mathist_.erase(mathist_.begin() + drop);
205 dmatHist_.erase(dmatHist_.begin() + drop);
206 errhist_.erase(errhist_.begin() + drop);
207 }
208 // The UNSHIFTED Fock matrix goes into the history: the level shift is a
209 // device for the next diagonalization only, and would otherwise enter
210 // the DIIS errors and the ADIIS energy model.
211 mathist_.push_back(H);
212 dmatHist_.push_back(dmat);
213 errhist_.push_back(diiserror_);
214 diis_.Update(drop, errormatrix);
215 error_history_.push_back(diiserror_);
216 if (Index(error_history_.size()) > kTrailingWindow) {
217 error_history_.erase(error_history_.begin());
218 }
219
220 // Second-order (TRAH) steps: from the start with second_order "always";
221 // with "auto" once (A)DIIS has failed repeatedly or the DIIS error has
222 // stalled. Once started they continue for the rest of this SCF, unless
223 // the trust radius collapses.
224 if (SecondOrderAvailable() && !trah_active_ &&
225 (opt_.second_order == "always" || SecondOrderStalled())) {
226 XTP_LOG(Log::warning, *log_)
227 << TimeStamp() << " Switching to second-order (TRAH) steps"
228 << (opt_.second_order == "always"
229 ? ""
230 : " ((A)DIIS failed repeatedly or the DIIS error stalled)")
231 << std::flush;
232 trah_active_ = true;
233 }
234 if (trah_active_) {
235 return SecondOrderStep(dmat, H, MOs, totE);
236 }
237
238 bool diis_error = false;
239 Eigen::MatrixXd H_guess = H;
240 // Below DIIS_start the density is already close to self-consistency (as
241 // after a warm start from the previous QM/MM iteration): no damping, a
242 // plain step from the current Fock matrix first and DIIS from two
243 // history entries on. Far from it (a cold start), the first steps are
244 // damped until the history holds three entries, as before.
245 const bool near_convergence = diiserror_ < opt_.diis_start;
246 const std::size_t min_history = near_convergence ? 1 : 2;
247 if ((diiserror_ < opt_.adiis_start || diiserror_ < opt_.diis_start) &&
248 opt_.usediis && mathist_.size() > min_history) {
249 Eigen::VectorXd coeffs;
250 // ADIIS above DIIS_start, and also below it whenever the energy went
251 // up: plain DIIS does not minimize the energy and can run away from a
252 // bad step.
253 const bool energy_rose = getDeltaE() > kEnergyRiseForADIIS;
254 if (diiserror_ > opt_.diis_start || energy_rose) {
255 coeffs = adiis_.CalcCoeff(dmatHist_, mathist_);
256 diis_error = !adiis_.Info();
257 XTP_LOG(Log::warning, *log_)
258 << TimeStamp() << " Using ADIIS for next guess" << std::flush;
259 } else {
260 coeffs = diis_.CalcCoeff();
261 diis_error = !diis_.Info();
262 XTP_LOG(Log::warning, *log_)
263 << TimeStamp() << " Using DIIS for next guess" << std::flush;
264 }
265 if (diis_error) {
266 ++consecutive_failures_;
267 XTP_LOG(Log::warning, *log_)
268 << TimeStamp() << " (A)DIIS failed using mixing instead"
269 << std::flush;
270 } else {
271 consecutive_failures_ = 0;
272 H_guess.setZero();
273 for (Index i = 0; i < coeffs.size(); i++) {
274 if (std::abs(coeffs(i)) < 1e-8) {
275 continue;
276 }
277 H_guess += coeffs(i) * mathist_[i];
278 }
279 }
280 }
281
282 if (opt_.mode != KSmode::fractional && nocclevels_ > 0 &&
283 nocclevels_ < MOs_old_energies.size()) {
284 const double gap =
285 MOs_old_energies(nocclevels_) - MOs_old_energies(nocclevels_ - 1);
286 if ((diiserror_ > opt_.levelshiftend && opt_.levelshift > 0.0) ||
287 gap < 1e-6) {
288 Levelshift(H_guess, MOs_old);
289 }
290 }
291
292 MOs = SolveFockmatrix(H_guess);
293
294 Eigen::MatrixXd dmatout = DensityMatrix(MOs);
295 // mixing_end, not ADIIS_start, decides on damping (as in the UKS path):
296 // the two are separate options.
297 if (diiserror_ > opt_.mixingend || !opt_.usediis || diis_error ||
298 (mathist_.size() <= 2 && !near_convergence)) {
299 usedmixing_ = true;
300 dmatout =
301 opt_.mixingparameter * dmat + (1.0 - opt_.mixingparameter) * dmatout;
302 XTP_LOG(Log::warning, *log_)
303 << TimeStamp() << " Using Mixing with alpha=" << opt_.mixingparameter
304 << std::flush;
305 } else {
306 usedmixing_ = false;
307 }
308 return dmatout;
309}

311bool ConvergenceAcc::HistoryIsAligned() const {
312 const std::vector<Eigen::MatrixXd>& errors = diis_.ErrorHistory();
313 if (errors.size() != mathist_.size() || dmatHist_.size() != mathist_.size() ||
314 errhist_.size() != mathist_.size()) {
315 return false;
316 }
317 const Eigen::MatrixXd& S = S_->Matrix();
318 for (std::size_t k = 0; k < mathist_.size(); ++k) {
319 const Eigen::MatrixXd e =
320 Sminusahalf.transpose() *
321 (mathist_[k] * dmatHist_[k] * S - S * dmatHist_[k] * mathist_[k]) *
322 Sminusahalf;
323 if ((e - errors[k]).cwiseAbs().maxCoeff() > 1e-12 ||
324 std::abs(e.cwiseAbs().maxCoeff() - errhist_[k]) > 1e-12) {
325 return false;
326 }
327 }
328 return true;
329}

331bool ConvergenceAcc::SecondOrderStalled() const {
332 if (consecutive_failures_ >= kMaxConsecutiveFailures) {
333 return true;
334 }
335 if (scf_iterations_ < kAutoStartIteration ||
336 Index(error_history_.size()) < kTrailingWindow) {
337 return false;
338 }
339 double mean_ratio = 0.0;
340 Index count = 0;
341 for (std::size_t i = 1; i < error_history_.size(); ++i) {
342 if (error_history_[i - 1] > 1e-12) {
343 mean_ratio += error_history_[i] / error_history_[i - 1];
344 ++count;
345 }
346 }
347 return count > 0 && mean_ratio / double(count) > 1.0 / kMeanRatioTolerance;
348}

350// One TRAH step from the current orbitals. The gradient (from H) and the
351// Fletcher check (against totE) belong to dmat; if dmat is not the density
352// of MOs (a guess or damped density), the orbital density is returned first
353// and the step follows on the next call.
354Eigen::MatrixXd ConvergenceAcc::SecondOrderStep(const Eigen::MatrixXd& dmat,
355 const Eigen::MatrixXd& H,
356 tools::EigenSystem& MOs,
357 double totE) {
358 const Eigen::MatrixXd dmat_mos = DensityMatrix(MOs);
359 const double mismatch = (dmat_mos - dmat).cwiseAbs().maxCoeff();
360 if (mismatch > kDensityMismatchTolerance) {
361 XTP_LOG(Log::warning, *log_)
362 << TimeStamp()
363 << " Density differs from that of the current orbitals (max "
364 "deviation "
365 << mismatch << "); returning the orbital density first" << std::flush;
366 usedmixing_ = false;
367 return dmat_mos;
368 }
369
370 const Eigen::MatrixXd& C = MOs.eigenvectors();
371 const Eigen::MatrixXd F_MO = C.transpose() * H * C;
372 const Index nocc = nocclevels_;
373 const Index nvirt = C.cols() - nocc;
374 const Eigen::VectorXd g = EnergyGradientScale * trah::OccVirt(F_MO, nocc);
375 // diagonal of the energy Hessian, 4 (F_aa - F_ii), for preconditioning
376 Eigen::VectorXd diag(nocc * nvirt);
377 constexpr double kMinGap = 1e-3;
378 for (Index i = 0; i < nocc; ++i) {
379 for (Index a = 0; a < nvirt; ++a) {
380 diag(i * nvirt + a) =
381 4.0 *
382 std::max(std::abs(F_MO(nocc + a, nocc + a) - F_MO(i, i)), kMinGap);
383 }
384 }
385 const tools::EigenSystem MOs_now = MOs;
386 trah::HessianProduct hessian = [&](const Eigen::MatrixXd& X) {
387 return orbital_hessian_(MOs_now, H, X);
388 };
389 trah::StepResult step;
390 try {
391 step = trah::Step(g, diag, hessian, trust_.Radius(), trah_options_,
392 &trah_subspace_);
393 } catch (const std::runtime_error& error) {
394 XTP_LOG(Log::error, *log_)
395 << TimeStamp()
396 << " Second-order step not possible, continuing without: "
397 << error.what() << std::flush;
398 trah_active_ = false;
399 trah_disabled_ = true;
400 usedmixing_ = false;
401 return dmat_mos;
402 }
403 XTP_LOG(Log::warning, *log_)
404 << TimeStamp() << " TRAH step: |g|=" << g.norm()
405 << ", |x|=" << step.x.norm() << ", trust radius=" << trust_.Radius()
406 << ", alpha=" << step.alpha << ", micro-iterations=" << step.iterations
407 << ", Hessian products=" << step.products
408 << (step.reused ? " (subspace reused)" : "")
409 << ", residual=" << step.residual
410 << (step.converged ? "" : " (not converged)")
411 << ", predicted dE=" << step.predicted << std::flush;
412
413 trah_pre_energy_ = totE;
414 trah_pre_MOs_ = MOs;
415 trah_pre_H_ = H;
416 trah_predicted_ = step.predicted;
417 trah_linear_ = step.linear;
418 trah_step_norm_ = step.x.norm();
419 trah_pending_ = true;
420
421 const Eigen::MatrixXd C_new = trah::RotateOrbitals(C, step.x, nocc);
422 MOs.eigenvectors() = C_new;
423 // diagonal of the old Fock matrix in the rotated orbitals, for the level
424 // shift gap check and the preconditioner until the next diagonalisation
425 MOs.eigenvalues() = (C_new.transpose() * H * C_new).diagonal();
426 usedmixing_ = false;
427 return DensityMatrix(MOs);
428}

466// Solve the generalized Roothaan-Hall problem
467//
468// F C = S C eps
469//
470// by symmetric orthogonalization: diagonalize X^T F X with X = S^{-1/2} and
471// back-transform the eigenvectors as C = X C' .
472tools::EigenSystem ConvergenceAcc::SolveFockmatrix(
473 const Eigen::MatrixXd& H) const {
474 // transform to orthogonal for
475 Eigen::MatrixXd H_ortho = Sminusahalf.transpose() * H * Sminusahalf;
476 if (removed_projector_.size() > 0) {
477 H_ortho += kRemovedShift * removed_projector_;
478 }
479 const SymmetricEigenSystem es = SymmetricEigen(H_ortho);
480 if (!es.values.allFinite()) {
481 throw std::runtime_error("Matrix Diagonalisation failed");
482 }
483
484 tools::EigenSystem result;
485 result.eigenvalues() = es.values;
486 result.eigenvectors() = Sminusahalf * es.vectors;
487 return result;
488}

490// Add a virtual-space level shift
491//
492// F <- F + S C_virt Delta C_virt^T S,
493//
494// implemented by placing the scalar shift on the virtual diagonal in the MO
495// basis built from the previous iteration. This leaves occupied orbitals
496// unchanged while opening the HOMO-LUMO gap during difficult SCF phases.
497void ConvergenceAcc::Levelshift(Eigen::MatrixXd& H,
498 const Eigen::MatrixXd& MOs_old) const {
499 if (opt_.levelshift < 1e-9) {
500 return;
501 }
502 Eigen::VectorXd virt = Eigen::VectorXd::Zero(H.rows());
503 for (Index i = nocclevels_; i < H.rows(); i++) {
504 virt(i) = opt_.levelshift;
505 }
506
507 XTP_LOG(Log::error, *log_)
508 << TimeStamp() << " Using levelshift:" << opt_.levelshift << " Hartree"
509 << std::flush;
510 Eigen::MatrixXd vir = S_->Matrix() * MOs_old * virt.asDiagonal() *
511 MOs_old.transpose() * S_->Matrix();
512 H += vir;
513 return;
514}

530// Closed-shell AO density matrix
531//
532// P = 2 C_occ C_occ^T,
533//
534// where each occupied spatial orbital contributes one alpha and one beta
535// electron.
536Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState(
537 const Eigen::MatrixXd& MOs) const {
538 const Eigen::MatrixXd occstates = MOs.leftCols(nocclevels_);
539 Eigen::MatrixXd dmatGS = 2.0 * occstates * occstates.transpose();
540 return dmatGS;
541}
542
543// Spin-resolved unrestricted AO density matrix for one spin channel,
544//
545// P^sigma = C_occ^sigma (C_occ^sigma)^T,
546//
547// with no factor of two because a single spin channel is represented.
548Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState_unres(
549 const Eigen::MatrixXd& MOs) const {
550 if (nocclevels_ == 0) {
551 return Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
552 }
553 Eigen::MatrixXd occstates = MOs.leftCols(nocclevels_);
554 Eigen::MatrixXd dmatGS = occstates * occstates.transpose();
555 return dmatGS;
556}

558// Fractionally occupied AO density matrix
559//
560// P = C f C^T,
561//
562// where f is a diagonal matrix of orbital occupations assembled from the
563// configured electron count.
564// Fractional-occupation AO density matrix
565//
566// P = C n C^T,
567//
568// where n is the diagonal matrix of orbital occupations provided in the
569// EigenSystem container.
570Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState_frac(
571 const tools::EigenSystem& MOs) const {
572 if (opt_.numberofelectrons == 0) {
573 return Eigen::MatrixXd::Zero(MOs.eigenvectors().rows(),
574 MOs.eigenvectors().rows());
575 }
576
577 Eigen::VectorXd occupation = Eigen::VectorXd::Zero(MOs.eigenvalues().size());
578 std::vector<std::vector<Index> > degeneracies;
579 double buffer = 1e-4;
580 degeneracies.push_back(std::vector<Index>{0});
581 for (Index i = 1; i < occupation.size(); i++) {
582 if (MOs.eigenvalues()(i) <
583 MOs.eigenvalues()(degeneracies[degeneracies.size() - 1][0]) + buffer) {
584 degeneracies[degeneracies.size() - 1].push_back(i);
585 } else {
586 degeneracies.push_back(std::vector<Index>{i});
587 }
588 }
589 Index numofelec = opt_.numberofelectrons;
590 for (const std::vector<Index>& deglevel : degeneracies) {
591 Index numofpossibleelectrons = 2 * Index(deglevel.size());
592 if (numofpossibleelectrons <= numofelec) {
593 for (Index i : deglevel) {
594 occupation(i) = 2;
595 }
596 numofelec -= numofpossibleelectrons;
597 } else {
598 double occ = double(numofelec) / double(deglevel.size());
599 for (Index i : deglevel) {
600 occupation(i) = occ;
601 }
602 break;
603 }
604 }
605 Eigen::MatrixXd dmatGS = MOs.eigenvectors() * occupation.asDiagonal() *
606 MOs.eigenvectors().transpose();
607 return dmatGS;
608}

611 * EXTENSION FOR SPIN-KS-DFT
612 *******************************************************/

614ConvergenceAcc::SpinDensity
615 ConvergenceAcc::DensityMatrixGroundState_restricted_open(
616 const Eigen::MatrixXd& MOs) const {
617
618 const Index n_docc =
619 std::min(opt_.number_alpha_electrons, opt_.number_beta_electrons);
620 const Index n_socc_alpha = opt_.number_alpha_electrons - n_docc;
621
622 SpinDensity result;
623 result.alpha = Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
624 result.beta = Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
625
626 if (n_docc > 0) {
627 const Eigen::MatrixXd docc = MOs.leftCols(n_docc);
628 const Eigen::MatrixXd d_docc = docc * docc.transpose();
629 result.alpha += d_docc;
630 result.beta += d_docc;
631 }
632
633 if (n_socc_alpha > 0) {
634 const Eigen::MatrixXd socc = MOs.middleCols(n_docc, n_socc_alpha);
635 result.alpha += socc * socc.transpose();
636 }
637
638 return result;
639}

641// Construct spin-resolved densities according to the configured occupation
642// model. For restricted open-shell cases the same spatial orbitals are split
643// into doubly and singly occupied subsets using the stored alpha/beta counts.
644ConvergenceAcc::SpinDensity ConvergenceAcc::DensityMatrixSpinResolved(
645 const tools::EigenSystem& MOs) const {
646
647 if (opt_.mode == KSmode::restricted_open) {
648 return DensityMatrixGroundState_restricted_open(MOs.eigenvectors());
649 } else if (opt_.mode == KSmode::closed) {
650 Eigen::MatrixXd d = DensityMatrixGroundState(MOs.eigenvectors());
651 return {0.5 * d, 0.5 * d};
652 } else if (opt_.mode == KSmode::open) {
653 Eigen::MatrixXd d = DensityMatrixGroundState_unres(MOs.eigenvectors());
654 return {d, Eigen::MatrixXd::Zero(d.rows(), d.cols())};
655 } else {
656 Eigen::MatrixXd d = DensityMatrixGroundState_frac(MOs);
657 return {0.5 * d, 0.5 * d};
658 }
659}

662 const tools::EigenSystem& MOs) const {
663 SpinDensity spin_dmat = DensityMatrixSpinResolved(MOs);
664 return spin_dmat.total();
665}
