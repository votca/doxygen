49// Perform one SCF acceleration step.
50//
51// The commutator residual
52//
53// R = X^T (F P S - S P F) X
54//
55// vanishes at self-consistency in a non-orthogonal AO basis. Its maximum
56// element is used as the DIIS error metric, while the history of Fock and
57// density matrices is passed to either DIIS or ADIIS to construct the next
58// extrapolated Fock matrix. When the extrapolation is deemed unsafe, linear
59// density mixing is used instead.
60Eigen::MatrixXd ConvergenceAcc::Iterate(const Eigen::MatrixXd& dmat,
61 Eigen::MatrixXd& H,
62 tools::EigenSystem& MOs, double totE) {
63 Eigen::MatrixXd H_guess = Eigen::MatrixXd::Zero(H.rows(), H.cols());
64
65 if (int(mathist_.size()) == opt_.histlength) {
66 totE_.erase(totE_.begin() + maxerrorindex_);
67 mathist_.erase(mathist_.begin() + maxerrorindex_);
68 dmatHist_.erase(dmatHist_.begin() + maxerrorindex_);
69 }
70
71 totE_.push_back(totE);
72 if (opt_.mode != KSmode::fractional && nocclevels_ > 0 &&
73 nocclevels_ < MOs.eigenvalues().size()) {
74 double gap =
75 MOs.eigenvalues()(nocclevels_) - MOs.eigenvalues()(nocclevels_ - 1);
76 if ((diiserror_ > opt_.levelshiftend && opt_.levelshift > 0.0) ||
77 gap < 1e-6) {
78 Levelshift(H, MOs.eigenvectors());
79 }
80 }
81 const Eigen::MatrixXd& S = S_->Matrix();
82 Eigen::MatrixXd errormatrix =
83 Sminusahalf.transpose() * (H * dmat * S - S * dmat * H) * Sminusahalf;
84 diiserror_ = errormatrix.cwiseAbs().maxCoeff();
85
86 mathist_.push_back(H);
87 dmatHist_.push_back(dmat);
88
89 if (opt_.maxout) {
90 if (diiserror_ > maxerror_) {
91 maxerror_ = diiserror_;
92 maxerrorindex_ = mathist_.size() - 1;
93 }
94 }
95
96 diis_.Update(maxerrorindex_, errormatrix);
97 bool diis_error = false;
98 XTP_LOG(Log::error, *log_)
99 << TimeStamp() << " DIIs error " << getDIIsError() << std::flush;
100
101 XTP_LOG(Log::error, *log_)
102 << TimeStamp() << " Delta Etot " << getDeltaE() << std::flush;
103
104 if ((diiserror_ < opt_.adiis_start || diiserror_ < opt_.diis_start) &&
105 opt_.usediis && mathist_.size() > 2) {
106 Eigen::VectorXd coeffs;
107 // use ADIIs if energy has risen a lot in current iteration
108
109 if (diiserror_ > opt_.diis_start ||
110 totE_.back() > 0.9 * totE_[totE_.size() - 2]) {
111 coeffs = adiis_.CalcCoeff(dmatHist_, mathist_);
112 diis_error = !adiis_.Info();
113 XTP_LOG(Log::warning, *log_)
114 << TimeStamp() << " Using ADIIS for next guess" << std::flush;
115
116 } else {
117 coeffs = diis_.CalcCoeff();
118 diis_error = !diis_.Info();
119 XTP_LOG(Log::warning, *log_)
120 << TimeStamp() << " Using DIIS for next guess" << std::flush;
121 }
122 if (diis_error) {
123 XTP_LOG(Log::warning, *log_)
124 << TimeStamp() << " (A)DIIS failed using mixing instead"
125 << std::flush;
126 H_guess = H;
127 } else {
128 for (Index i = 0; i < coeffs.size(); i++) {
129 if (std::abs(coeffs(i)) < 1e-8) {
130 continue;
131 }
132 H_guess += coeffs(i) * mathist_[i];
133 }
134 }
135
136 } else {
137 H_guess = H;
138 }
139
140 MOs = SolveFockmatrix(H_guess);
141
142 Eigen::MatrixXd dmatout = DensityMatrix(MOs);
143 if (diiserror_ > opt_.adiis_start || !opt_.usediis || diis_error ||
144 mathist_.size() <= 2) {
145 usedmixing_ = true;
146 dmatout =
147 opt_.mixingparameter * dmat + (1.0 - opt_.mixingparameter) * dmatout;
148 XTP_LOG(Log::warning, *log_)
149 << TimeStamp() << " Using Mixing with alpha=" << opt_.mixingparameter
150 << std::flush;
151 } else {
152 usedmixing_ = false;
153 }
154 return dmatout;
155}

188// Solve the generalized Roothaan-Hall problem
189//
190// F C = S C eps
191//
192// by symmetric orthogonalization: diagonalize X^T F X with X = S^{-1/2} and
193// back-transform the eigenvectors as C = X C' .
194tools::EigenSystem ConvergenceAcc::SolveFockmatrix(
195 const Eigen::MatrixXd& H) const {
196 // transform to orthogonal for
197 Eigen::MatrixXd H_ortho = Sminusahalf.transpose() * H * Sminusahalf;
198 Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> es(H_ortho);
199
200 if (es.info() != Eigen::ComputationInfo::Success) {
201 throw std::runtime_error("Matrix Diagonalisation failed. DiagInfo" +
202 std::to_string(es.info()));
203 }
204
205 tools::EigenSystem result;
206 result.eigenvalues() = es.eigenvalues();
207
208 result.eigenvectors() = Sminusahalf * es.eigenvectors();
209 return result;
210}

212// Add a virtual-space level shift
213//
214// F <- F + S C_virt Delta C_virt^T S,
215//
216// implemented by placing the scalar shift on the virtual diagonal in the MO
217// basis built from the previous iteration. This leaves occupied orbitals
218// unchanged while opening the HOMO-LUMO gap during difficult SCF phases.
219void ConvergenceAcc::Levelshift(Eigen::MatrixXd& H,
220 const Eigen::MatrixXd& MOs_old) const {
221 if (opt_.levelshift < 1e-9) {
222 return;
223 }
224 Eigen::VectorXd virt = Eigen::VectorXd::Zero(H.rows());
225 for (Index i = nocclevels_; i < H.rows(); i++) {
226 virt(i) = opt_.levelshift;
227 }
228
229 XTP_LOG(Log::error, *log_)
230 << TimeStamp() << " Using levelshift:" << opt_.levelshift << " Hartree"
231 << std::flush;
232 Eigen::MatrixXd vir = S_->Matrix() * MOs_old * virt.asDiagonal() *
233 MOs_old.transpose() * S_->Matrix();
234 H += vir;
235 return;
236}

252// Closed-shell AO density matrix
253//
254// P = 2 C_occ C_occ^T,
255//
256// where each occupied spatial orbital contributes one alpha and one beta
257// electron.
258Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState(
259 const Eigen::MatrixXd& MOs) const {
260 const Eigen::MatrixXd occstates = MOs.leftCols(nocclevels_);
261 Eigen::MatrixXd dmatGS = 2.0 * occstates * occstates.transpose();
262 return dmatGS;
263}

265// Spin-resolved unrestricted AO density matrix for one spin channel,
266//
267// P^sigma = C_occ^sigma (C_occ^sigma)^T,
268//
269// with no factor of two because a single spin channel is represented.
270Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState_unres(
271 const Eigen::MatrixXd& MOs) const {
272 if (nocclevels_ == 0) {
273 return Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
274 }
275 Eigen::MatrixXd occstates = MOs.leftCols(nocclevels_);
276 Eigen::MatrixXd dmatGS = occstates * occstates.transpose();
277 return dmatGS;
278}

280// Fractionally occupied AO density matrix
281//
282// P = C f C^T,
283//
284// where f is a diagonal matrix of orbital occupations assembled from the
285// configured electron count.
286// Fractional-occupation AO density matrix
287//
288// P = C n C^T,
289//
290// where n is the diagonal matrix of orbital occupations provided in the
291// EigenSystem container.
292Eigen::MatrixXd ConvergenceAcc::DensityMatrixGroundState_frac(
293 const tools::EigenSystem& MOs) const {
294 if (opt_.numberofelectrons == 0) {
295 return Eigen::MatrixXd::Zero(MOs.eigenvectors().rows(),
296 MOs.eigenvectors().rows());
297 }
298
299 Eigen::VectorXd occupation = Eigen::VectorXd::Zero(MOs.eigenvalues().size());
300 std::vector<std::vector<Index> > degeneracies;
301 double buffer = 1e-4;
302 degeneracies.push_back(std::vector<Index>{0});
303 for (Index i = 1; i < occupation.size(); i++) {
304 if (MOs.eigenvalues()(i) <
305 MOs.eigenvalues()(degeneracies[degeneracies.size() - 1][0]) + buffer) {
306 degeneracies[degeneracies.size() - 1].push_back(i);
307 } else {
308 degeneracies.push_back(std::vector<Index>{i});
309 }
310 }
311 Index numofelec = opt_.numberofelectrons;
312 for (const std::vector<Index>& deglevel : degeneracies) {
313 Index numofpossibleelectrons = 2 * Index(deglevel.size());
314 if (numofpossibleelectrons <= numofelec) {
315 for (Index i : deglevel) {
316 occupation(i) = 2;
317 }
318 numofelec -= numofpossibleelectrons;
319 } else {
320 double occ = double(numofelec) / double(deglevel.size());
321 for (Index i : deglevel) {
322 occupation(i) = occ;
323 }
324 break;
325 }
326 }
327 Eigen::MatrixXd dmatGS = MOs.eigenvectors() * occupation.asDiagonal() *
328 MOs.eigenvectors().transpose();
329 return dmatGS;
330}

335
336ConvergenceAcc::SpinDensity
337 ConvergenceAcc::DensityMatrixGroundState_restricted_open(
338 const Eigen::MatrixXd& MOs) const {
339
340 const Index n_docc =
341 std::min(opt_.number_alpha_electrons, opt_.number_beta_electrons);
342 const Index n_socc_alpha = opt_.number_alpha_electrons - n_docc;
343
344 SpinDensity result;
345 result.alpha = Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
346 result.beta = Eigen::MatrixXd::Zero(MOs.rows(), MOs.rows());
347
348 if (n_docc > 0) {
349 const Eigen::MatrixXd docc = MOs.leftCols(n_docc);
350 const Eigen::MatrixXd d_docc = docc * docc.transpose();
351 result.alpha += d_docc;
352 result.beta += d_docc;
353 }
354
355 if (n_socc_alpha > 0) {
356 const Eigen::MatrixXd socc = MOs.middleCols(n_docc, n_socc_alpha);
357 result.alpha += socc * socc.transpose();
358 }
359
360 return result;
361}

363// Construct spin-resolved densities according to the configured occupation
364// model. For restricted open-shell cases the same spatial orbitals are split
365// into doubly and singly occupied subsets using the stored alpha/beta counts.
366ConvergenceAcc::SpinDensity ConvergenceAcc::DensityMatrixSpinResolved(
367 const tools::EigenSystem& MOs) const {
368
369 if (opt_.mode == KSmode::restricted_open) {
370 return DensityMatrixGroundState_restricted_open(MOs.eigenvectors());
371 } else if (opt_.mode == KSmode::closed) {
372 Eigen::MatrixXd d = DensityMatrixGroundState(MOs.eigenvectors());
373 return {0.5 * d, 0.5 * d};
374 } else if (opt_.mode == KSmode::open) {
375 Eigen::MatrixXd d = DensityMatrixGroundState_unres(MOs.eigenvectors());
376 return {d, Eigen::MatrixXd::Zero(d.rows(), d.cols())};
377 } else {
378 Eigen::MatrixXd d = DensityMatrixGroundState_frac(MOs);
379 return {0.5 * d, 0.5 * d};
380 }
381}

383Eigen::MatrixXd ConvergenceAcc::DensityMatrix(
384 const tools::EigenSystem& MOs) const {
385 SpinDensity spin_dmat = DensityMatrixSpinResolved(MOs);
386 return spin_dmat.total();
387}
